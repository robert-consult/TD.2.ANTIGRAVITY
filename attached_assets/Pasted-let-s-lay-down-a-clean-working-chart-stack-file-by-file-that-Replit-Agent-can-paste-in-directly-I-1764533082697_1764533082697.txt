  "1M": "D",
};

  // Fallback: assume CMCMARKETS:<symbol> if not in map

export function getTradingViewInterval(period: ChartPeriod): string {
  return periodIntervalMap[period] || "D";
}

// client/src/modules/tradingview/TradingViewChart.tsx
import { useEffect, useRef, useState, useId } from "react";

declare global {
  interface Window {
    TradingView?: any;
  }
}

// -------- Shared tv.js loader (only once per app) --------
let tvScriptPromise: Promise<void> | null = null;

function loadTradingViewScript(): Promise<void> {
  if (typeof window === "undefined") return Promise.resolve();

  // Already loaded
  if (window.TradingView) return Promise.resolve();

  // Already loading
  if (tvScriptPromise) return tvScriptPromise;

  tvScriptPromise = new Promise<void>((resolve, reject) => {
    const existing = document.getElementById("tradingview-script");
    if (existing) {
      existing.addEventListener("load", () => resolve(), { once: true });
      existing.addEventListener(
        "error",
        () => reject(new Error("tv.js failed to load")),
        { once: true }
      );
      return;
    }

    const script = document.createElement("script");
    script.id = "tradingview-script";
    script.type = "text/javascript";
    script.src = "https://s3.tradingview.com/tv.js";
    script.async = true;

    script.onload = () => {
      resolve();
    };

    script.onerror = (err) => {
      console.error("Error loading TradingView tv.js:", err);
      reject(err instanceof Error ? err : new Error("tv.js failed to load"));
    };

    document.head.appendChild(script);
  });

  return tvScriptPromise;
}

// -------- Component --------

interface TradingViewChartProps {
  symbol: string;   // e.g. "CMCMARKETS:USDJPY"
  interval: string; // e.g. "5", "30", "240", "D"
  theme?: "light" | "dark";
  backgroundColor?: string;
  className?: string;
}

/**
 * Advanced Chart wrapper:
 * - loads tv.js once
 * - creates widget once
 * - uses chart().setSymbol(...) on symbol/interval changes
 */
export function TradingViewChart({
  symbol,
  interval,
  theme = "dark",
  backgroundColor = "#0F0F0F",
  className = "",
}: TradingViewChartProps) {
  const containerRef = useRef<HTMLDivElement | null>(null);
  const widgetRef = useRef<any>(null);
  const id = useId();
  const containerId = `tv-chart-${id.replace(/:/g, "")}`;
  const [ready, setReady] = useState(false);

  useEffect(() => {
    let cancelled = false;

    async function init() {
      // 1) Ensure tv.js is loaded
      await loadTradingViewScript();

      if (cancelled) return;
      if (!containerRef.current || !window.TradingView) return;

      // 2) If widget doesn't exist yet -> create it
      if (!widgetRef.current) {
        widgetRef.current = new window.TradingView.widget({
          autosize: true,
          symbol,
          interval,
          timezone: "Etc/UTC",
          theme,
          style: "1",
          locale: "en",
          enable_publishing: false,
          allow_symbol_change: true,
          container_id: containerId,
          hide_top_toolbar: false,
          hide_side_toolbar: false,
          withdateranges: true,
          studies: ["RSI@tv-basicstudies", "MASimple@tv-basicstudies"],
          backgroundColor,
          toolbar_bg: backgroundColor,
        });

        if (widgetRef.current.onChartReady) {
          widgetRef.current.onChartReady(() => {
            if (!cancelled) setReady(true);
          });
        }

        return;
      }

      // 3) If widget exists -> change symbol/interval dynamically
      try {
        const chart = widgetRef.current.chart?.();
        if (chart && typeof chart.setSymbol === "function") {
          chart.setSymbol(symbol, interval, () => {
            // optional: callback once symbol applied
          });
        } else {
          // Fallback: recreate widget if chart API not available
          try {
            widgetRef.current.remove?.();
          } catch (e) {
            console.warn("Error removing TradingView widget, recreating:", e);
          }
          widgetRef.current = null;
          init();
        }
      } catch (e) {
        console.error("Error changing TradingView symbol:", e);
      }
    }

    init();

    return () => {
      cancelled = true;
      // Optional: clean up widget on unmount
      try {
        widgetRef.current?.remove?.();
        widgetRef.current = null;
      } catch (e) {
        console.warn("Error cleaning TradingView widget:", e);
      }
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [symbol, interval, theme, backgroundColor, containerId]);

  return (
    <div
      id={containerId}
      ref={containerRef}
      className={className}
      style={{ height: "100%", width: "100%" }}
      data-tv-ready={ready ? "true" : "false"}
    />
  );
}

// client/src/pages/ChartScreen.tsx
import { useQuotes } from "@/hooks/use-quotes";
import { useState } from "react";
import { Button } from "@/components/ui/button";
import { Skeleton } from "@/components/ui/skeleton";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import { toast } from "@/hooks/use-toast";
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { apiRequest } from "@/lib/queryClient";

import {
  TradingViewChart,
  type ChartPeriod,
  getTradingViewSymbol,
  getTradingViewInterval,
} from "@/modules/tradingview";

interface ChartScreenProps {
  selectedSymbol: string;
}

export function ChartScreen({ selectedSymbol }: ChartScreenProps) {
  const [activePeriod, setActivePeriod] = useState<ChartPeriod>("1D");
  const { quotes, isLoading } = useQuotes();
  const currentQuote = quotes?.find((quote) => quote.symbol === selectedSymbol);

  // Map internal symbol + period to TradingView
  const tvSymbol = getTradingViewSymbol(selectedSymbol);
  const tvInterval = getTradingViewInterval(activePeriod);

  const [tradeDirection, setTradeDirection] = useState<"BUY" | "SELL" | null>(
    null
  );
  const [lots, setLots] = useState<number>(1);
  const [currentTradeType, setCurrentTradeType] = useState<
    "BUY" | "SELL" | null
  >(null);
  const queryClient = useQueryClient();

  const executeTrade = useMutation({
    mutationFn: async (data: {
      symbol: string;
      type: "BUY" | "SELL";
      orderType: string;
      lots: number;
      openPrice: number;
    }) => {
      return apiRequest("POST", "/api/trades", data);
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["/api/trades"] });
      queryClient.invalidateQueries({ queryKey: ["/api/trades/open"] });
      queryClient.invalidateQueries({ queryKey: ["/api/auth/current-user"] });

      toast({
        title: "Trade Executed",
        description: "Your market order has been placed.",
      });

      setTradeDirection(null);
      setCurrentTradeType(null);
    },
    onError: (error: any) => {
      console.error("Error executing trade:", error);

      toast({
        title: "Trade Failed",
        description:
          error?.message ||
          "There was a problem executing your trade. Please try again.",
        variant: "destructive",
      });

      setTradeDirection(null);
      setCurrentTradeType(null);
    },
  });

  const handleTrade = (type: "buy" | "sell") => {
    if (!currentQuote) {
      toast({
        title: "No quote available",
        description: "Please wait for market data to load and try again.",
        variant: "destructive",
      });
      return;
    }

    const tradeType: "BUY" | "SELL" = type.toUpperCase() === "BUY" ? "BUY" : "SELL";
    const price = tradeType === "BUY" ? currentQuote.ask : currentQuote.bid;

    setTradeDirection(tradeType);
    setCurrentTradeType(tradeType);

    executeTrade.mutate({
      symbol: selectedSymbol,
      type: tradeType,
      orderType: "MARKET",
      lots,
      openPrice: price,
    });
  };

  const isJpyPair = selectedSymbol.includes("JPY");
  const pricePrecision = isJpyPair ? 2 : 4;
  const pipMultiplier = isJpyPair ? 100 : 10000;

  return (
    <div className="h-full flex flex-col bg-[#0F0F0F]">
      {/* Top bar with title and quote information */}
      <div className="border-b border-neutral-800 px-4 py-2 flex items-center justify-between">
        <div>
          <h1 className="text-sm font-semibold text-white md:text-base">
            Chart: {selectedSymbol}
          </h1>
          <p className="text-xs text-gray-400">
            Interactive TradingView chart with advanced tools
          </p>
        </div>

        {/* Current Quote Summary */}
        <div className="text-xs md:text-sm text-right text-gray-300">
          {isLoading ? (
            <div className="flex items-center gap-2">
              <Skeleton className="h-4 w-24 bg-neutral-700" />
              <Skeleton className="h-4 w-24 bg-neutral-700" />
            </div>
          ) : currentQuote ? (
            <div className="space-y-0.5">
              <div className="flex gap-4 items-center justify-end">
                <span className="text-gray-400">Bid:</span>
                <span className="font-mono text-danger-500 font-medium">
                  {currentQuote.bid.toFixed(pricePrecision)}
                </span>
                <span className="text-gray-400">Ask:</span>
                <span className="font-mono text-success-500 font-medium">
                  {currentQuote.ask.toFixed(pricePrecision)}
                </span>
              </div>
              <div className="flex gap-4 items-center justify-end">
                <span className="text-gray-400">Spread:</span>
                <span className="font-mono text-white">
                  {(
                    (currentQuote.ask - currentQuote.bid) *
                    pipMultiplier
                  ).toFixed(1)}{" "}
                  pips
                </span>
              </div>
            </div>
          ) : (
            <span className="text-gray-500">No market data</span>
          )}
        </div>
      </div>

      {/* Main content area */}
      <div className="flex-1 flex flex-col lg:flex-row">
        {/* Chart area */}
        <div className="flex-1 min-h-0 relative bg-neutral-900">
          {/* TradingView chart */}
          <TradingViewChart
            symbol={tvSymbol}
            interval={tvInterval}
            theme="dark"
            backgroundColor="#0F0F0F"
            className="w-full h-[55vh] md:h-[70vh] lg:h-full"
          />

          {/* Chart period controls overlay */}
          <div className="absolute bottom-3 left-3 bg-black bg-opacity-80 rounded-full px-3 py-1.5 flex items-center gap-1">
            {(["1D", "1H", "1W", "1M"] as ChartPeriod[]).map((period) => (
              <button
                key={period}
                onClick={() => setActivePeriod(period)}
                className={`px-2.5 py-1 rounded-full text-xs font-medium transition-colors ${
                  activePeriod === period
                    ? "bg-blue-500 text-white"
                    : "bg-transparent text-gray-300 hover:bg-white/10"
                }`}
              >
                {period}
              </button>
            ))}
          </div>

          {/* Current price label overlay */}
          {currentQuote && (
            <div className="absolute top-4 right-4 bg-black bg-opacity-75 rounded p-2 border border-gray-800">
              <div className="grid grid-cols-2 gap-2 text-sm">
                <span className="text-gray-400">Bid:</span>
                <span className="font-mono text-danger-500 font-medium text-right">
                  {currentQuote.bid.toFixed(pricePrecision)}
                </span>
                <span className="text-gray-400">Ask:</span>
                <span className="font-mono text-success-500 font-medium text-right">
                  {currentQuote.ask.toFixed(pricePrecision)}
                </span>
                <span className="text-gray-400">Spread:</span>
                <span className="font-mono text-white text-right">
                  {(
                    (currentQuote.ask - currentQuote.bid) *
                    pipMultiplier
                  ).toFixed(1)}{" "}
                  pips
                </span>
              </div>
            </div>
          )}
        </div>

        {/* Trade panel */}
        <div className="w-full lg:w-80 border-t border-neutral-800 lg:border-t-0 lg:border-l bg-[#111111] p-4 space-y-4">
          <div>
            <h2 className="text-sm font-semibold text-white mb-1">
              Quick Trade
            </h2>
            <p className="text-xs text-gray-400">
              Market order at current bid/ask price.
            </p>
          </div>

          {/* Trade type buttons */}
          <div className="flex space-x-3">
            <Button
              className={`flex-1 font-bold ${
                tradeDirection === "SELL"
                  ? "bg-red-600 hover:bg-red-700"
                  : "bg-red-500 hover:bg-red-600"
              } text-white`}
              onClick={() => handleTrade("sell")}
              disabled={executeTrade.isPending}
            >
              {executeTrade.isPending && currentTradeType === "SELL" ? (
                <span className="animate-pulse">Processing...</span>
              ) : currentQuote && currentQuote.bid ? (
                `SELL @ ${currentQuote.bid.toFixed(pricePrecision)}`
              ) : (
                "SELL"
              )}
            </Button>

            <div className="flex items-center space-x-2">
              <span className="font-bold text-gray-400">LOTS</span>
              <Select
                value={lots.toString()}
                onValueChange={(value) => setLots(Number(value))}
                disabled={executeTrade.isPending}
              >
                <SelectTrigger className="w-16 bg-neutral-800 border-gray-700">
                  <SelectValue placeholder="1" />
                </SelectTrigger>
                <SelectContent className="max-h-[calc(5*2.25rem)] overflow-y-auto w-16 bg-neutral-900 border-gray-700">
                  {Array.from({ length: 50 }, (_, i) => i + 1).map((lot) => (
                    <SelectItem
                      key={lot}
                      value={lot.toString()}
                      className="text-white hover:bg-neutral-800"
                    >
                      {lot}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </div>

            <Button
              className={`flex-1 font-bold ${
                tradeDirection === "BUY"
                  ? "bg-lime-600 hover:bg-lime-700"
                  : "bg-lime-500 hover:bg-lime-600"
              } text-white`}
              onClick={() => handleTrade("buy")}
              disabled={executeTrade.isPending}
            >
              {executeTrade.isPending && currentTradeType === "BUY" ? (
                <span className="animate-pulse">Processing...</span>
              ) : currentQuote && currentQuote.ask ? (
                `BUY @ ${currentQuote.ask.toFixed(pricePrecision)}`
              ) : (
                "BUY"
              )}
            </Button>
          </div>
        </div>
      </div>
    </div>
  );
}

import { useState } from "react";
import { useAuth } from "@/hooks/use-auth";
import QuotesScreen from "./QuotesScreen";
import ChartScreen from "./ChartScreen";
import TradeScreen from "./TradeScreen";
import HistoryScreen from "./HistoryScreen";
import { Header } from "@/components/Header";
import { MobileNavigation, SideNavigation } from "@/components/Navigation";

export default function Dashboard() {
  const { user } = useAuth();
  const [activeTab, setActiveTab] = useState<"quotes" | "chart" | "trade" | "history">("quotes");
  const [selectedSymbol, setSelectedSymbol] = useState<string>("USDJPY");

  const handleSelectSymbol = (symbol: string) => {

    if (activeTab !== "chart" && activeTab !== "trade") {
      setActiveTab("chart");
    }
  };

  return (
    <div className="h-screen flex flex-col overflow-hidden bg-neutral-950 text-white">

            <QuotesScreen onSelectSymbol={handleSelectSymbol} />
          )}

          {activeTab === "chart" && (
            <ChartScreen selectedSymbol={selectedSymbol} />
          )}

          {activeTab === "trade" && (
            <TradeScreen selectedSymbol={selectedSymbol} />
          )}

          {activeTab === "history" && <HistoryScreen />}
        </main>
      </div>

      {/* Mobile bottom navigation */}
      <MobileNavigation activeTab={activeTab} setActiveTab={setActiveTab} />
    </div>
  );
