## 2) `server/db/ensureSchema.ts` ✅ (adds columns + creates new tables)

> Adds freeze columns and creates `user_login_history` + `user_account_events` tables + indexes.

```ts
/** Runtime schema migration for SQLite */
import Database from "better-sqlite3";

function getDb() {
  return new Database("./trading_app.db");
}

function hasColumn(db: Database.Database, table: string, col: string): boolean {
  const rows = db.prepare(`PRAGMA table_info(${table})`).all() as any[];
  return rows.some(r => r.name === col);
}

function addColumnIfMissing(db: Database.Database, table: string, col: string, ddl: string) {
  if (!hasColumn(db, table, col)) db.exec(ddl);
}

export function ensureUsersColumns() {
  const db = getDb();
  try {
    addColumnIfMissing(db, "users", "is_disabled", `ALTER TABLE users ADD COLUMN is_disabled INTEGER NOT NULL DEFAULT 0;`);
    addColumnIfMissing(db, "users", "is_frozen", `ALTER TABLE users ADD COLUMN is_frozen INTEGER NOT NULL DEFAULT 0;`);
    addColumnIfMissing(db, "users", "freeze_reason_code", `ALTER TABLE users ADD COLUMN freeze_reason_code TEXT;`);
    addColumnIfMissing(db, "users", "freeze_reason_text", `ALTER TABLE users ADD COLUMN freeze_reason_text TEXT;`);
    addColumnIfMissing(db, "users", "frozen_at", `ALTER TABLE users ADD COLUMN frozen_at INTEGER;`);
    addColumnIfMissing(db, "users", "frozen_by", `ALTER TABLE users ADD COLUMN frozen_by INTEGER;`);
  } finally {
    db.close();
  }
}

export function ensureUserLoginHistoryTable() {
  const db = getDb();
  try {
    db.exec(`
      CREATE TABLE IF NOT EXISTS user_login_history (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        user_id INTEGER,
        email TEXT NOT NULL,
        ip TEXT,
        user_agent TEXT,
        success INTEGER NOT NULL,
        failure_reason TEXT,
        created_at INTEGER NOT NULL DEFAULT (strftime('%s','now')),
        FOREIGN KEY (user_id) REFERENCES users(id)
      );
    `);
    db.exec(`CREATE INDEX IF NOT EXISTS idx_user_login_history_user_id_created_at ON user_login_history(user_id, created_at);`);
  } finally {
    db.close();
  }
}

export function ensureUserAccountEventsTable() {
  const db = getDb();
  try {
    db.exec(`
      CREATE TABLE IF NOT EXISTS user_account_events (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        user_id INTEGER NOT NULL,
        admin_id INTEGER,
        event_type TEXT NOT NULL,
        title TEXT NOT NULL,
        description TEXT,
        reason_code TEXT,
        reason_text TEXT,
        metadata TEXT,
        created_at INTEGER NOT NULL DEFAULT (strftime('%s','now')),
        FOREIGN KEY (user_id) REFERENCES users(id),
        FOREIGN KEY (admin_id) REFERENCES users(id)
      );
    `);
    db.exec(`CREATE INDEX IF NOT EXISTS idx_user_account_events_user_id_created_at ON user_account_events(user_id, created_at);`);
  } finally {
    db.close();
  }
}

function getClientIp(req: Request): string {
  const xff = req.headers["x-forwarded-for"];
  if (typeof xff === "string" && xff.trim().length) return xff.split(",")[0].trim();
  if (Array.isArray(xff) && xff.length) return String(xff[0]);
  return (req.ip || (req.socket && req.socket.remoteAddress) || "");
}

// User activity timeline (admin view)
const [timelineOpen, setTimelineOpen] = useState(false);
const [timelineUser, setTimelineUser] = useState<any | null>(null);

const timelineUrl = timelineUser ? `/api/admin/users/${timelineUser.id}/timeline?limit=200` : null;
const {
  data: timelineEvents = [],
  isLoading: timelineLoading,
  error: timelineError,
} = useQuery<any[]>({
  queryKey: [timelineUrl || "__no_timeline__"],
  queryFn: async () => {
    if (!timelineUrl) return [];
    const res = await fetch(timelineUrl);
    if (!res.ok) throw new Error("Failed to load timeline");
    return res.json();
  },
  enabled: !!timelineUrl && timelineOpen,
});
```

## ✅ Vertical Timeline Dialog (dot/line UI)

```tsx
{/* User Timeline Dialog */}
<Dialog
  open={timelineOpen}
  onOpenChange={(open) => {
    setTimelineOpen(open);
    if (!open) setTimelineUser(null);
  }}
>
  <DialogContent className="max-w-3xl bg-gray-900 text-white border-gray-700">
    <DialogHeader>
      <DialogTitle>User Activity Timeline</DialogTitle>
      <DialogDescription className="text-gray-400">
        {timelineUser ? `${timelineUser.username} (${timelineUser.email})` : ""}
      </DialogDescription>
    </DialogHeader>

    <div className="max-h-[70vh] overflow-y-auto pr-2">
      {timelineLoading && (
        <div className="text-gray-400 text-sm">Loading timeline…</div>
      )}

      {!timelineLoading && timelineError && (
        <div className="text-red-400 text-sm">Failed to load timeline.</div>
      )}

      {!timelineLoading && !timelineError && timelineEvents.length === 0 && (
        <div className="text-gray-400 text-sm">No activity yet.</div>
      )}

      {!timelineLoading && !timelineError && timelineEvents.length > 0 && (
        <div className="relative pl-6">
          {timelineEvents.map((e: any, idx: number) => {
            const isLast = idx === timelineEvents.length - 1;

            const ts = typeof e.at === "number"
              ? e.at
              : (typeof e.createdAt === "number" ? e.createdAt : 0);

            const d = ts ? new Date(ts * 1000) : null;
            const timeLabel = d && !isNaN(d.getTime()) ? d.toLocaleString() : "";

            return (
              <div key={e.id || `${e.type}-${idx}`} className="relative pb-6">
                {/* Dot marker */}
                <span className="absolute left-0 top-1.5 flex h-3 w-3 items-center justify-center">
                  <span className="h-2 w-2 rounded-full bg-white/70" />
                </span>

                {/* Vertical line */}
                {!isLast && (
                  <span className="absolute left-[5px] top-3 h-full w-px bg-gray-700" />
                )}

                <div className="space-y-1">
                  <div className="flex flex-wrap items-center gap-x-2 gap-y-1">
                    <span className="text-sm font-semibold">
                      {e.title || e.type}
                    </span>
                    {timeLabel && (
                      <span className="text-xs text-gray-400">{timeLabel}</span>
                    )}
                    {e.source && (
                      <span className="text-xs text-gray-500">• {e.source}</span>
                    )}
                  </div>

                  {e.description && (
                    <div className="text-sm text-gray-300">{e.description}</div>
                  )}

                  {e.metadata && (
                    <pre className="text-xs text-gray-400 bg-gray-950/60 border border-gray-800 rounded p-2 overflow-x-auto">
                      {JSON.stringify(e.metadata, null, 2)}
                    </pre>
                  )}
                </div>
              </div>
            );
          })}
        </div>
      )}
    </div>
  </DialogContent>
</Dialog>
```
