function normalizeEpochMs(value: unknown, fallbackMs: number): number {
  const n = typeof value === "number" ? value : Number(value);
  if (!Number.isFinite(n)) return fallbackMs;
  return n < 1e12 ? n * 1000 : n;
}

async function loadPersistedValkeyQuoteRows(symbols: string[]): Promise<PersistedValkeyQuoteRow[]> {
  if (!symbols.length) return [];
  const v = getValkey();
  if (!v) return [];

  const keys = symbols.map((s) => `q:v1:${normalizeSymbol(s)}`);
  try {
    const values = await v.mget(...keys);
    const rows: PersistedValkeyQuoteRow[] = [];
    for (const raw of values) {
      if (!raw) continue;
      try {
        const parsed = JSON.parse(raw);
        if (!parsed?.symbol) continue;
        rows.push({
          symbol: normalizeSymbol(parsed.symbol),
          bid: typeof parsed.bid === "number" ? parsed.bid : parsed.bid == null ? null : Number(parsed.bid),
          ask: typeof parsed.ask === "number" ? parsed.ask : parsed.ask == null ? null : Number(parsed.ask),
          price: typeof parsed.price === "number" ? parsed.price : parsed.price == null ? null : Number(parsed.price),
          lastApiUpdate: normalizeEpochMs(parsed.lastApiUpdate, Date.now()),
          isStale: Boolean(parsed.isStale),
        });
      } catch {
        // ignore malformed entries
      }
    }
    return rows;
  } catch {
    return [];
  }
}

function updateSnapshotCache(quotes: any[], options: { markSuccess?: boolean } = {}) {
  const now = Date.now();
  for (const quote of quotes) {
    if (!quote?.symbol) continue;
    const symbol = normalizeSymbol(String(quote.symbol));
    const updatedAt = normalizeEpochMs(quote.lastUpdated, now);
    const consecutiveFailures = typeof quote.consecutiveFailures === "number" ? quote.consecutiveFailures : 0;
    quoteSnapshotCache.set(symbol, {
      symbol,
      price: quote.price,
      bid: quote.bid,
      ask: quote.ask,
      lastUpdated: updatedAt,
      isStale: Boolean(quote.isStale),
      consecutiveFailures,
    });
  }
  if (options.markSuccess !== false) {
    lastSuccessfulApiCall = now;
    consecutiveApiFailures = 0;
  }
}

async function fetchForgeQuotes(symbols: string[]) {
  const nowMs = Date.now();
  const staleThresholdMs =
    Number.isFinite(dynamicConfig.staleThresholdMs) && dynamicConfig.staleThresholdMs > 0
      ? dynamicConfig.staleThresholdMs
      : DEFAULT_STALE_MS;

  const formattedSymbols = formatSymbolsForForgeAPI(symbols);
  if (!formattedSymbols) return [];
  const response = await axios.get(
    `https://api.1forge.com/quotes?pairs=${formattedSymbols}&api_key=${API_KEY}`,
  );
  let quotesData = response.data;
  if (quotesData && typeof quotesData === "object" && !Array.isArray(quotesData)) {
    const errMsg = extractForgeErrorMessage(quotesData);
    if (errMsg) throw new Error(errMsg);
    if (quotesData.quotes) quotesData = quotesData.quotes;
  }

  if (!Array.isArray(quotesData) || quotesData.length === 0) {
    throw new Error("Empty 1Forge response");
  }

  return quotesData
    .map((quote: any) => {
      const formattedSymbol = quote.s ? quote.s.replace("/", "") : quote.symbol ? quote.symbol.replace("/", "") : null;
      const price = quote.p ?? quote.price ?? quote.mid ?? 0;
      const bid = quote.b ?? quote.bid ?? price * 0.9999;
      const ask = quote.a ?? quote.ask ?? price * 1.0001;
      const apiTimestamp = typeof quote.t === "number" ? quote.t : typeof quote.timestamp === "number" ? quote.timestamp : null;
      const lastUpdated = apiTimestamp != null ? (apiTimestamp > 1e12 ? apiTimestamp : apiTimestamp * 1000) : Date.now();
      const ageMs = nowMs - lastUpdated;
      const marketOpen = isMarketOpenForSymbol(formattedSymbol, new Date(nowMs));
      return {
        symbol: formattedSymbol,
        price,
        bid,
        ask,
        timestamp: Math.floor(lastUpdated / 1000),
        lastUpdated,
        isStale: marketOpen && ageMs > staleThresholdMs,
      };
    })
    .filter((q: any) => q.symbol && q.price);
}

      bid: row.bid,
      ask: row.ask,
      lastUpdated,
      isStale: Boolean(row.isStale),
      consecutiveFailures: 0,
    });
    recoveredForPublish.push({
      symbol: row.symbol,
      price: row.price,
      bid: row.bid,
      ask: row.ask,
      timestamp: Math.floor(lastUpdated / 1000),
      lastUpdated,
      isStale: true,
      consecutiveFailures: consecutiveApiFailures,
    });
  }

  const stillMissing = missing.filter((sym) => !valkeyFound.has(sym));
  if (stillMissing.length) {
    const dbRows = await loadPersistedDbQuoteRows(stillMissing);
    if (dbRows.length) {
      recoveredForCache.push(...dbRows);
      for (const row of dbRows) {
        const lastUpdated = normalizeEpochMs(row.lastUpdated, now);
        recoveredForPublish.push({
          symbol: row.symbol,
          price: row.price,
          bid: row.bid,
          ask: row.ask,
          timestamp: Math.floor(lastUpdated / 1000),
          lastUpdated,
          isStale: true,
          consecutiveFailures: consecutiveApiFailures,
        });

    }
  }

  if (recoveredForCache.length) {
    updateSnapshotCache(recoveredForCache, { markSuccess: false });
  }

  // Never synthesize random prices on API failures; if we cannot recover anything, skip the batch.
  const recovered = recoveredForPublish.filter((q) => {
    if (!q?.symbol) return false;
    return typeof q.price === "number" || typeof q.bid === "number" || typeof q.ask === "number";
  });

async function persistSnapshotToValkey(rows: LiveQuoteRow[], meta: { seq: number; asOf: number; source: string }) {
  const v = getValkey();
  if (!v) return;
  const snapshotTtl = Number.isFinite(QUOTE_SNAPSHOT_TTL_SEC) && QUOTE_SNAPSHOT_TTL_SEC > 0 ? QUOTE_SNAPSHOT_TTL_SEC : 30;
  const symbolTtl = Number.isFinite(QUOTE_SYMBOL_TTL_SEC) && QUOTE_SYMBOL_TTL_SEC > 0 ? QUOTE_SYMBOL_TTL_SEC : 60;
  const payload = {
    seq: meta.seq,
    asOf: meta.asOf,
    source: meta.source,
    rows,
  };
  try {
    const pipeline = v.pipeline();
    pipeline.set(QUOTE_SNAPSHOT_KEY, JSON.stringify(payload), "EX", snapshotTtl);
    for (const row of rows) {
      pipeline.set(`q:v1:${row.symbol}`, JSON.stringify(row), "EX", symbolTtl);
    }
    await pipeline.exec();
  } catch {
    // ignore cache write failures
  }
}

async function writeToRollingBufferBatch(rows: any[]) {

    if (!q?.symbol) return Promise.resolve(false);
    return writeToRollingBuffer(q.symbol, {
      bid: typeof q.bid === "number" ? q.bid : null,
      ask: typeof q.ask === "number" ? q.ask : null,
      price: typeof q.price === "number" ? q.price : null,
      lastApiUpdate: typeof q.lastUpdated === "number" ? q.lastUpdated : Date.now(),
    });
  });
  await Promise.all(promises);
}

async function persistDailyClose(rows: any[]) {
  if (!rows.length) return;
  const client = await dbClient.connect();
  const nowMs = Date.now();
  const nowSec = Math.floor(nowMs / 1000);
  try {
    await client.query("BEGIN");
    for (const q of rows) {
      if (!q?.symbol) continue;
      const lastUpdatedMs = typeof q.lastUpdated === "number" ? q.lastUpdated : nowMs;
      const bid = typeof q.bid === "number" ? q.bid : null;
      const ask = typeof q.ask === "number" ? q.ask : null;
      const mid = bid != null && ask != null ? (bid + ask) / 2 : typeof q.price === "number" ? q.price : null;
      if (!Number.isFinite(mid) || (mid as number) <= 0) continue;
      const sessionDay = computeSessionDayForQuote(lastUpdatedMs, {
        rolloverTz: dynamicConfig.rolloverTz,
        rolloverTime: dynamicConfig.rolloverTime,
      });
      const result = await client.query(
        `
        INSERT INTO market_daily_close (symbol, session_day, close, close_ts_ms, updated_at)
        VALUES ($1, $2, $3, $4, $5)
        ON CONFLICT (symbol, session_day) DO NOTHING
        RETURNING symbol
        `,
        [q.symbol, sessionDay, mid, lastUpdatedMs, nowSec],
      );
      // If a new session close was inserted, cache it as prevClose for next session
      if (result.rowCount && result.rowCount > 0) {
        await cachePrevClose(q.symbol, mid as number);
      }
    }
    await client.query("COMMIT");
  } catch (error) {
    await client.query("ROLLBACK");
    console.warn("[market_daily_close] upsert failed:", error);
  } finally {
    client.release();
  }
}

async function maybePersistDailyClose(rows: any[]) {

async function notifyAccountsForSymbols(rows: any[], reason: string) {
  const affectedSymbols = uniqueSymbols(rows);

  const now = Date.now();
  if (Number.isFinite(ACCOUNT_RECALC_BATCH_INTERVAL_MS) && ACCOUNT_RECALC_BATCH_INTERVAL_MS > 0) {

      FROM trades t
      JOIN symbol_configs sc ON t.symbol_id = sc.id

      `,
      [affectedSymbols],
    );
    const userIds = userIdsRes.rows.map((r: any) => Number(r.userId)).filter((n: number) => Number.isFinite(n));

async function pullBatch() {
  await refreshDynamicSet();
  const wanted = [...new Set([...dynamicSet])];
  if (wanted.length === 0) return;

  if (!API_KEY) {
    const simulated = generateSimulatedQuotes(wanted);
    await handleQuoteBatch(simulated, "simulated", false);
    return;
  }

  // Prefer a single batched request for all enabled symbols.
  try {
    const batchQuotes = await fetchForgeQuotes(wanted);
    await handleQuoteBatch(batchQuotes, "1forge", false);
    return;
  } catch (e: any) {
    consecutiveApiFailures++;
    console.error("[1Forge] API failure (batched):", e?.message || e);
  }

  // Fallback: chunk requests if the batch is rejected (URL length / API limits).
  const chunks: string[][] = [];
  for (let i = 0; i < wanted.length; i += MAX_PER_REQ) {
    chunks.push(wanted.slice(i, i + MAX_PER_REQ));
  }

  for (const slice of chunks) {
    try {
      const sliceQuotes = await fetchForgeQuotes(slice);
      await handleQuoteBatch(sliceQuotes, "1forge", false);
    } catch (e: any) {
      consecutiveApiFailures++;
      console.error("[1Forge] API failure:", e?.message || e);
      const fallbackQuotes = await buildFallbackQuotes(slice);
      if (!fallbackQuotes.length) {
        console.warn("[1Forge] No fallback cache available; skipping quote persistence.");
        continue;
      }
      await handleQuoteBatch(fallbackQuotes, "fallback_cache", true, { markSuccess: false, reason: "QUOTE_STALE" });
    }
  }
}

function schedulePoll() {
  pollTimerId = setTimeout(async () => {

    schedulePoll();
  }, dynamicConfig.pollIntervalMs);
}

    if (!unsubscribeLiveBus) {
      unsubscribeLiveBus = onLiveEvent((event) => {
        if (!event || typeof event !== "object") return;
        if (event.type === "symbols:updated") {
          void refreshSymbolsAndPull(String(event?.payload?.action ?? "updated"));
          return;
        }
        if (event.type === "feed:config-updated") {
          void reloadFeedConfig();
          return;
        }
      });
    }

    void pullBatch();
    schedulePoll();
    started = true;
  })().catch((err) => {

    started = false;
    throw err;
  });

  return startPromise;
}

function toQuoteCore(row: any, asOf: number): QuoteCore | null {
  if (!row?.symbol) return null;
  const symbol = normalizeSymbol(String(row.symbol));
  const bid = typeof row.bid === "number" ? row.bid : row.bid == null ? null : Number(row.bid);
  const ask = typeof row.ask === "number" ? row.ask : row.ask == null ? null : Number(row.ask);
  const price =
    typeof row.price === "number"
      ? row.price
      : bid != null && ask != null
        ? (bid + ask) / 2
        : row.price == null
          ? null
          : Number(row.price);
  const lastUpdatedRaw = row.lastUpdated ?? row.lastApiUpdate ?? row.updatedAt ?? asOf;
  const lastApiRaw = row.lastApiUpdate ?? row.lastUpdated ?? row.updatedAt ?? asOf;
  const lastUpdated = Number.isFinite(Number(lastUpdatedRaw)) ? Number(lastUpdatedRaw) : asOf;
  const lastApiUpdate = Number.isFinite(Number(lastApiRaw)) ? Number(lastApiRaw) : asOf;
  return {
    symbol,
    bid,
    ask,
    price,
    lastUpdated,
    lastApiUpdate,
    isStale: Boolean(row.isStale),
  };
}

export function applyQuoteUpdate(rows: Array<Partial<QuoteCore> & { symbol: string }>, meta?: { seq?: number; asOf?: number }) {
  const asOf = Number.isFinite(meta?.asOf) ? Number(meta?.asOf) : Date.now();
  if (Number.isFinite(meta?.seq)) lastSeq = Number(meta?.seq);
  lastAsOf = asOf;
  for (const row of rows) {
    const core = toQuoteCore(row, asOf);

  }

export async function getValkeyQuoteRows(symbols: string[]): Promise<QuoteCore[]> {
  if (!symbols.length) return [];
  const v = getValkey();
  if (!v) return [];
  const keys = symbols.map((symbol) => `q:v1:${normalizeSymbol(symbol)}`);
  try {
    const values = await v.mget(...keys);
    const asOf = Date.now();
    const rows: QuoteCore[] = [];
    for (const raw of values) {
      if (!raw) continue;
      try {
        const parsed = JSON.parse(raw);
        const core = toQuoteCore(parsed, asOf);
        if (core) rows.push(core);
      } catch {
        // ignore malformed entries
      }
    }
    return rows;
  } catch {
    return [];
  }
}

export async function getValkeySnapshot(symbols?: string[]) {
  const snapshot = await valkeyGetJson<QuoteSnapshot>(QUOTE_SNAPSHOT_KEY);
  if (!snapshot?.rows?.length) return null;
  const asOf = Number.isFinite(snapshot.asOf) ? Number(snapshot.asOf) : Date.now();
  const wanted = symbols?.length ? new Set(symbols.map(normalizeSymbol)) : null;
  const rows: QuoteCore[] = [];
  for (const row of snapshot.rows as any[]) {
    const core = toQuoteCore(row, asOf);
    if (!core) continue;
    if (wanted && !wanted.has(core.symbol)) continue;
    rows.push(core);
  }
  return {
    seq: snapshot.seq,
    asOf: snapshot.asOf,
    source: snapshot.source,
    rows,
  };
}

// Get a fresh database connection for each query to avoid locking issues
// Fallback chain: Hub → Valkey Snapshot → Rolling Buffer → DB → prevClose
export async function getLatestQuoteRow(symbol: string): Promise<any | null> {
  const sym = normalizeSymbol(symbol);

  // 1. Try in-memory hub (fastest)
  const hubQuote = getQuote(sym);
  if (hubQuote) {
    return {
      symbol: sym,
      bid: hubQuote.bid,
      ask: hubQuote.ask,
      price: hubQuote.price,
      lastApiUpdate: hubQuote.lastApiUpdate,
      isStale: hubQuote.isStale,
      prevClose: hubQuote.prevClose,
    };
  }

  // 2. Try Valkey snapshot
  const valkeyRows = await getValkeyQuoteRows([sym]);
  if (valkeyRows.length) {
    const cached = valkeyRows[0];
    return {
      symbol: sym,
      bid: cached.bid,
      ask: cached.ask,
      price: cached.price,
      lastApiUpdate: cached.lastApiUpdate,
      isStale: cached.isStale,
    };
  }

  // 3. Try rolling buffer (30-second history)
  const rollingQuote = await getFromRollingBuffer(sym);
  if (rollingQuote && rollingQuote.price != null) {
    return {
      symbol: sym,
      bid: rollingQuote.bid,
      ask: rollingQuote.ask,
      price: rollingQuote.price,
      lastApiUpdate: rollingQuote.lastApiUpdate,
      isStale: true, // Mark as stale since we're falling back
    };
  }

  // 4. Try database
  const row = await db.query.quotes.findFirst({
    where: eq(quotes.symbol, sym),
  });
  if (row) return row;

  // 5. Last resort: use prevClose as price (better than nothing)
  const prevClose = await getCachedPrevClose(sym);
  if (prevClose != null) {
    return {
      symbol: sym,
      bid: prevClose,
      ask: prevClose,
      price: prevClose,
      lastApiUpdate: Date.now(),
      isStale: true,
      isPrevCloseFallback: true,
    };
  }

  return null;
}

export async function getExecutionQuote(symbol: string, side: "BUY" | "SELL", action: "OPEN" | "CLOSE"): Promise<ExecutionQuote> {
  const sym = normalizeSymbol(symbol);
  const row = await getLatestQuoteRow(sym);
  if (!row) throw new Error(`QUOTE_NOT_FOUND:${sym}`);

  const bid = toNum(row.bid);
  const ask = toNum(row.ask);
  const px = toNum(row.price);

  const usableBid = bid ?? px;
  const usableAsk = ask ?? px;
  if (usableBid === null || usableAsk === null) throw new Error(`QUOTE_INVALID:${sym}`);

  const mid = (usableBid + usableAsk) / 2;
  const spread = usableAsk - usableBid;

  // Handle both timestamp formats (seconds or milliseconds)
  const lastApiRaw = toNum((row as any).lastApiUpdate ?? (row as any).last_api_update) ?? toNum((row as any).updatedAt ?? (row as any).updated_at);
  const lastApiMs =
    lastApiRaw === null ? null : (lastApiRaw < 1e12 ? lastApiRaw * 1000 : lastApiRaw);

  const now = Date.now();
  const isStale =
    Boolean((row as any).isStale ?? (row as any).is_stale) ||
    lastApiMs === null ||
    (now - lastApiMs) > STALE_AFTER_MS;

  const marketOpen = isMarketOpenForSymbol(sym, new Date());

  // Institutional: BUY opens at ask, closes at bid. SELL opens at bid, closes at ask.
  const execPrice =
    action === "OPEN"
      ? (side === "BUY" ? usableAsk : usableBid)
      : (side === "BUY" ? usableBid : usableAsk);

  const quoteTs = new Date(lastApiMs ?? now);

  return {
    symbol: sym,
    bid: usableBid,
    ask: usableAsk,
    mid,
    spread,
    quoteTs,
    source: process.env.QUOTE_SOURCE ?? (process.env.FORGE_KEY ? "1forge" : "quotes_db"),
    isStale,
    marketOpen,
    execPrice,
  };
}

// FX conversion (quoteCurrency -> USD) using available pairs in quotes table
export async function getConversionRate(fromCcy: string, toCcy: string): Promise<number> {
  const from = fromCcy.toUpperCase();
  const to = toCcy.toUpperCase();
  if (from === to) return 1;

  const direct = await getLatestQuoteRow(from + to);
  if (direct) {
    const b = toNum(direct.bid) ?? toNum(direct.price);
    const a = toNum(direct.ask) ?? toNum(direct.price);
    if (b !== null && a !== null) return (a + b) / 2;
  }

  const inverse = await getLatestQuoteRow(to + from);
  if (inverse) {
    const b = toNum(inverse.bid) ?? toNum(inverse.price);
    const a = toNum(inverse.ask) ?? toNum(inverse.price);
    if (b !== null && a !== null) return 1 / ((a + b) / 2);
  }

  // Bridge via USD
  if (from !== "USD" && to !== "USD") {
    return (await getConversionRate(from, "USD")) * (await getConversionRate("USD", to));
  }

  // Fallback rates for common currencies if quote not found
  const fallbackRates: Record<string, number> = {
    'JPY': 0.0067,  // 1 JPY = ~0.0067 USD
    'EUR': 1.09,    // 1 EUR = ~1.09 USD
    'GBP': 1.27,    // 1 GBP = ~1.27 USD
    'AUD': 0.65,    // 1 AUD = ~0.65 USD
    'USD': 1.0,
  };

  if (to === "USD" && fallbackRates[from]) {
    return fallbackRates[from];
  }
  if (from === "USD" && fallbackRates[to]) {
    return 1 / fallbackRates[to];
  }

  throw new Error(`FX_CONVERSION_MISSING:${from}->${to}`);
}

/**
 * Precision-aware price comparison utilities for forex trading.
 * Converts prices to integer ticks to avoid floating-point precision issues.
 * Handles truncated decimals (e.g., 0.67 = 0.6700) correctly.
 */
function getPrecision(symbol: string): number {
  // JPY pairs have 2 decimal precision (pip = 0.01), others have 4 (pip = 0.0001)
  return symbol.includes("JPY") ? 2 : 4;
}

function toTicks(price: number, precision: number): number {
  // Convert to integer ticks for precise comparison
  // e.g., 0.6700 with precision 4 → 6700, 0.6698 → 6698
  const multiplier = Math.pow(10, precision);
  return Math.round(price * multiplier);
}

function ticksToPrice(ticks: number, precision: number): number {
  return ticks / Math.pow(10, precision);
}

// Precision-aware comparison: returns true if priceA < priceB
function priceLessThan(priceA: number, priceB: number, symbol: string): boolean {
  const precision = getPrecision(symbol);
  return toTicks(priceA, precision) < toTicks(priceB, precision);
}

// Precision-aware comparison: returns true if priceA > priceB
function priceGreaterThan(priceA: number, priceB: number, symbol: string): boolean {
  const precision = getPrecision(symbol);
  return toTicks(priceA, precision) > toTicks(priceB, precision);
}

// Precision-aware comparison: returns true if priceA <= priceB
function priceLessThanOrEqual(priceA: number, priceB: number, symbol: string): boolean {
  const precision = getPrecision(symbol);
  return toTicks(priceA, precision) <= toTicks(priceB, precision);
}

function convertQuestionMarks(sql: string): string {
  let out = "";
  let index = 1;
  let inSingle = false;
  let inDouble = false;
  let inLineComment = false;
  let inBlockComment = false;

  for (let i = 0; i < sql.length; i++) {
    const ch = sql[i];
    const next = i + 1 < sql.length ? sql[i + 1] : "";

    if (inLineComment) {
      out += ch;
      if (ch === "\n") inLineComment = false;
      continue;
    }
    if (inBlockComment) {
      out += ch;
      if (ch === "*" && next === "/") {
        out += next;
        i++;
        inBlockComment = false;
      }
      continue;
    }

    if (!inSingle && !inDouble) {
      if (ch === "-" && next === "-") {
        out += ch + next;
        i++;
        inLineComment = true;
        continue;
      }
      if (ch === "/" && next === "*") {
        out += ch + next;
        i++;
        inBlockComment = true;
        continue;
      }
    }

    if (ch === "'" && !inDouble) {
      out += ch;
      if (inSingle && next === "'") {
        out += next;
        i++;
      } else {
        inSingle = !inSingle;
      }
      continue;
    }

    if (ch === "\"" && !inSingle) {
      out += ch;
      inDouble = !inDouble;
      continue;
    }

    if (!inSingle && !inDouble && ch === "?") {
      out += `$${index++}`;
      continue;
    }

    out += ch;
  }

  return out;
}

async function queryAll<T = any>(sql: string, params: any[] = []): Promise<T[]> {
  const text = convertQuestionMarks(sql);
  const result = await dbClient.query(text, params);

}

import axios from "axios";
import WebSocket from "ws";
import type { MarketDataProviderConfig, ProviderQuote, QuoteStream, QuoteStreamStatus } from "../types";

function chunk<T>(arr: T[], size: number): T[][] {
  const out: T[][] = [];
  for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size));
  return out;
}

function num(v: any): number | null {
  if (v == null) return null;
  const n = Number(v);
  return Number.isFinite(n) ? n : null;
}

export async function fetchTwelveDataQuotesRest(params: {
  config: MarketDataProviderConfig;
  providerSymbols: string[];
  providerToCanonical: Map<string, string>;
}): Promise<ProviderQuote[]> {
  const { config, providerSymbols, providerToCanonical } = params;
  const rest = config.rest;
  if (!rest) throw new Error("TwelveData REST config missing");
  const apiKeyEnv = rest.apiKeyEnv || "TWELVEDATA_API_KEY";
  const apiKey = process.env[apiKeyEnv];
  if (!apiKey) throw new Error(`Missing env ${apiKeyEnv}`);

  const maxBatch = config.capabilities?.maxBatchSymbols ?? 120;
  const batches = chunk(providerSymbols, Math.max(1, maxBatch));
  const base = rest.baseUrl.replace(/\/$/, "");
  const url = `${base}${rest.quotePath.startsWith("/") ? rest.quotePath : `/${rest.quotePath}`}`;

  const out: ProviderQuote[] = [];
  for (const batch of batches) {
    try {
      const r = await axios.get(url, {
        timeout: rest.timeoutMs ?? 8000,
        params: {
          [rest.symbolParam]: batch.join(","),
          [rest.apiKeyParam]: apiKey,
        },
      });

      const data = r.data;
      const now = Date.now();

      // Twelve Data: single symbol -> object; batch -> dictionary keyed by symbols (common pattern)
      if (Array.isArray(data)) {
        for (const row of data) {
          const providerSymbol = String(row?.symbol ?? "").trim();
          const canonical = providerToCanonical.get(providerSymbol) ?? providerSymbol;
          const price = num(row?.price ?? row?.close ?? row?.last);
          const bid = num(row?.bid);
          const ask = num(row?.ask);
          const tsSec = num(row?.timestamp);
          const tsMs = tsSec != null ? (tsSec < 1e12 ? Math.floor(tsSec * 1000) : Math.floor(tsSec)) : now;
          out.push({
            symbol: canonical,
            bid,
            ask,
            price,
            lastApiUpdate: tsMs,
            isStale: false,
            source: config.key,
            transport: "rest",
          });
        }
        continue;
      }

      if (data && typeof data === "object") {
        // Single row
        if (typeof (data as any).symbol === "string") {
          const providerSymbol = String((data as any).symbol).trim();
          const canonical = providerToCanonical.get(providerSymbol) ?? providerSymbol;
          const price = num((data as any).price ?? (data as any).close ?? (data as any).last);
          const bid = num((data as any).bid);
          const ask = num((data as any).ask);
          const tsSec = num((data as any).timestamp);
          const tsMs = tsSec != null ? (tsSec < 1e12 ? Math.floor(tsSec * 1000) : Math.floor(tsSec)) : now;
          out.push({
            symbol: canonical,
            bid,
            ask,
            price,
            lastApiUpdate: tsMs,
            isStale: false,
            source: config.key,
            transport: "rest",
          });
          continue;
        }

        // Batch dictionary
        for (const [k, v] of Object.entries(data)) {
          if (!v || typeof v !== "object") continue;
          const providerSymbol = String((v as any).symbol ?? k).trim();
          const canonical = providerToCanonical.get(providerSymbol) ?? providerSymbol;
          const price = num((v as any).price ?? (v as any).close ?? (v as any).last);
          const bid = num((v as any).bid);
          const ask = num((v as any).ask);
          const tsSec = num((v as any).timestamp);
          const tsMs = tsSec != null ? (tsSec < 1e12 ? Math.floor(tsSec * 1000) : Math.floor(tsSec)) : now;
          out.push({
            symbol: canonical,
            bid,
            ask,
            price,
            lastApiUpdate: tsMs,
            isStale: false,
            source: config.key,
            transport: "rest",
          });
        }
      }
    } catch (e) {
      // eslint-disable-next-line no-console
      console.warn("[marketdata:twelvedata] REST batch failed:", e);
    }
  }

  return out;
}

function wsSubscribePayload(symbolsCsv: string) {
  return JSON.stringify({ action: "subscribe", params: { symbols: symbolsCsv } });
}

function wsUnsubscribePayload(symbolsCsv: string) {
  return JSON.stringify({ action: "unsubscribe", params: { symbols: symbolsCsv } });
}

function wsHeartbeatPayload() {
  return JSON.stringify({ action: "heartbeat" });
}

export function createTwelveDataQuoteStream(params: {
  config: MarketDataProviderConfig;
  providerSymbols: string[];
  providerToCanonical: Map<string, string>;
  onQuote: (q: ProviderQuote) => void;
  onStatus?: (s: QuoteStreamStatus) => void;
}): QuoteStream {
  const { config, providerSymbols, providerToCanonical, onQuote, onStatus } = params;
  const wsCfg = config.ws;
  if (!wsCfg) throw new Error("TwelveData WS config missing");

  const apiKeyEnv = wsCfg.apiKeyEnv || "TWELVEDATA_API_KEY";
  const apiKey = process.env[apiKeyEnv];
  if (!apiKey) throw new Error(`Missing env ${apiKeyEnv}`);

  let ws: WebSocket | null = null;
  let connected = false;
  let stopping = false;
  let hbTimer: NodeJS.Timeout | null = null;
  let reconnectTimer: NodeJS.Timeout | null = null;

  let desired = new Set(providerSymbols.map((s) => s.trim()).filter(Boolean));
  let subscribed = new Set<string>();

  const reconnectMin = wsCfg.reconnectMinMs ?? 1000;
  const reconnectMax = wsCfg.reconnectMaxMs ?? 15000;
  let reconnectDelay = reconnectMin;

  const heartbeatMs = wsCfg.heartbeatIntervalMs ?? 10000;
  const maxPerSub = wsCfg.maxSymbolsPerSubscribe ?? 500;

  const send = (payload: string) => {
    if (!ws || ws.readyState !== WebSocket.OPEN) return;
    ws.send(payload);
  };

  const clearTimers = () => {
    if (hbTimer) clearInterval(hbTimer);
    hbTimer = null;
    if (reconnectTimer) clearTimeout(reconnectTimer);
    reconnectTimer = null;
  };

  const scheduleReconnect = (why: string) => {
    if (stopping) return;
    clearTimers();
    onStatus?.({ status: "error", providerKey: config.key, error: `reconnect:${why}` });
    if (reconnectTimer) return;
    const delay = reconnectDelay;
    reconnectDelay = Math.min(reconnectMax, Math.floor(reconnectDelay * 1.6));
    reconnectTimer = setTimeout(() => {
      reconnectTimer = null;
      connect();
    }, delay);
  };

  const subscribeAll = () => {
    const list = [...desired];
    if (!list.length) return;
    for (let i = 0; i < list.length; i += maxPerSub) {
      const slice = list.slice(i, i + maxPerSub);
      send(wsSubscribePayload(slice.join(",")));
    }
  };

  const connect = () => {
    if (stopping) return;
    try {
      onStatus?.({ status: "connecting", providerKey: config.key });
      const url = `${wsCfg.url}?apikey=${encodeURIComponent(apiKey)}`;
      ws = new WebSocket(url);

      ws.on("open", () => {
        connected = true;
        reconnectDelay = reconnectMin;
        onStatus?.({ status: "open", providerKey: config.key });

        // subscribe
        subscribeAll();

        // heartbeat
        hbTimer = setInterval(() => {
          try {
            send(wsHeartbeatPayload());
          } catch {
            // ignore
          }
        }, heartbeatMs);
      });

      ws.on("message", (raw) => {
        try {
          const msg = JSON.parse(String(raw));

          // Price event examples: event/type == "price"; fields: symbol, price, timestamp
          const eventType = String((msg as any).event ?? (msg as any).type ?? "").toLowerCase();
          if (eventType === "price" || (msg as any).price != null) {
            const providerSymbol = String((msg as any).symbol ?? "").trim();
            if (!providerSymbol) return;
            const canonical = providerToCanonical.get(providerSymbol) ?? providerSymbol;
            const price = num((msg as any).price);
            const tsSec = num((msg as any).timestamp);
            const tsMs = tsSec != null ? (tsSec < 1e12 ? Math.floor(tsSec * 1000) : Math.floor(tsSec)) : Date.now();
            onQuote({
              symbol: canonical,
              bid: null,
              ask: null,
              price,
              lastApiUpdate: tsMs,
              isStale: false,
              source: config.key,
              transport: "ws",
            });
            return;
          }

          // Subscribe status event: best-effort tracking (not relied on for correctness)
          if (eventType.includes("subscribe")) {
            const ok = Array.isArray((msg as any).success) ? (msg as any).success : null;
            if (ok) {
              for (const s of ok) subscribed.add(String(s));
            }
            return;
          }
        } catch {
          // ignore invalid payloads
        }
      });

      ws.on("close", (code, reason) => {
        connected = false;
        subscribed = new Set();
        onStatus?.({ status: "closed", providerKey: config.key, code, reason: String(reason ?? "") });
        scheduleReconnect(`close:${code}`);
      });

      ws.on("error", (err) => {
        connected = false;
        subscribed = new Set();
        onStatus?.({ status: "error", providerKey: config.key, error: String((err as any)?.message ?? err) });
        scheduleReconnect("error");
      });
    } catch (e: any) {
      scheduleReconnect(e?.message ?? "exception");
    }
  };

  // Start immediately
  connect();

  return {
    providerKey: config.key,
    transport: "ws",
    updateSymbols: (providerSymbolsNext: string[]) => {
      const next = new Set(providerSymbolsNext.map((s) => s.trim()).filter(Boolean));

      const toUnsub: string[] = [];
      for (const s of desired) if (!next.has(s)) toUnsub.push(s);

      const toSub: string[] = [];

 * Twelve Data REST quote fetch.
 * Supports batching by passing comma-delimited symbols in the `symbol` query param.

export async function fetchTwelveDataQuotesRest(params: {
  providerConfig: MarketDataProviderConfig;
  providerSymbols: string[]; // provider-native symbol ids (e.g. EUR/USD)
  canonicalByProviderSymbol: Map<string, string>; // providerSym -> canonicalSym
}): Promise<ProviderQuote[]> {
  const cfg = params.providerConfig;
  if (!cfg.rest) throw new Error("TD_REST_CONFIG_MISSING");

  const apiKey = cfg.rest.apiKeyEnv ? process.env[cfg.rest.apiKeyEnv] : undefined;
  if (!apiKey) throw new Error("TD_API_KEY_MISSING");

  const max = Math.max(1, cfg.capabilities?.maxBatchSymbols ?? 120);
  const batches = chunk(params.providerSymbols, max);
  const out: ProviderQuote[] = [];

  for (const batch of batches) {
    if (!batch.length) continue;
    const symbols = batch.join(",");
    const url = `${cfg.rest.baseUrl}${cfg.rest.quotePath}`;
    const nowMs = Date.now();

    const resp = await axios.get(url, {
      timeout: cfg.rest.timeoutMs ?? 8000,
      params: {
        [cfg.rest.symbolParam]: symbols,
        [cfg.rest.apiKeyParam]: apiKey,
      },
    });

    const data = resp.data;

    // Twelve Data: single symbol => object with "symbol".
    // Batch => may return object keyed by symbol, or array. We parse flexibly.
    const rows: any[] = [];
    if (Array.isArray(data)) {
      rows.push(...data);
    } else if (data && typeof data === "object") {
      if (typeof (data as any).symbol === "string") {
        rows.push(data);
      } else {
        // keyed object: { "AAPL": { ... }, "EUR/USD": { ... } }
        for (const v of Object.values(data)) {
          if (v && typeof v === "object") rows.push(v);
        }
      }
    }

    for (const row of rows) {
      const providerSymbol = String(row?.symbol ?? row?.instrument ?? "").trim();
      if (!providerSymbol) continue;
      const canonical = params.canonicalByProviderSymbol.get(providerSymbol) ?? providerSymbol;
      const parsed = parseQuoteRow(row, nowMs);
      out.push({
        symbol: canonical,
        bid: parsed.bid,
        ask: parsed.ask,
        price: parsed.price,
        lastApiUpdate: parsed.lastApiUpdate,
        isStale: false,
        source: cfg.key,
        transport: "rest",
      });
    }
  }

  return out;
}

    this.apiKey = apiKey;

    this.connect();
  }

      connected: Boolean(this.ws && this.ws.readyState === WebSocket.OPEN),
      subscribed: this.subscribed.size,
      desired: this.desired.size,
    };
  }

    this.desired = next;

    if (!this.ws || this.ws.readyState !== WebSocket.OPEN) return;

    const toAdd: string[] = [];
    const toRemove: string[] = [];

    for (const s of next) if (!this.subscribed.has(s)) toAdd.push(s);
    for (const s of this.subscribed) if (!next.has(s)) toRemove.push(s);

    if (toRemove.length) this.sendUnsubscribe(toRemove);
    if (toAdd.length) this.sendSubscribe(toAdd);
  }

    this.reconnectTimer = null;
    if (this.heartbeatTimer) clearInterval(this.heartbeatTimer);

    } catch {}
  }

    const ws = new WebSocket(url);
    this.ws = ws;

    ws.on("open", () => {

      this.subscribed.clear();

      // subscribe to desired symbols
      const desiredList = [...this.desired];
      if (desiredList.length) this.sendSubscribe(desiredList);

      const intervalMs = wsCfg.heartbeatIntervalMs ?? 10_000;
      if (this.heartbeatTimer) clearInterval(this.heartbeatTimer);
      this.heartbeatTimer = setInterval(() => {

    });

    }
  }

  private sendHeartbeat() {

    this.sendJson({ action: "heartbeat" });
  }

import axios from "axios";
import type { MarketDataProviderConfig, ProviderQuote } from "../types";

function toNum(v: any): number | null {
  if (v === null || v === undefined) return null;
  const n = Number(v);
  return Number.isFinite(n) ? n : null;
}

export async function fetchOneForgeQuotesRest(params: {
  providerConfig: MarketDataProviderConfig;
  providerSymbols: string[]; // e.g. EURUSD
  canonicalByProviderSymbol: Map<string, string>;
}): Promise<ProviderQuote[]> {
  const cfg = params.providerConfig;
  if (!cfg.rest) throw new Error("1FORGE_REST_CONFIG_MISSING");

  const apiKey = cfg.rest.apiKeyEnv ? process.env[cfg.rest.apiKeyEnv] : undefined;
  if (!apiKey) throw new Error("1FORGE_API_KEY_MISSING");

  const url = `${cfg.rest.baseUrl}${cfg.rest.quotePath}`;
  const nowMs = Date.now();

  const resp = await axios.get(url, {
    timeout: cfg.rest.timeoutMs ?? 8000,
    params: {
      [cfg.rest.symbolParam]: params.providerSymbols.join(","),
      [cfg.rest.apiKeyParam]: apiKey,
    },
  });

  const data = resp.data;
  if (!Array.isArray(data)) return [];

  const out: ProviderQuote[] = [];
  for (const row of data) {
    const providerSymbol = String(row?.symbol ?? "").trim();
    if (!providerSymbol) continue;
    const canonical = params.canonicalByProviderSymbol.get(providerSymbol) ?? providerSymbol;

    const bid = toNum(row?.bid);
    const ask = toNum(row?.ask);
    const price = toNum(row?.price) ?? (bid != null && ask != null ? (bid + ask) / 2 : null);
    const tsSec = toNum(row?.timestamp);
    const lastApiUpdate = tsSec != null ? tsSec * 1000 : nowMs;

    out.push({
      symbol: canonical,
      bid,
      ask,
      price,
      lastApiUpdate,
      isStale: false,
      source: cfg.key,
      transport: "rest",
    });
  }

  return out;
}

import type { MarketDataProviderConfig, ProviderQuote } from "../types";

// Very lightweight simulated quotes, used when no provider key is configured.
// Not intended for production trading.

function hash32(str: string): number {
  let h = 2166136261;
  for (let i = 0; i < str.length; i++) {
    h ^= str.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return h >>> 0;
}

function basePrice(symbol: string): number {
  const h = hash32(symbol);
  // Spread base across a reasonable range
  return 1 + (h % 10_000) / 1_000; // 1.0 .. 11.0
}

export async function fetchSimulatedQuotesRest(params: {
  providerConfig: MarketDataProviderConfig;
  providerSymbols: string[]; // canonical symbols
  canonicalByProviderSymbol: Map<string, string>;
}): Promise<ProviderQuote[]> {
  const cfg = params.providerConfig;
  const nowMs = Date.now();
  return params.providerSymbols.map((providerSymbol) => {
    const canonical = params.canonicalByProviderSymbol.get(providerSymbol) ?? providerSymbol;
    const base = basePrice(canonical);
    const drift = (Math.sin(nowMs / 10_000 + base) * 0.002) + (Math.random() - 0.5) * 0.001;
    const mid = base * (1 + drift);
    const spread = Math.max(0.00005, mid * 0.0002);
    const bid = Number((mid - spread / 2).toFixed(5));
    const ask = Number((mid + spread / 2).toFixed(5));

    return {
      symbol: canonical,
      bid,
      ask,
      price: Number(((bid + ask) / 2).toFixed(5)),
      lastApiUpdate: nowMs,
      isStale: false,
      source: cfg.key,
      transport: "rest",
    };
  });
}

Building providerManager.ts
For the providerManager.ts, I'll interface with the database to get system and symbol configurations. We’ll import the DB using @db and use Drizzle for querying with EQ and SQL. Symbol configs will include properties like id, symbol, enabled, and providerSymbolMap. The providerManager will track configurations and an active stream, ensuring settings are loaded on module load. Functions such as reloadProviderConfigs and listProviderConfigs will handle updates and listings. We'll fetch the active provider state, defaulting to 'twelvedata'.

Handling fallback mode and managing active streams
The default mode will fallback to a specific setting. For the ensureQuoteStream, we'll make sure the providerSymbols are mapped to their canonical symbol using a lookup map. If the mode is ws/hybrid and the provider supports ws, we'll start or restart the stream; otherwise, we'll stop. We'll manage updates to symbol mappings during updates. fetchQuotesRestActive will handle provider-specific fetchers, throwing an error for unsupported types besides 'twelvedata' for WS. The provider manager will prevent multiple starts to avoid redundancy.

import { eq } from "drizzle-orm";
import { db } from "@db";
import { systemConfig, symbolConfigs } from "@shared/schema";
import { loadProviderConfigsFromDisk } from "./providerLoader";
import type {
  MarketDataMode,
  MarketDataProviderConfig,
  ProviderQuote,
  QuoteStream,
  QuoteStreamStatus,
} from "./types";
import { fetchTwelveDataQuotesRest, createTwelveDataQuoteStream } from "./providers/twelvedata";
import { fetchOneForgeQuotesRest } from "./providers/oneforge";
import { fetchSimulatedQuotesRest } from "./providers/simulated";

let providerCatalog: Map<string, MarketDataProviderConfig> = new Map();

function normalizeMode(mode: any): MarketDataMode {
  const m = String(mode ?? "").toLowerCase();
  if (m === "ws" || m === "rest" || m === "hybrid") return m;
  return "hybrid";
}

function loadCatalog() {
  const configs = loadProviderConfigsFromDisk();
  const m = new Map<string, MarketDataProviderConfig>();
  for (const c of configs) {
    if (!c?.key) continue;
    m.set(String(c.key), c);
  }
  providerCatalog = m;
}

// initial load
loadCatalog();

export function reloadProviderCatalog(): { ok: true; loaded: number; keys: string[] } {
  loadCatalog();
  return { ok: true, loaded: providerCatalog.size, keys: [...providerCatalog.keys()] };
}

export function listProviderCatalog(): MarketDataProviderConfig[] {
  return [...providerCatalog.values()].sort((a, b) => a.key.localeCompare(b.key));
}

export function getProviderConfig(key: string): MarketDataProviderConfig | null {
  return providerCatalog.get(key) ?? null;
}

export async function getActiveMarketDataConfig(): Promise<{ providerKey: string; mode: MarketDataMode; provider: MarketDataProviderConfig }>{
  const [cfgRow] = await db.select().from(systemConfig).where(eq(systemConfig.id, 1)).limit(1);
  const envKey = process.env.MARKET_DATA_PROVIDER_KEY;
  const envMode = process.env.MARKET_DATA_PROVIDER_MODE;

  const keyRaw = String(envKey ?? cfgRow?.marketDataProviderKey ?? "").trim();
  const modeRaw = String(envMode ?? cfgRow?.marketDataProviderMode ?? "").trim();

  let providerKey = keyRaw || "twelvedata";
  let provider = providerCatalog.get(providerKey);

  if (!provider) {
    // fall back to first available provider
    const first = providerCatalog.values().next().value as MarketDataProviderConfig | undefined;
    if (first) {
      providerKey = first.key;
      provider = first;
    } else {
      // no configs on disk — enforce simulated fallback
      providerKey = "simulated";
      provider = {
        key: "simulated",
        label: "Simulated",
        defaultMode: "rest",
        capabilities: { rest: true, ws: false, batch: true, maxBatchSymbols: 1000 },
        rest: { baseUrl: "local://simulated", quotePath: "/simulated", symbolParam: "symbols", apiKeyParam: "", apiKeyEnv: "", timeoutMs: 0 },
      };
    }
  }

  const mode = normalizeMode(modeRaw || provider.defaultMode);
  return { providerKey, mode, provider };
}

export type ActiveSymbolRow = {

export async function getEnabledSymbolsWithProviderMaps(): Promise<ActiveSymbolRow[]> {
  const rows = await db
    .select({
      symbol: symbolConfigs.symbol,
      providerSymbolMap: symbolConfigs.providerSymbolMap as any,
    })
    .from(symbolConfigs)
    .where(eq(symbolConfigs.enabled, true));

  return rows.map((r: any) => ({
    symbol: String(r.symbol).toUpperCase(),
    providerSymbolMap: (r.providerSymbolMap && typeof r.providerSymbolMap === "object") ? (r.providerSymbolMap as any) : null,
  }));
}

export function defaultProviderSymbol(canonicalSymbol: string, providerKey: string): string {
  const s = String(canonicalSymbol).trim().toUpperCase();
  if (!s) return s;

  if (providerKey === "twelvedata") {
    if (s.includes("/")) return s;

    // FX/metals/crypto pairs: EURUSD -> EUR/USD, XAUUSD -> XAU/USD, BTCUSD -> BTC/USD
    if (/^[A-Z]{6}$/.test(s)) {
      return `${s.slice(0, 3)}/${s.slice(3, 6)}`;
    }
    return s;
  }

  // 1Forge uses the legacy canonical style for FX pairs.
  return s;
}

export function resolveProviderSymbol(args: {
  canonicalSymbol: string;
  providerKey: string;
  providerSymbolMap: Record<string, string> | null;
}): string {
  const map = args.providerSymbolMap;
  if (map && typeof map[args.providerKey] === "string" && map[args.providerKey].trim()) {
    return String(map[args.providerKey]).trim();
  }
  return defaultProviderSymbol(args.canonicalSymbol, args.providerKey);
}

export async function fetchQuotesViaActiveProviderRest(args: {

  providerSymbols: string[];

  if (key === "twelvedata") {
    return fetchTwelveDataQuotesRest({
      providerConfig: args.provider,

      providerSymbols: args.providerSymbols,
      canonicalByProviderSymbol: args.canonicalByProviderSymbol,
    });
  }

  // unknown provider — attempt simulated as safe fallback
  return fetchSimulatedQuotesRest({

      key: "simulated",
      label: "Simulated",
      defaultMode: "rest",
      capabilities: { rest: true, ws: false, batch: true, maxBatchSymbols: 1000 },
      rest: { baseUrl: "local://simulated", quotePath: "/simulated", symbolParam: "symbols", apiKeyParam: "", apiKeyEnv: "", timeoutMs: 0 },
    },
    providerSymbols: args.providerSymbols.map((s) => args.canonicalByProviderSymbol.get(s) ?? s),
    canonicalByProviderSymbol: args.canonicalByProviderSymbol,
  });
}

// ---------------- WS stream management (currently Twelve Data) ----------------

let activeStream: QuoteStream | null = null;
let activeStreamProviderKey: string | null = null;

export async function ensureActiveProviderWs(args: {
  providerKey: string;
  provider: MarketDataProviderConfig;
  mode: MarketDataMode;
  providerSymbols: string[];
  canonicalByProviderSymbol: Map<string, string>;
  onQuote: (q: ProviderQuote) => void;
  onStatus?: (s: QuoteStreamStatus) => void;
}): Promise<{ ok: true; started: boolean; running: boolean }>{
  const wantsWs = args.mode === "ws" || args.mode === "hybrid";
  const canWs = Boolean(args.provider.capabilities?.ws && args.provider.ws);

  if (!wantsWs || !canWs) {
    await stopActiveProviderWs();
    return { ok: true, started: false, running: false };
  }

  const needRestart = !activeStream || activeStreamProviderKey !== args.providerKey;

  if (needRestart) {
    await stopActiveProviderWs();

    if (args.providerKey === "twelvedata") {
      activeStream = createTwelveDataQuoteStream({
        providerConfig: args.provider,
        onStatus: args.onStatus,
        onQuote: (q) => {
          // q.symbol is provider symbol; map to canonical for downstream
          const canonical = args.canonicalByProviderSymbol.get(q.symbol) ?? q.symbol;
          args.onQuote({ ...q, symbol: canonical });
        },
      });
      activeStreamProviderKey = args.providerKey;
      activeStream.updateSymbols(args.providerSymbols);
      return { ok: true, started: true, running: true };
    }

    // Future providers can be added here.
    return { ok: true, started: false, running: false };
  }

  // already running; update symbols + mapping used by callback via closure above
  activeStream.updateSymbols(args.providerSymbols);
  return { ok: true, started: false, running: true };
}

export async function stopActiveProviderWs(): Promise<void> {
  const s = activeStream;
  activeStream = null;
  activeStreamProviderKey = null;
  try {
    await s?.stop();
  } catch {}
}

export function getActiveWsState(): { providerKey: string | null; connected: boolean; subscribed: number; desired: number } {
  const state = activeStream?.getState();
  return {
    providerKey: activeStreamProviderKey,
    connected: Boolean(state?.connected),
    subscribed: Number(state?.subscribed ?? 0),
    desired: Number(state?.desired ?? 0),
  };
}

const lastEmitByKey = new Map<string, number>();
const LIVEBUS_CHANNEL = process.env.LIVEBUS_CHANNEL || "livebus:events";
const LIVEBUS_VALKEY_ENABLED = !["0", "false"].includes(
  String(process.env.LIVEBUS_VALKEY_ENABLED || "").toLowerCase()
);
const LIVEBUS_ORIGIN = `pid:${process.pid}-${Math.random().toString(36).slice(2, 8)}`;

let valkeySubscriber: Redis | null = null;
let valkeySubscribePending = false;
let valkeySubscribed = false;

function shouldEmit(event: LiveEvent): boolean {
  if (event.type === "account:updated" && typeof event.userId === "number") {
    const key = `account:updated:${event.userId}`;
    const now = Date.now();
    const last = lastEmitByKey.get(key) ?? 0;
    const minIntervalMs = Number(process.env.LIVE_ACCOUNT_THROTTLE_MS ?? 1000);
    if (now - last < minIntervalMs) return false;
    lastEmitByKey.set(key, now);
  }
  return true;
}

function emitLocal(event: LiveEvent) {
  if (!shouldEmit(event)) return;
  emitter.emit("event", event);
}

function handleValkeyMessage(message: string) {
  try {
    const parsed = JSON.parse(message);
    if (!parsed || typeof parsed !== "object") return;
    if (parsed.__origin && parsed.__origin === LIVEBUS_ORIGIN) return;
    const { __origin, ...rest } = parsed;
    const event = rest as LiveEvent;
    event.ts = typeof event.ts === "number" ? event.ts : Date.now();
    emitLocal(event);
  } catch {
    // ignore invalid payloads
  }
}

function ensureValkeySubscriber() {
  if (!LIVEBUS_VALKEY_ENABLED || valkeySubscribed || valkeySubscribePending) return;
  const base = getValkey();
  if (!base) return;
  valkeySubscribePending = true;
  valkeySubscriber = base.duplicate();
  valkeySubscriber.on("message", (channel, message) => {
    if (channel !== LIVEBUS_CHANNEL) return;
    handleValkeyMessage(message);
  });
  valkeySubscriber.on("error", () => {
    // silent by design
  });
  valkeySubscriber
    .subscribe(LIVEBUS_CHANNEL)
    .then(() => {
      valkeySubscribed = true;
    })
    .catch(() => {
      valkeySubscribed = false;
    })
    .finally(() => {
      valkeySubscribePending = false;
    });
}

export function publishLiveEvent(event: LiveEvent) {
  const ev: LiveEvent = { ...event, ts: event.ts ?? Date.now() };
  emitLocal(ev);

  if (!LIVEBUS_VALKEY_ENABLED) return;
  const v = getValkey();
  if (!v) return;
  const payload = JSON.stringify({ ...ev, __origin: LIVEBUS_ORIGIN });
  v.publish(LIVEBUS_CHANNEL, payload).catch(() => {
    // ignore publish errors, local emit already happened
  });
}

export function onLiveEvent(handler: (event: LiveEvent) => void) {
  ensureValkeySubscriber();
  emitter.on("event", handler);
  return () => emitter.off("event", handler);
}
