## A) Immediate fixes for the two cards (Login Activity + Sessions) and GMT/UTC timezone standards

### A1) Fix “Invalid Date” in Login Activity card (field mismatch)

Your backend returns login history timestamps as `eventAt`, but `ProfileSettings.tsx` renders `login.createdAt * 1000`, so it becomes `undefined * 1000 → NaN → Invalid Date`.

#### Patch 1 — `client/src/pages/ProfileSettings.tsx` (surgical change)

**Find** the login history timestamp rendering:

```tsx
{new Date(login.createdAt * 1000).toLocaleString()}
```

**Replace** with a robust parser that supports `eventAt` (Date string), `createdAt` (sec), or `createdAtMs` (ms):

```tsx
const parseLoginTsMs = (login: any): number | null => {
  const v = login?.createdAtMs ?? login?.eventAtMs ?? login?.eventAt ?? login?.createdAt;
  if (v == null) return null;

  if (typeof v === "number") {
    // Heuristic: seconds vs milliseconds
    return v < 2_000_000_000 ? v * 1000 : v;
  }

  // string (ISO) or Date-like
  const ms = Date.parse(String(v));
  return Number.isFinite(ms) ? ms : null;
};
```

Then render:

```tsx
{(() => {
  const ms = parseLoginTsMs(login);
  return ms ? new Date(ms).toLocaleString() : "Unknown";
})()}
```

#### Patch 3 — add backward-compatible alias (recommended)

**Full file replacement:** `server/routes/meSessions.ts`
(Adds `/logout-others` alias that routes to the same logic.)

```ts
// server/routes/meSessions.ts
import type { Express, Request, Response } from "express";
import express from "express";
import { getActiveUserSessions, revokeUserSession, revokeOtherUserSessions } from "../security/sessionTrail";

export function registerMeSessionsRoutes(app: Express) {
  const router = express.Router();

  const ensureAuth = (req: Request, res: Response, next: any) => {
    if (!req.session?.userId) return res.status(401).json({ message: "Not authenticated" });
    next();
  };

  // List active sessions/devices
  router.get("/", ensureAuth, async (req, res) => {
    const userId = req.session.userId!;
    const currentSessionId = req.sessionID;

    const sessions = await getActiveUserSessions(userId, currentSessionId);
    res.json(
      sessions.map((s) => ({
        ...s,
        createdAtMs: s.createdAt?.getTime?.() ?? null,
        lastSeenAtMs: s.lastActiveAt?.getTime?.() ?? null,
      }))
    );
  });

  // Revoke one session by ID
  router.post("/:sessionId/revoke", ensureAuth, async (req, res) => {
    const userId = req.session.userId!;
    const { sessionId } = req.params;

    await revokeUserSession({
      userId,
      sessionId,
      actorAdminId: null,
      req,
      sessionStore: (req as any).sessionStore,
    });

    res.json({ ok: true });
  });

  // Revoke all other sessions
  router.post("/revoke-others", ensureAuth, async (req, res) => {
    const userId = req.session.userId!;
    const currentSessionId = req.sessionID;

    const result = await revokeOtherUserSessions({
      userId,
      currentSessionId,
      actorAdminId: null,
      req,
      sessionStore: (req as any).sessionStore,
    });

    res.json({ ok: true, ...result });
  });

  // Backward-compatible alias (UI previously used this)
  router.post("/logout-others", ensureAuth, async (req, res) => {
    const userId = req.session.userId!;
    const currentSessionId = req.sessionID;

    const result = await revokeOtherUserSessions({
      userId,
      currentSessionId,
      actorAdminId: null,
      req,
      sessionStore: (req as any).sessionStore,
    });

    res.json({ ok: true, ...result });
  });

  app.use("/api/me/sessions", router);
}
```

### A3) Enforce revoked sessions (hedge-fund-grade)

Today, a revoked session may remain usable if the cookie persists and the session store entry isn’t destroyed for some reason. Hedge-fund-grade requires **server-side revocation enforcement** on every authenticated request.

**Step 2: replace `ensureAuth` with async revoked check**
Find the existing `ensureAuth` in `server/routes.ts` and replace it with:

```ts
const ensureAuth = async (req: Request, res: Response, next: NextFunction) => {
  if (!req.session.userId) return res.status(401).json({ message: "Not authenticated" });

  // Hedge-fund-grade: revoked-session enforcement
  try {
    const sid = req.sessionID;
    const uid = req.session.userId;

    const row = await db.query.userSessions.findFirst({
      where: (t, { and, eq }) => and(eq(t.sessionId, sid), eq(t.userId, uid)),
      columns: { revokedAt: true },
    });

    if (row?.revokedAt) {
      req.session.destroy(() => {});
      return res.status(401).json({ message: "Session revoked" });
    }
  } catch {
    // Fail-open to avoid global outage if DB hiccups;
    // the revoke route still destroys store entry in the normal path.
  }

  next();
};
```

**Replace:**

```ts
function formatTzLabel(z: any) {
  const utc = formatUtcOffset(tzOffsetMinutes(z.name)); // "UTC+02:00"
  const gmt = utc.replace(/^UTC/, "GMT");               // "GMT+02:00"
  const cities = z.mainCities?.slice(0, 3)?.join(", ");
  return `${utc} / ${gmt} • ${z.name}${cities ? ` • ${cities}` : ""}`;
}
```

This keeps everything in UTC/GMT standards while still preserving the IANA zone ID and major cities. ([npm][1])

#### Patch 6 — Use city search in UI (recommended component)

Create a dedicated searchable selector that calls your existing endpoint:

* `GET /api/meta/timezones/search?q=<user input>`

**New file:** `client/src/components/TimezoneSelect.tsx`

```tsx
import * as React from "react";
import { Button } from "@/components/ui/button";
import { Popover, PopoverContent, PopoverTrigger } from "@/components/ui/popover";
import { Command, CommandEmpty, CommandInput, CommandItem, CommandList } from "@/components/ui/command";
import { Check, ChevronsUpDown } from "lucide-react";

type TzOption = { name: string; label: string };

export function TimezoneSelect({
  value,
  onChange,
  disabled,
}: {
  value: string;
  onChange: (tz: string) => void;
  disabled?: boolean;
}) {
  const [open, setOpen] = React.useState(false);
  const [q, setQ] = React.useState("");
  const [options, setOptions] = React.useState<TzOption[]>([]);
  const [loading, setLoading] = React.useState(false);

  const fetchOptions = React.useCallback(async (query: string) => {
    setLoading(true);
    try {
      const url = `/api/meta/timezones/search?q=${encodeURIComponent(query)}`;
      const res = await fetch(url, { credentials: "include" });
      const data = (await res.json()) as TzOption[];
      setOptions(Array.isArray(data) ? data : []);
    } finally {
      setLoading(false);
    }
  }, []);

  React.useEffect(() => {
    const t = setTimeout(() => fetchOptions(q), 200);
    return () => clearTimeout(t);
  }, [q, fetchOptions]);

  const selected = React.useMemo(() => {
    return options.find((o) => o.name === value)?.label ?? value ?? "Select timezone";
  }, [options, value]);

  return (
    <Popover open={open} onOpenChange={setOpen}>
      <PopoverTrigger asChild>
        <Button variant="outline" role="combobox" disabled={disabled} className="w-full justify-between">
          <span className="truncate">{selected}</span>
          <ChevronsUpDown className="ml-2 h-4 w-4 opacity-50" />
        </Button>
      </PopoverTrigger>

      <PopoverContent className="w-[420px] p-0" align="start">
        <Command>
          <CommandInput
            placeholder="Search timezone (city, country, UTC offset, IANA)…"
            value={q}
            onValueChange={setQ}
          />
          <CommandList>
            <CommandEmpty>{loading ? "Loading…" : "No results"}</CommandEmpty>
            {options.map((o) => (
              <CommandItem
                key={o.name}
                value={o.name}
                onSelect={() => {
                  onChange(o.name);
                  setOpen(false);
                }}
              >
                <Check className={`mr-2 h-4 w-4 ${o.name === value ? "opacity-100" : "opacity-0"}`} />
                <span className="truncate">{o.label}</span>
              </CommandItem>
            ))}
          </CommandList>
        </Command>
      </PopoverContent>
    </Popover>
  );
}
```
