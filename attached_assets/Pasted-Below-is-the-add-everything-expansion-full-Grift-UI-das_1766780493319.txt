/**
 * Helper: add a missing column if it does not exist.
 * SQLite supports ALTER TABLE ADD COLUMN (no IF NOT EXISTS), so we check PRAGMA first.
 */
function ensureColumn(table: string, column: string, ddl: string) {
  const cols = sqlite.prepare(`PRAGMA table_info(${table})`).all() as any[];
  const exists = cols.some((c) => String(c.name).toLowerCase() === column.toLowerCase());
  if (!exists) sqlite.exec(`ALTER TABLE ${table} ADD COLUMN ${ddl}`);
}

// server/grift/griftEngine.ts
import { sqlite } from "@db";
import type { GriftConfig, GriftRuleCode, GriftSeverity } from "./griftTypes";

function clamp(n: number, min: number, max: number) {
  return Math.max(min, Math.min(max, n));
}

function tierFor(score: number, cfg: GriftConfig): "LOW" | "MED" | "HIGH" | "CRITICAL" {
  if (score >= cfg.tierCritical) return "CRITICAL";
  if (score >= cfg.tierHigh) return "HIGH";
  if (score >= cfg.tierMed) return "MED";
  return "LOW";
}

function now() { return Date.now(); }

function severityFromPoints(points: number): GriftSeverity {
  if (points >= 80) return "CRITICAL";
  if (points >= 55) return "HIGH";
  if (points >= 25) return "MED";
  return "LOW";
}

function makeDedupeKey(rule: GriftRuleCode, primaryUserId: number, secondaryUserId: number | null, bucketMs: number) {
  const bucket = Math.floor(now() / bucketMs);
  return `${rule}:${primaryUserId}:${secondaryUserId ?? 0}:${bucket}`;
}

function haversineKm(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const toRad = (d: number) => (d * Math.PI) / 180;
  const R = 6371;
  const dLat = toRad(lat2 - lat1);
  const dLon = toRad(lon2 - lon1);
  const a =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLon / 2) ** 2;
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}

function upsertDevice(deviceId: string, ip?: string, userAgent?: string, country?: string) {
  const ts = now();
  const existing = sqlite.prepare(`SELECT device_id FROM grift_devices WHERE device_id=?`).get(deviceId);
  if (!existing) {
    sqlite.prepare(`
      INSERT INTO grift_devices (device_id, first_seen_at, last_seen_at, first_ip, last_ip, first_user_agent, last_user_agent, first_country, last_country, trust_level)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, 'NEW')
    `).run(deviceId, ts, ts, ip ?? null, ip ?? null, userAgent ?? null, userAgent ?? null, country ?? null, country ?? null);
  } else {
    sqlite.prepare(`
      UPDATE grift_devices
      SET last_seen_at=?, last_ip=?, last_user_agent=?, last_country=?
      WHERE device_id=?
    `).run(ts, ip ?? null, userAgent ?? null, country ?? null, deviceId);
  }
}

function upsertDeviceUserLink(deviceId: string, userId: number) {
  const ts = now();
  const row = sqlite.prepare(`SELECT seen_count FROM grift_device_users WHERE device_id=? AND user_id=?`).get(deviceId, userId) as any;
  if (!row) {
    sqlite.prepare(`
      INSERT INTO grift_device_users (device_id, user_id, first_seen_at, last_seen_at, seen_count, link_strength)
      VALUES (?, ?, ?, ?, 1, 0)
    `).run(deviceId, userId, ts, ts);
  } else {
    sqlite.prepare(`
      UPDATE grift_device_users
      SET last_seen_at=?, seen_count=seen_count+1
      WHERE device_id=? AND user_id=?
    `).run(ts, deviceId, userId);
  }
}

function recomputeUserAggregates(userId: number, cfg: GriftConfig) {
  const ts = now();
  const ms7d = 7 * 24 * 60 * 60 * 1000;
  const ms30d = 30 * 24 * 60 * 60 * 1000;
  const since7d = ts - ms7d;
  const since30d = ts - ms30d;

  const openSignals = sqlite.prepare(`
    SELECT COUNT(*) as c, COALESCE(SUM(points),0) as p
    FROM grift_signals
    WHERE status='OPEN' AND (primary_user_id=? OR secondary_user_id=?)
  `).get(userId, userId) as any;

  const scoreCurrent = clamp(Number(openSignals?.p ?? 0), 0, 100);
  const tier = tierFor(scoreCurrent, cfg);

  const score7d = sqlite.prepare(`
    SELECT COALESCE(SUM(points),0) as p
    FROM grift_signals
    WHERE created_at>=? AND status='OPEN' AND (primary_user_id=? OR secondary_user_id=?)
  `).get(since7d, userId, userId) as any;

  const score30d = sqlite.prepare(`
    SELECT COALESCE(SUM(points),0) as p
    FROM grift_signals
    WHERE created_at>=? AND status='OPEN' AND (primary_user_id=? OR secondary_user_id=?)
  `).get(since30d, userId, userId) as any;

  const devices7d = sqlite.prepare(`
    SELECT COUNT(DISTINCT device_id) as c
    FROM grift_observations
    WHERE ts>=? AND user_id=? AND device_id IS NOT NULL
  `).get(since7d, userId) as any;

  const ips7d = sqlite.prepare(`
    SELECT COUNT(DISTINCT ip) as c
    FROM grift_observations
    WHERE ts>=? AND user_id=? AND ip IS NOT NULL
  `).get(since7d, userId) as any;

  const uas7d = sqlite.prepare(`
    SELECT COUNT(DISTINCT user_agent) as c
    FROM grift_observations
    WHERE ts>=? AND user_id=? AND user_agent IS NOT NULL
  `).get(since7d, userId) as any;

  const countries7d = sqlite.prepare(`
    SELECT COUNT(DISTINCT country) as c
    FROM grift_observations
    WHERE ts>=? AND user_id=? AND country IS NOT NULL
  `).get(since7d, userId) as any;

  const asns7d = sqlite.prepare(`
    SELECT COUNT(DISTINCT asn) as c
    FROM grift_observations
    WHERE ts>=? AND user_id=? AND asn IS NOT NULL
  `).get(since7d, userId) as any;

  const linkedAccounts30d = sqlite.prepare(`
    SELECT COUNT(DISTINCT du2.user_id) as c
    FROM grift_device_users du1
    JOIN grift_device_users du2 ON du1.device_id = du2.device_id
    WHERE du1.user_id=? AND du2.user_id != ?
      AND du1.last_seen_at >= ?
      AND du2.last_seen_at >= ?
  `).get(userId, userId, since30d, since30d) as any;

  const hedgePairs7d = sqlite.prepare(`
    SELECT COUNT(*) as c
    FROM grift_signals
    WHERE created_at>=? AND rule_code='HEDGE_PAIR' AND status='OPEN'
      AND (primary_user_id=? OR secondary_user_id=?)
  `).get(since7d, userId, userId) as any;

  const existing = sqlite.prepare(`SELECT user_id FROM grift_user_scores WHERE user_id=?`).get(userId);

  if (!existing) {
    sqlite.prepare(`
      INSERT INTO grift_user_scores (
        user_id, score_current, tier,
        score_7d, score_30d,
        devices_7d, ips_7d, user_agents_7d, countries_7d, asns_7d,
        linked_accounts_30d, hedge_pairs_7d, open_signals_count,
        last_evaluated_at
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `).run(
      userId, scoreCurrent, tier,
      Number(score7d?.p ?? 0), Number(score30d?.p ?? 0),
      Number(devices7d?.c ?? 0),
      Number(ips7d?.c ?? 0),
      Number(uas7d?.c ?? 0),
      Number(countries7d?.c ?? 0),
      Number(asns7d?.c ?? 0),
      Number(linkedAccounts30d?.c ?? 0),
      Number(hedgePairs7d?.c ?? 0),
      Number(openSignals?.c ?? 0),
      ts
    );
  } else {
    sqlite.prepare(`
      UPDATE grift_user_scores
      SET
        score_current=?,
        tier=?,
        score_7d=?,
        score_30d=?,
        devices_7d=?,
        ips_7d=?,
        user_agents_7d=?,
        countries_7d=?,
        asns_7d=?,
        linked_accounts_30d=?,
        hedge_pairs_7d=?,
        open_signals_count=?,
        last_evaluated_at=?
      WHERE user_id=?
    `).run(
      scoreCurrent, tier,
      Number(score7d?.p ?? 0),
      Number(score30d?.p ?? 0),
      Number(devices7d?.c ?? 0),
      Number(ips7d?.c ?? 0),
      Number(uas7d?.c ?? 0),
      Number(countries7d?.c ?? 0),
      Number(asns7d?.c ?? 0),
      Number(linkedAccounts30d?.c ?? 0),

export const griftEngine = {
  getConfig,

  onLoginSuccess(input: {
    userId: number;
    deviceId?: string;
    ip?: string;
    userAgent?: string;
    country?: string;
    region?: string;
    city?: string;
    latitude?: number | null;
    longitude?: number | null;
    asn?: number | null;
    org?: string | null;
  }) {
    const cfg = getConfig();
    if (!cfg.enabled) return;

    const ts = insertObservation({
      userId: input.userId,
      eventType: "LOGIN_SUCCESS",
      deviceId: input.deviceId,
      ip: input.ip,
      userAgent: input.userAgent,
      country: input.country,
      region: input.region,
      city: input.city,
      latitude: input.latitude ?? null,
      longitude: input.longitude ?? null,
      asn: input.asn ?? null,
      org: input.org ?? null,
    });

    const bucketHour = 60 * 60 * 1000;

    // DEVICE LINKAGE (multi-account)
    if (input.deviceId) {
      upsertDevice(input.deviceId, input.ip, input.userAgent, input.country);
      upsertDeviceUserLink(input.deviceId, input.userId);

      const since = ts - cfg.multiAccountWindowDays * 24 * 60 * 60 * 1000;
      const others = sqlite.prepare(`
        SELECT DISTINCT user_id as userId
        FROM grift_device_users
        WHERE device_id=? AND last_seen_at>=? AND user_id != ?
        LIMIT 25
      `).all(input.deviceId, since, input.userId) as any[];

      if (others.length > 0) {
        for (const o of others) {
          const otherUserId = Number(o.userId);
          const points = clamp(cfg.scoreMultiAccountDevice + 10 * Math.max(0, others.length - 1), 0, 75);
          const dedupeKey = makeDedupeKey("MULTI_ACCOUNT_DEVICE", input.userId, otherUserId, bucketHour);

          insertSignal({
            ruleCode: "MULTI_ACCOUNT_DEVICE",
            severity: severityFromPoints(points),
            primaryUserId: input.userId,
            secondaryUserId: otherUserId,
            deviceId: input.deviceId,
            ip: input.ip,
            userAgent: input.userAgent,
            country: input.country,
            region: input.region,
            city: input.city,
            latitude: input.latitude ?? null,
            longitude: input.longitude ?? null,
            asn: input.asn ?? null,
            org: input.org ?? null,
            points,
            dedupeKey,
            evidence: {
              deviceId: input.deviceId,
              primaryUserId: input.userId,
              secondaryUserId: otherUserId,
              windowDays: cfg.multiAccountWindowDays,
            },
          });
        }
      }
    }

    // CHURN METRICS
    const churnSince = ts - cfg.churnWindowHours * 60 * 60 * 1000;

    const ipCount = sqlite.prepare(`
      SELECT COUNT(DISTINCT ip) as c
      FROM grift_observations
      WHERE user_id=? AND ts>=? AND ip IS NOT NULL
    `).get(input.userId, churnSince) as any;

    if (Number(ipCount?.c ?? 0) >= cfg.ipUniqueThreshold) {
      const points = cfg.scoreIpChurn;
      const dedupeKey = makeDedupeKey("IP_CHURN", input.userId, null, bucketHour);
      insertSignal({
        ruleCode: "IP_CHURN",
        severity: severityFromPoints(points),
        primaryUserId: input.userId,
        deviceId: input.deviceId,
        ip: input.ip,
        userAgent: input.userAgent,
        country: input.country,
        points,
        dedupeKey,
        evidence: { uniqueIps: Number(ipCount?.c ?? 0), windowHours: cfg.churnWindowHours, threshold: cfg.ipUniqueThreshold },
      });
    }

    const uaCount = sqlite.prepare(`
      SELECT COUNT(DISTINCT user_agent) as c
      FROM grift_observations
      WHERE user_id=? AND ts>=? AND user_agent IS NOT NULL
    `).get(input.userId, churnSince) as any;

    if (Number(uaCount?.c ?? 0) >= cfg.uaUniqueThreshold) {
      const points = cfg.scoreUaChurn;
      const dedupeKey = makeDedupeKey("UA_CHURN", input.userId, null, bucketHour);
      insertSignal({
        ruleCode: "UA_CHURN",
        severity: severityFromPoints(points),
        primaryUserId: input.userId,
        deviceId: input.deviceId,
        ip: input.ip,
        userAgent: input.userAgent,
        country: input.country,
        points,
        dedupeKey,
        evidence: { uniqueUserAgents: Number(uaCount?.c ?? 0), windowHours: cfg.churnWindowHours, threshold: cfg.uaUniqueThreshold },
      });
    }

    const devCount = sqlite.prepare(`
      SELECT COUNT(DISTINCT device_id) as c
      FROM grift_observations
      WHERE user_id=? AND ts>=? AND device_id IS NOT NULL
    `).get(input.userId, churnSince) as any;

    if (Number(devCount?.c ?? 0) >= cfg.deviceUniqueThreshold) {
      const points = cfg.scoreDeviceChurn;
      const dedupeKey = makeDedupeKey("DEVICE_CHURN", input.userId, null, bucketHour);
      insertSignal({
        ruleCode: "DEVICE_CHURN",
        severity: severityFromPoints(points),
        primaryUserId: input.userId,
        deviceId: input.deviceId,
        ip: input.ip,
        userAgent: input.userAgent,
        country: input.country,
        points,
        dedupeKey,
        evidence: { uniqueDevices: Number(devCount?.c ?? 0), windowHours: cfg.churnWindowHours, threshold: cfg.deviceUniqueThreshold },
      });
    }

    const asnCount = sqlite.prepare(`
      SELECT COUNT(DISTINCT asn) as c
      FROM grift_observations
      WHERE user_id=? AND ts>=? AND asn IS NOT NULL
    `).get(input.userId, churnSince) as any;

    if (Number(asnCount?.c ?? 0) >= cfg.asnUniqueThreshold) {
      const points = cfg.scoreAsnVolatility;
      const dedupeKey = makeDedupeKey("ASN_VOLATILITY", input.userId, null, bucketHour);
      insertSignal({
        ruleCode: "ASN_VOLATILITY",
        severity: severityFromPoints(points),
        primaryUserId: input.userId,
        deviceId: input.deviceId,
        ip: input.ip,
        userAgent: input.userAgent,
        country: input.country,
        asn: input.asn ?? null,
        org: input.org ?? null,
        points,
        dedupeKey,
        evidence: { uniqueAsns: Number(asnCount?.c ?? 0), windowHours: cfg.churnWindowHours, threshold: cfg.asnUniqueThreshold },
      });
    }

    // GEO VELOCITY: compare previous observation with lat/lng
    if (input.latitude != null && input.longitude != null) {
      const prev = sqlite.prepare(`
        SELECT ts, latitude, longitude, country, city
        FROM grift_observations
        WHERE user_id=? AND ts < ? AND latitude IS NOT NULL AND longitude IS NOT NULL
        ORDER BY ts DESC
        LIMIT 1
      `).get(input.userId, ts) as any;

      if (prev?.latitude != null && prev?.longitude != null && prev?.ts != null) {
        const dtHours = (ts - Number(prev.ts)) / (1000 * 60 * 60);
        if (dtHours > 0 && dtHours <= cfg.geoVelocityMaxHours) {
          const distKm = haversineKm(Number(prev.latitude), Number(prev.longitude), Number(input.latitude), Number(input.longitude));
          const kmh = distKm / dtHours;

          if (distKm >= cfg.geoVelocityMinDistanceKm && kmh >= cfg.geoVelocityKmhThreshold) {
            const points = cfg.scoreGeoVelocity;
            const dedupeKey = makeDedupeKey("GEO_VELOCITY", input.userId, null, bucketHour);

            insertSignal({
              ruleCode: "GEO_VELOCITY",
              severity: severityFromPoints(points),
              primaryUserId: input.userId,
              deviceId: input.deviceId,
              ip: input.ip,
              userAgent: input.userAgent,
              country: input.country,
              region: input.region,
              city: input.city,
              latitude: input.latitude,
              longitude: input.longitude,
              points,
              dedupeKey,
              evidence: {
                from: { ts: Number(prev.ts), lat: Number(prev.latitude), lng: Number(prev.longitude), country: prev.country, city: prev.city },
                to: { ts, lat: Number(input.latitude), lng: Number(input.longitude), country: input.country, city: input.city },
                distKm,
                dtHours,
                kmh,
                thresholds: { minDistanceKm: cfg.geoVelocityMinDistanceKm, minKmh: cfg.geoVelocityKmhThreshold, maxHours: cfg.geoVelocityMaxHours },
              },
            });
          }
        }
      }
    }

    // CONCURRENT_SESSIONS heuristic: another login within concurrentWindowMinutes with different device or IP
    const concSince = ts - cfg.concurrentWindowMinutes * 60 * 1000;
    const conc = sqlite.prepare(`
      SELECT ts, device_id, ip, country
      FROM grift_observations
      WHERE user_id=? AND ts>=? AND ts < ?
      ORDER BY ts DESC
      LIMIT 1
    `).get(input.userId, concSince, ts) as any;

    if (conc) {
      const devDiff = input.deviceId && conc.device_id && String(input.deviceId) !== String(conc.device_id);
      const ipDiff = input.ip && conc.ip && String(input.ip) !== String(conc.ip);
      const cDiff = input.country && conc.country && String(input.country) !== String(conc.country);

      if (devDiff || ipDiff || cDiff) {
        const points = cfg.scoreConcurrentSessions;
        const dedupeKey = makeDedupeKey("CONCURRENT_SESSIONS", input.userId, null, bucketHour);
        insertSignal({
          ruleCode: "CONCURRENT_SESSIONS",
          severity: severityFromPoints(points),
          primaryUserId: input.userId,
          deviceId: input.deviceId,
          ip: input.ip,
          userAgent: input.userAgent,
          country: input.country,
          points,
          dedupeKey,
          evidence: {
            windowMinutes: cfg.concurrentWindowMinutes,
            previous: conc,
            current: { ts, deviceId: input.deviceId, ip: input.ip, country: input.country },
            differences: { devDiff, ipDiff, cDiff },
          },
        });
      }
    }

    recomputeUserAggregates(input.userId, cfg);
  },

  onTradeSubmit(input: {
    userId: number;
    tradeId: number;
    symbol: string;
    side: "BUY" | "SELL";
    deviceId?: string;
    ip?: string;
  }) {
    const cfg = getConfig();
    if (!cfg.enabled) return;

    if (!input.symbol || !input.side) return;

    const ts = insertTradeObservation({
      userId: input.userId,
      tradeId: input.tradeId,
      symbol: input.symbol,
      side: input.side,
      deviceId: input.deviceId,
      ip: input.ip,
    });

    const since = ts - cfg.hedgeWindowMinutes * 60 * 1000;
    const bucketMs = 10 * 60 * 1000;

    const candidates = sqlite.prepare(`
      SELECT user_id as userId, trade_id as tradeId, side as side, device_id as deviceId, ip as ip, ts as ts
      FROM grift_trade_observations
      WHERE symbol=? AND ts>=? AND user_id != ?
      ORDER BY ts DESC
      LIMIT 100
    `).all(input.symbol, since, input.userId) as any[];

    if (!candidates.length) {
      const cfg2 = getConfig();
      recomputeUserAggregates(input.userId, cfg2);
      return;
    }

    const opp = candidates.find((c) => String(c.side).toUpperCase() !== input.side);
    if (!opp) {
      const cfg2 = getConfig();
      recomputeUserAggregates(input.userId, cfg2);
      return;
    }

    const otherUserId = Number(opp.userId);

    // linkage logic per config
    let deviceMatch = false;
    let ipMatch = false;

    if (input.deviceId && opp.deviceId && String(input.deviceId) === String(opp.deviceId)) deviceMatch = true;
    if (input.ip && opp.ip && String(input.ip) === String(opp.ip)) ipMatch = true;

    const linkageOk =
      (cfg.hedgeRequireDeviceMatch ? deviceMatch : true) ||
      (cfg.hedgeAllowIpMatch ? ipMatch : false);

    if (!linkageOk) {
      recomputeUserAggregates(input.userId, cfg);
      return;
    }

    const points = cfg.scoreHedgePair;
    const dedupeKey = makeDedupeKey("HEDGE_PAIR", input.userId, otherUserId, bucketMs);

    insertSignal({
      ruleCode: "HEDGE_PAIR",
      severity: severityFromPoints(points),
      primaryUserId: input.userId,
      secondaryUserId: otherUserId,
      deviceId: input.deviceId,
      ip: input.ip,
      symbol: input.symbol,
      tradeId: input.tradeId,
      points,
      dedupeKey,
      evidence: {
        symbol: input.symbol,
        thisUserId: input.userId,
        otherUserId,
        thisSide: input.side,
        otherSide: String(opp.side).toUpperCase(),
        hedgeWindowMinutes: cfg.hedgeWindowMinutes,
        linkage: { deviceMatch, ipMatch, requireDevice: !!cfg.hedgeRequireDeviceMatch, allowIp: !!cfg.hedgeAllowIpMatch },
        thisTradeId: input.tradeId,
        otherTradeId: Number(opp.tradeId),
      },
    });

    recomputeUserAggregates(input.userId, cfg);
    recomputeUserAggregates(otherUserId, cfg);
  }
};

// client/src/lib/deviceId.ts
const STORAGE_KEY = "tc_device_id_v1";

function fallbackUuid(): string {
  const s: string[] = [];
  const hex = "0123456789abcdef";
  for (let i = 0; i < 36; i++) s.push(hex[Math.floor(Math.random() * 16)]);
  s[14] = "4";
  // @ts-ignore
  s[19] = hex[(parseInt(s[19], 16) & 0x3) | 0x8];
  s[8] = s[13] = s[18] = s[23] = "-";
  return s.join("");
}

export function getDeviceId(): string {
  try {
    const existing = localStorage.getItem(STORAGE_KEY);
    if (existing && existing.length >= 16) return existing;

    const id =
      typeof crypto !== "undefined" && "randomUUID" in crypto
        ? (crypto as any).randomUUID()
        : fallbackUuid();

    localStorage.setItem(STORAGE_KEY, id);
    return id;
  } catch {
    return typeof crypto !== "undefined" && "randomUUID" in crypto
      ? (crypto as any).randomUUID()
      : fallbackUuid();
  }
}

// client/src/lib/api.ts
import { getDeviceId } from "./deviceId";

export async function apiFetch<T>(
  url: string,
  opts?: {
    method?: string;
    body?: any;
    headers?: Record<string, string>;
  }
): Promise<T> {
  const method = opts?.method ?? "GET";
  const headers: Record<string, string> = {
    ...(opts?.headers ?? {}),
    "X-Device-Id": getDeviceId(),
  };

  let body: any = undefined;
  if (opts?.body !== undefined) {
    headers["Content-Type"] = "application/json";
    body = JSON.stringify(opts.body);
  }

  const res = await fetch(url, {
    method,
    headers,
    body,
    credentials: "include",
  });

  if (!res.ok) {
    const text = await res.text().catch(() => "");
    throw new Error(`API ${res.status}: ${text || res.statusText}`);
  }

  return (await res.json()) as T;
}
