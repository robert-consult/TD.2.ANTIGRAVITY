export function resolveSecret(value: string | undefined): string | undefined {
  if (!value) return value;
  const v = String(value).trim();
  if (v.toLowerCase().startsWith("env:")) {
    const key = v.slice(4).trim();
    return process.env[key];
  }
  return v;
}

import { db } from "@db";
import { and, eq, isNull } from "drizzle-orm";
import { marketDataProviders, systemConfig } from "@shared/schema";
import { MarketDataProviderUploadSchema } from "@shared/marketDataProviders";
import type { MarketDataProvider } from "./providerTypes";
import { buildProviderFromRow } from "./providerRegistry";

type ProviderRow = typeof marketDataProviders.$inferSelect;

let cachedActive: { providerKey: string; provider: MarketDataProvider; loadedAtMs: number } | null = null;

export async function getActiveProvider(): Promise<MarketDataProvider> {
  // Fast path cache (2s)
  if (cachedActive && Date.now() - cachedActive.loadedAtMs < 2000) return cachedActive.provider;

  const rows = await db
    .select()
    .from(marketDataProviders)
    .where(and(eq(marketDataProviders.isActive, true), isNull(marketDataProviders.deletedAt)))
    .limit(1);

  // Backward-compatible fallback: if none active, pick first enabled
  const activeRow = rows[0] ?? (await db
    .select()
    .from(marketDataProviders)
    .where(and(eq(marketDataProviders.isEnabled, true), isNull(marketDataProviders.deletedAt)))
    .limit(1))[0];

  if (!activeRow) {
    throw new Error("No market data providers configured. Add one in System Config > Market Data > Providers.");
  }

  const cfg = safeParseProviderConfig(activeRow.configJson, activeRow.driver);
  const provider = buildProviderFromRow(activeRow, cfg);

  cachedActive = { providerKey: activeRow.providerKey, provider, loadedAtMs: Date.now() };
  return provider;
}

export async function activateProvider(providerKey: string): Promise<void> {
  // single-writer semantics
  await db.transaction(async (tx) => {
    await tx.update(marketDataProviders).set({ isActive: false }).where(eq(marketDataProviders.isActive, true));
    await tx.update(marketDataProviders).set({ isActive: true }).where(eq(marketDataProviders.providerKey, providerKey));
    await tx.update(systemConfig).set({ marketDataActiveProviderKey: providerKey }).where(eq(systemConfig.id, 1));
  });

  // invalidate cache immediately
  cachedActive = null;
}

function safeParseProviderConfig(configJson: string, driverHint?: string) {
  let obj: any;
  try {
    obj = JSON.parse(configJson || "{}");
  } catch {
    obj = {};
  }
  if (driverHint && !obj.driver) obj.driver = driverHint;
  const parsed = MarketDataProviderUploadSchema.parse(obj);
  return parsed;
}
server/marketdata/providerRegistry.ts
import type { MarketDataProvider } from "./providerTypes";
import type { MarketDataProviderUpload } from "@shared/marketDataProviders";
import { TwelveDataProvider } from "./providers/twelvedata";
import { OneForgeProvider } from "./providers/oneforge";
import { GenericRestV1Provider } from "./providers/genericRestV1";

export function buildProviderFromRow(
  row: { providerKey: string; displayName: string; driver: string },
  cfg: MarketDataProviderUpload
): MarketDataProvider {
  switch (cfg.driver) {
    case "twelvedata":
      return new TwelveDataProvider({ providerKey: row.providerKey, displayName: row.displayName, cfg });
    case "oneforge":
      return new OneForgeProvider({ providerKey: row.providerKey, displayName: row.displayName, cfg });
    case "generic_rest_v1":
      return new GenericRestV1Provider({ providerKey: row.providerKey, displayName: row.displayName, cfg });
    default:
      throw new Error(`Unsupported provider driver: ${(cfg as any).driver}`);
  }
}
