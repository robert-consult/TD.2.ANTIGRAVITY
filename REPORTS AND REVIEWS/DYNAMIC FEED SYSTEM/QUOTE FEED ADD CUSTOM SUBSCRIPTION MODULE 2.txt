Here is the consolidated implementation plan, design document, and code patch based on the merged chains of thought. This solution introduces a "Dynamic Quote Feed" system that allows for a global baseline feed while supporting user-specific custom subscriptions.

1. Product Requirements Document (PRD)
Feature: Dynamic Quote Feed & Subscription Management Goal: Transition from a static, global quote feed to a dynamic model where traders receive a union of a "Baseline" feed and their own "Custom" subscriptions.

Core Requirements
Quote Modes (Per User):

Basic: User receives only the system-defined "Baseline" feed.

Custom: User receives only their specifically subscribed symbols.

Both (Hybrid): User receives the Baseline feed + their Custom subscriptions.

Data Distribution:

Upstream (Fetch): The system must subscribe to the union of the Baseline list and ALL active custom subscriptions to ensure data availability.

Downstream (Client): Each connected client receives only the data permitted by their Quote Mode and Subscription list.

Admin Interface:

New "Quote Subscriptions" mini-tab under the Instruments page.

Capabilities: Manage the Baseline symbol list, view/edit user Quote Modes, and manage User Subscription lists.

2. Algorithm & Architecture
Database Schema Changes
users table: Add quote_mode column (basic | custom | both, default: basic).

quote_baseline table: Stores the global list of symbols (replacing static config).

user_subscriptions table: Stores user_id <-> symbol relationships.

Backend Logic (The "Union" Strategy)
Ingestion Layer:

Periodically (or on change) calculate Set(Baseline) U Set(All_User_Subs).

Ensure the market data provider (WebSocket/REST) is watching this superset.

Distribution Layer (WebSocket):

On client connect, load user's quote_mode and subscriptions.

Construct the user's AllowedSet.

When a quote arrives for Symbol X:

If X is in Baseline, broadcast to all Basic and Both users.

If X is in a user's Custom list, send to that specific connection.

Optimization: For the MVP, we can filter at the socket level: if (userAllowed(socket, symbol)) send(socket, data).

import { quoteBaseline, userSubscriptions, users } from "@db/schema";
import { eq } from "drizzle-orm";

export class FeedManager {
  // Get the effective list of symbols a user should receive
  static async getUserAllowedSymbols(userId: number): Promise<Set<string>> {
    const user = await db.query.users.findFirst({
      where: eq(users.id, userId),
      columns: { quoteMode: true },
    });

    if (!user) return new Set();

    const allowed = new Set<string>();

    // 1. Load Baseline if applicable
    if (user.quoteMode === "basic" || user.quoteMode === "both") {
      const baseline = await db.select({ symbol: quoteBaseline.symbol }).from(quoteBaseline);
      baseline.forEach((b) => allowed.add(b.symbol));
    }

    // 2. Load Custom Subs if applicable
    if (user.quoteMode === "custom" || user.quoteMode === "both") {
      const subs = await db.select({ symbol: userSubscriptions.symbol })
        .from(userSubscriptions)
        .where(eq(userSubscriptions.userId, userId));
      subs.forEach((s) => allowed.add(s.symbol));
    }

    return allowed;
  }

  // Get the UNION of all needed symbols (for the upstream fetcher)
  static async getGlobalFetchList(): Promise<string[]> {
    const baseline = await db.select({ symbol: quoteBaseline.symbol }).from(quoteBaseline);
    const custom = await db.select({ symbol: userSubscriptions.symbol }).from(userSubscriptions);
    
    const union = new Set([
        ...baseline.map(b => b.symbol),
        ...custom.map(c => c.symbol)
    ]);
    
    return Array.from(union);
  }
}

import { db } from "@db";
import { quoteBaseline, userSubscriptions, users } from "@db/schema";
import { eq } from "drizzle-orm";

export class FeedManager {
  // Cache for the global fetch list to avoid heavy DB queries on every tick
  private static globalFetchListCache: Set<string> | null = null;

  /**
   * Returns the set of symbols a specific user is allowed to see.
   */
  static async getUserAllowedSymbols(userId: number): Promise<Set<string>> {
    const user = await db.query.users.findFirst({
      where: eq(users.id, userId),
      columns: { quoteMode: true },
    });

    if (!user) return new Set();

    const allowed = new Set<string>();

    // 1. Add Baseline (if mode is Basic or Both)
    if (user.quoteMode === "basic" || user.quoteMode === "both") {
      const baseline = await db.select({ symbol: quoteBaseline.symbol })
        .from(quoteBaseline)
        .where(eq(quoteBaseline.isActive, true));
      baseline.forEach((b) => allowed.add(b.symbol));
    }

    // 2. Add Custom Subs (if mode is Custom or Both)
    if (user.quoteMode === "custom" || user.quoteMode === "both") {
      const subs = await db.select({ symbol: userSubscriptions.symbol })
        .from(userSubscriptions)
        .where(eq(userSubscriptions.userId, userId));
      subs.forEach((s) => allowed.add(s.symbol));
    }

    return allowed;
  }

  /**
   * Returns the UNION of all symbols the system needs to fetch from upstream.
   * (Baseline U All_User_Custom_Subs)
   */
  static async getGlobalFetchList(forceRefresh = false): Promise<string[]> {
    if (!forceRefresh && this.globalFetchListCache) {
      return Array.from(this.globalFetchListCache);
    }

    const baseline = await db.select({ symbol: quoteBaseline.symbol })
      .from(quoteBaseline)
      .where(eq(quoteBaseline.isActive, true));

    const custom = await db.select({ symbol: userSubscriptions.symbol })
      .from(userSubscriptions);

    const union = new Set([
      ...baseline.map(b => b.symbol),
      ...custom.map(c => c.symbol)
    ]);

    this.globalFetchListCache = union;
    return Array.from(union);
  }

  /**
   * Call this whenever subscriptions change to invalidate cache.
   */
  static invalidateCache() {
    this.globalFetchListCache = null;
    // Optional: Trigger upstream re-subscription logic here
  }
}
