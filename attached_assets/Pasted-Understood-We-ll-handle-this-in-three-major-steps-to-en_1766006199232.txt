#### **Backend (Express Route Patch: `usersController.js`)**

```js
// Toggle Freeze
exports.freezeUser = async (req, res) => {
  const { id, reason } = req.body;
  await prisma.user.update({
    where: { id },
    data: { isFrozen: true, freezeReason: reason || 'No reason provided' }
  });
  res.json({ success: true, status: 'frozen' });
};

// Unfreeze
exports.unfreezeUser = async (req, res) => {
  const { id } = req.body;
  const user = await prisma.user.findUnique({ where: { id } });
  let newState = { isFrozen: false };
  if (!user.isDisabled) newState.status = 'Active';
  await prisma.user.update({ where: { id }, data: newState });
  res.json({ success: true, status: 'active' });
};

// Disable
exports.disableUser = async (req, res) => {
  const { id, reason } = req.body;
  await prisma.user.update({
    where: { id },
    data: { isDisabled: true, disableReason: reason || 'No reason provided' }
  });
  res.json({ success: true, status: 'disabled' });
};

// Enable
exports.enableUser = async (req, res) => {
  const { id } = req.body;
  const user = await prisma.user.findUnique({ where: { id } });
  let updateData = { isDisabled: false };
  // logic branch
  if (user.isFrozen) updateData.status = 'Frozen';
  else updateData.status = 'Active';
  await prisma.user.update({ where: { id }, data: updateData });
  res.json({ success: true, status: updateData.status });
};
```

exports.getLoginHistory = async (req, res) => {
  const logs = await prisma.loginHistory.findMany({
    include: { user: { select: { email: true, username: true } } },
    orderBy: { createdAt: 'desc' },
  });
  res.json(logs);
};
