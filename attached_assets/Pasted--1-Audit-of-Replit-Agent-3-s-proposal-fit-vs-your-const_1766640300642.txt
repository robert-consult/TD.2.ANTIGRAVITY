## 3) Profile Settings (top-right menu) — prop-grade, low-friction standard (research-backed)

For the **trader-facing profile settings**, prop/fintech-grade patterns converge on:

* **Progressive disclosure / risk-based step-up:** don’t force phone/KYC at signup; only request when needed (fits your model). Plaid’s onboarding overview explicitly describes KYC as a later verification step after signup in many fintech flows.
* **Security-first “Privacy & Security” section:** 2FA with backup codes is standard in trading products. TradingView explicitly guides enabling 2FA in account settings and emphasizes saving backup codes.
* **Account activity log:** exchanges commonly expose a security/audit event list (login success/fail, password changes, API key events). Deribit’s account settings reference event types like login success/fail and security-key/API-key actions.
* **KYC requirements are jurisdiction- and capability-dependent:** if you later integrate a payments/KYC provider, required fields vary by region and account capabilities (Stripe Connect documents this directly).

**Minimum sections for your Profile Settings page (trader self-service):**

1. **Account**

   * Email (verified status indicator)
   * Username (usually immutable or limited changes)
   * Display name (can map to your existing `users.name`)
2. **Contact**

   * Phone (locked behind eligibility if you want; otherwise “optional until invited to funding”)

3. **Security**

   * Change password
   * 2FA (Authenticator app first; SMS only if you choose)
   * Active sessions / recent logins (read-only)
4. **Notifications**

   * Trade fills, margin alerts, rule-violation alerts, weekly summary
5. **Verification & Payout Profile (GATED)**

   * Legal first/last name, DOB, address, docs upload
   * **Preferred Payment Currency** (your wording) defaulted to region; only appears once the user is in the funding pipeline

```tsx
import axios from "axios";
import { useQuery } from "@tanstack/react-query";

import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";

type FullUser = {
  id: number;
  email: string;
  username: string;
  name?: string | null;
  phone?: string | null;
  isAdmin: boolean;
  isDisabled?: boolean;
  isFrozen?: boolean;
  createdAt?: number | null; // unix seconds in this app
};

type AdminActionRow = {
  id: number;
  adminId: number | null;
  userId: number | null;
  actionType: string;

  reasonText?: string | null;
  ip?: string | null;

};

type LoginHistoryRow = {
  id: number;
  userId: number | null;
  email: string | null;
  username: string | null;
  ipAddress: string | null;
  userAgent: string | null;
  success: boolean;
  failureReason: string | null;
  createdAt: number; // unix seconds
};

type ExportManifestRow = {
  id: number;
  exported_at_utc_ms: number;

  row_count: number;
  sha256_manifest: string;
  sha256_payload: string;
  note?: string | null;
};

type FeedKind = "SIGNUP" | "LOGIN" | "ADMIN_ACTION" | "EXPORT";

type FeedItem = {
  id: string;
  kind: FeedKind;
  tsMs: number;
  userEmail?: string | null;
  actorEmail?: string | null;
  ip?: string | null;
  title: string;
  note?: string | null;
  severity: "INFO" | "WARN";
};

function toMsFromUnixSeconds(v: any): number | null {
  if (v == null) return null;
  if (v instanceof Date) return v.getTime();
  const n = Number(v);
  if (!Number.isFinite(n)) return null;
  // heuristic: app stores seconds for most createdAt
  return n > 10_000_000_000 ? n : n * 1000;
}

function fmt(tsMs: number): string {
  try {
    return new Date(tsMs).toLocaleString();
  } catch {
    return String(tsMs);
  }
}

export default function AdminAuditTrail() {
  const [q, setQ] = useState("");
  const [kind, setKind] = useState<FeedKind | "ALL">("ALL");
  const limit = 250;

  const usersQ = useQuery<FullUser[]>({
    queryKey: ["/api/admin/users/full"],
    queryFn: () => axios.get("/api/admin/users/full").then(r => r.data),
  });

  const actionsQ = useQuery<AdminActionRow[]>({
    queryKey: ["/api/admin/audit-log", limit],
    queryFn: () => axios.get(`/api/admin/audit-log?limit=${limit}`).then(r => r.data),
  });

  const loginsQ = useQuery<LoginHistoryRow[]>({
    queryKey: ["/api/admin/login-history", limit],
    queryFn: () => axios.get(`/api/admin/login-history?limit=${limit}`).then(r => r.data),
  });

  const manifestsQ = useQuery<ExportManifestRow[]>({
    queryKey: ["/api/admin/export-manifests", 100],
    queryFn: () => axios.get(`/api/admin/export-manifests?limit=100`).then(r => r.data),
  });

  const isLoading = usersQ.isLoading || actionsQ.isLoading || loginsQ.isLoading || manifestsQ.isLoading;

  const feed = useMemo<FeedItem[]>(() => {
    const users = usersQ.data ?? [];
    const actions = actionsQ.data ?? [];
    const logins = loginsQ.data ?? [];
    const manifests = manifestsQ.data ?? [];

    const byId = new Map<number, FullUser>();
    for (const u of users) byId.set(u.id, u);

    const items: FeedItem[] = [];

    // Signups (derived from users.createdAt)
    for (const u of users) {
      const ms = toMsFromUnixSeconds(u.createdAt);
      if (!ms) continue;
      items.push({
        id: `signup:${u.id}:${ms}`,
        kind: "SIGNUP",
        tsMs: ms,
        userEmail: u.email,
        actorEmail: null,
        ip: null,
        title: "User signup",
        note: `@${u.username}`,
        severity: "INFO",
      });
    }

    // Login events
    for (const l of logins) {
      const ms = toMsFromUnixSeconds(l.createdAt) ?? Date.now();
      items.push({
        id: `login:${l.id}`,
        kind: "LOGIN",
        tsMs: ms,
        userEmail: l.email ?? (l.userId ? byId.get(l.userId)?.email : null) ?? null,
        actorEmail: null,
        ip: l.ipAddress,
        title: l.success ? "Login success" : "Login failed",
        note: l.success ? null : (l.failureReason ?? "Unknown failure"),
        severity: l.success ? "INFO" : "WARN",
      });
    }

    // Admin actions
    for (const a of actions) {
      const ms = toMsFromUnixSeconds(a.createdAt) ?? Date.now();
      const targetEmail =
        (a.userId ? byId.get(a.userId)?.email : null) ?? null;
      const actorEmail =
        (a.adminId ? byId.get(a.adminId)?.email : null) ?? null;

      items.push({
        id: `admin:${a.id}`,
        kind: "ADMIN_ACTION",
        tsMs: ms,
        userEmail: targetEmail,
        actorEmail,
        ip: a.ip ?? null,
        title: a.actionType,
        note: [a.reasonCode, a.reasonText].filter(Boolean).join(" • ") || null,
        severity: "WARN",
      });
    }

    // Export manifests (tamper-evidence / integrity)
    for (const m of manifests) {
      items.push({
        id: `export:${m.id}`,
        kind: "EXPORT",
        tsMs: Number(m.exported_at_utc_ms) || Date.now(),
        userEmail: null,
        actorEmail: null,
        ip: null,
        title: `Export manifest: ${m.export_type}`,
        note: `rows=${m.row_count}`,
        severity: "INFO",
      });
    }

    items.sort((a, b) => b.tsMs - a.tsMs);
    return items;
  }, [usersQ.data, actionsQ.data, loginsQ.data, manifestsQ.data]);

  const filtered = useMemo(() => {
    const needle = q.trim().toLowerCase();
    return feed.filter(it => {
      if (kind !== "ALL" && it.kind !== kind) return false;
      if (!needle) return true;

      const hay = [
        it.title,
        it.note ?? "",
        it.userEmail ?? "",
        it.actorEmail ?? "",
        it.ip ?? "",
      ].join(" ").toLowerCase();

      return hay.includes(needle);
    });
  }, [feed, q, kind]);

  return (
    <div className="space-y-4">
      <Card className="bg-neutral-800 border-gray-700">
        <CardHeader className="flex flex-row items-center justify-between">
          <CardTitle className="text-white">Audit Trail</CardTitle>
          <div className="flex items-center gap-2">
            <Badge variant="outline" className="border-gray-600 text-gray-200">
              {filtered.length} events
            </Badge>
            <Button
              variant="outline"
              className="border-gray-600 text-gray-200 hover:bg-neutral-700"
              onClick={() => {
                usersQ.refetch();
                actionsQ.refetch();
                loginsQ.refetch();
                manifestsQ.refetch();
              }}
              disabled={isLoading}
            >
              Refresh
            </Button>
          </div>
        </CardHeader>

        <CardContent className="space-y-3">
          <div className="flex flex-col md:flex-row md:items-center gap-2">
            <Input
              className="bg-neutral-700 border-gray-600 text-white placeholder:text-gray-400"
              placeholder="Search email / IP / action / reason…"
              value={q}
              onChange={(e) => setQ(e.target.value)}
            />
            <div className="flex items-center gap-2">
              <Button
                variant={kind === "ALL" ? "default" : "outline"}
                className={kind === "ALL" ? "" : "border-gray-600 text-gray-200 hover:bg-neutral-700"}
                onClick={() => setKind("ALL")}
              >
                All
              </Button>
              <Button
                variant={kind === "SIGNUP" ? "default" : "outline"}
                className={kind === "SIGNUP" ? "" : "border-gray-600 text-gray-200 hover:bg-neutral-700"}
                onClick={() => setKind("SIGNUP")}
              >
                Signups
              </Button>
              <Button
                variant={kind === "LOGIN" ? "default" : "outline"}
                className={kind === "LOGIN" ? "" : "border-gray-600 text-gray-200 hover:bg-neutral-700"}
                onClick={() => setKind("LOGIN")}

                onClick={() => setKind("ADMIN_ACTION")}
              >
                Admin
              </Button>
              <Button
                variant={kind === "EXPORT" ? "default" : "outline"}
                className={kind === "EXPORT" ? "" : "border-gray-600 text-gray-200 hover:bg-neutral-700"}

                Exports

                  <TableHead className="text-gray-300">Type</TableHead>
                  <TableHead className="text-gray-300">User</TableHead>
                  <TableHead className="text-gray-300">Actor</TableHead>
```

```tsx
import { useQuery } from "@tanstack/react-query";

  username: string;
  email: string;

  win_rate: number;
  profit: number;
  profit_percent: number;

  username: string;
  name?: string | null;

function daysSince(unixSeconds?: number | null): number | null {
  if (!unixSeconds) return null;
  const ms = unixSeconds * 1000;
  const d = (Date.now() - ms) / (1000 * 60 * 60 * 24);
  return Number.isFinite(d) ? Math.floor(d) : null;
}

export default function KycQueue() {
  // Adjustable eligibility thresholds (UI-only; backend enforcement comes later)
  const [minTrades, setMinTrades] = useState(30);
  const [minProfitPct, setMinProfitPct] = useState(5);
  const [minWinRate, setMinWinRate] = useState(45);
  const [search, setSearch] = useState("");

  const statsQ = useQuery<TraderStatsRow[]>({
    queryKey: ["/api/admin/trader-stats", 30],
    queryFn: () => axios.get("/api/admin/trader-stats?days=30").then(r => r.data),
  });

  const usersQ = useQuery<FullUser[]>({
    queryKey: ["/api/admin/users/full"],
    queryFn: () => axios.get("/api/admin/users/full").then(r => r.data),
  });

  const candidates = useMemo(() => {
    const stats = statsQ.data ?? [];
    const users = usersQ.data ?? [];
    const byId = new Map<number, FullUser>();
    for (const u of users) byId.set(u.id, u);

    const needle = search.trim().toLowerCase();

    return stats
      .map(s => {
        const u = byId.get(s.user_id);
        const signupDays = daysSince(u?.createdAt ?? null);
        const eligible =
          s.total_trades >= minTrades &&
          s.profit_percent >= minProfitPct &&
          s.win_rate >= minWinRate &&
          !(u?.isDisabled) &&
          !(u?.isFrozen);

        return {
          ...s,
          signupDays,
          name: u?.name ?? null,
          phone: u?.phone ?? null,
          isDisabled: u?.isDisabled ?? false,
          isFrozen: u?.isFrozen ?? false,
          eligible,
        };
      })
      .filter(row => {
        if (!needle) return true;
        const hay = [
          row.email,
          row.username,
          row.name ?? "",
          String(row.user_id),
        ].join(" ").toLowerCase();
        return hay.includes(needle);
      })
      // show eligible first, then by profit%
      .sort((a, b) => {
        if (a.eligible !== b.eligible) return a.eligible ? -1 : 1;
        return (b.profit_percent ?? 0) - (a.profit_percent ?? 0);
      });
  }, [statsQ.data, usersQ.data, minTrades, minProfitPct, minWinRate, search]);

  const eligibleCount = candidates.filter(c => c.eligible).length;

      <Card className="bg-neutral-800 border-gray-700">

            </Badge>
            <Badge variant="outline" className="border-gray-600 text-gray-200">
              Total: {candidates.length}
            </Badge>
          </div>
        </CardHeader>

        <CardContent className="space-y-3">
          <div className="text-sm text-gray-300">
            This queue is intentionally <span className="font-semibold">step-up gated</span>: traders can use the platform freely; only top performers get invited into KYC/phone verification and the “real capital” pipeline.
          </div>

            <div>
              <div className="text-xs text-gray-400 mb-1">Min trades</div>
              <Input
                className="bg-neutral-700 border-gray-600 text-white"
                type="number"
                value={minTrades}
                onChange={(e) => setMinTrades(Number(e.target.value || 0))}
              />
            </div>
            <div>
              <div className="text-xs text-gray-400 mb-1">Min profit %</div>
              <Input
                className="bg-neutral-700 border-gray-600 text-white"
                type="number"
                value={minProfitPct}
                onChange={(e) => setMinProfitPct(Number(e.target.value || 0))}
              />
            </div>
            <div>
              <div className="text-xs text-gray-400 mb-1">Min win rate %</div>
              <Input
                className="bg-neutral-700 border-gray-600 text-white"
                type="number"
                value={minWinRate}
                onChange={(e) => setMinWinRate(Number(e.target.value || 0))}

            </div>
            <div>
              <div className="text-xs text-gray-400 mb-1">Search</div>
```

```tsx
import axios from "axios";
import { useQuery } from "@tanstack/react-query";

import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";

type FullUser = {
  id: number;
  email: string;
  username: string;
  createdAt?: number | null; // unix seconds
};

type LoginHistoryRow = {
  id: number;
  success: boolean;
  createdAt: number; // unix seconds
};

type AdminActionRow = {
  id: number;
  createdAt: number; // unix seconds
};

function withinDays(unixSeconds: number, days: number): boolean {
  const cutoffMs = Date.now() - days * 24 * 60 * 60 * 1000;
  return unixSeconds * 1000 >= cutoffMs;
}

export default function AdminComplianceAnalytics() {
  const usersQ = useQuery<FullUser[]>({
    queryKey: ["/api/admin/users/full"],
    queryFn: () => axios.get("/api/admin/users/full").then(r => r.data),
  });

  const loginsQ = useQuery<LoginHistoryRow[]>({
    queryKey: ["/api/admin/login-history", 500],
    queryFn: () => axios.get("/api/admin/login-history?limit=500").then(r => r.data),
  });

  const actionsQ = useQuery<AdminActionRow[]>({
    queryKey: ["/api/admin/audit-log", 500],
    queryFn: () => axios.get("/api/admin/audit-log?limit=500").then(r => r.data),
  });

  const stats = useMemo(() => {
    const users = usersQ.data ?? [];
    const logins = loginsQ.data ?? [];
    const actions = actionsQ.data ?? [];

    const totalUsers = users.length;

    const signups7 = users.filter(u => u.createdAt && withinDays(u.createdAt, 7)).length;
    const signups30 = users.filter(u => u.createdAt && withinDays(u.createdAt, 30)).length;

    const failedLogins7 = logins.filter(l => !l.success && withinDays(l.createdAt, 7)).length;
    const logins7 = logins.filter(l => withinDays(l.createdAt, 7)).length;

    const adminActions7 = actions.filter(a => withinDays(a.createdAt, 7)).length;

    return { totalUsers, signups7, signups30, failedLogins7, logins7, adminActions7 };
  }, [usersQ.data, loginsQ.data, actionsQ.data]);

  return (
    <Card className="bg-neutral-800 border-gray-700">
      <CardHeader className="flex flex-row items-center justify-between">
        <CardTitle className="text-white">Compliance & Growth Analytics (Additive)</CardTitle>
        <Badge variant="outline" className="border-gray-600 text-gray-200">
          Rolling windows
        </Badge>
      </CardHeader>
      <CardContent>
        <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
          <Card className="bg-neutral-900 border-gray-700">
            <CardHeader>
              <div className="text-gray-400 text-xs">Total users</div>
              <div className="text-white text-2xl font-semibold">{stats.totalUsers}</div>
            </CardHeader>
          </Card>

          <Card className="bg-neutral-900 border-gray-700">
            <CardHeader>
              <div className="text-gray-400 text-xs">Signups (7d / 30d)</div>
              <div className="text-white text-2xl font-semibold">
                {stats.signups7} <span className="text-gray-400 text-sm font-normal">/ {stats.signups30}</span>
              </div>
            </CardHeader>
          </Card>

          <Card className="bg-neutral-900 border-gray-700">
            <CardHeader>
              <div className="text-gray-400 text-xs">Login attempts (7d)</div>
              <div className="text-white text-2xl font-semibold">{stats.logins7}</div>
            </CardHeader>
          </Card>

          <Card className="bg-neutral-900 border-gray-700">
            <CardHeader>
              <div className="text-gray-400 text-xs">Failed logins (7d)</div>
              <div className="text-white text-2xl font-semibold">{stats.failedLogins7}</div>
            </CardHeader>
          </Card>

          <Card className="bg-neutral-900 border-gray-700">
            <CardHeader>
              <div className="text-gray-400 text-xs">Admin actions (7d)</div>
              <div className="text-white text-2xl font-semibold">{stats.adminActions7}</div>
            </CardHeader>
          </Card>

          <Card className="bg-neutral-900 border-gray-700">
            <CardHeader>
              <div className="text-gray-400 text-xs">Next step</div>
              <div className="text-gray-300 text-sm">
                Add KYC-case statuses + phone verification events to extend these metrics without changing tab structure.
              </div>
            </CardHeader>
          </Card>
        </div>
      </CardContent>
    </Card>
  );
}
```
