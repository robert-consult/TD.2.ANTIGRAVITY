import type { DenyCode, PolicyDecision } from "./types";

export function deny(
  d: Omit<NonNullable<PolicyDecision["deny"]>, "messageKey"> & { messageKey?: string },
  extra?: Partial<PolicyDecision>
): PolicyDecision {
  return {
    allow: false,
    derived: extra?.derived as any, // caller must fill derived
    deny: {
      messageKey: d.messageKey ?? `deny.${d.code}`,
      ...d,
    },
    obligations: extra?.obligations ?? [],
    ui: extra?.ui,
  };
}

export function denyHttp(code: DenyCode): 401 | 403 | 409 | 429 {
  switch (code) {
    case "EMAIL_SEND_COOLDOWN":
    case "EMAIL_SEND_RATE_LIMIT":
    case "SMS_RATE_LIMIT":
      return 429;
    case "ACTION_NOT_ALLOWED_IN_STATE":
    case "EMAIL_UNVERIFIED_GRACE_EXPIRED":
    case "EMAIL_REVERIFY_OVERDUE":
    case "SMS_NOT_ELIGIBLE":
    case "KYC_NOT_SELECTED":
    case "KYC_STATE_INVALID":
      return 403;
    case "ADMIN_SUSPENDED":
      return 403;
    default:
      return 403;
  }
}

function dayKey(d: Date): string {
  // YYYY-MM-DD in server local time
  const yyyy = d.getFullYear();
  const mm = String(d.getMonth() + 1).padStart(2, "0");
  const dd = String(d.getDate()).padStart(2, "0");
  return `${yyyy}-${mm}-${dd}`;
}

function addDays(d: Date, days: number): Date {
  return new Date(d.getTime() + days * 86400000);
}

function diffDays(a: Date, b: Date): number {
  return (a.getTime() - b.getTime()) / 86400000;
}

export function resolveEmailInitialDueAt(ctx: DecisionContext, cfg: PolicyConfig): Date {
  if (ctx.user.emailInitialDueAt) return ctx.user.emailInitialDueAt;
  return addDays(ctx.user.createdAt, cfg.emailInitialGraceDays);
}

export function resolveAccountState(ctx: DecisionContext, cfg: PolicyConfig): AccountState {
  if (ctx.user.suspendedAt) return "SUSPENDED_ADMIN";

  const emailInitialDueAt = resolveEmailInitialDueAt(ctx, cfg);

  if (!ctx.user.emailVerifiedAt) {
    return ctx.now < emailInitialDueAt ? "ACTIVE_UNVERIFIED_GRACE" : "LOCKED_EMAIL_UNVERIFIED";
  }

  if (ctx.user.emailReverifyDueAt) {
    const overdueAt = addDays(ctx.user.emailReverifyDueAt, cfg.emailReverifyOverdueGraceDays);
    if (ctx.now >= overdueAt) return "LOCKED_EMAIL_REVERIFY_OVERDUE";
  }

  return "ACTIVE_VERIFIED";
}

export function isSelectedForKyc(ctx: DecisionContext): boolean {
  return !!ctx.user.selectedAt || ctx.user.contenderTier === "SELECTED_REAL_CAPITAL";
}

export function isSmsEligibleByPerformance(ctx: DecisionContext, cfg: PolicyConfig): boolean {
  // Hard prerequisite: must be ACTIVE_VERIFIED (NOT locked)
  const st = resolveAccountState(ctx, cfg);
  if (st !== "ACTIVE_VERIFIED") return false;

  const ageDays = diffDays(ctx.now, ctx.user.createdAt);
  const agg = ctx.user.tradingAgg;

  const balancePct = agg.startingBalance > 0 ? agg.balanceNow / agg.startingBalance : 0;

  const path1 =
    ageDays >= cfg.contenderPath1.minAccountAgeDays &&
    balancePct >= cfg.contenderPath1.minBalancePctOfStart &&
    agg.positionsCountLifetime >= cfg.contenderPath1.minLifetimeTrades;

  const recentTradeOk =
    !!agg.lastTradeAt && diffDays(ctx.now, agg.lastTradeAt) <= cfg.contenderPath2.requireRecentTradeDays;

  const path2 =
    ageDays >= cfg.contenderPath2.minAccountAgeDays &&
    recentTradeOk &&
    agg.returnLast90 >= cfg.contenderPath2.minReturnLast90 &&
    agg.positionsCountLast90 >= cfg.contenderPath2.minTradesLast90;

  return path1 || path2;
}

function canSendEmail(ctx: DecisionContext, cfg: PolicyConfig): { ok: true } | { ok: false; why: "cooldown" | "cap" } {
  const now = ctx.now;
  const k = dayKey(now);
  const c = ctx.user.counters.email;

  if (c.lastSentAt) {
    const sec = (now.getTime() - c.lastSentAt.getTime()) / 1000;
    if (sec < cfg.reminderResendCooldownSec) return { ok: false, why: "cooldown" };
  }

  // reset day counter if day key differs
  const countToday = c.sentDayKey === k ? c.sentCountDay : 0;
  if (countToday >= cfg.reminderMaxPerDay) return { ok: false, why: "cap" };

  return { ok: true };
}

function canSendSms(ctx: DecisionContext, cfg: PolicyConfig): { ok: true } | { ok: false; why: "cap" } {
  const now = ctx.now;
  const k = dayKey(now);
  const c = ctx.user.counters.sms;
  const countToday = c.sentDayKey === k ? c.sentCountDay : 0;
  if (countToday >= cfg.smsDailySendCap) return { ok: false, why: "cap" };
  return { ok: true };
}

function isOtpLocked(ctx: DecisionContext): boolean {
  const o = ctx.user.counters.otp;
  return !!o.lockedUntil && ctx.now < o.lockedUntil;
}

function isExposureIncreasing(action: PolicyAction): boolean {
  return (
    action === "TRADE_OPEN_OR_INCREASE" ||
    action === "TRADE_MODIFY_SLTP" ||
    action === "TRADE_PLACE_PENDING" ||
    action === "TRADE_CANCEL_PENDING"
  );
}

export function evaluatePolicy(ctx: DecisionContext, cfg: PolicyConfig): PolicyDecision {
  const obligations: PolicyDecision["obligations"] = [];

  const emailInitialDueAt = resolveEmailInitialDueAt(ctx, cfg);
  const accountState = resolveAccountState(ctx, cfg);

  const smsEligible = isSmsEligibleByPerformance(ctx, cfg);
  const selectedForKyc = isSelectedForKyc(ctx);

  const derived = {
    accountState,
    contenderTier: ctx.user.contenderTier,
    emailInitialDueAt,
    emailReverifyDueAt: ctx.user.emailReverifyDueAt,
    isEmailInitialOverdue: !ctx.user.emailVerifiedAt && ctx.now >= emailInitialDueAt,
    isEmailReverifyOverdue: !!ctx.user.emailReverifyDueAt && ctx.now >= addDays(ctx.user.emailReverifyDueAt, cfg.emailReverifyOverdueGraceDays),
    isSmsEligibleByPerformance: smsEligible,
    isSelectedForKyc: selectedForKyc,
  };

  const denyAndLog = (
    code: Parameters<typeof denyHttp>[0],
    nextStep?: PolicyDecision["deny"]["nextStep"]
  ): PolicyDecision => {
    const httpStatus = denyHttp(code);
    obligations.push({
      type: "ACCOUNT_ACTION_DENIED",
      data: {
        code,
        action: ctx.action,
        accountState,
        contenderTier: ctx.user.contenderTier,
        emailInitialDueAt: emailInitialDueAt.toISOString(),
        emailReverifyDueAt: ctx.user.emailReverifyDueAt?.toISOString() ?? null,
        correlationId: ctx.request.correlationId,
      },
    });
    return {
      allow: false,
      derived,
      deny: { code, messageKey: `deny.${code}`, httpStatus, nextStep },
      obligations,
      ui: {
        showLockedBanner:
          code === "EMAIL_UNVERIFIED_GRACE_EXPIRED" || code === "EMAIL_REVERIFY_OVERDUE" || code === "ADMIN_SUSPENDED",
        redirectTo:
          code === "EMAIL_UNVERIFIED_GRACE_EXPIRED" || code === "EMAIL_REVERIFY_OVERDUE"
            ? "/verify-email"
            : undefined,
      },
    };
  };

  // 1) admin suspension overrides all
  if (accountState === "SUSPENDED_ADMIN") {
    // Optional: allow verify-email flow even when suspended. Keep strict by default.
    if (ctx.action === "EMAIL_RESEND_VERIFICATION") {
      // still allow email sending only if throttles pass
    } else {
      return denyAndLog("ADMIN_SUSPENDED", "CONTACT_SUPPORT");
    }
  }

  // 2) enforcement: locked states block exposure-increasing actions
  if (isExposureIncreasing(ctx.action)) {
    if (accountState === "LOCKED_EMAIL_UNVERIFIED") return denyAndLog("EMAIL_UNVERIFIED_GRACE_EXPIRED", "VERIFY_EMAIL");
    if (accountState === "LOCKED_EMAIL_REVERIFY_OVERDUE") return denyAndLog("EMAIL_REVERIFY_OVERDUE", "REVERIFY_EMAIL");
  }

  // 3) reduce-only safety (allowed in locked states if configured)
  if (ctx.action === "TRADE_CLOSE_OR_REDUCE") {
    // allow in all except admin-suspended by policy choice; your doc says "Optional (policy)".
    if (accountState === "SUSPENDED_ADMIN") return denyAndLog("ADMIN_SUSPENDED", "CONTACT_SUPPORT");
    if (!cfg.allowReduceOnlyWhenLocked) {
      if (accountState === "LOCKED_EMAIL_UNVERIFIED") return denyAndLog("EMAIL_UNVERIFIED_GRACE_EXPIRED", "VERIFY_EMAIL");
      if (accountState === "LOCKED_EMAIL_REVERIFY_OVERDUE") return denyAndLog("EMAIL_REVERIFY_OVERDUE", "REVERIFY_EMAIL");
    }
  }

  // 4) Email resend verification throttles
  if (ctx.action === "EMAIL_RESEND_VERIFICATION") {
    const ok = canSendEmail(ctx, cfg);
    if (!ok.ok) {
      return denyAndLog(ok.why === "cooldown" ? "EMAIL_SEND_COOLDOWN" : "EMAIL_SEND_RATE_LIMIT", "WAIT_COOLDOWN");
    }
    obligations.push({
      type: "EMAIL_VERIFICATION_EMAIL_SENT",
      data: { correlationId: ctx.request.correlationId, userId: ctx.user.id, email: ctx.user.email },
    });
    return { allow: true, derived, obligations };
  }

  // 5) Phone/SMS verify gates (only eligible contenders; block if locked)
  if (ctx.action === "PHONE_VERIFY_START" || ctx.action === "PHONE_VERIFY_CONFIRM") {
    if (accountState !== "ACTIVE_VERIFIED") {
      return denyAndLog(
        accountState === "LOCKED_EMAIL_UNVERIFIED" ? "EMAIL_UNVERIFIED_GRACE_EXPIRED" : "EMAIL_REVERIFY_OVERDUE",
        accountState === "LOCKED_EMAIL_UNVERIFIED" ? "VERIFY_EMAIL" : "REVERIFY_EMAIL"
      );
    }
    if (!smsEligible) return denyAndLog("SMS_NOT_ELIGIBLE", "BECOME_CONTENDER");
    if (isOtpLocked(ctx)) return denyAndLog("SMS_OTP_TOO_MANY_ATTEMPTS", "WAIT_COOLDOWN");

    if (ctx.action === "PHONE_VERIFY_START") {
      const s = canSendSms(ctx, cfg);
      if (!s.ok) return denyAndLog("SMS_RATE_LIMIT", "WAIT_COOLDOWN");
      obligations.push({ type: "SMS_OTP_SENT", data: { correlationId: ctx.request.correlationId, userId: ctx.user.id } });
      return { allow: true, derived, obligations };
    }

    // confirm allowed (rate-limits handled in OTP validator)
    return { allow: true, derived, obligations };
  }

  // 6) KYC gates (selected only)
  if (ctx.action === "KYC_VIEW" || ctx.action === "KYC_SUBMIT" || ctx.action === "PREFERRED_PAYMENT_CURRENCY_SET") {
    if (!selectedForKyc) return denyAndLog("KYC_NOT_SELECTED", "GET_SELECTED");
    if (ctx.action === "PREFERRED_PAYMENT_CURRENCY_SET" && ctx.user.kycStatus === "REJECTED") {
      return denyAndLog("KYC_STATE_INVALID", "CONTACT_SUPPORT");
    }
    return { allow: true, derived, obligations };
  }

  // 7) Sessions view can be Tier2+ (your earlier design); enforce as: must be ACTIVE_VERIFIED + smsEligible OR already phone verified
  if (ctx.action === "SESSIONS_VIEW") {
    const tier2 = ctx.user.phoneVerifiedAt || smsEligible; // eligible/verified users get sessions
    if (!tier2) return denyAndLog("ACTION_NOT_ALLOWED_IN_STATE");
    return { allow: true, derived, obligations };
  }

  // Default allow
  return { allow: true, derived, obligations };
}

  userId: number;
  correlationId: string;
  eventType: AuditEventType;

  actorType: "USER" | "ADMIN" | "SYSTEM";
  actorUserId: number;
  sessionId?: string;
  ip?: string;
  userAgent?: string;
};

export async function appendAudit(adapter: AuditAdapter, e: AuditWrite): Promise<void> {
  const prevHash = await adapter.getPrevHashForUser(e.userId);

  // hash-chain (tamper-evident) – safe even if you later disable it
  const payload = JSON.stringify({
    userId: e.userId,
    correlationId: e.correlationId,
    eventType: e.eventType,
    data: e.data,
    actorType: e.actorType,
    actorUserId: e.actorUserId,
    sessionId: e.sessionId ?? null,
    ip: e.ip ?? null,
    userAgent: e.userAgent ?? null,
    prevHash,
    ts: new Date().toISOString(),
  });

  const eventHash = crypto.createHash("sha256").update(payload).digest("hex");

  await adapter.insertEvent({
    userId: e.userId,
    correlationId: e.correlationId,
    eventType: e.eventType,

    actorType: e.actorType,
    actorUserId: e.actorUserId,
    sessionId: e.sessionId ?? null,

    userAgent: e.userAgent ?? null,
    prevHash,

    createdAt: new Date(),
  });

import type { Request, Response, NextFunction } from "express";
import type { PolicyAction, DecisionContext } from "../policy/types";
import { POLICY_CONFIG } from "../policy/config";
import { policyAllowed, PolicyError } from "../policy/enforce";

/**
 * IMPORTANT:
 * You must wire "loadUserPolicyContext" to YOUR DB.
 * It must return the full DecisionContext.user object including counters + aggregates.
 */
export type LoadUserPolicyContext = (args: {
  userId: number;
  now: Date;
}) => Promise<DecisionContext["user"]>;

export function requirePolicy(loadUser: LoadUserPolicyContext, action: PolicyAction) {
  return async (req: Request, res: Response, next: NextFunction) => {
    try {
      const now = new Date();
      const userId = (req as any).user?.id ?? (req as any).session?.userId;
      if (!userId) return res.status(401).json({ error: "auth.required" });

      const user = await loadUser({ userId, now });

      const ctx: DecisionContext = {
        now,
        action,
        request: {
          correlationId: req.correlationId || "missing-correlation-id",
          actorType: "USER",
          actorUserId: userId,
          sessionId: (req as any).sessionID,
          ip: req.ip,
          userAgent: req.header("user-agent") ?? undefined,
        },
        user,
        payload: {
          // optional: pass through details for richer audit
          route: req.originalUrl,
          method: req.method,
        },
      };

      const decision = policyAllowed(ctx, POLICY_CONFIG);

      // Attach for handlers/UI responses (optional)
      (req as any).policyDecision = decision;

      if (!decision.allow) {
        return res.status(decision.deny!.httpStatus).json({
          error: decision.deny!.messageKey,
          deny: decision.deny,
          derived: decision.derived,
          correlationId: req.correlationId,
        });
      }

      next();
    } catch (err: any) {
      if (err instanceof PolicyError) {
        return res.status(err.decision.deny?.httpStatus ?? 403).json({
          error: err.decision.deny?.messageKey ?? "policy.denied",
          deny: err.decision.deny,
          derived: err.decision.derived,
          correlationId: (req as any).correlationId,
        });
      }
      next(err);
    }
  };
}

This job runner calls the same policy logic. You can run it daily with cron (or a lightweight interval in Replit).

import { POLICY_CONFIG } from "../policy/config";
import { evaluatePolicy, resolveEmailInitialDueAt, resolveAccountState, isSmsEligibleByPerformance } from "../policy/engine";
import type { DecisionContext } from "../policy/types";

/**
 * Provide adapters that query users with relevant fields + counters + aggregates.
 */
export async function runDailyPolicyJobs(deps: {
  // list candidate users for reminders/locks (you can do it in SQL filters)
  listUsersForPolicyScan: () => Promise<Array<{ userId: number }>>;
  loadUserPolicyContext: (args: { userId: number; now: Date }) => Promise<DecisionContext["user"]>;
  sendEmailVerification: (args: { userId: number; email: string; correlationId: string; kind: "INITIAL" | "REVERIFY" }) => Promise<void>;
  bumpEmailCounters: (args: { userId: number; now: Date }) => Promise<void>;
  lockUser: (args: { userId: number; now: Date; reason: "EMAIL_UNVERIFIED" | "EMAIL_REVERIFY_OVERDUE" }) => Promise<void>;
  setContenderTier: (args: { userId: number; tier: "NONE" | "CANDIDATE_SMS_REQUIRED" }) => Promise<void>;
  appendAuditObligations: (args: { userId: number; correlationId: string; actorUserId: number; obligations: Array<{ type: any; data: any }> }) => Promise<void>;
}) {
  const now = new Date();
  const scan = await deps.listUsersForPolicyScan();

  for (const { userId } of scan) {
    // SYSTEM actor for jobs
    const correlationId = `job-${now.toISOString()}-${userId}`;

    const user = await deps.loadUserPolicyContext({ userId, now });

    // A) initial verify reminders + lock
    const initialDue = resolveEmailInitialDueAt({ now, action: "JOB_SEND_EMAIL_REMINDER", request: { correlationId, actorType: "SYSTEM", actorUserId: 0 }, user }, POLICY_CONFIG);
    const daysSinceSignup = Math.floor((now.getTime() - user.createdAt.getTime()) / 86400000);

    if (!user.emailVerifiedAt) {
      // Reminder schedule: Day 0,7,12,13
      if (POLICY_CONFIG.initialVerifyReminderDaysAfterSignup.includes(daysSinceSignup)) {
        const ctx: DecisionContext = {
          now,
          action: "EMAIL_RESEND_VERIFICATION",
          request: { correlationId, actorType: "SYSTEM", actorUserId: 0 },
          user,
        };
        const decision = evaluatePolicy(ctx, POLICY_CONFIG);
        if (decision.allow) {
          await deps.sendEmailVerification({ userId, email: user.email, correlationId, kind: "INITIAL" });
          await deps.bumpEmailCounters({ userId, now });
          await deps.appendAuditObligations({ userId, correlationId, actorUserId: 0, obligations: [
            { type: "REMINDER_SENT", data: { channel: "EMAIL", kind: "INITIAL_VERIFY", day: daysSinceSignup } }
          ]});
        }
      }

      // Lock on/after day 14 if still unverified
      if (now >= initialDue) {
        const st = resolveAccountState({ now, action: "JOB_ENFORCE_LOCKS", request: { correlationId, actorType: "SYSTEM", actorUserId: 0 }, user }, POLICY_CONFIG);
        if (st === "LOCKED_EMAIL_UNVERIFIED") {
          await deps.lockUser({ userId, now, reason: "EMAIL_UNVERIFIED" });
          await deps.appendAuditObligations({ userId, correlationId, actorUserId: 0, obligations: [
            { type: "ACCOUNT_LOCKED_EMAIL_UNVERIFIED", data: { initialDueAt: initialDue.toISOString() } }
          ]});
        }
      }
    }

    // B) monthly reverify reminders + lock
    if (user.emailVerifiedAt && user.emailReverifyDueAt) {
      const tDue = user.emailReverifyDueAt;
      const daysToDue = Math.floor((tDue.getTime() - now.getTime()) / 86400000);

      const offsets = POLICY_CONFIG.reverifyReminderOffsetsDays;
      // If today matches any offset bucket, attempt reminder
      if (offsets.includes(daysToDue)) {
        const ctx: DecisionContext = {
          now,
          action: "EMAIL_RESEND_VERIFICATION",
          request: { correlationId, actorType: "SYSTEM", actorUserId: 0 },
          user,
        };
        const decision = evaluatePolicy(ctx, POLICY_CONFIG);
        if (decision.allow) {
          await deps.sendEmailVerification({ userId, email: user.email, correlationId, kind: "REVERIFY" });
          await deps.bumpEmailCounters({ userId, now });
          await deps.appendAuditObligations({ userId, correlationId, actorUserId: 0, obligations: [
            { type: "EMAIL_REVERIFY_REMINDER_SENT", data: { dueAt: tDue.toISOString(), offsetDays: daysToDue } }
          ]});
        }
      }

      // Lock if overdue beyond grace
      const overdueAt = new Date(tDue.getTime() + POLICY_CONFIG.emailReverifyOverdueGraceDays * 86400000);
      if (now >= overdueAt) {
        const st = resolveAccountState({ now, action: "JOB_ENFORCE_LOCKS", request: { correlationId, actorType: "SYSTEM", actorUserId: 0 }, user }, POLICY_CONFIG);
        if (st === "LOCKED_EMAIL_REVERIFY_OVERDUE") {
          await deps.lockUser({ userId, now, reason: "EMAIL_REVERIFY_OVERDUE" });
          await deps.appendAuditObligations({ userId, correlationId, actorUserId: 0, obligations: [
            { type: "ACCOUNT_LOCKED_REVERIFY_OVERDUE", data: { dueAt: tDue.toISOString(), graceDays: POLICY_CONFIG.emailReverifyOverdueGraceDays } }
          ]});
        }
      }
    }

    // C) contender eligibility evaluation (sets tier to CANDIDATE_SMS_REQUIRED)
    // Only set if currently NONE and eligible; do not downgrade automatically (optional policy).
    if (user.contenderTier === "NONE") {
      const eligible = isSmsEligibleByPerformance({ now, action: "JOB_EVALUATE_CONTENDER_ELIGIBILITY", request: { correlationId, actorType: "SYSTEM", actorUserId: 0 }, user }, POLICY_CONFIG);
      if (eligible) {
        await deps.setContenderTier({ userId, tier: "CANDIDATE_SMS_REQUIRED" });
        await deps.appendAuditObligations({ userId, correlationId, actorUserId: 0, obligations: [
          { type: "CONTENDER_ELIGIBLE", data: { reason: "PERFORMANCE_THRESHOLD_MET" } }
        ]});
      }
    }
  }
}

  };

export function usePolicySnapshot() {
  const [snap, setSnap] = useState<PolicySnapshot | null>(null);
  const [loading, setLoading] = useState(true);

  async function fetchSnap() {
    setLoading(true);
    const res = await fetch("/api/policy/snapshot", { headers: { "x-correlation-id": crypto.randomUUID() } });
    if (!res.ok) {
      setLoading(false);
      setSnap(null);
      return;
    }
    const data = await res.json();
    setSnap(data);
    setLoading(false);
  }

  useEffect(() => { fetchSnap(); }, []);

  return { snap, loading, refresh: fetchSnap };
}

import type { AccountState, DecisionContext, PolicyConfig, PolicyDecision, PolicyAction } from "./types";

const MS_DAY = 86400000;

function addDays(d: Date, days: number): Date {
  return new Date(d.getTime() + days * MS_DAY);
}
function dayKey(d: Date): string {
  const yyyy = d.getFullYear();
  const mm = String(d.getMonth() + 1).padStart(2, "0");
  const dd = String(d.getDate()).padStart(2, "0");
  return `${yyyy}-${mm}-${dd}`;
}
function diffDays(now: Date, past: Date): number {
  return (now.getTime() - past.getTime()) / MS_DAY;
}
function isExposureIncreasing(action: PolicyAction): boolean {
  return (
    action === "TRADE_OPEN_OR_INCREASE" ||
    action === "TRADE_MODIFY_SLTP" ||
    action === "TRADE_PLACE_PENDING" ||
    action === "TRADE_CANCEL_PENDING"
  );
}

export function resolveEmailInitialDueAt(ctx: DecisionContext, cfg: PolicyConfig): Date {
  return ctx.user.emailInitialDueAt ?? addDays(ctx.user.createdAt, cfg.emailInitialGraceDays);
}

export function resolveAccountState(ctx: DecisionContext, cfg: PolicyConfig): AccountState {
  if (ctx.user.suspendedAt) return "SUSPENDED_ADMIN";

  const emailInitialDueAt = resolveEmailInitialDueAt(ctx, cfg);

  if (!ctx.user.emailVerifiedAt) {
    return ctx.now < emailInitialDueAt ? "ACTIVE_UNVERIFIED_GRACE" : "LOCKED_EMAIL_UNVERIFIED";
  }

  if (ctx.user.emailReverifyDueAt) {
    const overdueAt = addDays(ctx.user.emailReverifyDueAt, cfg.emailReverifyOverdueGraceDays);
    if (ctx.now >= overdueAt) return "LOCKED_EMAIL_REVERIFY_OVERDUE";
  }

  return "ACTIVE_VERIFIED";
}

export function isSelectedForKyc(ctx: DecisionContext): boolean {
  return !!ctx.user.selectedAt || ctx.user.contenderTier === "SELECTED_REAL_CAPITAL";
}

export function isSmsEligibleByPerformance(ctx: DecisionContext, cfg: PolicyConfig): boolean {
  const st = resolveAccountState(ctx, cfg);
  if (st !== "ACTIVE_VERIFIED") return false;

  const ageDays = diffDays(ctx.now, ctx.user.createdAt);
  const a = ctx.user.tradingAgg;

  const balancePct = a.startingBalance > 0 ? a.balanceNow / a.startingBalance : 0;

  const path1 =
    ageDays >= cfg.contenderPath1.minAccountAgeDays &&
    balancePct >= cfg.contenderPath1.minBalancePctOfStart &&
    a.positionsCountLifetime >= cfg.contenderPath1.minLifetimeTrades;

  const recentTradeOk =
    !!a.lastTradeAt && diffDays(ctx.now, a.lastTradeAt) <= cfg.contenderPath2.requireRecentTradeDays;

  const path2 =
    ageDays >= cfg.contenderPath2.minAccountAgeDays &&
    recentTradeOk &&
    a.returnLast90 >= cfg.contenderPath2.minReturnLast90 &&
    a.positionsCountLast90 >= cfg.contenderPath2.minTradesLast90;

  return path1 || path2;
}

function canSendEmail(ctx: DecisionContext, cfg: PolicyConfig): { ok: true } | { ok: false; why: "cooldown" | "cap" } {
  const now = ctx.now;
  const k = dayKey(now);
  const c = ctx.user.counters.email;

  if (c.lastSentAt) {
    const sec = (now.getTime() - c.lastSentAt.getTime()) / 1000;
    if (sec < cfg.reminderResendCooldownSec) return { ok: false, why: "cooldown" };
  }

  const countToday = c.sentDayKey === k ? c.sentCountDay : 0;
  if (countToday >= cfg.reminderMaxPerDay) return { ok: false, why: "cap" };
  return { ok: true };
}

function canSendSms(ctx: DecisionContext, cfg: PolicyConfig): { ok: true } | { ok: false; why: "cap" } {
  const now = ctx.now;
  const k = dayKey(now);
  const c = ctx.user.counters.sms;
  const countToday = c.sentDayKey === k ? c.sentCountDay : 0;
  if (countToday >= cfg.smsDailySendCap) return { ok: false, why: "cap" };
  return { ok: true };
}

function isOtpLocked(ctx: DecisionContext): boolean {
  const o = ctx.user.counters.otp;
  return !!o.lockedUntil && ctx.now < o.lockedUntil;
}

function deny(code: PolicyDecision["deny"]["code"], ctx: DecisionContext, derived: PolicyDecision["derived"], nextStep?: PolicyDecision["deny"]["nextStep"], httpStatus: 403 | 429 = 403): PolicyDecision {
  return {
    allow: false,
    derived,
    deny: { code, messageKey: `deny.${code}`, httpStatus, nextStep },
    obligations: [{
      type: "ACCOUNT_ACTION_DENIED",
      data: {
        code,
        action: ctx.action,
        correlationId: ctx.request.correlationId,
        accountState: derived.accountState,
        contenderTier: derived.contenderTier,
        emailInitialDueAt: derived.emailInitialDueAt.toISOString(),
        emailReverifyDueAt: derived.emailReverifyDueAt?.toISOString() ?? null,
      },
    }],
  };
}

export function evaluatePolicy(ctx: DecisionContext, cfg: PolicyConfig): PolicyDecision {
  const obligations: PolicyDecision["obligations"] = [];

  const emailInitialDueAt = resolveEmailInitialDueAt(ctx, cfg);
  const accountState = resolveAccountState(ctx, cfg);

  const smsEligible = isSmsEligibleByPerformance(ctx, cfg);
  const selected = isSelectedForKyc(ctx);

  const derived: PolicyDecision["derived"] = {
    accountState,
    contenderTier: ctx.user.contenderTier,
    emailInitialDueAt,
    emailReverifyDueAt: ctx.user.emailReverifyDueAt,
    isEmailInitialOverdue: !ctx.user.emailVerifiedAt && ctx.now >= emailInitialDueAt,
    isEmailReverifyOverdue: !!ctx.user.emailReverifyDueAt && ctx.now >= addDays(ctx.user.emailReverifyDueAt, cfg.emailReverifyOverdueGraceDays),
    isSmsEligibleByPerformance: smsEligible,
    isSelectedForKyc: selected,
  };

  // Admin suspension
  if (accountState === "SUSPENDED_ADMIN") {
    // keep strict
    return deny("ADMIN_SUSPENDED", ctx, derived, "CONTACT_SUPPORT", 403);
  }

  // Locked states block exposure-increasing actions
  if (isExposureIncreasing(ctx.action)) {
    if (accountState === "LOCKED_EMAIL_UNVERIFIED") return deny("EMAIL_UNVERIFIED_GRACE_EXPIRED", ctx, derived, "VERIFY_EMAIL", 403);
    if (accountState === "LOCKED_EMAIL_REVERIFY_OVERDUE") return deny("EMAIL_REVERIFY_OVERDUE", ctx, derived, "REVERIFY_EMAIL", 403);
  }

  // Reduce-only close allowed when locked (policy choice)
  if (ctx.action === "TRADE_CLOSE_OR_REDUCE") {
    if (!cfg.allowReduceOnlyWhenLocked) {
      if (accountState === "LOCKED_EMAIL_UNVERIFIED") return deny("EMAIL_UNVERIFIED_GRACE_EXPIRED", ctx, derived, "VERIFY_EMAIL", 403);
      if (accountState === "LOCKED_EMAIL_REVERIFY_OVERDUE") return deny("EMAIL_REVERIFY_OVERDUE", ctx, derived, "REVERIFY_EMAIL", 403);
    }
  }

  // Email resend throttles
  if (ctx.action === "EMAIL_RESEND_VERIFICATION") {
    const ok = canSendEmail(ctx, cfg);
    if (!ok.ok) {
      return deny(ok.why === "cooldown" ? "EMAIL_SEND_COOLDOWN" : "EMAIL_SEND_RATE_LIMIT", ctx, derived, "WAIT_COOLDOWN", 429);
    }
    obligations.push({
      type: "EMAIL_VERIFICATION_EMAIL_SENT",
      data: { correlationId: ctx.request.correlationId, userId: ctx.user.id, email: ctx.user.email },
    });
    return { allow: true, derived, obligations };
  }

  // SMS verify gates
  if (ctx.action === "PHONE_VERIFY_START" || ctx.action === "PHONE_VERIFY_CONFIRM") {
    if (accountState !== "ACTIVE_VERIFIED") {
      return deny(accountState === "LOCKED_EMAIL_UNVERIFIED" ? "EMAIL_UNVERIFIED_GRACE_EXPIRED" : "EMAIL_REVERIFY_OVERDUE", ctx, derived,
        accountState === "LOCKED_EMAIL_UNVERIFIED" ? "VERIFY_EMAIL" : "REVERIFY_EMAIL", 403);
    }
    if (!smsEligible) return deny("SMS_NOT_ELIGIBLE", ctx, derived, "BECOME_CONTENDER", 403);
    if (isOtpLocked(ctx)) return deny("SMS_OTP_TOO_MANY_ATTEMPTS", ctx, derived, "WAIT_COOLDOWN", 429);

    if (ctx.action === "PHONE_VERIFY_START") {
      const s = canSendSms(ctx, cfg);
      if (!s.ok) return deny("SMS_RATE_LIMIT", ctx, derived, "WAIT_COOLDOWN", 429);
      obligations.push({ type: "SMS_OTP_SENT", data: { correlationId: ctx.request.correlationId, userId: ctx.user.id } });
      return { allow: true, derived, obligations };
    }

    return { allow: true, derived, obligations };
  }

  // KYC gates (selected only)
  if (ctx.action === "KYC_VIEW" || ctx.action === "KYC_SUBMIT" || ctx.action === "PREFERRED_PAYMENT_CURRENCY_SET") {
    if (!selected) return deny("KYC_NOT_SELECTED", ctx, derived, "GET_SELECTED", 403);
    if (ctx.action === "PREFERRED_PAYMENT_CURRENCY_SET" && ctx.user.kycStatus === "REJECTED") {
      return deny("KYC_STATE_INVALID", ctx, derived, "CONTACT_SUPPORT", 409);
    }
    return { allow: true, derived, obligations };
  }

  // Sessions view (Tier-2+ rule: phone verified OR performance eligible)
  if (ctx.action === "SESSIONS_VIEW") {
    const tier2 = !!ctx.user.phoneVerifiedAt || smsEligible;
    if (!tier2) return deny("ACTION_NOT_ALLOWED_IN_STATE", ctx, derived, undefined, 403);
    return { allow: true, derived, obligations };
  }

  return { allow: true, derived, obligations };
}

  userId: number;
  correlationId: string;
  actorType: "USER" | "ADMIN" | "SYSTEM";
  actorUserId: number;
  sessionId?: string;
  ip?: string;

export function requirePolicy(loadUser: LoadUserPolicyContext, append: AppendObligations, action: PolicyAction) {
  return async (req: Request, res: Response, next: NextFunction) => {
    const now = new Date();
    const userId = (req as any).user?.id ?? (req as any).session?.userId;
    if (!userId) return res.status(401).json({ error: "auth.required" });

    const user = await loadUser({ userId, now });

    const ctx: DecisionContext = {
      now,
      action,
      request: {
        correlationId: req.correlationId || "missing-correlation-id",
        actorType: "USER",
        actorUserId: userId,
        sessionId: (req as any).sessionID,
        ip: req.ip,
        userAgent: req.header("user-agent") ?? undefined,

      user,
      payload: { route: req.originalUrl, method: req.method },
    };

    const decision = evaluatePolicy(ctx, POLICY_CONFIG);
    (req as any).policyDecision = decision;

    // Always persist obligations (denies included)
    if (decision.obligations?.length) {
      await append({
        userId,
        correlationId: ctx.request.correlationId,
        actorType: ctx.request.actorType,
        actorUserId: ctx.request.actorUserId,
        sessionId: ctx.request.sessionId,
        ip: ctx.request.ip,
        userAgent: ctx.request.userAgent,
        obligations: decision.obligations,
      });
    }

    if (!decision.allow) {
      return res.status(decision.deny!.httpStatus).json({
        error: decision.deny!.messageKey,
        deny: decision.deny,
        derived: decision.derived,
        correlationId: ctx.request.correlationId,
      });
    }

    next();

/**
 * Computes last-90 return from daily equity snapshots.
 * If missing history, returns 0 (conservative).
 */
async function computeReturnLast90(userId: number, now: Date, equityNow: number): Promise<number> {
  const ninetyAgo = new Date(now.getTime() - 90 * MS_DAY);
  const ninetyKey = dayKey(ninetyAgo);

  // closest snapshot on or before ninetyKey
  const snap = await db
    .select({ equity: userEquityDaily.equity })
    .from(userEquityDaily)
    .where(and(eq(userEquityDaily.userId, userId), sql`${userEquityDaily.dayKey} <= ${ninetyKey}`))
    .orderBy(desc(userEquityDaily.dayKey))
    .limit(1);

  if (!snap.length) return 0;
  const base = Number(snap[0].equity);
  if (!base || base <= 0) return 0;
  return (equityNow - base) / base;
}

export async function verifyEmailByToken(token: string): Promise<{ userId: number; kind: "INITIAL" | "REVERIFY" } | null> {
  const tokenHash = sha256Hex(token);
  const now = nowMs();

  const rows = await db
    .select({
      id: emailVerificationTokens.id,
      userId: emailVerificationTokens.userId,
      kind: emailVerificationTokens.kind,
      expiresAtMs: emailVerificationTokens.expiresAtMs,
      usedAtMs: emailVerificationTokens.usedAtMs,
    })
    .from(emailVerificationTokens)
    .where(eq(emailVerificationTokens.tokenHash, tokenHash))
    .limit(1);

  if (!rows.length) return null;
  const r = rows[0] as any;

  if (r.usedAtMs) return null;
  if (Number(r.expiresAtMs) < now) return null;

  // mark used
  await db
    .update(emailVerificationTokens)
    .set({ usedAtMs: now } as any)
    .where(eq(emailVerificationTokens.id, r.id));

  return { userId: Number(r.userId), kind: r.kind };
}

export async function consumeSmsOtp(userId: number, phoneNumber: string, otp: string): Promise<{ ok: boolean; reason?: string }> {
  const otpHash = sha256Hex(otp);
  const now = nowMs();

  const row = await db
    .select({
      id: smsOtpTokens.id,
      expiresAtMs: smsOtpTokens.expiresAtMs,
      consumedAtMs: smsOtpTokens.consumedAtMs,
      otpHash: smsOtpTokens.otpHash,
    })
    .from(smsOtpTokens)
    .where(and(eq(smsOtpTokens.userId, userId), eq(smsOtpTokens.phoneNumber, phoneNumber)))
    .orderBy(desc(smsOtpTokens.id))
    .limit(1);

  if (!row.length) return { ok: false, reason: "otp.not_found" };
  const r = row[0] as any;
  if (r.consumedAtMs) return { ok: false, reason: "otp.consumed" };
  if (Number(r.expiresAtMs) < now) return { ok: false, reason: "otp.expired" };
  if (String(r.otpHash) !== otpHash) return { ok: false, reason: "otp.invalid" };

  await db.update(smsOtpTokens).set({ consumedAtMs: now } as any).where(eq(smsOtpTokens.id, r.id));
  await db.update(userProfile).set({ phoneNumber, updatedAtMs: now } as any).where(eq(userProfile.userId, userId));
  await db.update(userPolicyState).set({ phoneVerifiedAtMs: now, otpFailCount: 0, otpLockedUntilMs: null, updatedAtMs: now } as any).where(eq(userPolicyState.userId, userId));

  return { ok: true };
}

export async function registerOtpFailure(userId: number, now: Date) {
  const row = await db.select().from(userPolicyState).where(eq(userPolicyState.userId, userId)).limit(1);
  const p = row[0] as any;
  const fail = Number(p.otpFailCount ?? 0) + 1;

  let lockedUntilMs: number | null = p.otpLockedUntilMs ?? null;
  if (fail >= POLICY_CONFIG.otpMaxAttempts) {
    lockedUntilMs = now.getTime() + POLICY_CONFIG.otpLockMinutes * 60 * 1000;
  }

  await db.update(userPolicyState).set({
    otpFailCount: fail,
    otpLockedUntilMs: lockedUntilMs,
    updatedAtMs: now.getTime(),
  } as any).where(eq(userPolicyState.userId, userId));
}

export async function appendUserAudit(args: {
  userId: number;
  correlationId: string;
  eventType: string;
  actorType: "USER" | "ADMIN" | "SYSTEM";
  actorUserId: number;
  sessionId?: string;
  ip?: string;
  userAgent?: string;
  data: any;
}) {

    .select({ eventHash: userAudit.eventHash })
    .from(userAudit)
    .where(eq(userAudit.userId, args.userId))
    .orderBy(desc(userAudit.id))
    .limit(1);

  const prevHash = prev.length ? (prev[0] as any).eventHash : null;

  const payload = JSON.stringify({
    userId: args.userId,
    correlationId: args.correlationId,
    eventType: args.eventType,
    actorType: args.actorType,
    actorUserId: args.actorUserId,
    sessionId: args.sessionId ?? null,
    ip: args.ip ?? null,
    userAgent: args.userAgent ?? null,
    data: args.data,
    prevHash,
    ts: new Date().toISOString(),
  });

  const eventHash = sha256Hex(payload);

  await db.insert(userAudit).values({
    userId: args.userId,
    correlationId: args.correlationId,
    eventType: args.eventType,
    actorType: args.actorType,
    actorUserId: args.actorUserId,
    sessionId: args.sessionId ?? null,
    ip: args.ip ?? null,
    userAgent: args.userAgent ?? null,
    data: JSON.stringify(args.data ?? {}),
    prevHash,
    eventHash,
    createdAtMs: nowMs(),
  } as any);
}

import { POLICY_CONFIG } from "../policy/config";
import { evaluatePolicy, resolveEmailInitialDueAt, resolveAccountState, isSmsEligibleByPerformance } from "../policy/engine";
import type { DecisionContext } from "../policy/types";

const MS_DAY = 86400000;

export async function runDailyPolicyJobs(deps: {
  listActiveUserIds: () => Promise<number[]>;
  loadUserPolicyContext: (args: { userId: number; now: Date }) => Promise<DecisionContext["user"]>;

  // actions
  ensureEmailInitialDueAt: (userId: number, createdAt: Date) => Promise<void>;
  sendEmailVerification: (args: { userId: number; email: string; kind: "INITIAL" | "REVERIFY"; correlationId: string }) => Promise<void>;
  bumpEmailCounters: (userId: number, now: Date) => Promise<void>;
  lockUser: (userId: number, now: Date, reason: "EMAIL_UNVERIFIED" | "EMAIL_REVERIFY_OVERDUE") => Promise<void>;
  setContenderTier: (userId: number, tier: "CANDIDATE_SMS_REQUIRED") => Promise<void>;
  upsertDailyEquity: (userId: number, now: Date, equity: number) => Promise<void>;

  appendPolicyObligations: (args: { userId: number; correlationId: string; actorType: "SYSTEM"; actorUserId: number; obligations: any[] }) => Promise<void>;
}) {
  const now = new Date();
  const ids = await deps.listActiveUserIds();

  for (const userId of ids) {
    const correlationId = `job-${now.toISOString()}-${userId}`;
    const user = await deps.loadUserPolicyContext({ userId, now });

    // Ensure initial due date stored (optional but useful for reporting consistency)
    await deps.ensureEmailInitialDueAt(userId, user.createdAt);

    // Upsert daily equity snapshot (powers returnLast90)
    await deps.upsertDailyEquity(userId, now, user.tradingAgg.balanceNow);

    const daysSinceSignup = Math.floor((now.getTime() - user.createdAt.getTime()) / MS_DAY);

    // A) initial verify reminders & lock
    if (!user.emailVerifiedAt) {
      if (POLICY_CONFIG.initialVerifyReminderDaysAfterSignup.includes(daysSinceSignup)) {
        const ctx: DecisionContext = {
          now,
          action: "EMAIL_RESEND_VERIFICATION",
          request: { correlationId, actorType: "SYSTEM", actorUserId: 0 },
          user,
        };
        const dec = evaluatePolicy(ctx, POLICY_CONFIG);
        if (dec.allow) {
          await deps.sendEmailVerification({ userId, email: user.email, kind: "INITIAL", correlationId });
          await deps.bumpEmailCounters(userId, now);
          await deps.appendPolicyObligations({
            userId,
            correlationId,
            actorType: "SYSTEM",
            actorUserId: 0,
            obligations: [{ type: "REMINDER_SENT", data: { channel: "EMAIL", kind: "INITIAL_VERIFY", day: daysSinceSignup } }],
          });
        }
      }

      const due = resolveEmailInitialDueAt({ now, action: "JOB_ENFORCE_LOCKS", request: { correlationId, actorType: "SYSTEM", actorUserId: 0 }, user }, POLICY_CONFIG);
      if (now >= due) {
        const st = resolveAccountState({ now, action: "JOB_ENFORCE_LOCKS", request: { correlationId, actorType: "SYSTEM", actorUserId: 0 }, user }, POLICY_CONFIG);
        if (st === "LOCKED_EMAIL_UNVERIFIED") {
          await deps.lockUser(userId, now, "EMAIL_UNVERIFIED");
          await deps.appendPolicyObligations({
            userId,
            correlationId,
            actorType: "SYSTEM",
            actorUserId: 0,
            obligations: [{ type: "ACCOUNT_LOCKED_EMAIL_UNVERIFIED", data: { dueAt: due.toISOString() } }],
          });
        }
      }
    }

    // B) reverify reminders & lock
    if (user.emailVerifiedAt && user.emailReverifyDueAt) {
      const tDue = user.emailReverifyDueAt;
      const daysToDue = Math.floor((tDue.getTime() - now.getTime()) / MS_DAY);

      if (POLICY_CONFIG.reverifyReminderOffsetsDays.includes(daysToDue)) {
        const ctx: DecisionContext = {
          now,
          action: "EMAIL_RESEND_VERIFICATION",
          request: { correlationId, actorType: "SYSTEM", actorUserId: 0 },
          user,
        };
        const dec = evaluatePolicy(ctx, POLICY_CONFIG);
        if (dec.allow) {
          await deps.sendEmailVerification({ userId, email: user.email, kind: "REVERIFY", correlationId });
          await deps.bumpEmailCounters(userId, now);
          await deps.appendPolicyObligations({
            userId,
            correlationId,
            actorType: "SYSTEM",
            actorUserId: 0,
            obligations: [{ type: "EMAIL_REVERIFY_REMINDER_SENT", data: { dueAt: tDue.toISOString(), offsetDays: daysToDue } }],
          });
        }
      }

      const overdueAt = new Date(tDue.getTime() + POLICY_CONFIG.emailReverifyOverdueGraceDays * MS_DAY);
      if (now >= overdueAt) {
        const st = resolveAccountState({ now, action: "JOB_ENFORCE_LOCKS", request: { correlationId, actorType: "SYSTEM", actorUserId: 0 }, user }, POLICY_CONFIG);
        if (st === "LOCKED_EMAIL_REVERIFY_OVERDUE") {
          await deps.lockUser(userId, now, "EMAIL_REVERIFY_OVERDUE");
          await deps.appendPolicyObligations({
            userId,
            correlationId,
            actorType: "SYSTEM",
            actorUserId: 0,
            obligations: [{ type: "ACCOUNT_LOCKED_REVERIFY_OVERDUE", data: { dueAt: tDue.toISOString() } }],
          });
        }
      }
    }

    // C) contender eligibility evaluation (sets CANDIDATE_SMS_REQUIRED)
    if (user.contenderTier === "NONE") {
      const eligible = isSmsEligibleByPerformance({ now, action: "JOB_EVALUATE_CONTENDER_ELIGIBILITY", request: { correlationId, actorType: "SYSTEM", actorUserId: 0 }, user }, POLICY_CONFIG);
      if (eligible) {
        await deps.setContenderTier(userId, "CANDIDATE_SMS_REQUIRED");
        await deps.appendPolicyObligations({
          userId,
          correlationId,
          actorType: "SYSTEM",
          actorUserId: 0,
          obligations: [{ type: "CONTENDER_ELIGIBLE", data: { reason: "PERFORMANCE_THRESHOLD_MET" } }],
        });
      }
    }
  }
}

import { useEffect, useState } from "react";

export type PolicySnapshot = {
  derived: {
    accountState: string;
    contenderTier: string;
    emailInitialDueAt: string;
    emailReverifyDueAt: string | null;
    isEmailInitialOverdue: boolean;
    isEmailReverifyOverdue: boolean;
    isSmsEligibleByPerformance: boolean;
    isSelectedForKyc: boolean;
  };
  features: {
    canSendEmailVerification: boolean;
    canStartPhoneVerify: boolean;
    canViewSessions: boolean;
    canViewKyc: boolean;
  };
  correlationId?: string;
};

export function usePolicySnapshot() {
  const [snap, setSnap] = useState<PolicySnapshot | null>(null);
  const [loading, setLoading] = useState(true);

  async function fetchSnap() {
    setLoading(true);
    const res = await fetch("/api/policy/snapshot", {
      headers: { "x-correlation-id": window.crypto.randomUUID() },
    });
    if (!res.ok) {
      setSnap(null);
      setLoading(false);
      return;
    }
    setSnap(await res.json());
    setLoading(false);
  }

  useEffect(() => { fetchSnap(); }, []);

  return { snap, loading, refresh: fetchSnap };
}
