```ts
// server/services/crypto.ts
import crypto from "crypto";

export function sha256Hex(input: string): string {
  return crypto.createHash("sha256").update(input).digest("hex");
}

export function randomToken(bytes = 32): string {
  return crypto.randomBytes(bytes).toString("hex");
}

// Simple AES-256-GCM encryption for MFA secrets (no external libs)
const key = (() => {
  const raw = process.env.MFA_ENCRYPTION_KEY ?? "";
  if (raw.length < 32) return null;
  return crypto.createHash("sha256").update(raw).digest(); // 32 bytes
})();

export function encryptString(plaintext: string): string {
  if (!key) throw new Error("MFA_ENCRYPTION_KEY is missing/too short");
  const iv = crypto.randomBytes(12);
  const cipher = crypto.createCipheriv("aes-256-gcm", key, iv);
  const enc = Buffer.concat([cipher.update(plaintext, "utf8"), cipher.final()]);
  const tag = cipher.getAuthTag();
  return Buffer.concat([iv, tag, enc]).toString("base64");
}

export function decryptString(ciphertextB64: string): string {
  if (!key) throw new Error("MFA_ENCRYPTION_KEY is missing/too short");
  const buf = Buffer.from(ciphertextB64, "base64");
  const iv = buf.subarray(0, 12);
  const tag = buf.subarray(12, 28);
  const enc = buf.subarray(28);
  const decipher = crypto.createDecipheriv("aes-256-gcm", key, iv);
  decipher.setAuthTag(tag);
  const dec = Buffer.concat([decipher.update(enc), decipher.final()]);
  return dec.toString("utf8");
}
```

```ts
// server/services/tiering.ts
import { db } from "../db";
import { userVerification, userKycProfiles } from "../../shared/schema";
import { and, eq, gte, sql } from "drizzle-orm";

/**
 * NOTE:
 * - This function is "best-effort" because your current schema for balances may differ.
 * - It uses trade counts and basic time windows, plus a pluggable "equityReturn90d" callback.
 * - Replit Agent 3 should wire `getReturnLast90d` to your actual equity/balance time-series if present.
 */

export async function recomputeContenderTier(userId: number, opts?: {
  // provide if you have a reliable 90d return calculator
  getReturnLast90d?: (userId: number) => Promise<number>; // e.g., 0.10 for +10%
  getBalancePctOfStart?: (userId: number) => Promise<number>; // e.g., 1.20
}) {
  const now = Date.now();
  const dayKey = new Date(now).toISOString().slice(0, 10);

  const uv = await db.select().from(userVerification).where(eq(userVerification.userId, userId)).limit(1);
  if (!uv[0]) {
    await db.insert(userVerification).values({
      userId,
      contenderTier: "NONE",
      emailResendDayKey: dayKey,
      smsSendDayKey: dayKey,
    });
  }

  // --- trade activity metrics (assumes `trades` table exists with userId + createdAt/closedAt)
  // Replace `trades`/columns to match your repo.
  const tradesLifetime = await db.get<{ n: number }>(sql`
    SELECT COUNT(1) as n FROM trades WHERE user_id = ${userId}
  `);

  const tradesLast90 = await db.get<{ n: number }>(sql`
    SELECT COUNT(1) as n FROM trades
    WHERE user_id = ${userId}
      AND COALESCE(closed_at, created_at) >= ${now - 90 * 24 * 3600 * 1000}
  `);

  const firstTrade = await db.get<{ t: number | null }>(sql`
    SELECT MIN(created_at) as t FROM trades WHERE user_id = ${userId}
  `);

  const lastTrade = await db.get<{ t: number | null }>(sql`
    SELECT MAX(COALESCE(closed_at, created_at)) as t FROM trades WHERE user_id = ${userId}
  `);

  const accountAgeDays = firstTrade?.t ? Math.floor((now - firstTrade.t) / (24 * 3600 * 1000)) : 0;
  const lastTradeDaysAgo = lastTrade?.t ? Math.floor((now - lastTrade.t) / (24 * 3600 * 1000)) : 9999;

  const balancePctOfStart = opts?.getBalancePctOfStart ? await opts.getBalancePctOfStart(userId) : 1.0;
  const returnLast90 = opts?.getReturnLast90d ? await opts.getReturnLast90d(userId) : 0.0;

  // Eligibility Path 1 (>=30d, >120%, >=30 trades)
  const path1 =
    accountAgeDays >= 30 &&
    balancePctOfStart >= 1.2 &&
    (tradesLifetime?.n ?? 0) >= 30;

  // Eligibility Path 2 (>=90d, active recently, >=10% return, >=20 trades in 90d)
  const path2 =
    accountAgeDays >= 90 &&
    lastTradeDaysAgo <= 14 &&
    returnLast90 >= 0.10 &&
    (tradesLast90?.n ?? 0) >= 20;

  const contenderTier = (path1 || path2) ? "CANDIDATE_SMS_REQUIRED" : "CANDIDATE_EMAIL_ONLY";

  await db
    .update(userVerification)
    .set({
      contenderTier,
      contenderEligibleAt: (path1 || path2) ? now : null,
      updatedAt: now,
    })
    .where(eq(userVerification.userId, userId));

  return { contenderTier, path1, path2, accountAgeDays, tradesLifetime: tradesLifetime?.n ?? 0, tradesLast90: tradesLast90?.n ?? 0 };
}
```

```ts
    .where(eq(userVerification.userId, user.id));

  await appendIdentityAudit({
    userId: user.id,
    email: user.email,
    username: user.username,
    category: "EMAIL",
    type: "EMAIL_VERIFIED",
    title: "Email verified",

function generateRecoveryCodes(n = 10) {
  const codes: string[] = [];

profileSettingsRouter.post("/api/profile/mfa/enable", requireAuth, async (req, res) => {
  const user = req.user as any;
  const { code } = req.body ?? {};
  const c = String(code ?? "").trim();

  const row = await db.select().from(userMfa).where(eq(userMfa.userId, user.id)).limit(1);
  const pending = row[0]?.totpPendingSecretEnc;
  if (!pending) return res.status(400).json({ ok: false, message: "No pending MFA setup" });

  const ok = speakeasy.totp.verify({ secret: secretBase32, encoding: "base32", token: c, window: 1 });

    totpSecretEnc: pending,
    totpPendingSecretEnc: null,
    enabledAt: Date.now(),
    disabledAt: null,
    recoveryCodesHashJson: JSON.stringify(recoveryHashes),
    updatedAt: Date.now(),
```

```tsx
// client/src/pages/AdminTierManagement.tsx
import { useMemo, useState } from "react";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
import { apiRequest } from "@/lib/queryClient";

type Row = {
  userId: number;
  email: string;
  username: string;
  userTier: "CANDIDATE" | "PERFORMER" | "SELECTED";
  contenderTier?: string | null;
  smsVerifiedAt?: number | null;
};

export default function AdminTierManagement() {
  const qc = useQueryClient();
  const [q, setQ] = useState("");

  const { data, isLoading } = useQuery({
    queryKey: ["admin-tier-users"],
    queryFn: async () => {
      const res = await apiRequest("GET", "/api/admin/user-profiles");
      // user-profiles should include tier fields; if not, extend it
      return (await res.json()) as Row[];
    },
  });

  const rows = useMemo(() => {
    const all = (data ?? []) as any[];
    if (!q.trim()) return all;
    const s = q.toLowerCase();
    return all.filter((r) => (r.email ?? "").toLowerCase().includes(s) || (r.username ?? "").toLowerCase().includes(s) || String(r.userId).includes(s));
  }, [data, q]);

  const setTier = useMutation({
    mutationFn: async (payload: { userId: number; userTier: Row["userTier"] }) =>
      (await apiRequest("POST", `/api/admin/users/${payload.userId}/tier`, { userTier: payload.userTier })).json(),
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["admin-tier-users"] });
      qc.invalidateQueries({ queryKey: ["admin-audit-trail"] });
    },
  });

  const recompute = useMutation({
    mutationFn: async (userId: number) => (await apiRequest("POST", `/api/admin/users/${userId}/tier/recompute`, {})).json(),
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["admin-tier-users"] });
      qc.invalidateQueries({ queryKey: ["admin-audit-trail"] });
    },
  });

  return (
    <div className="space-y-4">
      <Card className="bg-neutral-850 border-gray-800">
        <CardHeader>
          <CardTitle>Tier Progression Management</CardTitle>
        </CardHeader>
        <CardContent className="flex flex-col sm:flex-row gap-2 sm:items-center">
          <Input placeholder="Search email/username/userId..." value={q} onChange={(e) => setQ(e.target.value)} className="bg-neutral-900 border-gray-700" />
          <div className="text-xs text-gray-400">
            Admin can set tiers; recompute runs performance eligibility checks.
          </div>
        </CardContent>
      </Card>

      <div className="overflow-x-auto">
        {isLoading ? (
          <div className="p-6 text-gray-400">Loading...</div>
        ) : (
          <Table>
            <TableHeader>
              <TableRow className="border-b border-gray-700">
                <TableHead className="text-gray-400">User</TableHead>
                <TableHead className="text-gray-400">Tier</TableHead>
                <TableHead className="text-gray-400">Contender Tier</TableHead>
                <TableHead className="text-gray-400">Actions</TableHead>
              </TableRow>
            </TableHeader>
            <TableBody>
              {rows.map((r: any) => (
                <TableRow key={r.userId} className="border-b border-gray-800">
                  <TableCell>
                    <div className="font-medium text-gray-200">{r.email}</div>
                    <div className="text-xs text-gray-400">@{r.username} (#{r.userId})</div>
                  </TableCell>
                  <TableCell className="text-xs text-gray-200">{r.userTier ?? "CANDIDATE"}</TableCell>
                  <TableCell className="text-xs text-gray-400">{r.contenderTier ?? "—"}</TableCell>
                  <TableCell className="flex flex-wrap gap-2 py-3">
                    <Button size="sm" variant="outline" className="bg-neutral-900 border-gray-700" onClick={() => recompute.mutate(r.userId)} disabled={recompute.isPending}>
                      Recompute
                    </Button>
                    <Button size="sm" className="bg-slate-700 hover:bg-slate-800" onClick={() => setTier.mutate({ userId: r.userId, userTier: "CANDIDATE" })} disabled={setTier.isPending}>
                      Candidate
                    </Button>
                    <Button size="sm" className="bg-yellow-700 hover:bg-yellow-800" onClick={() => setTier.mutate({ userId: r.userId, userTier: "PERFORMER" })} disabled={setTier.isPending}>
                      Performer
                    </Button>
                    <Button size="sm" className="bg-emerald-700 hover:bg-emerald-800" onClick={() => setTier.mutate({ userId: r.userId, userTier: "SELECTED" })} disabled={setTier.isPending}>
                      Selected
                    </Button>
                  </TableCell>
                </TableRow>
              ))}
            </TableBody>
          </Table>
        )}
      </div>
    </div>
  );
}
```
