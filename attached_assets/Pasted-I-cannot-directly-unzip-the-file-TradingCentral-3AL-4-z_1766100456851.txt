```sql
-- --------------------------------------------------------
-- INSTITUTIONAL TRADE AUDIT TRAIL (OATS-Compliant)
-- --------------------------------------------------------

CREATE TABLE IF NOT EXISTS audit_events (
    -- 1. IMMUTABILITY & SEQUENCE
    audit_event_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_timestamp     TIMESTAMPTZ NOT NULL DEFAULT NOW(), -- Server UTC
    sequence_index      BIGSERIAL NOT NULL,                 -- For strict ordering reconstruction

    -- 2. CORE CONTEXT
    event_category      VARCHAR(20) NOT NULL, -- ORDER, RISK, EXECUTION, ADMIN, SYSTEM
    event_code          VARCHAR(50) NOT NULL, -- e.g., ORDER_SUBMITTED, RISK_CHECK_FAIL
    actor_type          VARCHAR(10) NOT NULL, -- USER, ADMIN, SYSTEM
    user_id             INT,                  -- Changed to INT to match likely standard user IDs, change to UUID if needed
    session_id          VARCHAR(64),          -- Provenance: ties to login session
    ip_address          VARCHAR(45),          -- Supports IPv6
    
    -- 3. LIFECYCLE IDENTIFIERS
    order_id            UUID,                 -- Unique intent ID
    client_order_id     VARCHAR(64),          -- Frontend correlation ID
    execution_id        UUID,                 -- Unique fill ID
    position_id         UUID,                 -- Unique holding ID
    
    -- 4. ECONOMICS
    symbol              VARCHAR(20),
    side                VARCHAR(4),           -- BUY, SELL
    order_type          VARCHAR(20),          -- LIMIT, MARKET, STOP
    quantity_lots       NUMERIC(10, 2),
    
    price_requested     NUMERIC(18, 5),       -- The price the user clicked/sent
    price_limit         NUMERIC(18, 5),
    price_stop          NUMERIC(18, 5),
    price_filled        NUMERIC(18, 5),
    
    -- 5. MARKET CONTEXT (Best Execution)
    quote_bid_at_time   NUMERIC(18, 5),
    quote_ask_at_time   NUMERIC(18, 5),
    slippage_pips       NUMERIC(10, 2),
    
    -- 6. RISK EVIDENCE
    risk_check_name     VARCHAR(50),          -- e.g., MAX_LOTS_PER_USER
    risk_limit_val      NUMERIC(18, 2),       -- The limit enforced
    risk_actual_val     NUMERIC(18, 2),       -- The value attempted
    risk_result         VARCHAR(10),          -- PASS, FAIL, OVERRIDE
    
    -- 7. DATA INTEGRITY
    notes               TEXT,                 -- Human readable summary
    payload_snapshot    JSONB                 -- Full raw state for forensic replay
);

-- Indexes for performance
CREATE INDEX idx_audit_user_time ON audit_events(user_id, event_timestamp DESC);
CREATE INDEX idx_audit_order_id ON audit_events(order_id);
CREATE INDEX idx_audit_event_category ON audit_events(event_category);

```

### **Part 2: Backend Implementation (Node.js/TypeScript)**

**File:** `server/src/services/audit.service.ts`
*(This service handles the asynchronous writing of logs so it doesn't block trading)*

```typescript
import { db } from "../db"; // Your DB connection import
import { sql } from "drizzle-orm"; // Or your raw SQL query builder

export interface AuditEventPayload {
  event_category: 'ORDER' | 'RISK' | 'EXECUTION' | 'ADMIN' | 'SYSTEM';
  event_code: string;
  actor_type: 'USER' | 'ADMIN' | 'SYSTEM';
  user_id?: number;
  session_id?: string;
  ip_address?: string;
  order_id?: string;
  symbol?: string;
  side?: string;
  quantity_lots?: number;
  price_requested?: number;
  price_filled?: number;
  risk_check_name?: string;
  risk_result?: 'PASS' | 'FAIL';
  payload_snapshot?: any;
  notes?: string;
}

export class AuditService {
  /**
   * Log an event to the audit trail.
   * Fire-and-forget (non-blocking) usually, or await if strict compliance required.
   */
  static async log(payload: AuditEventPayload) {
    try {
      // Adjust this generic INSERT to match your ORM (Drizzle, Prisma, TypeORM)
      await db.execute(sql`
        INSERT INTO audit_events (
          event_category, event_code, actor_type, user_id, session_id, ip_address,
          order_id, symbol, side, quantity_lots, price_requested, price_filled,
          risk_check_name, risk_result, payload_snapshot, notes
        ) VALUES (
          ${payload.event_category}, ${payload.event_code}, ${payload.actor_type}, 
          ${payload.user_id || null}, ${payload.session_id || null}, ${payload.ip_address || null},
          ${payload.order_id || null}, ${payload.symbol || null}, ${payload.side || null}, 
          ${payload.quantity_lots || null}, ${payload.price_requested || null}, ${payload.price_filled || null},
          ${payload.risk_check_name || null}, ${payload.risk_result || null}, 
          ${JSON.stringify(payload.payload_snapshot)}, ${payload.notes || null}
        )
      `);
    } catch (error) {
      console.error("CRITICAL: Failed to write audit log:", error);
      // In a real institution, this would trigger a P1 alert.
    }
  }
}
```

**File:** `server/src/routes/audit.routes.ts`
*(API to feed the new grid)*

```typescript
import { Router } from "express";
import { db } from "../db";
import { sql } from "drizzle-orm";

const router = Router();

router.get("/api/audit-logs", async (req, res) => {
  try {
    const { userId, symbol, startDate, endDate } = req.query;
    
    // Basic query construction
    let query = sql`SELECT * FROM audit_events WHERE 1=1`;
    
    if (userId) query.append(sql` AND user_id = ${userId}`);
    if (symbol) query.append(sql` AND symbol = ${symbol}`);
    if (startDate) query.append(sql` AND event_timestamp >= ${startDate}`);
    
    query.append(sql` ORDER BY event_timestamp DESC LIMIT 500`);

    const logs = await db.execute(query);
    res.json(logs.rows);
  } catch (err) {
    res.status(500).json({ error: "Failed to fetch audit logs" });
  }
});

export default router;
```

```tsx
import React, { useState, useEffect } from 'react';

  event_code: string;
  symbol: string;
  side: string;
  quantity_lots: number;
```
