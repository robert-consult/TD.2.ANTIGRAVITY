```ts
// server/grift/griftGeo.ts
export function haversineKm(a: { lat: number; lon: number }, b: { lat: number; lon: number }) {
  const R = 6371;
  const dLat = ((b.lat - a.lat) * Math.PI) / 180;
  const dLon = ((b.lon - a.lon) * Math.PI) / 180;
  const lat1 = (a.lat * Math.PI) / 180;
  const lat2 = (b.lat * Math.PI) / 180;

  const s =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.sin(dLon / 2) * Math.sin(dLon / 2) * Math.cos(lat1) * Math.cos(lat2);

  const c = 2 * Math.atan2(Math.sqrt(s), Math.sqrt(1 - s));
  return R * c;
}

export function kmh(distanceKm: number, deltaMs: number) {
  const hours = deltaMs / (1000 * 60 * 60);
  if (hours <= 0) return Infinity;
  return distanceKm / hours;
}
```

```ts
function createSignal(args: {
  ts: number;
  ruleCode: string;
  primaryUserId: number;
  secondaryUserId?: number | null;
  deviceId?: string | null;
  ip?: string | null;
  userAgent?: string | null;
  country?: string | null;
  region?: string | null;
  city?: string | null;
  latitude?: number | null;
  longitude?: number | null;
  asn?: number | null;
  org?: string | null;
  symbol?: string | null;
  tradeId?: number | null;
  points: number;
  evidence: any;
  status?: GriftSignalStatus;
  bucketMinutes: number;
}) {
  const severity = severityFromPoints(args.points);
  const st: GriftSignalStatus = args.status ?? "OPEN";

  const b = timeBucket(args.ts, args.bucketMinutes);

  const dedupeKey = [
    args.ruleCode,
    args.primaryUserId,
    args.secondaryUserId ?? "",
    args.deviceId ?? "",
    args.ip ?? "",
    args.symbol ?? "",
    b,
  ].join("|");

  const evidenceJson = JSON.stringify(args.evidence ?? {});
  const updatedAt = args.ts;

  try {
    const info = sqlite.prepare(`
      INSERT INTO grift_signals (
        created_at, updated_at,
        rule_code, severity, status,
        primary_user_id, secondary_user_id,
        device_id, ip, user_agent,
        country, region, city, latitude, longitude,
        asn, org,
        symbol, trade_id,
        points, dedupe_key, evidence_json
      ) VALUES (
        ?, ?,
        ?, ?, ?,
        ?, ?,
        ?, ?, ?,
        ?, ?, ?, ?, ?,
        ?, ?,
        ?, ?,
        ?, ?, ?
      )
    `).run(
      args.ts, updatedAt,
      args.ruleCode, severity, st,
      args.primaryUserId, args.secondaryUserId ?? null,
      args.deviceId ?? null, args.ip ?? null, args.userAgent ?? null,
      args.country ?? null, args.region ?? null, args.city ?? null,
      args.latitude ?? null, args.longitude ?? null,
      args.asn ?? null, args.org ?? null,
      args.symbol ?? null, args.tradeId ?? null,
      args.points, dedupeKey, evidenceJson
    );

    return { created: true, id: Number(info.lastInsertRowid), dedupeKey };
  } catch {
    // If dedupe hit, we still update updated_at/evidence/points (but do not reopen status)
    sqlite.prepare(`
      UPDATE grift_signals
      SET updated_at=?, points=?, evidence_json=?
      WHERE dedupe_key=?
    `).run(updatedAt, args.points, evidenceJson, dedupeKey);

    const row = sqlite.prepare(`SELECT id FROM grift_signals WHERE dedupe_key=?`).get(dedupeKey) as any;
    return { created: false, id: row?.id ? Number(row.id) : null, dedupeKey };
  }
}

function detectMultiAccountDevice(userId: number, deviceId: string, cfg: GriftConfig) {
  const since = now() - cfg.multiAccountWindowDays * 24 * 60 * 60 * 1000;

  const rows = sqlite.prepare(`
    SELECT user_id as userId, last_seen_at as lastSeenAt, seen_count as seenCount
    FROM grift_device_users
    WHERE device_id=? AND last_seen_at >= ?
    ORDER BY last_seen_at DESC
  `).all(deviceId, since) as any[];

  const distinctUsers = Array.from(new Set(rows.map((r) => Number(r.userId))));
  if (distinctUsers.length <= 1) return;

  const other = distinctUsers.find((u) => u !== userId) ?? null;

  createSignal({
    ts: now(),
    ruleCode: "MULTI_ACCOUNT_DEVICE",
    primaryUserId: userId,
    secondaryUserId: other,
    deviceId,
    points: cfg.scoreMultiAccountDevice,
    bucketMinutes: cfg.multiAccountWindowDays * 24 * 60, // day-window dedupe
    evidence: {
      deviceId,
      windowDays: cfg.multiAccountWindowDays,
      users: rows,
      distinctUsers,
    },
  });
}

function detectChurn(userId: number, cfg: GriftConfig) {
  const since = now() - cfg.churnWindowHours * 60 * 60 * 1000;

  const obs = sqlite.prepare(`
    SELECT ts, ip, user_agent as ua, device_id as deviceId, country, asn
    FROM grift_observations
    WHERE user_id=? AND ts >= ?
    ORDER BY ts DESC
    LIMIT 500
  `).all(userId, since) as any[];

  const uniq = (arr: any[]) => Array.from(new Set(arr.filter(Boolean).map(String)));

  const ips = uniq(obs.map((o) => o.ip));
  const uas = uniq(obs.map((o) => o.ua));
  const devices = uniq(obs.map((o) => o.deviceId));
  const countries = uniq(obs.map((o) => o.country));
  const asns = Array.from(new Set(obs.map((o) => (o.asn == null ? null : Number(o.asn))).filter((x) => x != null)));

  if (ips.length >= cfg.ipUniqueThreshold) {
    createSignal({
      ts: now(),
      ruleCode: "IP_CHURN",
      primaryUserId: userId,
      ip: ips[0] ?? null,
      points: cfg.scoreIpChurn,
      bucketMinutes: cfg.churnWindowHours * 60,
      evidence: { windowHours: cfg.churnWindowHours, uniqueIps: ips, count: ips.length, recent: obs.slice(0, 50) },
    });
  }

  if (uas.length >= cfg.uaUniqueThreshold) {
    createSignal({
      ts: now(),
      ruleCode: "UA_CHURN",
      primaryUserId: userId,
      userAgent: uas[0] ?? null,
      points: cfg.scoreUaChurn,
      bucketMinutes: cfg.churnWindowHours * 60,
      evidence: { windowHours: cfg.churnWindowHours, uniqueUserAgents: uas, count: uas.length, recent: obs.slice(0, 50) },
    });
  }

  if (devices.length >= cfg.deviceUniqueThreshold) {
    createSignal({
      ts: now(),
      ruleCode: "DEVICE_CHURN",
      primaryUserId: userId,
      deviceId: devices[0] ?? null,
      points: cfg.scoreDeviceChurn,
      bucketMinutes: cfg.churnWindowHours * 60,
      evidence: { windowHours: cfg.churnWindowHours, uniqueDevices: devices, count: devices.length, recent: obs.slice(0, 50) },
    });
  }

  if (asns.length >= cfg.asnUniqueThreshold) {
    createSignal({
      ts: now(),
      ruleCode: "ASN_VOLATILITY",
      primaryUserId: userId,
      asn: asns[0] ?? null,
      points: cfg.scoreAsnVolatility,
      bucketMinutes: cfg.churnWindowHours * 60,
      evidence: { windowHours: cfg.churnWindowHours, uniqueAsns: asns, count: asns.length, recent: obs.slice(0, 50) },
    });
  }

  // Countries changing can be a leading indicator, but we keep it as evidence only (no separate rule)
  if (countries.length >= 2) {
    // Optional: fold into IP_CHURN evidence already. Keeping as is (no extra signal).
  }
}

function detectGeoVelocity(userId: number, cfg: GriftConfig) {
  const since = now() - cfg.geoVelocityMaxHours * 60 * 60 * 1000;

  const obs = sqlite.prepare(`
    SELECT ts, latitude as lat, longitude as lon, country, city, ip
    FROM grift_observations
    WHERE user_id=? AND ts >= ?
      AND latitude IS NOT NULL AND longitude IS NOT NULL
    ORDER BY ts DESC
    LIMIT 50
  `).all(userId, since) as any[];

  if (obs.length < 2) return;

  // Find max velocity pair among consecutive points in time order
  const sorted = [...obs].sort((a, b) => Number(a.ts) - Number(b.ts));
  let best: any = null;

  for (let i = 1; i < sorted.length; i++) {
    const a = sorted[i - 1];
    const b = sorted[i];
    const d = haversineKm({ lat: Number(a.lat), lon: Number(a.lon) }, { lat: Number(b.lat), lon: Number(b.lon) });
    const v = kmh(d, Number(b.ts) - Number(a.ts));

    if (!best || v > best.v) best = { a, b, d, v };
  }

  if (!best) return;

  if (best.d >= cfg.geoVelocityMinDistanceKm && best.v >= cfg.geoVelocityKmhThreshold) {
    createSignal({
      ts: now(),
      ruleCode: "GEO_VELOCITY",
      primaryUserId: userId,
      ip: best.b.ip ?? null,
      latitude: Number(best.b.lat),
      longitude: Number(best.b.lon),
      country: best.b.country ?? null,
      city: best.b.city ?? null,
      points: cfg.scoreGeoVelocity,
      bucketMinutes: cfg.geoVelocityMaxHours * 60,
      evidence: {
        thresholdKmh: cfg.geoVelocityKmhThreshold,
        minDistanceKm: cfg.geoVelocityMinDistanceKm,
        maxHours: cfg.geoVelocityMaxHours,
        distanceKm: best.d,
        velocityKmh: best.v,
        from: best.a,
        to: best.b,
      },
    });
  }
}

function detectConcurrentSessions(userId: number, cfg: GriftConfig) {
  const since = now() - cfg.concurrentWindowMinutes * 60 * 1000;

  const obs = sqlite.prepare(`
    SELECT ts, device_id as deviceId, ip
    FROM grift_observations
    WHERE user_id=? AND ts >= ?
    ORDER BY ts DESC
    LIMIT 300
  `).all(userId, since) as any[];

  // Distinct session “contexts” = device + ip (ip optional)
  const keys = new Set<string>();
  for (const o of obs) {
    const d = o.deviceId ? String(o.deviceId) : "";
    const ip = o.ip ? String(o.ip) : "";
    if (!d && !ip) continue;
    keys.add(`${d}|${ip}`);
  }

  if (keys.size >= 2) {
    createSignal({
      ts: now(),
      ruleCode: "CONCURRENT_SESSIONS",
      primaryUserId: userId,
      deviceId: obs[0]?.deviceId ?? null,
      ip: obs[0]?.ip ?? null,
      points: cfg.scoreConcurrentSessions,
      bucketMinutes: cfg.concurrentWindowMinutes,
      evidence: {
        windowMinutes: cfg.concurrentWindowMinutes,
        distinctContexts: Array.from(keys),
        sample: obs.slice(0, 50),
      },
    });
  }
}

function detectHedgePairOnTrade(args: {
  userId: number;
  tradeId: number;
  symbol: string;
  side: string;
  deviceId: string | null;
  ip: string | null;
  cfg: GriftConfig;
}) {
  const { userId, tradeId, symbol, side, deviceId, ip, cfg } = args;
  const since = now() - cfg.hedgeWindowMinutes * 60 * 1000;

  // Look for opposite side, other user, same symbol, within window
  const rows = sqlite.prepare(`
    SELECT ts, user_id as userId, trade_id as tradeId, side, device_id as deviceId, ip
    FROM grift_trade_observations
    WHERE symbol=? AND ts >= ?
      AND user_id != ?
    ORDER BY ts DESC
    LIMIT 200
  `).all(symbol, since, userId) as any[];

  const opposite = (s: string) => (String(s).toUpperCase() === "BUY" ? "SELL" : "BUY");

  const opp = opposite(side);
  const candidates = rows.filter((r) => String(r.side).toUpperCase() === opp);

  const filtered = candidates.filter((r) => {
    if (cfg.hedgeRequireDeviceMatch === 1) {
      if (!deviceId) return false;
      return String(r.deviceId || "") === String(deviceId);
    }
    if (cfg.hedgeAllowIpMatch === 1 && ip) {
      return String(r.ip || "") === String(ip);
    }
    return true;
  });

  if (filtered.length === 0) return;

  // Take best (most recent) match
  const hit = filtered[0];
  const otherUserId = Number(hit.userId);

  createSignal({
    ts: now(),
    ruleCode: "HEDGE_PAIR",
    primaryUserId: userId,
    secondaryUserId: otherUserId,
    deviceId: pick(deviceId, null),
    ip: pick(ip, null),
    symbol,
    tradeId,
    points: cfg.scoreHedgePair,
    bucketMinutes: cfg.hedgeWindowMinutes,
    evidence: {
      windowMinutes: cfg.hedgeWindowMinutes,
      symbol,
      thisTrade: { userId, tradeId, side, deviceId, ip },
      matchedTrade: hit,
      requireDeviceMatch: cfg.hedgeRequireDeviceMatch === 1,
      allowIpMatch: cfg.hedgeAllowIpMatch === 1,
    },
  });
}

export function recomputeUserAggregates(userId: number, cfg: GriftConfig) {
  const ts = now();

  // Open signals affecting user
  const openSignals = sqlite.prepare(`
    SELECT id, points, created_at
    FROM grift_signals
    WHERE status='OPEN' AND (primary_user_id=? OR secondary_user_id=?)
    ORDER BY created_at DESC
  `).all(userId, userId) as any[];

  const openPointsSum = Math.min(100, openSignals.reduce((s, r) => s + Number(r.points || 0), 0));
  const scoreCurrent = openPointsSum;

  const since7d = ts - 7 * 24 * 60 * 60 * 1000;
  const since30d = ts - 30 * 24 * 60 * 60 * 1000;

  const score7d = sqlite.prepare(`
    SELECT COALESCE(SUM(points), 0) as s
    FROM grift_signals
    WHERE created_at >= ? AND (primary_user_id=? OR secondary_user_id=?)
  `).get(since7d, userId, userId) as any;

  const score30d = sqlite.prepare(`
    SELECT COALESCE(SUM(points), 0) as s
    FROM grift_signals
    WHERE created_at >= ? AND (primary_user_id=? OR secondary_user_id=?)
  `).get(since30d, userId, userId) as any;

  const obs7d = sqlite.prepare(`
    SELECT device_id as deviceId, ip, user_agent as ua, country, asn
    FROM grift_observations
    WHERE user_id=? AND ts >= ?
  `).all(userId, since7d) as any[];

  const uniq = (arr: any[]) => Array.from(new Set(arr.filter(Boolean).map(String)));
  const devices7d = uniq(obs7d.map((o) => o.deviceId)).length;
  const ips7d = uniq(obs7d.map((o) => o.ip)).length;
  const uas7d = uniq(obs7d.map((o) => o.ua)).length;
  const countries7d = uniq(obs7d.map((o) => o.country)).length;
  const asns7d = Array.from(new Set(obs7d.map((o) => (o.asn == null ? null : Number(o.asn))).filter((x) => x != null))).length;

  // Linked accounts via shared device (30d)
  const linked = sqlite.prepare(`
    SELECT COUNT(DISTINCT du2.user_id) as c
    FROM grift_device_users du1
    JOIN grift_device_users du2 ON du1.device_id = du2.device_id
    WHERE du1.user_id=? AND du2.user_id != ?
      AND du1.last_seen_at >= ? AND du2.last_seen_at >= ?
  `).get(userId, userId, since30d, since30d) as any;

  // Hedge pairs (7d)
  const hedgePairs = sqlite.prepare(`
    SELECT COUNT(DISTINCT CASE
      WHEN primary_user_id=? THEN secondary_user_id
      ELSE primary_user_id
    END) as c
    FROM grift_signals
    WHERE rule_code='HEDGE_PAIR'
      AND created_at >= ?
      AND secondary_user_id IS NOT NULL
      AND (primary_user_id=? OR secondary_user_id=?)
  `).get(userId, since7d, userId, userId) as any;

  const openCount = openSignals.length;
  const tier = tierFromScore(scoreCurrent, cfg);

  sqlite.prepare(`
    INSERT INTO grift_user_scores (
      user_id, score_current, tier,
      score_7d, score_30d,
      devices_7d, ips_7d, user_agents_7d, countries_7d, asns_7d,
      linked_accounts_30d, hedge_pairs_7d,
      open_signals_count, last_evaluated_at
    ) VALUES (
      ?, ?, ?,
      ?, ?,
      ?, ?, ?, ?, ?,
      ?, ?,
      ?, ?
    )
    ON CONFLICT(user_id) DO UPDATE SET
      score_current=excluded.score_current,
      tier=excluded.tier,
      score_7d=excluded.score_7d,
      score_30d=excluded.score_30d,
      devices_7d=excluded.devices_7d,
      ips_7d=excluded.ips_7d,
      user_agents_7d=excluded.user_agents_7d,
      countries_7d=excluded.countries_7d,
      asns_7d=excluded.asns_7d,
      linked_accounts_30d=excluded.linked_accounts_30d,
      hedge_pairs_7d=excluded.hedge_pairs_7d,
      open_signals_count=excluded.open_signals_count,
      last_evaluated_at=excluded.last_evaluated_at
  `).run(
    userId, scoreCurrent, tier,
    Number(score7d?.s ?? 0), Number(score30d?.s ?? 0),
    devices7d, ips7d, uas7d, countries7d, asns7d,
    Number(linked?.c ?? 0), Number(hedgePairs?.c ?? 0),
    openCount, ts
  );
}

function collectContext(req: any, body?: any) {
  const ip = getClientIp(req);

function runDetectorsForUser(userId: number, cfg: GriftConfig, deviceId: string | null) {

  if (deviceId) detectMultiAccountDevice(userId, deviceId, cfg);

}

export const griftEngine = {
  getConfig,

  onLoginSuccess(req: any, userId: number) {
    const cfg = getConfig();
    if (cfg.enabled !== 1) return;

    const ts = now();
    const ctx = collectContext(req);

    insertObservation({
      ts,
      userId,
      eventType: "LOGIN_SUCCESS",
      deviceId: ctx.deviceId,
      ip: ctx.ip,
      userAgent: ctx.ua,
      country: ctx.country,
      region: ctx.region,
      city: ctx.city,
      latitude: ctx.latitude,
      longitude: ctx.longitude,
      asn: ctx.asn,
      org: ctx.org,
    });

    if (ctx.deviceId) {
      upsertDevice(ctx.deviceId, { ip: ctx.ip, ua: ctx.ua, country: ctx.country });
      upsertDeviceUser(ctx.deviceId, userId);
    }

    runDetectorsForUser(userId, cfg, ctx.deviceId);
  },

  onSessionPing(req: any, userId: number, body?: any) {
    const cfg = getConfig();
    if (cfg.enabled !== 1) return;

    const ts = now();
    const ctx = collectContext(req, body);

    insertObservation({
      ts,
      userId,
      eventType: "SESSION_PING",
      deviceId: ctx.deviceId,
      ip: ctx.ip,
      userAgent: ctx.ua,
      country: ctx.country,
      region: ctx.region,
      city: ctx.city,
      latitude: ctx.latitude,
      longitude: ctx.longitude,
      asn: ctx.asn,
      org: ctx.org,
    });

    if (ctx.deviceId) {
      upsertDevice(ctx.deviceId, { ip: ctx.ip, ua: ctx.ua, country: ctx.country });
      upsertDeviceUser(ctx.deviceId, userId);
    }

    // Concurrency is best detected from ping volume
    runDetectorsForUser(userId, cfg, ctx.deviceId);
  },

  onTradeSubmit(req: any, userId: number, tradeId: number, symbol: string, side: "BUY" | "SELL") {
    const cfg = getConfig();
    if (cfg.enabled !== 1) return;

    const ts = now();
    const ctx = collectContext(req);

    insertTradeObs({
      ts,
      userId,
      tradeId,
      symbol,
      side,
      deviceId: ctx.deviceId,
      ip: ctx.ip,
    });

    // Also treat trade submit as observation
    insertObservation({
      ts,
      userId,
      eventType: "TRADE_SUBMIT",
      deviceId: ctx.deviceId,
      ip: ctx.ip,
      userAgent: ctx.ua,
      country: ctx.country,
      region: ctx.region,
      city: ctx.city,
      latitude: ctx.latitude,
      longitude: ctx.longitude,
      asn: ctx.asn,
      org: ctx.org,
    });

    if (ctx.deviceId) {
      upsertDevice(ctx.deviceId, { ip: ctx.ip, ua: ctx.ua, country: ctx.country });
      upsertDeviceUser(ctx.deviceId, userId);
      detectMultiAccountDevice(userId, ctx.deviceId, cfg);
    }

    detectHedgePairOnTrade({
      userId,
      tradeId,
      symbol,
      side,
      deviceId: ctx.deviceId,
      ip: ctx.ip,
      cfg,
    });

    runDetectorsForUser(userId, cfg, ctx.deviceId);
  },

  recomputeUser(userId: number) {
    const cfg = getConfig();
    runDetectorsForUser(userId, cfg, null);
    return true;
  },
};
```

```ts
// client/src/lib/griftPing.ts
import { getDeviceId } from "./deviceId";

export function startGriftPing(opts?: { intervalMs?: number }) {
  const intervalMs = Math.max(30_000, Math.min(5 * 60_000, opts?.intervalMs ?? 60_000));

  let stopped = false;
  let timer: any = null;

  const send = async () => {
    try {
      // optional: if you store auth state, gate here; otherwise the server returns 401 safely.
      await fetch("/api/grift/ping", {
        method: "POST",
        credentials: "include",
        headers: {
          "Content-Type": "application/json",
          "X-Device-Id": getDeviceId(),
        },
        body: JSON.stringify({}),
      });
    } catch {
      // ignore
    }
  };

  const loop = async () => {
    if (stopped) return;
    await send();
    if (stopped) return;
    timer = setTimeout(loop, intervalMs);
  };

  loop();

  return () => {
    stopped = true;
    if (timer) clearTimeout(timer);
  };
}
```
