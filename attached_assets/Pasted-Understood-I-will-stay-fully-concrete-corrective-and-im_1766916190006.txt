## 🔧 PATCH 1 — CAPTCHA (LEFT-TO-RIGHT SLIDER)

### 1. Frontend Component (Cloudflare Turnstile)

**NEW FILE**
`client/src/components/CaptchaTurnstile.tsx`

```tsx
import { useEffect, useRef } from "react";

declare global {
  interface Window {
    turnstile?: any;
  }
}

export function CaptchaTurnstile({
  siteKey,
  onToken,
}: {
  siteKey: string;
  onToken: (token: string) => void;
}) {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!siteKey || !ref.current) return;

    const scriptId = "cf-turnstile";
    if (!document.getElementById(scriptId)) {
      const s = document.createElement("script");
      s.id = scriptId;
      s.src = "https://challenges.cloudflare.com/turnstile/v0/api.js";
      s.async = true;
      s.defer = true;
      document.head.appendChild(s);
    }

    const interval = setInterval(() => {
      if (window.turnstile && ref.current) {
        clearInterval(interval);
        window.turnstile.render(ref.current, {
          sitekey: siteKey,
          callback: onToken,
          appearance: "interaction-only",
        });
      }
    }, 100);

    return () => clearInterval(interval);
  }, [siteKey]);

  return <div ref={ref} className="mt-3" />;
}
```

### 2. Phone Input Component

**NEW FILE**
`client/src/components/PhoneInput.tsx`

```tsx
import { parsePhoneNumberFromString } from "libphonenumber-js";
import { useState } from "react";

export function PhoneInput({
  country,
  onChange,
}: {
  country: string;
  onChange: (e164: string | null) => void;
}) {
  const [raw, setRaw] = useState("");

  function handleChange(v: string) {
    setRaw(v);
    try {
      const parsed = parsePhoneNumberFromString(v, country);
      onChange(parsed?.isValid() ? parsed.number : null);
    } catch {
      onChange(null);
    }
  }

  return (
    <input
      className="input"
      placeholder="Phone number"
      value={raw}
      onChange={(e) => handleChange(e.target.value)}
    />
  );
}
```
