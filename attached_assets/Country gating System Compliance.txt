import type { Request } from "express";
import { buildGeoContext } from "./sessionTrail";

export function normalizeIso2(v: any): string | null {
  if (!v) return null;
  const s = String(v).trim().toUpperCase();
  if (s.length !== 2) return null;
  // "T1", "XX" sometimes used by proxies/vendors as "unknown"
  if (s === "T1" || s === "XX") return null;
  return s;
}

export function parseIso2CsvToSet(csv: any): Set<string> {
  const raw = typeof csv === "string" ? csv : "";
  const parts = raw
    .split(",")
    .map((x) => normalizeIso2(x))
    .filter((x): x is string => !!x);
  return new Set(parts);
}

/**
 * Resolve request country via (1) edge/CDN headers, then (2) geoip-lite fallback.
 * If you are behind Cloudflare, ensure CF-IPCountry is enabled.
 */
export function resolveIpCountryIso2(req: Request, ip: string): string | null {
  // Cloudflare
  const cf = normalizeIso2(req.headers["cf-ipcountry"]);
  if (cf) return cf;

  // Vercel / some proxies
  const vercel = normalizeIso2(req.headers["x-vercel-ip-country"]);
  if (vercel) return vercel;

  // Generic "x-country" if you ever add it at the edge
  const xCountry = normalizeIso2(req.headers["x-country"]);
  if (xCountry) return xCountry;

  // Fallback to geoip-lite (best effort)
  try {
    const geo = buildGeoContext(ip);
    return normalizeIso2((geo as any)?.countryCode);
  } catch {
    return null;
  }
}

export function checkJurisdictionAccess(params: {
  action: "SIGNUP" | "LOGIN";
  sysRow: any;
  ipCountryIso2: string | null;
  declaredCountryIso2: string | null;
}):
  | { blocked: false }
  | {
      blocked: true;
      reasonCode: "JURISDICTION_RESTRICTED";
      message: string;
      matched: {
        restrictedList: string[];
        ipCountryIso2: string | null;
        declaredCountryIso2: string | null;
        matchedBy: Array<"IP" | "DECLARED">;
      };
    } {
  const { action, sysRow, ipCountryIso2, declaredCountryIso2 } = params;

  const restricted = parseIso2CsvToSet(sysRow?.jurisdictionRestrictedIso2Csv);
  const message =
    typeof sysRow?.jurisdictionRestrictedMessage === "string" && sysRow.jurisdictionRestrictedMessage.trim()
      ? String(sysRow.jurisdictionRestrictedMessage)
      : "This jurisdiction is not supported due to regulatory restrictions.";

  const blockSignup = Boolean(sysRow?.jurisdictionBlockSignup);
  const blockLogin = Boolean(sysRow?.jurisdictionBlockLogin);
  const blockByIp = Boolean(sysRow?.jurisdictionBlockByIp);
  const blockByDeclared = Boolean(sysRow?.jurisdictionBlockByCountrySelection);

  const shouldEnforce =
    (action === "SIGNUP" && blockSignup) ||
    (action === "LOGIN" && blockLogin);

  if (!shouldEnforce) return { blocked: false };

  const matchedBy: Array<"IP" | "DECLARED"> = [];
  if (blockByIp && ipCountryIso2 && restricted.has(ipCountryIso2)) matchedBy.push("IP");
  if (blockByDeclared && declaredCountryIso2 && restricted.has(declaredCountryIso2)) matchedBy.push("DECLARED");

  if (matchedBy.length === 0) return { blocked: false };

  return {
    blocked: true,
    reasonCode: "JURISDICTION_RESTRICTED",
    message,
    matched: {
      restrictedList: Array.from(restricted),
      ipCountryIso2,
      declaredCountryIso2,
      matchedBy,
    },
  };
}

import type { IncomingHttpHeaders } from "http";
import geoip from "geoip-lite";

export type JurisdictionDecision =
  | { blocked: false }
  | {
      blocked: true;
      reasonCode: "JURISDICTION_RESTRICTED";
      message: string;
      matched: {
        ipCountryIso2: string | null;
        declaredCountryIso2: string | null;
      };

export function normalizeIso2(v: unknown): string | null {
  if (!v) return null;
  const s = String(v).trim().toUpperCase();
  if (!/^[A-Z]{2}$/.test(s)) return null;
  return s;
}

export function parseIso2CsvToSet(csv: unknown): Set<string> {
  const set = new Set<string>();
  String(csv ?? "")
    .split(/[,\s;]+/)
    .map((x) => x.trim().toUpperCase())
    .filter((x) => /^[A-Z]{2}$/.test(x))
    .forEach((x) => set.add(x));
  return set;
}

/**
 * Prefer upstream geo headers when available (Cloudflare/Vercel/etc),
 * fall back to geoip-lite lookup on the resolved client IP.
 */
export function resolveIpCountryIso2(args: { headers: IncomingHttpHeaders; ip: string }): string | null {
  const h = args.headers;

  const headerCandidates = [
    h["cf-ipcountry"],              // Cloudflare
    (h as any)["CF-IPCountry"],
    h["x-vercel-ip-country"],       // Vercel
    h["x-country-code"],            // common custom header
    h["x-geo-country"],             // common custom header
  ];

  for (const c of headerCandidates) {
    const iso2 = normalizeIso2(Array.isArray(c) ? c[0] : c);
    // Cloudflare uses "T1" for Tor, "XX" for unknown. Treat as unknown.
    if (iso2 && iso2 !== "XX" && iso2 !== "T1") return iso2;
  }

  const geo = geoip.lookup(args.ip);
  return normalizeIso2((geo as any)?.country);
}

export function checkJurisdictionAccess(params: {
  action: "SIGNUP" | "LOGIN";
  sysRow: any;
  ipCountryIso2?: string | null;
  declaredCountryIso2?: string | null;
}): JurisdictionDecision {
  const restrictedCsv = String(params.sysRow?.jurisdictionRestrictedIso2Csv ?? "KP,IR,CU,SY");
  const restrictedMessage = String(
    params.sysRow?.jurisdictionRestrictedMessage ??
      "This jurisdiction is not supported due to regulatory restrictions."
  );

  const blockSignup = Boolean(params.sysRow?.jurisdictionBlockSignup ?? false);
  const blockLogin = Boolean(params.sysRow?.jurisdictionBlockLogin ?? false);
  const byIp = Boolean(params.sysRow?.jurisdictionBlockByIp ?? true);
  const byDeclared = Boolean(params.sysRow?.jurisdictionBlockByCountrySelection ?? true);

  const apply = params.action === "SIGNUP" ? blockSignup : blockLogin;
  if (!apply) return { blocked: false };

  const restrictedSet = parseIso2CsvToSet(restrictedCsv);

  const ipIso2 = normalizeIso2(params.ipCountryIso2);
  const declaredIso2 = normalizeIso2(params.declaredCountryIso2);

  const ipHit = byIp && ipIso2 && restrictedSet.has(ipIso2);
  const declaredHit = byDeclared && declaredIso2 && restrictedSet.has(declaredIso2);

  if (!ipHit && !declaredHit) return { blocked: false };

  return {
    blocked: true,
    reasonCode: "JURISDICTION_RESTRICTED",
    message: restrictedMessage,
    matched: {
      ipCountryIso2: ipHit ? ipIso2 : null,
      declaredCountryIso2: declaredHit ? declaredIso2 : null,
    },
  };
}

// server/middleware/jurisdictionGuard.ts
import type { NextFunction, Request, Response } from "express";
import { eq } from "drizzle-orm";
import { db } from "@db";
import { systemConfig, users } from "../../shared/schema";
import { getClientIp } from "../security/sessionTrail";
import { revokeSession } from "../security/sessionTrail";
import { checkJurisdictionAccess, resolveIpCountryIso2 } from "../security/jurisdictionAccess";

/**
 * If jurisdictionBlockLogin is enabled, enforce it on every authenticated API call.
 * This ensures a user cannot keep using the app on an existing session after policy changes.
 *
 * Admin sessions are excluded by default to avoid locking operators out.
 */
export function jurisdictionGuard(req: Request, res: Response, next: NextFunction) {
  const session: any = req.session;
  if (!session?.userId) return next();

  // Do not lock out admins/operators by default (adjust if you want stricter behavior)
  if (session?.isAdmin || session?.realAdminId || session?.isImpersonating) return next();

  const sysRow = db.select().from(systemConfig).where(eq(systemConfig.id, 1)).get() as any;
  const blockLogin = Boolean(sysRow?.jurisdictionBlockLogin ?? false);
  if (!blockLogin) return next();

  const user = db
    .select({ id: users.id, email: users.email, countryIso2: users.countryIso2 })
    .from(users)
    .where(eq(users.id, session.userId))
    .get() as any;

  if (!user) return next();

  const ip = getClientIp(req);
  const ipCountryIso2 = resolveIpCountryIso2({ headers: req.headers, ip });

  const decision = checkJurisdictionAccess({
    action: "LOGIN",
    sysRow,
    ipCountryIso2,
    declaredCountryIso2: user.countryIso2,
  });

  if (!decision.blocked) return next();

  // Revoke the active session so subsequent requests are denied even if cookie persists
  try {
    revokeSession({
      actorUserId: 0, // "system"
      targetUserId: session.userId,
      sessionId: req.sessionID,
      reason: decision.reasonCode,
    });
  } catch {
    // If revoke fails, still hard-fail access
  }

  try {
    req.session.destroy(() => {});
  } catch {
    // ignore
  }

  return res.status(403).json({
    code: decision.reasonCode,
    message: decision.message,
    matched: decision.matched,
  });
}

import { db } from "@db";
import { systemConfig } from "@shared/schema";

const DEFAULT_RESTRICTED_ISO2 = "KP,IR,CU,SY";
const DEFAULT_RESTRICTED_MESSAGE =
  "This jurisdiction is not supported due to regulatory restrictions.";

function normIso2(code: string) {
  return String(code || "").trim().toUpperCase();
}

export function parseRestrictedCountriesCsv(csv?: string | null): string[] {
  return String(csv || "")
    .split(",")
    .map((s) => normIso2(s))
    .filter((s) => /^[A-Z]{2}$/.test(s));
}

type JurisdictionPolicyCache = {
  countries: string[];
  message: string;
  blockSignup: boolean;
  blockLogin: boolean;
  byIp: boolean;
  byCountrySelection: boolean;
  fetchedAtMs: number;
};

let cachedPolicy: JurisdictionPolicyCache | null = null;
// small TTL to keep runtime overhead low but still react to admin changes quickly
const POLICY_TTL_MS = 15_000;

export function getJurisdictionPolicy(): Omit<JurisdictionPolicyCache, "fetchedAtMs"> {
  const now = Date.now();
  if (cachedPolicy && now - cachedPolicy.fetchedAtMs < POLICY_TTL_MS) {
    const { fetchedAtMs, ...rest } = cachedPolicy;
    return rest;
  }

  const row = db.select().from(systemConfig).where(eq(systemConfig.id, 1)).get() as any;

  const countries = parseRestrictedCountriesCsv(
    row?.jurisdictionRestrictedIso2Csv ?? DEFAULT_RESTRICTED_ISO2
  );

  const message = String(row?.jurisdictionRestrictedMessage ?? DEFAULT_RESTRICTED_MESSAGE);

  const blockSignup = Boolean(row?.jurisdictionBlockSignup ?? true);
  const blockLogin = Boolean(row?.jurisdictionBlockLogin ?? true);
  const byIp = Boolean(row?.jurisdictionBlockByIp ?? true);
  const byCountrySelection = Boolean(row?.jurisdictionBlockByCountrySelection ?? true);

  cachedPolicy = {
    countries,
    message,
    blockSignup,
    blockLogin,
    byIp,
    byCountrySelection,
    fetchedAtMs: now,
  };

  const { fetchedAtMs, ...rest } = cachedPolicy;
  return rest;
}

// Keep backward compatibility where used
export function getRestrictedCountryPolicy(): { countries: string[]; message: string } {
  const p = getJurisdictionPolicy();
  return { countries: p.countries, message: p.message };
}

export function isRestrictedCountry(code: string): boolean {
  const { countries } = getRestrictedCountryPolicy();
  return countries.includes(normIso2(code));
}

function normIso2(code: any): string {

function evaluateJurisdictionBlock(params: {
  mode: "SIGNUP" | "LOGIN" | "WAITLIST";
  declaredIso2?: string | null; // signup country (SIGNUP) or user.profile country (LOGIN)
  ipCountryIso2?: string | null;
}) {
  const jp = getJurisdictionPolicy();

  const declared = normIso2(params.declaredIso2);
  const ipIso2 = normIso2(params.ipCountryIso2);

  const blockedByIp = jp.byIp && ipIso2 && jp.countries.includes(ipIso2);
  const blockedByDeclared = jp.byCountrySelection && declared && jp.countries.includes(declared);

  if (params.mode === "SIGNUP" && !jp.blockSignup) return { blocked: false as const };
  if (params.mode === "WAITLIST" && !jp.blockSignup) return { blocked: false as const };
  if (params.mode === "LOGIN" && !jp.blockLogin) return { blocked: false as const };

  // For WAITLIST we only have IP; declared is not available
  if (params.mode === "WAITLIST") {
    if (blockedByIp) {
      return { blocked: true as const, blockedBy: "IP" as const, message: jp.message, ipIso2 };
    }
    return { blocked: false as const };
  }

  // For SIGNUP/LOGIN: block if either enabled mode indicates restriction
  if (blockedByIp) {
    return { blocked: true as const, blockedBy: "IP" as const, message: jp.message, ipIso2, declared };
  }
  if (blockedByDeclared) {
    return { blocked: true as const, blockedBy: "DECLARED" as const, message: jp.message, ipIso2, declared };
  }
  return { blocked: false as const };
}

import { Badge } from "@/components/ui/badge";
import { Button } from "@/components/ui/button";
import { Textarea } from "@/components/ui/textarea";
import { Popover, PopoverContent, PopoverTrigger } from "@/components/ui/popover";
import { Command, CommandInput, CommandItem, CommandList } from "@/components/ui/command";

type CountryRow = { code: string; name: string };

function normIso2(v: any): string {
  const s = String(v ?? "").trim().toUpperCase();
  return /^[A-Z]{2}$/.test(s) ? s : "";
}

function parseCsv(csv: string): string[] {
  return String(csv || "")
    .split(",")
    .map((x) => normIso2(x))
    .filter(Boolean);
}

function toCsv(list: string[]): string {
  const uniq = Array.from(new Set(list.map((x) => normIso2(x)).filter(Boolean)));
  return uniq.join(",");
}

export default function JurisdictionControlsCard(props: {
  config: any;
  setConfig: React.Dispatch<React.SetStateAction<any>>;
  setConfigChanged: React.Dispatch<React.SetStateAction<boolean>>;
}) {
  const { config, setConfig, setConfigChanged } = props;

  const [open, setOpen] = useState(false);

  const { data } = useQuery({
    queryKey: ["/api/meta/countries"],
    queryFn: async () => {
      const res = await axios.get("/api/meta/countries");
      return (res.data?.rows ?? []) as CountryRow[];
    },
    staleTime: 10 * 60 * 1000,
  });

  const countries = data ?? [];

  const restrictedList = useMemo(
    () => parseCsv(config?.jurisdictionRestrictedIso2Csv ?? ""),
    [config?.jurisdictionRestrictedIso2Csv]
  );

  const restrictedSet = useMemo(() => new Set(restrictedList), [restrictedList]);

  function updateField<K extends string>(key: K, value: any) {
    setConfig((prev: any) => ({ ...prev, [key]: value }));
    setConfigChanged(true);
  }

  function addRestricted(code: string) {
    const iso2 = normIso2(code);
    if (!iso2) return;
    const next = toCsv([...restrictedList, iso2]);
    updateField("jurisdictionRestrictedIso2Csv", next);
  }

  function removeRestricted(code: string) {
    const iso2 = normIso2(code);
    const next = toCsv(restrictedList.filter((c) => c !== iso2));
    updateField("jurisdictionRestrictedIso2Csv", next);
  }

  const labelFor = (iso2: string) => {
    const found = countries.find((c) => c.code.toUpperCase() === iso2);
    return found ? `${found.name} (${iso2})` : iso2;
  };

  return (
    <Card>
      <CardHeader>
        <CardTitle>Jurisdiction Controls (Sanctions / Restricted Countries)</CardTitle>
      </CardHeader>

      <CardContent className="space-y-6">
        <div className="grid gap-4 md:grid-cols-2">
          <div className="flex items-center justify-between gap-3 rounded-md border p-3">
            <div>
              <Label className="block">Block Signups</Label>
              <div className="text-sm text-muted-foreground">
                Deny new registrations from restricted jurisdictions.
              </div>
            </div>
            <Switch
              checked={Boolean(config?.jurisdictionBlockSignup)}
              onCheckedChange={(v) => updateField("jurisdictionBlockSignup", v)}
            />
          </div>

          <div className="flex items-center justify-between gap-3 rounded-md border p-3">
            <div>
              <Label className="block">Block Logins</Label>
              <div className="text-sm text-muted-foreground">
                Deny login and revoke active sessions from restricted jurisdictions.
              </div>
            </div>
            <Switch
              checked={Boolean(config?.jurisdictionBlockLogin)}
              onCheckedChange={(v) => updateField("jurisdictionBlockLogin", v)}
            />
          </div>

          <div className="flex items-center justify-between gap-3 rounded-md border p-3">
            <div>
              <Label className="block">Enforce By IP Geolocation</Label>
              <div className="text-sm text-muted-foreground">
                Uses server-side IP geo (VPNs may trigger false positives).
              </div>
            </div>
            <Switch
              checked={Boolean(config?.jurisdictionBlockByIp)}
              onCheckedChange={(v) => updateField("jurisdictionBlockByIp", v)}
            />
          </div>

          <div className="flex items-center justify-between gap-3 rounded-md border p-3">
            <div>
              <Label className="block">Enforce By Country Selected on Signup</Label>
              <div className="text-sm text-muted-foreground">
                Uses the country selected on signup (stored on the user profile).
              </div>
            </div>
            <Switch
              checked={Boolean(config?.jurisdictionBlockByCountrySelection)}
              onCheckedChange={(v) => updateField("jurisdictionBlockByCountrySelection", v)}
            />
          </div>
        </div>

        <div className="space-y-3">
          <Label>Restricted Countries</Label>

          <div className="flex flex-wrap gap-2">
            {restrictedList.length === 0 ? (
              <div className="text-sm text-muted-foreground">No restricted countries configured.</div>
            ) : (
              restrictedList.map((iso2) => (
                <Badge key={iso2} variant="secondary" className="flex items-center gap-2">

          </div>

          <div className="flex flex-col gap-3 md:flex-row">

export function csvToIso2List(csv: string | null | undefined): string[] {
  if (!csv) return [];
  return String(csv)
    .split(',')
    .map((s) => s.trim().toUpperCase())
    .filter((s) => s.length === 2);
}

export function iso2ListToCsv(list: string[] | null | undefined): string {
  if (!list || !Array.isArray(list)) return '';
  const dedup = Array.from(new Set(list.map((s) => String(s).trim().toUpperCase()).filter((s) => s.length === 2)));
  return dedup.join(',');
}

export type JurisdictionBlockReasonCode =
  | "JURISDICTION_RESTRICTED_IP_GEO"
  | "JURISDICTION_RESTRICTED_SELECTED"
  | "JURISDICTION_RESTRICTED_BOTH";

export type JurisdictionDecision =
  | { allowed: true }
  | {
      allowed: false;
      code: "JURISDICTION_RESTRICTED";
      httpStatus: number;
      message: string;
      reasonCode: JurisdictionBlockReasonCode;
      blockedBy: Array<"IP_GEO" | "COUNTRY_SELECTED">;
      ipCountryIso2?: string;
      selectedCountryIso2?: string;
    };

export function evaluateSignupJurisdiction(args: { ipCountryIso2?: string | null; selectedCountryIso2?: string | null; }): JurisdictionDecision {
  const policy = getJurisdictionRestrictionPolicy();
  if (!policy.jurisdictionBlockSignup) return { allowed: true };

  const restricted = new Set(policy.countries.map((c) => c.toUpperCase()));
  const blockedBy: Array<"IP_GEO" | "COUNTRY_SELECTED"> = [];

  const ip = (args.ipCountryIso2 ?? "").trim().toUpperCase();
  const sel = (args.selectedCountryIso2 ?? "").trim().toUpperCase();

  if (policy.jurisdictionEnforceByIpGeo && ip && restricted.has(ip)) blockedBy.push("IP_GEO");
  if (policy.jurisdictionEnforceBySignupCountry && sel && restricted.has(sel)) blockedBy.push("COUNTRY_SELECTED");

  if (!blockedBy.length) return { allowed: true };

  const reasonCode: JurisdictionBlockReasonCode =
    blockedBy.length === 2 ? "JURISDICTION_RESTRICTED_BOTH" :
    blockedBy[0] === "IP_GEO" ? "JURISDICTION_RESTRICTED_IP_GEO" :
    "JURISDICTION_RESTRICTED_SELECTED";

  return {
    allowed: false,
    code: "JURISDICTION_RESTRICTED",
    httpStatus: 403,
    message: policy.message,
    reasonCode,
    blockedBy,
    ipCountryIso2: ip || undefined,
    selectedCountryIso2: sel || undefined,
  };
}

export function evaluateLoginJurisdiction(args: { ipCountryIso2?: string | null; userCountryIso2?: string | null; }): JurisdictionDecision {
  const policy = getJurisdictionRestrictionPolicy();
  if (!policy.jurisdictionBlockLogin) return { allowed: true };

  const restricted = new Set(policy.countries.map((c) => c.toUpperCase()));
  const blockedBy: Array<"IP_GEO" | "COUNTRY_SELECTED"> = [];

  const ip = (args.ipCountryIso2 ?? "").trim().toUpperCase();
  const usr = (args.userCountryIso2 ?? "").trim().toUpperCase();

  if (policy.jurisdictionEnforceByIpGeo && ip && restricted.has(ip)) blockedBy.push("IP_GEO");
  if (policy.jurisdictionEnforceBySignupCountry && usr && restricted.has(usr)) blockedBy.push("COUNTRY_SELECTED");

  if (!blockedBy.length) return { allowed: true };

  const reasonCode: JurisdictionBlockReasonCode =

function readHeaderIso2(req: Request): string | undefined {
  const raw =
    req.get("cf-ipcountry") ||
    req.get("x-vercel-ip-country") ||

/**
 * Applies jurisdiction login restrictions to already-authenticated sessions.
 * If admin toggles blocking after a user is logged in, this will terminate them on next API call.
 */
export async function jurisdictionSessionGuard(req: Request, res: Response, next: NextFunction) {
  try {
    const sess: any = req.session;

    // Only applies to authenticated sessions
    if (!sess?.userId) return next();

    // Do not block real admin sessions (but impersonation sessions have isAdmin=false already)
    if (sess.isAdmin && !sess.isImpersonating) return next();

    const policy = getJurisdictionRestrictionPolicy();
    if (!policy.jurisdictionBlockLogin) return next();

    // Resolve user country if needed
    let userCountryIso2: string | undefined =
      typeof sess.userCountryIso2 === "string" ? String(sess.userCountryIso2).trim().toUpperCase() : undefined;

    if (policy.jurisdictionEnforceBySignupCountry && !userCountryIso2) {
      const row = db
        .select({ countryIso2: users.countryIso2 })
        .from(users)
        .where(eq(users.id, sess.userId))
        .get() as any;

      userCountryIso2 = row?.countryIso2 ? String(row.countryIso2).trim().toUpperCase() : undefined;
      sess.userCountryIso2 = userCountryIso2;
    }

    // Resolve IP country if needed
    let ipCountryIso2: string | undefined =
      typeof sess.ipCountryIso2 === "string" ? String(sess.ipCountryIso2).trim().toUpperCase() : undefined;

    if (policy.jurisdictionEnforceByIpGeo) {
      ipCountryIso2 = readHeaderIso2(req) ?? (() => {
        const ip = getClientIp(req);
        const geo = buildGeoContext(ip);
        return geo.countryCode ? String(geo.countryCode).trim().toUpperCase() : undefined;
      })();

      sess.ipCountryIso2 = ipCountryIso2;
    }

    const decision = evaluateLoginJurisdiction({
      ipCountryIso2,
      userCountryIso2,
    });

    if (decision.allowed) return next();

    // Revoke session in user_sessions (best-effort) + destroy cookie session
    try {
      revokeSession({
        actorUserId: Number(sess.userId),
        targetUserId: Number(sess.userId),
        sessionId: req.sessionID,
        reason: decision.reasonCode,
      });
    } catch {}

    try {
      req.session.destroy(() => {});
    } catch {}

    return res.status(decision.httpStatus).json({
      message: decision.message,
      code: decision.code,
      reasonCode: decision.reasonCode,
      blockedBy: decision.blockedBy,
      ipCountryIso2: decision.ipCountryIso2 ?? null,
      userCountryIso2: decision.selectedCountryIso2 ?? null,
    });
  } catch (e) {
    console.error("[Jurisdiction] session guard error:", e);
    return next(); // fail-open for middleware errors only (policy eval itself fails closed at login/signup)
  }
}

// POST /api/admin/system-config/jurisdiction-enforcement/revoke-active
// Revokes all currently-active sessions whose session geo or user-selected country is now restricted.
// This forces logout without waiting for the user’s next API call.
adminSystemConfigRouter.post("/jurisdiction-enforcement/revoke-active", (req, res) => {
  try {
    const adminUserId = Number((req.session as any)?.userId || 0);
    const policy = getJurisdictionRestrictionPolicy();

    if (!policy.jurisdictionBlockLogin) {
      return res.status(400).json({
        ok: false,
        error: "jurisdictionBlockLogin is disabled. Enable it first to enforce login blocking.",
      });
    }

    const restricted = (policy.countries || []).map((c) => String(c).toUpperCase());
    if (restricted.length === 0) {
      return res.json({ ok: true, revoked: 0, restricted: [] });
    }

    const rows = db
      .select({
        sessionId: userSessions.sessionId,
        userId: userSessions.userId,
        sessionCountry: userSessions.countryCode,
        userCountry: users.countryIso2,
        isAdmin: users.isAdmin,
      })
      .from(userSessions)
      .leftJoin(users, eq(users.id, userSessions.userId))
      .where(isNull(userSessions.revokedAt))
      .all();

    let revoked = 0;

    for (const r of rows) {
      if (!r.sessionId || !r.userId) continue;
      if (r.isAdmin) continue; // avoid locking admins out by accident

      const sessionIso2 = r.sessionCountry ? String(r.sessionCountry).toUpperCase() : null;
      const userIso2 = r.userCountry ? String(r.userCountry).toUpperCase() : null;

      const blockedByIp =
        policy.jurisdictionEnforceByIpGeo && sessionIso2 && restricted.includes(sessionIso2);

      const blockedByUserCountry =
        policy.jurisdictionEnforceBySignupCountry && userIso2 && restricted.includes(userIso2);

      if (!blockedByIp && !blockedByUserCountry) continue;

      try {
        revokeSession({
          actorUserId: adminUserId || 0,
          targetUserId: r.userId,
          sessionId: r.sessionId,
          reason: "JURISDICTION_POLICY_ENFORCEMENT",
        });
        revoked++;
      } catch {
        // ignore individual revoke failure
      }
    }

    return res.json({ ok: true, revoked, restricted });
  } catch (e: any) {
    return res.status(400).json({ ok: false, error: e?.message || "Failed to revoke sessions." });
  }
});

import type { Request, Response, NextFunction } from "express";
import { db } from "@db";
import { users } from "@shared/schema";
import { eq } from "drizzle-orm";
import { evaluateLoginJurisdiction } from "../policy/jurisdictionControl";
import { getClientIp, buildGeoContext, revokeSession } from "../security/sessionTrail";

export function jurisdictionSessionGuard(req: Request, res: Response, next: NextFunction) {
  try {
    const userId = Number((req.session as any)?.userId || 0);
    if (!userId) return next();

    // Do not lock admins out of admin remediation
    if ((req.session as any)?.isAdmin) return next();

    const ip = getClientIp(req);
    const geo = buildGeoContext(ip);
    const ipCountryIso2 = geo?.countryCode ? String(geo.countryCode).toUpperCase() : null;

    let userCountryIso2 = (req.session as any)?.userCountryIso2
      ? String((req.session as any).userCountryIso2).toUpperCase()
      : null;

    // Backfill from DB if session doesn’t have it (older sessions)
    if (!userCountryIso2) {
      const row = db
        .select({ countryIso2: users.countryIso2 })
        .from(users)
        .where(eq(users.id, userId))
        .get();
      userCountryIso2 = row?.countryIso2 ? String(row.countryIso2).toUpperCase() : null;
    }

    const decision = evaluateLoginJurisdiction({ ipCountryIso2, userCountryIso2 });

    if (decision.allowed) return next();

    // Force revoke + destroy session (blocks active logins)
    const sessionId = req.sessionID;
    if (sessionId) {
      try {
        revokeSession({
          actorUserId: 0, // system
          targetUserId: userId,
          sessionId,
          reason: `JURISDICTION_BLOCKED:${decision.reasonCode || decision.code || "RESTRICTED"}`,
        });
      } catch {
        // ignore
      }
    }

    try {
      req.session.destroy(() => {});
    } catch {
      // ignore
    }

    return res.status(decision.httpStatus || 403).json({
      message: decision.message,
      code: decision.code,
      reasonCode: decision.reasonCode,
      blockedBy: decision.blockedBy,
      ipCountryIso2: decision.ipCountryIso2,
      userCountryIso2: decision.selectedCountryIso2,
    });
  } catch {
    // Fail open (do not accidentally break the platform)
    return next();
  }
}

// server/legal/regionRules.ts
import { db } from "@db";
import { eq } from "drizzle-orm";
import { systemConfig } from "@shared/schema";

export type RestrictedCountryPolicy = {
  restrictedIso2: string[];
  message: string;
};

export type JurisdictionRestrictionPolicy = RestrictedCountryPolicy & {
  enforceBySignupCountry: boolean;
  enforceByIpGeo: boolean;
  blockSignup: boolean;
  blockLogin: boolean;
  updatedAtMs: number;
  updatedBy?: string | null;
};

const DEFAULT_RESTRICTED = ["KP", "IR", "CU", "SY"];
const DEFAULT_MESSAGE = "This jurisdiction is not supported due to regulatory restrictions.";

const CACHE_TTL_MS = 30_000;
let cached: { at: number; value: JurisdictionRestrictionPolicy } | null = null;

function toBool(value: unknown, fallback: boolean): boolean {
  if (typeof value === "boolean") return value;
  if (value === 1 || value === "1" || String(value).toLowerCase() === "true") return true;
  if (value === 0 || value === "0" || String(value).toLowerCase() === "false") return false;
  return fallback;
}

export function parseRestrictedCountriesCsv(csv: string): string[] {
  const tokens = String(csv || "")
    .split(/[,\s]+/g)
    .map((s) => s.trim().toUpperCase())
    .filter(Boolean);

  const out: string[] = [];
  const seen = new Set<string>();

  for (const t of tokens) {
    if (!/^[A-Z]{2}$/.test(t)) continue;
    if (seen.has(t)) continue;
    seen.add(t);
    out.push(t);
  }

  return out.length ? out : DEFAULT_RESTRICTED.slice();
}

export function invalidateJurisdictionRestrictionPolicyCache() {
  cached = null;
}

export function getJurisdictionRestrictionPolicy(): JurisdictionRestrictionPolicy {
  const now = Date.now();
  if (cached && now - cached.at < CACHE_TTL_MS) return cached.value;

  const row = db
    .select()
    .from(systemConfig)
    .where(eq(systemConfig.id, 1))
    .get() as any;

  const restrictedIso2 = parseRestrictedCountriesCsv(String(row?.jurisdictionRestrictedIso2Csv ?? ""));
  const message = String(row?.jurisdictionRestrictedMessage ?? DEFAULT_MESSAGE);

  const policy: JurisdictionRestrictionPolicy = {
    restrictedIso2,
    message,
    enforceBySignupCountry: toBool(row?.jurisdictionEnforceBySignupCountry, true),
    enforceByIpGeo: toBool(row?.jurisdictionEnforceByIpGeo, false),
    blockSignup: toBool(row?.jurisdictionBlockSignup, true),
    blockLogin: toBool(row?.jurisdictionBlockLogin, true),
    updatedAtMs: row?.updatedAt ? new Date(row.updatedAt as any).getTime() : 0,
    updatedBy: row?.updatedBy ?? null,
  };

  cached = { at: now, value: policy };
  return policy;
}

// Backwards-compatible exports used elsewhere in the legal system:
export function getRestrictedCountryPolicy(): RestrictedCountryPolicy {
  const p = getJurisdictionRestrictionPolicy();
  return { restrictedIso2: p.restrictedIso2, message: p.message };
}

// IMPORTANT: this now respects the admin toggle "enforce by signup country"
export function isRestrictedCountry(countryCode: string): boolean {
  const p = getJurisdictionRestrictionPolicy();
  if (!p.enforceBySignupCountry) return false;
  const iso2 = String(countryCode || "").trim().toUpperCase();
  return iso2.length === 2 && p.restrictedIso2.includes(iso2);
}

// server/policy/jurisdictionControl.ts
import { getJurisdictionRestrictionPolicy } from "../legal/regionRules";

export type JurisdictionDecision = {
  allowed: boolean;
  code?: "JURISDICTION_RESTRICTED";
  message?: string;
  matches?: {
    signupCountryIso2?: string | null;
    ipCountryIso2?: string | null;
    restrictedBySignupCountry?: boolean;
    restrictedByIpGeo?: boolean;
  };
};

function normIso2(v: unknown): string | null {
  const s = String(v ?? "").trim().toUpperCase();
  return /^[A-Z]{2}$/.test(s) ? s : null;
}

export function evaluateSignupJurisdiction(args: {
  signupCountryIso2?: string | null;
  ipCountryIso2?: string | null;
}): JurisdictionDecision {
  const p = getJurisdictionRestrictionPolicy();

  const signupIso2 = normIso2(args.signupCountryIso2);
  const ipIso2 = normIso2(args.ipCountryIso2);

  const restrictedBySignupCountry =
    p.enforceBySignupCountry && !!signupIso2 && p.restrictedIso2.includes(signupIso2);

  const restrictedByIpGeo =
    p.enforceByIpGeo && !!ipIso2 && p.restrictedIso2.includes(ipIso2);

  const hit = restrictedBySignupCountry || restrictedByIpGeo;

  if (p.blockSignup && hit) {
    return {
      allowed: false,
      code: "JURISDICTION_RESTRICTED",
      message: p.message,
      matches: {
        signupCountryIso2: signupIso2,
        ipCountryIso2: ipIso2,
        restrictedBySignupCountry,
        restrictedByIpGeo,
      },
    };
  }

  return {
    allowed: true,
    matches: {
      signupCountryIso2: signupIso2,
      ipCountryIso2: ipIso2,
      restrictedBySignupCountry,
      restrictedByIpGeo,
    },
  };
}

export function evaluateLoginJurisdiction(args: {
  signupCountryIso2?: string | null;
  ipCountryIso2?: string | null;
}): JurisdictionDecision {
  const p = getJurisdictionRestrictionPolicy();

  const signupIso2 = normIso2(args.signupCountryIso2);
  const ipIso2 = normIso2(args.ipCountryIso2);

  const restrictedBySignupCountry =
    p.enforceBySignupCountry && !!signupIso2 && p.restrictedIso2.includes(signupIso2);

  const restrictedByIpGeo =
    p.enforceByIpGeo && !!ipIso2 && p.restrictedIso2.includes(ipIso2);

  const hit = restrictedBySignupCountry || restrictedByIpGeo;

  if (p.blockLogin && hit) {
    return {
      allowed: false,
      code: "JURISDICTION_RESTRICTED",
      message: p.message,
      matches: {
        signupCountryIso2: signupIso2,
        ipCountryIso2: ipIso2,
        restrictedBySignupCountry,
        restrictedByIpGeo,
      },
    };
  }

  return {
    allowed: true,
    matches: {
      signupCountryIso2: signupIso2,
      ipCountryIso2: ipIso2,
      restrictedBySignupCountry,
      restrictedByIpGeo,
    },
  };
}

adminSystemConfigRouter.post(
  "/jurisdiction-enforcement/revoke-active",
  requireAdmin,
  async (req: Request, res: Response) => {
    try {
      const actor = String(req.session?.email || "admin");

      const rows = await db
        .select({
          sessionId: userSessions.sessionId,
          userId: userSessions.userId,

              adminId: null,
              adminEmail: actor,
              reason: "JURISDICTION_ENFORCEMENT",

            revoked++;

            // continue
          }
        }
      }

      return res.json({ ok: true, revoked, scanned: rows.length });

import React, { useMemo, useState, useCallback } from "react";
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
import { Checkbox } from "@/components/ui/checkbox";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Input } from "@/components/ui/input";
import { Textarea } from "@/components/ui/textarea";
import { useToast } from "@/hooks/use-toast";
import { useMutation, useQuery } from "@tanstack/react-query";
import { apiRequest } from "@/lib/queryClient";
import { X, ShieldAlert, RefreshCcw } from "lucide-react";

type CountryRow = { iso2: string; name: string };

type SystemConfigLike = {
  jurisdictionRestrictedIso2Csv: string;
  jurisdictionRestrictedMessage: string;
  jurisdictionEnforceByIpGeo: boolean;
  jurisdictionEnforceBySignupCountry: boolean;
  jurisdictionBlockSignup: boolean;
  jurisdictionBlockLogin: boolean;
};

function normalizeIso2List(raw: string): string[] {
  const toks = String(raw || "")
    .split(/[,\s]+/g)
    .map((s) => s.trim().toUpperCase())
    .filter(Boolean);

  const out: string[] = [];
  const seen = new Set<string>();

  for (const t of toks) {
    if (!/^[A-Z]{2}$/.test(t)) continue;
    if (seen.has(t)) continue;
    seen.add(t);
    out.push(t);
  }
  return out;
}

function toCsv(list: string[]): string {
  return list.join(",");
}

export function JurisdictionControlsCard(props: {
  config: SystemConfigLike;
  setConfig: React.Dispatch<React.SetStateAction<any>>;
  setConfigChanged: (v: boolean) => void;
  configChanged: boolean;
  onSave: () => void;
  saving: boolean;
}) {
  const { toast } = useToast();

  const { data: countries } = useQuery<CountryRow[]>({
    queryKey: ["/api/meta/countries"],
  });

  const countryNameByIso2 = useMemo(() => {
    const map = new Map<string, string>();
    (countries || []).forEach((c) => map.set(String(c.iso2).toUpperCase(), c.name));
    return map;
  }, [countries]);

  const restrictedList = useMemo(
    () => normalizeIso2List(props.config.jurisdictionRestrictedIso2Csv),
    [props.config.jurisdictionRestrictedIso2Csv]
  );

  const [addIso2, setAddIso2] = useState("");

  const setField = useCallback(
    (patch: Partial<SystemConfigLike>) => {
      props.setConfig((prev: any) => (prev ? { ...prev, ...patch } : prev));
      props.setConfigChanged(true);
    },
    [props]
  );

  const removeIso2 = useCallback(
    (iso2: string) => {
      const next = restrictedList.filter((x) => x !== iso2);
      setField({ jurisdictionRestrictedIso2Csv: toCsv(next) });
    },
    [restrictedList, setField]
  );

  const addCountry = useCallback(() => {
    const iso2 = String(addIso2 || "").trim().toUpperCase();
    if (!/^[A-Z]{2}$/.test(iso2)) {
      toast({ title: "Invalid ISO2", description: "Use a 2-letter ISO code (e.g., IR, KP).", variant: "destructive" });
      return;
    }
    const next = Array.from(new Set([...restrictedList, iso2]));
    setField({ jurisdictionRestrictedIso2Csv: toCsv(next) });
    setAddIso2("");
  }, [addIso2, restrictedList, setField, toast]);

  const revokeMutation = useMutation({
    mutationFn: async () => {
      const res = await apiRequest("POST", "/api/admin/system-config/jurisdiction-enforcement/revoke-active");
      return await res.json();
    },
    onSuccess: (data: any) => {
      toast({
        title: "Enforcement applied",
        description: `Scanned ${data?.scanned ?? "?"} sessions, revoked ${data?.revoked ?? 0}.`,
      });
    },
    onError: (e: any) => {
      toast({
        title: "Enforcement failed",
        description: String(e?.message || e),
        variant: "destructive",
      });
    },
  });

  return (
    <Card className="mt-6">
      <CardHeader>
        <CardTitle className="flex items-center gap-2">
          <ShieldAlert className="h-5 w-5" />
          Jurisdiction Controls (Sanctions / Restricted Countries)
        </CardTitle>
        <CardDescription>
          Block signups and/or logins for specific jurisdictions. Enforcement can be based on IP geolocation, signup-selected country, or both.
          VPNs can affect IP-based enforcement.
        </CardDescription>
      </CardHeader>

      <CardContent className="space-y-6">
        <div className="space-y-2">
          <div className="text-sm font-medium">Restricted Countries (ISO2)</div>

          <div className="flex gap-2">
            <Input
              value={props.config.jurisdictionRestrictedIso2Csv}
              onChange={(e) => setField({ jurisdictionRestrictedIso2Csv: e.target.value })}
              placeholder="e.g., KP,IR,CU,SY"
            />
          </div>

          <div className="flex gap-2">
            <Input
              value={addIso2}
              onChange={(e) => setAddIso2(e.target.value)}
              placeholder="Add ISO2 (e.g., IR)"
              className="max-w-[220px]"
            />
            <Button type="button" onClick={addCountry} variant="secondary">
              Add
            </Button>
          </div>

          <div className="flex flex-wrap gap-2 pt-2">
            {restrictedList.length === 0 ? (
              <div className="text-sm text-muted-foreground">No restricted countries configured.</div>
            ) : (
              restrictedList.map((iso2) => (
                <Badge key={iso2} variant="outline" className="flex items-center gap-2">
                  <span>
                    {iso2}
                    {countryNameByIso2.get(iso2) ? ` — ${countryNameByIso2.get(iso2)}` : ""}
                  </span>
                  <button
                    type="button"
                    onClick={() => removeIso2(iso2)}
                    className="rounded hover:bg-muted p-0.5"
                    aria-label={`Remove ${iso2}`}
                  >
                    <X className="h-3 w-3" />
                  </button>
                </Badge>
              ))
            )}
          </div>
        </div>

        <div className="space-y-2">
          <div className="text-sm font-medium">Block Message</div>
          <Textarea
            value={props.config.jurisdictionRestrictedMessage}
            onChange={(e) => setField({ jurisdictionRestrictedMessage: e.target.value })}
            placeholder="Message shown to users who are blocked."
          />
        </div>

        <div className="grid gap-3 md:grid-cols-2">
          <label className="flex items-start gap-2">
            <Checkbox
              checked={props.config.jurisdictionEnforceBySignupCountry}
              onCheckedChange={(v) => setField({ jurisdictionEnforceBySignupCountry: Boolean(v) })}
            />
            <span className="text-sm">
              <span className="font-medium">Enforce by signup-selected country</span>

                Uses the country the user selects during signup (stored on the user profile).

          </label>

          <label className="flex items-start gap-2">
            <Checkbox
              checked={props.config.jurisdictionEnforceByIpGeo}
              onCheckedChange={(v) => setField({ jurisdictionEnforceByIpGeo: Boolean(v) })}
            />
            <span className="text-sm">

import { useState, useEffect, useMemo, useCallback } from "react";
import { AlertTriangle, Globe, CheckCircle, XCircle } from "lucide-react";
import { Alert, AlertDescription, AlertTitle } from "@/components/ui/alert";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Label } from "@/components/ui/label";
import { fetchWithIdentity } from "@/lib/fetchWithIdentity";
import { useQuery } from "@tanstack/react-query";

type CountryRow = { iso2: string; name: string };

type AvailabilityResp = {
  ok?: boolean;
  available?: boolean;
  enforce?: boolean;
  code?: string;
  reason?: string | null;
  countryCode?: string;
  termsExist?: boolean;
  signupAllowed?: boolean;
  restricted?: boolean;
  scopeKey?: string | null;
  fallbackUsed?: boolean;
  enforced?: boolean;
  message?: string;
  selected?: {
    globalTarget: { docSet: string; docType: string; jurisdictionType: string; jurisdictionKey: string };
    addendumTarget: { docSet: string; docType: string; jurisdictionType: string; jurisdictionKey: string };
  };
  modes?: { global: string; addendum: string };
  warnings?: string[];
};

interface SignupAvailabilityGateProps {
  onCountryChange: (country: string, available: boolean) => void;
  selectedCountry?: string;
}

export function SignupAvailabilityGate({ onCountryChange, selectedCountry }: SignupAvailabilityGateProps) {
  const [country, setCountry] = useState(selectedCountry || "");
  const [loading, setLoading] = useState(false);
  const [data, setData] = useState<AvailabilityResp | null>(null);
  const [error, setError] = useState<string | null>(null);

  const { data: countries } = useQuery<CountryRow[]>({
    queryKey: ["/api/meta/countries"],
  });

  const sortedCountries = useMemo(() => {
    const list = (countries || []).map((c) => ({
      code: String(c.iso2).toUpperCase(),
      name: c.name,
    }));
    list.sort((a, b) => a.name.localeCompare(b.name));
    return list;
  }, [countries]);

  const countryIso2 = useMemo(() => String(country || "").trim().toUpperCase(), [country]);

  useEffect(() => {
    if (!countryIso2 || countryIso2.length !== 2) {
      setData(null);
      return;
    }

    let cancelled = false;

    (async () => {
      setLoading(true);
      setError(null);
      try {
        const res = await fetchWithIdentity(`/api/legal/doc1/availability?country=${countryIso2}`);
        const j = (await res.json()) as AvailabilityResp;
        if (cancelled) return;

        setData(j);
        const isAvailable = Boolean(j.signupAllowed);
        onCountryChange(countryIso2, isAvailable);
      } catch {
        if (cancelled) return;
        setError("Unable to verify jurisdiction. Please try again.");
        onCountryChange(countryIso2, false);
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();

    return () => {
      cancelled = true;
    };
  }, [countryIso2, onCountryChange]);

  const handleCountryChange = (value: string) => {
    setCountry(value);
  };

  const available = Boolean(data?.signupAllowed);
  const restricted = Boolean(data?.restricted);
  const fallbackUsed = Boolean(data?.fallbackUsed);

  return (
    <div className="space-y-4">
      <div>
        <Label htmlFor="country">Your Country</Label>
        <Select value={country} onValueChange={handleCountryChange}>
          <SelectTrigger id="country" className="w-full">
            <SelectValue placeholder="Select your country" />
          </SelectTrigger>
          <SelectContent className="max-h-[300px]">
            {sortedCountries.map((c) => (
              <SelectItem key={c.code} value={c.code}>
                {c.name}
              </SelectItem>
            ))}
          </SelectContent>
        </Select>
      </div>

      {loading && (
        <div className="flex items-center gap-2 text-muted-foreground">
          <Globe className="w-4 h-4 animate-spin" />
          <span>Checking jurisdiction...</span>
        </div>
      )}

      {error && (
        <Alert variant="destructive">
          <AlertTriangle className="h-4 w-4" />
          <AlertTitle>Error</AlertTitle>
          <AlertDescription>{error}</AlertDescription>
        </Alert>
      )}

      {data && !loading && (
        <>
          {restricted && (
            <Alert variant="destructive">
              <XCircle className="h-4 w-4" />
              <AlertTitle>Jurisdiction Not Supported</AlertTitle>
              <AlertDescription>
                {data.message || "This jurisdiction is not supported due to regulatory restrictions."}
              </AlertDescription>
            </Alert>
          )}

          {!available && !restricted && (
            <Alert variant="destructive">
              <AlertTriangle className="h-4 w-4" />
              <AlertTitle>Terms Not Available</AlertTitle>
              <AlertDescription>
                {data.message || "Terms of service are not yet available for your jurisdiction. Please check back later."}
              </AlertDescription>
            </Alert>
          )}

          {available && fallbackUsed && (
            <Alert>
              <Globe className="h-4 w-4" />
              <AlertTitle>Using Global Terms</AlertTitle>
              <AlertDescription>
                {data.message || "Using global terms. Region-specific terms may be added later."}
              </AlertDescription>
            </Alert>
          )}

          {available && !fallbackUsed && !restricted && (
            <Alert className="border-green-200 bg-green-50 dark:border-green-800 dark:bg-green-950">
              <CheckCircle className="h-4 w-4 !text-green-600" />
              <AlertTitle className="text-green-800 dark:text-green-200">Jurisdiction Verified</AlertTitle>
              <AlertDescription className="text-green-700 dark:text-green-300">
                Terms of service are available for your location.
              </AlertDescription>
            </Alert>
          )}
        </>
      )}
    </div>
  );
}
