## Fused summary of what this deliverable implements

1. **Account state UX + logic (Active / Frozen / Disabled mini-tabs)**

   * Disabled users always appear in **Disabled** (even if also frozen).
   * Frozen users appear in **Frozen** only when **not disabled**.
   * Button rules:

     * **Frozen tab:** `Unfreeze` + `Disable`
     * **Disabled tab:** `Enable` (restores prior frozen state; if they were frozen then disabled, enable returns them to Frozen)
     * **Active tab:** `Freeze` + `Disable`

2. **User Timeline upgrade (vertical timeline with dot/line UI)**

   * Includes **LOGIN**, **LOGOUT**, and **SESSION_LENGTH** (computed) plus trade events.
   * CSV export includes: **Login Time, Login IP, Session Length, Logout Time** (plus event basics).

3. **Risk guardrails (backend, bug-resistant)**

   * Keeps: **max 3 concurrent trades per symbol** and **max 5 total concurrent trades** (counting OPEN + PENDING).
   * Adds: **max 50 lots concurrently across all open/pending trades** (platform-wide default).
   * Global limits are adjusted in **Trade Settings** and **persist**; **per-user overrides** can exceed global limits.

4. **Trade Audit upgrades (tamper-evident + correlation)**

   * Adds **correlationId** on every order attempt.
   * Adds **order_intent_audit** to capture **RECEIVED** and **DECISION (PASS/REJECT)**, including reject reason/check.
   * Adds **trade_audit** events for market order creation, pending placement, cancels, manual closes, and target updates.
   * Hash-chained entries via **SHA-256** (`prev_hash` → `event_hash`) for tamper-evidence.

  const rows = db.prepare(`PRAGMA table_info(${table})`).all() as Array<{ name: string }>;
  return rows.some((r) => r.name === column);
}

function ensureColumn(db: DB, table: string, ddlFragment: string, columnName: string) {
  if (columnExists(db, table, columnName)) return;
  db.exec(`ALTER TABLE ${table} ADD COLUMN ${ddlFragment}`);
}

import crypto from "node:crypto";
import type { DB } from "../db/db";

function stableStringify(value: any): string {
  if (value === null || value === undefined) return "null";
  if (typeof value !== "object") return JSON.stringify(value);
  if (Array.isArray(value)) return `[${value.map(stableStringify).join(",")}]`;
  const keys = Object.keys(value).sort();
  const entries = keys
    .filter((k) => value[k] !== undefined)
    .map((k) => `${JSON.stringify(k)}:${stableStringify(value[k])}`);
  return `{${entries.join(",")}}`;
}

function sha256Hex(input: string): string {
  return crypto.createHash("sha256").update(input).digest("hex");
}

export type AuditContext = {
  correlationId: string;
  actorType: "USER" | "ADMIN" | "SYSTEM";
  actorUserId?: string | null;
  sessionId?: string | null;
  ip?: string | null;
  userAgent?: string | null;
};

export function getOrCreateCorrelationId(existing?: string | null): string {
  return existing && existing.trim().length > 0 ? existing : crypto.randomUUID();
}

export function writeTradeAudit(db: DB, args: {
  tradeId: string;
  userId?: string | null;
  correlationId?: string | null;
  eventType: string;
  eventCategory?: string;
  ctx: AuditContext;
  eventAt?: number;
  payload?: any;
}) {
  const eventAt = args.eventAt ?? Date.now();
  const last = db.prepare(
    `SELECT event_hash FROM trade_audit WHERE trade_id = ? ORDER BY id DESC LIMIT 1`
  ).get(args.tradeId) as { event_hash?: string } | undefined;

  const prevHash = last?.event_hash ?? "GENESIS";
  const envelope = {
    tradeId: args.tradeId,
    correlationId: args.correlationId ?? args.ctx.correlationId ?? null,
    userId: args.userId ?? null,
    eventType: args.eventType,
    eventCategory: args.eventCategory ?? "TRADE",
    actorType: args.ctx.actorType,
    actorUserId: args.ctx.actorUserId ?? null,
    sessionId: args.ctx.sessionId ?? null,
    ip: args.ctx.ip ?? null,
    userAgent: args.ctx.userAgent ?? null,
    eventAt,
    payload: args.payload ?? null,
  };

  const eventHash = sha256Hex(prevHash + "\n" + stableStringify(envelope));

  db.prepare(`
    INSERT INTO trade_audit (
      trade_id, correlation_id, user_id, event_type, event_category,
      actor_type, actor_user_id, session_id, ip, user_agent,
      event_at, prev_hash, event_hash, payload_json
    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
  `).run(
    args.tradeId,
    envelope.correlationId,
    envelope.userId,
    envelope.eventType,
    envelope.eventCategory,
    envelope.actorType,
    envelope.actorUserId,
    envelope.sessionId,
    envelope.ip,
    envelope.userAgent,
    envelope.eventAt,
    prevHash,
    eventHash,
    JSON.stringify(envelope.payload ?? null)
  );
}

import type { Request, Response, NextFunction } from "express";
import { getOrCreateCorrelationId, type AuditContext } from "../audit/auditTrail";

declare global {
  namespace Express {
    interface Request {
      auditContext?: AuditContext;
    }
  }
}

export function attachAuditContext(req: Request, res: Response, next: NextFunction) {
  const headerCorr = req.header("x-correlation-id");
  const bodyCorr = (req.body && (req.body.correlationId || req.body.correlation_id)) || null;

  const correlationId = getOrCreateCorrelationId((headerCorr || bodyCorr) as string | null);

  // express-session puts an id on req.sessionID if configured
  const sessionId = (req as any).sessionID ?? null;

  req.auditContext = {
    correlationId,
    actorType: "USER",
    actorUserId: null,
    sessionId,
    ip: req.ip ?? null,
    userAgent: req.get("user-agent") ?? null,
  };

  res.setHeader("x-correlation-id", correlationId);
  next();
}

import type { Request, Response, NextFunction } from "express";
import type { DB } from "../db/db";

export function requireAuth(db: DB) {
  return (req: Request, res: Response, next: NextFunction) => {
    const userId = (req.session as any)?.userId as string | undefined;
    if (!userId) return res.status(401).json({ error: "UNAUTHENTICATED" });

    const user = db.prepare(`SELECT id, role, is_frozen, is_disabled FROM users WHERE id = ?`).get(userId) as any;
    if (!user) return res.status(401).json({ error: "UNAUTHENTICATED" });

    (req as any).user = user;
    next();
  };
}

export function requireAdmin() {
  return (req: Request, res: Response, next: NextFunction) => {
    const user = (req as any).user;
    if (!user || user.role !== "ADMIN") return res.status(403).json({ error: "FORBIDDEN" });
    next();
  };
}

import type { Request, Response, NextFunction } from "express";
import type { DB } from "../db/db";
import { writeOrderIntentAudit } from "../audit/auditTrail";

type TradeSettings = {
  max_trades_total: number;
  max_trades_per_symbol: number;
  max_lots_total: number;
  quote_max_age_ms: number;
};

function getTradeSettings(db: DB): TradeSettings {
  const row = db.prepare(`SELECT max_trades_total, max_trades_per_symbol, max_lots_total, quote_max_age_ms FROM trade_settings WHERE id = 1`).get() as any;
  return {
    max_trades_total: Number(row?.max_trades_total ?? 5),
    max_trades_per_symbol: Number(row?.max_trades_per_symbol ?? 3),
    max_lots_total: Number(row?.max_lots_total ?? 50),
    quote_max_age_ms: Number(row?.quote_max_age_ms ?? 15000),
  };
}

function getUserOverrides(db: DB, userId: string): Partial<TradeSettings> {
  const row = db.prepare(`SELECT risk_overrides_json FROM users WHERE id = ?`).get(userId) as any;
  if (!row?.risk_overrides_json) return {};
  try {
    const o = JSON.parse(row.risk_overrides_json);
    return {
      max_trades_total: o.max_trades_total ?? o.maxTradesTotal,
      max_trades_per_symbol: o.max_trades_per_symbol ?? o.maxTradesPerSymbol,
      max_lots_total: o.max_lots_total ?? o.maxLotsTotal,
      quote_max_age_ms: o.quote_max_age_ms ?? o.quoteMaxAgeMs,
    };
  } catch {
    return {};
  }
}

function effectiveSettings(global: TradeSettings, overrides: Partial<TradeSettings>): TradeSettings {
  return {
    max_trades_total: Number(overrides.max_trades_total ?? global.max_trades_total),
    max_trades_per_symbol: Number(overrides.max_trades_per_symbol ?? global.max_trades_per_symbol),
    max_lots_total: Number(overrides.max_lots_total ?? global.max_lots_total),
    quote_max_age_ms: Number(overrides.quote_max_age_ms ?? global.quote_max_age_ms),
  };
}

export function riskMiddleware(db: DB) {
  return (req: Request, res: Response, next: NextFunction) => {
    const user = (req as any).user as { id: string; is_frozen: number; is_disabled: number };
    const ctx = req.auditContext!;
    const body = req.body || {};

    const symbol = String(body.symbol || "");
    const side = String(body.side || "");
    const orderType = String(body.orderType || body.order_type || "");
    const lots = Number(body.lots ?? 0);

    const requestedPrice = body.requestedPrice != null ? Number(body.requestedPrice) : null;
    const triggerPrice = body.triggerPrice != null ? Number(body.triggerPrice) : null;
    const takeProfit = body.takeProfit != null ? Number(body.takeProfit) : null;
    const stopLoss = body.stopLoss != null ? Number(body.stopLoss) : null;

    const quoteBid = body.quoteBid != null ? Number(body.quoteBid) : null;
    const quoteAsk = body.quoteAsk != null ? Number(body.quoteAsk) : null;
    const quoteAt = body.quoteAt != null ? Number(body.quoteAt) : null;

    // 1) Log RECEIVED intent (pre-risk)
    try {
      writeOrderIntentAudit(db, {
        correlationId: ctx.correlationId,
        userId: user.id,
        symbol, side, orderType,
        lots,
        requestedPrice, triggerPrice, takeProfit, stopLoss,
        quoteBid, quoteAsk, quoteAt,
        quoteIsStale: false,
        eventType: "RECEIVED",
        ctx,
        payload: { path: req.path, method: req.method }
      });
    } catch {
      // do not block trading if audit insert fails
    }

    const reject = (rejectCheck: string, rejectReason: string, extra?: any) => {
      try {
        writeOrderIntentAudit(db, {
          correlationId: ctx.correlationId,
          userId: user.id,
          symbol, side, orderType,
          lots,
          requestedPrice, triggerPrice, takeProfit, stopLoss,
          quoteBid, quoteAsk, quoteAt,
          quoteIsStale: extra?.quoteIsStale ?? false,
          eventType: "DECISION",
          decision: "REJECT",
          rejectCheck,
          rejectReason,
          ctx,
          payload: extra ?? null,
        });
      } catch { /* ignore */ }

      return res.status(400).json({ error: "RISK_REJECT", check: rejectCheck, reason: rejectReason, correlationId: ctx.correlationId });
    };

    // 2) Account state checks
    if (user.is_disabled) return reject("ACCOUNT_DISABLED", "Account is disabled");
    if (user.is_frozen) return reject("ACCOUNT_FROZEN", "Account is frozen");

    // 3) Quote staleness check (optional but supported)
    const global = getTradeSettings(db);
    const overrides = getUserOverrides(db, user.id);
    const limits = effectiveSettings(global, overrides);

    if (quoteAt != null) {
      const age = Date.now() - quoteAt;
      if (age > limits.quote_max_age_ms) {
        return reject("STALE_QUOTE", `Quote age ${age}ms exceeds limit ${limits.quote_max_age_ms}ms`, { quoteIsStale: true, quoteAgeMs: age });
      }
    }

    // 4) Concurrency: count OPEN + PENDING
    const totalConcurrent = db.prepare(`
      SELECT COUNT(*) as c
      FROM trades
      WHERE user_id = ? AND status IN ('OPEN','PENDING')
    `).get(user.id) as any;

    if (Number(totalConcurrent?.c ?? 0) >= limits.max_trades_total) {
      return reject("MAX_TRADES_TOTAL", `Max total concurrent trades exceeded (${limits.max_trades_total})`, { current: Number(totalConcurrent?.c ?? 0) });
    }

    const perSymbolConcurrent = db.prepare(`
      SELECT COUNT(*) as c
      FROM trades
      WHERE user_id = ? AND symbol = ? AND status IN ('OPEN','PENDING')
    `).get(user.id, symbol) as any;

    if (Number(perSymbolConcurrent?.c ?? 0) >= limits.max_trades_per_symbol) {
      return reject("MAX_TRADES_PER_SYMBOL", `Max concurrent trades per symbol exceeded (${limits.max_trades_per_symbol})`, { symbol, current: Number(perSymbolConcurrent?.c ?? 0) });
    }

    // 5) NEW: total concurrent lots (OPEN + PENDING)
    const lotsRow = db.prepare(`
      SELECT COALESCE(SUM(lots), 0) as s
      FROM trades
      WHERE user_id = ? AND status IN ('OPEN','PENDING')
    `).get(user.id) as any;

    const existingLots = Number(lotsRow?.s ?? 0);
    const afterLots = existingLots + lots;

    if (afterLots > limits.max_lots_total) {
      return reject("MAX_LOTS_TOTAL", `Max concurrent lots exceeded (${limits.max_lots_total})`, { existingLots, requestedLots: lots, afterLots });
    }

    // PASS decision (write it once here; trade_id will be linked later)
    try {
      writeOrderIntentAudit(db, {
        correlationId: ctx.correlationId,
        userId: user.id,
        symbol, side, orderType,
        lots,
        requestedPrice, triggerPrice, takeProfit, stopLoss,
        quoteBid, quoteAsk, quoteAt,
        quoteIsStale: false,
        eventType: "DECISION",
        decision: "PASS",
        ctx,
      });
    } catch { /* ignore */ }

    next();
  };
}

import type { Request, Response, NextFunction } from "express";
import type { DB } from "../db/db";

export function recordLogin(db: DB, userId: string, req: Request) {
  const sessionId = (req as any).sessionID ?? null;
  const ip = req.ip ?? null;
  const ua = req.get("user-agent") ?? null;
  const now = Date.now();

  if (sessionId) {
    db.prepare(`
      INSERT INTO login_sessions (session_id, user_id, login_at, login_ip, user_agent)
      VALUES (?, ?, ?, ?, ?)
    `).run(sessionId, userId, now, ip, ua);

    db.prepare(`
      INSERT INTO timeline_events (user_id, event_type, event_at, ip, ref_id, meta_json)
      VALUES (?, 'LOGIN', ?, ?, ?, ?)
    `).run(userId, now, ip, sessionId, JSON.stringify({ userAgent: ua }));
  }
}

export function recordLogout(db: DB, userId: string, req: Request) {
  const sessionId = (req as any).sessionID ?? null;
  const ip = req.ip ?? null;
  const now = Date.now();

  if (!sessionId) return;

  db.prepare(`
    UPDATE login_sessions
    SET logout_at = ?
    WHERE user_id = ? AND session_id = ? AND logout_at IS NULL
  `).run(now, userId, sessionId);

  db.prepare(`
    INSERT INTO timeline_events (user_id, event_type, event_at, ip, ref_id, meta_json)
    VALUES (?, 'LOGOUT', ?, ?, ?, ?)

import { Router } from "express";
import type { DB } from "../db/db";
import crypto from "node:crypto";
import { recordLogin, recordLogout } from "../middleware/sessionTracking";

function hashPassword(pw: string): string {
  // Minimal (replace with bcrypt/argon2 in production)
  return crypto.createHash("sha256").update(pw).digest("hex");
}

export function authRoutes(db: DB) {
  const r = Router();

  r.post("/login", (req, res) => {
    const { email, password } = req.body || {};
    if (!email || !password) return res.status(400).json({ error: "BAD_REQUEST" });

    const user = db.prepare(`SELECT id, password_hash, role, is_frozen, is_disabled FROM users WHERE email = ?`).get(String(email)) as any;
    if (!user) return res.status(401).json({ error: "INVALID_CREDENTIALS" });

    if (user.password_hash !== hashPassword(String(password))) {
      return res.status(401).json({ error: "INVALID_CREDENTIALS" });
    }

    (req.session as any).userId = user.id;
    recordLogin(db, user.id, req);

    res.json({ ok: true, user: { id: user.id, role: user.role, isFrozen: !!user.is_frozen, isDisabled: !!user.is_disabled } });
  });

  r.post("/logout", (req, res) => {
    const userId = (req.session as any)?.userId;
    if (userId) recordLogout(db, userId, req);

    req.session.destroy((err) => {
      if (err) return res.status(500).json({ error: "LOGOUT_FAILED" });
      res.json({ ok: true });
    });
  });

  r.get("/me", (req, res) => {
    const userId = (req.session as any)?.userId;
    if (!userId) return res.status(401).json({ error: "UNAUTHENTICATED" });
    const user = db.prepare(`SELECT id, email, role, is_frozen, is_disabled FROM users WHERE id = ?`).get(userId) as any;
    if (!user) return res.status(401).json({ error: "UNAUTHENTICATED" });
    res.json({ id: user.id, email: user.email, role: user.role, isFrozen: !!user.is_frozen, isDisabled: !!user.is_disabled });
  });

  return r;
}

export function tradeRoutes(db: DB) {

  r.post("/trades", attachAuditContext, requireAuth(db), riskMiddleware(db), (req, res) => {
    const user = (req as any).user as { id: string };
    const ctx = req.auditContext!;
    const b = req.body || {};

    const tradeId = uuid();
    const now = Date.now();

    const symbol = String(b.symbol);
    const side = String(b.side);
    const orderType = String(b.orderType || "MARKET");
    const lots = Number(b.lots);

    const requestedPrice = b.requestedPrice != null ? Number(b.requestedPrice) : null;
    const triggerPrice = b.triggerPrice != null ? Number(b.triggerPrice) : null;
    const takeProfit = b.takeProfit != null ? Number(b.takeProfit) : null;
    const stopLoss = b.stopLoss != null ? Number(b.stopLoss) : null;

    const status = orderType === "MARKET" ? "OPEN" : "PENDING";
    const entryPrice = orderType === "MARKET" ? requestedPrice : null;

    db.prepare(`
      INSERT INTO trades (
        id, user_id, symbol, side, order_type, status, lots,
        requested_price, trigger_price, entry_price, take_profit, stop_loss,
        opened_at, correlation_id
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `).run(
      tradeId, user.id, symbol, side, orderType, status, lots,
      requestedPrice, triggerPrice, entryPrice, takeProfit, stopLoss,
      status === "OPEN" ? now : null,
      ctx.correlationId
    );

    // Link correlation decision rows to trade_id (best effort)
    try {
      db.prepare(`
        UPDATE order_intent_audit
        SET trade_id = COALESCE(trade_id, ?)
        WHERE correlation_id = ? AND user_id = ?
      `).run(tradeId, ctx.correlationId, user.id);

      // Optional: add a DECISION payload that now includes tradeId (append-only)
      writeOrderIntentAudit(db, {
        correlationId: ctx.correlationId,
        tradeId,
        userId: user.id,
        symbol, side, orderType, lots,
        requestedPrice, triggerPrice, takeProfit, stopLoss,
        eventType: "DECISION",
        decision: "PASS",
        ctx,
        payload: { linkedTradeId: true }
      });
    } catch { /* ignore */ }

    // Trade audit events
    writeTradeAudit(db, {
      tradeId,
      userId: user.id,
      correlationId: ctx.correlationId,
      eventType: "ORDER_PLACED",
      ctx: { ...ctx, actorType: "USER", actorUserId: user.id },

    });

    if (status === "OPEN") {
      writeTradeAudit(db, {
        tradeId,
        userId: user.id,
        correlationId: ctx.correlationId,
        eventType: "ORDER_FILLED",
        ctx: { ...ctx, actorType: "SYSTEM" },
        eventAt: now,
        payload: { entryPrice: entryPrice, filledAt: now }
      });

        INSERT INTO timeline_events (user_id, event_type, event_at, ip, ref_id, meta_json)
        VALUES (?, 'TRADE_OPEN', ?, ?, ?, ?)

    if (status === "active") where += " AND is_disabled = 0 AND is_frozen = 0";

import { Router } from "express";
import type { DB } from "../db/db";
import { requireAuth, requireAdmin } from "../middleware/auth";

export function userTimelineRoutes(db: DB) {
  const r = Router();

  // Returns unified timeline with session length computed
  r.get("/admin/users/:id/timeline", requireAuth(db), requireAdmin(), (req, res) => {
    const userId = String(req.params.id);

    const events = db.prepare(`
      SELECT id, event_type, event_at, ip, ref_id, meta_json
      FROM timeline_events
      WHERE user_id = ?
      ORDER BY event_at DESC
      LIMIT 1000
    `).all(userId) as any[];

    // Pull sessions to compute session length
    const sessions = db.prepare(`
      SELECT session_id, login_at, login_ip, logout_at
      FROM login_sessions
      WHERE user_id = ?
      ORDER BY id DESC
      LIMIT 200
    `).all(userId) as any[];

    const sessionMap = new Map<string, any>();
    for (const s of sessions) sessionMap.set(s.session_id, s);

    const enriched = events.map((e) => {
      if (e.event_type === "LOGIN" || e.event_type === "LOGOUT") {
        const sess = e.ref_id ? sessionMap.get(e.ref_id) : null;
        const loginAt = sess?.login_at ?? null;
        const logoutAt = sess?.logout_at ?? null;
        const sessionLengthMs = (loginAt && logoutAt) ? (logoutAt - loginAt) : null;
        return {
          ...e,
          loginAt,
          loginIp: sess?.login_ip ?? e.ip ?? null,
          logoutAt,
          sessionLengthMs,
        };
      }
      return e;
    });

    res.json(enriched);
  });

  return r;
}

export async function api<T>(path: string, opts: RequestInit = {}): Promise<T> {
  const res = await fetch(`/api${path}`, {
    ...opts,
    headers: {
      "Content-Type": "application/json",
      ...(opts.headers || {}),
    },
    credentials: "include",
  });

  const text = await res.text();
  const data = text ? JSON.parse(text) : null;

  if (!res.ok) {
    const msg = data?.error || `HTTP_${res.status}`;
    throw Object.assign(new Error(msg), { data, status: res.status });
  }
  return data as T;
}

export function toCsv(rows: Array<Record<string, any>>, columns: string[]): string {
  const esc = (v: any) => {
    if (v === null || v === undefined) return "";
    const s = String(v);
    if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`;
    return s;
  };
  const header = columns.map(esc).join(",");
  const lines = rows.map(r => columns.map(c => esc(r[c])).join(","));
  return [header, ...lines].join("\n");
}

export function downloadCsv(filename: string, csv: string) {
  const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
}

      await api(`/admin/users/${userId}/${type}`, { method: "POST", body: JSON.stringify(body || {}) });
      await load();
    } finally {
      setBusy(null);
    }
  };

  const header = useMemo(() => {
    if (tab === "active") return "Active";
    if (tab === "frozen") return "Frozen";
    return "Disabled";
  }, [tab]);

  return (
    <div>
      <h2>User Management</h2>

      <div style={{ display: "flex", gap: 10, margin: "10px 0" }}>
        <button onClick={() => setTab("active")} disabled={tab === "active"}>Active</button>
        <button onClick={() => setTab("frozen")} disabled={tab === "frozen"}>Frozen</button>
        <button onClick={() => setTab("disabled")} disabled={tab === "disabled"}>Disabled</button>
      </div>

      <div style={{ border: "1px solid #eee", borderRadius: 8, padding: 10 }}>
        <div style={{ marginBottom: 8, color: "#333" }}>
          <strong>{header}</strong> accounts
        </div>

        <table style={{ width: "100%", borderCollapse: "collapse" }}>
          <thead>
            <tr style={{ textAlign: "left", borderBottom: "1px solid #eee" }}>
              <th style={{ padding: 8 }}>Email</th>
              <th style={{ padding: 8 }}>Role</th>
              <th style={{ padding: 8 }}>State</th>
              <th style={{ padding: 8 }}>Actions</th>
            </tr>
          </thead>
          <tbody>
            {rows.map(u => {
              const state = u.is_disabled ? "DISABLED" : (u.is_frozen ? "FROZEN" : "ACTIVE");
              return (
                <tr key={u.id} style={{ borderBottom: "1px solid #f5f5f5" }}>
                  <td style={{ padding: 8 }}>{u.email}</td>
                  <td style={{ padding: 8 }}>{u.role}</td>
                  <td style={{ padding: 8 }}>
                    {state}
                    {u.freeze_reason ? <span style={{ marginLeft: 8, color: "#666" }}>({u.freeze_reason})</span> : null}
                  </td>
                  <td style={{ padding: 8, display: "flex", gap: 8, flexWrap: "wrap" }}>
                    <button onClick={() => onOpenTimeline(u.id)}>Timeline</button>

                    {/* Active tab actions */}
                    {!u.is_disabled && !u.is_frozen && (
                      <>
                        <button
                          disabled={busy === `${u.id}:freeze`}
                          onClick={() => action(u.id, "freeze", { reason: "Admin freeze" })}
                        >
                          Freeze
                        </button>
                        <button
                          disabled={busy === `${u.id}:disable`}
                          onClick={() => action(u.id, "disable")}
                        >
                          Disable
                        </button>
                      </>
                    )}

                    {/* Frozen tab actions: Unfreeze + Disable */}
                    {!u.is_disabled && u.is_frozen && (
                      <>
                        <button
                          disabled={busy === `${u.id}:unfreeze`}
                          onClick={() => action(u.id, "unfreeze")}
                        >
                          Unfreeze
                        </button>
                        <button
                          disabled={busy === `${u.id}:disable`}
                          onClick={() => action(u.id, "disable")}
                        >
                          Disable
                        </button>
                      </>
                    )}

                    {/* Disabled tab actions: Enable only (preserves frozen state) */}
                    {u.is_disabled ? (
                      <button
                        disabled={busy === `${u.id}:enable`}
                        onClick={() => action(u.id, "enable")}
                      >
                        Enable
                      </button>
                    ) : null}
                  </td>
                </tr>
              );
            })}
          </tbody>
        </table>

        {!rows.length && <div style={{ padding: 10, color: "#666" }}>No users in this tab.</div>}
      </div>
    </div>
  );
}

type IntentRow = any;

export function AdminTradeAudit() {

  const [userId, setUserId] = useState("");

  const [loading, setLoading] = useState(false);

    if (correlationId) p.set("correlationId", correlationId);
    if (eventType) p.set("eventType", eventType);
    if (decision) p.set("decision", decision);

    setLoading(true);

      if (tab === "trade") {
        const data = await api<TradeAuditRow[]>(`/admin/trade-audit?${query}`);
        setRows(data);

        const data = await api<IntentRow[]>(`/admin/order-intents?${query}`);

      setLoading(false);

      ...r,
      event_at: r.event_at ? new Date(r.event_at).toISOString() : "",

        <input placeholder="tradeId" value={tradeId} onChange={e => setTradeId(e.target.value)} />
        <input placeholder="correlationId" value={correlationId} onChange={e => setCorrelationId(e.target.value)} />
        <input placeholder="eventType" value={eventType} onChange={e => setEventType(e.target.value)} />
        {tab === "intent" ? (
          <select value={decision} onChange={e => setDecision(e.target.value as any)}>
            <option value="">Decision (any)</option>
            <option value="PASS">PASS</option>
            <option value="REJECT">REJECT</option>
          </select>
        ) : (
          <div />
        )}
      </div>

      {loading ? <div>Loading…</div> : null}

      <div style={{ border: "1px solid #eee", borderRadius: 8, overflow: "auto" }}>
        <table style={{ width: "100%", borderCollapse: "collapse", fontSize: 12 }}>
