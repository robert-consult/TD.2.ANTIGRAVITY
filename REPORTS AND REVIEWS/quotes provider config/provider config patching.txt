  key: z.string().min(1),

  kind: z.string().min(1),

      baseUrl: z.string().url(),
      quoteEndpoint: z.string().optional(),

    .optional(),

export async function loadProviderConfigs(configDir: string = DEFAULT_DIR): Promise<ProviderConfig[]> {
  let ents: any[] = [];
  try {
    ents = await fs.readdir(configDir, { withFileTypes: true });
  } catch {
    return [];
  }

  const configs: ProviderConfig[] = [];
  for (const ent of ents) {
    if (!ent.isFile()) continue;
    if (!ent.name.endsWith(".json")) continue;
    const file = path.join(configDir, ent.name);
    try {
      const raw = await fs.readFile(file, "utf8");
      const json = JSON.parse(raw);
      const parsed = ProviderConfigSchema.parse(json);
      configs.push(parsed);
    } catch (e: any) {
      // Ignore invalid provider configs; keep system booting.
      console.warn(`[MarketData] Failed to load provider config ${file}: ${e?.message || e}`);
    }
  }

  // Stable sort by key for deterministic outputs
  configs.sort((a, b) => a.key.localeCompare(b.key));
  return configs;
}

// Heuristic mapping when providerSymbolMap isn't set.
export function guessProviderSymbol(providerKey: string, canonicalSymbol: string): string {
  const sym = canonicalizeSymbol(canonicalSymbol);

  // Common FX pairs: EURUSD -> EUR/USD
  if (/^[A-Z]{6}$/.test(sym)) {
    return `${sym.slice(0, 3)}/${sym.slice(3)}`;
  }

  // Indices/CFDs often come in as already normalized; keep as-is.
  // Twelve Data accepts many symbols exactly as their canonical (e.g., AAPL, SPY).
  return sym;
}

export function mapToProviderSymbol(
  providerKey: string,
  canonicalSymbol: string,
  providerSymbolMap: any | null | undefined,
): string {
  const sym = canonicalizeSymbol(canonicalSymbol);
  const mapObj = (providerSymbolMap && typeof providerSymbolMap === "object") ? providerSymbolMap : null;
  const mapped = mapObj && typeof mapObj[providerKey] === "string" ? String(mapObj[providerKey]) : "";
  if (mapped.trim()) return mapped.trim();
  return guessProviderSymbol(providerKey, sym);
}

export function mapFromProviderSymbol(providerSymbol: string): string {
  // Our canonical normalization removes slashes.
  return canonicalizeSymbol(providerSymbol);

function normalizeEpochMs(raw: any): number {
  const n = toNum(raw);
  if (n === null) return nowMs();
  return n < 1e12 ? n * 1000 : n;

function pickBidAskPrice(obj: any): { bid: number | null; ask: number | null; price: number | null } {
  const bid = toNum(obj?.bid ?? obj?.b);
  const ask = toNum(obj?.ask ?? obj?.a);
  const price = toNum(obj?.price ?? obj?.p ?? obj?.close ?? obj?.last);
  return { bid, ask, price };

function extractRows(payload: any): Array<{ sym: string; row: any }> {
  if (!payload) return [];

  // Batch map response: { "AAPL": {...}, "MSFT": {...} }
  if (payload && typeof payload === "object" && !Array.isArray(payload)) {
    // Error wrapper: { status:"error", message:"..." }
    const status = typeof payload.status === "string" ? payload.status : "";
    if (status && status !== "ok") return [];

    // Sometimes payload contains 'data' wrapper
    if (payload.data && typeof payload.data === "object") {
      return extractRows(payload.data);
    }

    const keys = Object.keys(payload);
    // Single symbol returns quote object with `symbol`
    if (keys.includes("symbol") || keys.includes("instrument_name")) {
      const sym = String(payload.symbol ?? payload.instrument_name ?? "");
      return sym ? [{ sym, row: payload }] : [];
    }

    // Assume map keyed by symbol
    const out: Array<{ sym: string; row: any }> = [];
    for (const k of keys) {
      const v = (payload as any)[k];
      if (v && typeof v === "object") out.push({ sym: k, row: v });
    }
    return out;
  }

  // Array response (rare)
  if (Array.isArray(payload)) {
    return payload
      .map((row) => {
        const sym = String(row?.symbol ?? row?.instrument_name ?? "");
        return sym ? { sym, row } : null;
      })
      .filter(Boolean) as any;
  }

  return [];
}

export function createTwelveDataRestFetcher(cfg: ProviderConfig, apiKey: string) {

  const baseUrl = cfg.rest.baseUrl.replace(/\/+$/, "");

export function createTwelveDataWsStream(cfg: ProviderConfig, apiKey: string, onQuote: QuoteStreamHandler): QuoteStream {
  if (!cfg.ws?.url) {
    throw new Error("TWELVEDATA_WS_MISSING_URL");
  }

  let ws: WebSocket | null = null;
  let running = false;
  let desiredSymbols: string[] = [];
  let subscribed = new Set<string>();
  let heartbeatTimer: NodeJS.Timeout | null = null;
  let reconnectAttempt = 0;
  let closing = false;
  let pendingResub: NodeJS.Timeout | null = null;

  const buildUrl = () => {
    const u = new URL(cfg.ws!.url);
    u.searchParams.set("apikey", apiKey);
    return u.toString();
  };

  const heartbeatIntervalMs = cfg.ws.heartbeatIntervalMs ?? 10_000;
  const resubscribeDebounceMs = cfg.ws.resubscribeDebounceMs ?? 500;

  const sendJson = (obj: any) => {
    if (!ws || ws.readyState !== WebSocket.OPEN) return;
    try {
      ws.send(JSON.stringify(obj));
    } catch {
      // ignore
    }
  };

  const startHeartbeat = () => {
    stopHeartbeat();
    heartbeatTimer = setInterval(() => {
      // docs have typo 'hearbeat' in one place; support both.
      sendJson({ action: "heartbeat" });
      sendJson({ action: "hearbeat" });
    }, heartbeatIntervalMs);
  };

  const stopHeartbeat = () => {
    if (heartbeatTimer) {
      clearInterval(heartbeatTimer);
      heartbeatTimer = null;
    }
  };

  const scheduleResubscribe = () => {
    if (pendingResub) clearTimeout(pendingResub);
    pendingResub = setTimeout(() => {
      pendingResub = null;
      reconcileSubscriptions().catch(() => {});
    }, resubscribeDebounceMs);
  };

  const reconcileSubscriptions = async () => {
    if (!ws || ws.readyState !== WebSocket.OPEN) return;

    const want = new Set(desiredSymbols);

    const toUnsub: string[] = [];
    for (const s of subscribed) {
      if (!want.has(s)) toUnsub.push(s);
    }

    const toSub: string[] = [];
    for (const s of want) {
      if (!subscribed.has(s)) toSub.push(s);
    }

    if (toUnsub.length) {
      sendJson({ action: "unsubscribe", params: { symbols: toUnsub.join(",") } });
      for (const s of toUnsub) subscribed.delete(s);
    }

    if (toSub.length) {
      sendJson({ action: "subscribe", params: { symbols: toSub.join(",") } });
      for (const s of toSub) subscribed.add(s);
    }
  };

  const connect = async () => {
    if (closing) return;
    const url = buildUrl();
    ws = new WebSocket(url);

    ws.on("open", () => {
      reconnectAttempt = 0;
      startHeartbeat();
      scheduleResubscribe();
    });

    ws.on("message", (buf) => {
      try {
        const raw = buf.toString("utf8");
        const msg = JSON.parse(raw);

        // data messages are usually like:
        // { event: "price", symbol: "EUR/USD", price: "1.234", timestamp: 1234567890 }
        const symbol = typeof msg.symbol === "string" ? msg.symbol : "";
        const event = typeof msg.event === "string" ? msg.event : "";
        if (!symbol || (!event && msg.price == null && msg.bid == null && msg.ask == null)) return;

        const canonical = mapFromProviderSymbol(symbol);

        const price = toNum(msg.price ?? msg.p);
        const bid = toNum(msg.bid ?? msg.b);
        const ask = toNum(msg.ask ?? msg.a);

        const tsRaw = msg.timestamp ?? msg.t ?? msg.time ?? Date.now();
        const tsNum = toNum(tsRaw);
        const tsMs = tsNum == null ? Date.now() : (tsNum < 1e12 ? tsNum * 1000 : tsNum);

        const q: ProviderQuote = {
          symbol: canonical,
          bid,
          ask,
          price,
          lastApiUpdate: tsMs,
          isStale: false,
          source: cfg.key,
          transport: "ws",
          rawSymbol: symbol,
        };

        onQuote(q);
      } catch {
        // ignore bad frames
      }
    });

    ws.on("close", async () => {
      stopHeartbeat();
      ws = null;
      subscribed = new Set<string>();
      if (!closing) {
        reconnectAttempt++;
        const backoffMs = Math.min(30_000, 250 * (2 ** Math.min(8, reconnectAttempt)));
        await sleep(backoffMs);
        await connect();
      }
    });

    ws.on("error", () => {
      // close will trigger reconnect
      try {
        ws?.close();
      } catch {}
    });
  };

  return {
    async start(symbols: string[]) {
      desiredSymbols = [...new Set(symbols)];
      running = true;
      closing = false;
      await connect();
    },

    async updateSymbols(symbols: string[]) {
      desiredSymbols = [...new Set(symbols)];
      if (!running) return;
      scheduleResubscribe();
    },

    async stop() {
      closing = true;
      running = false;
      if (pendingResub) {
        clearTimeout(pendingResub);
        pendingResub = null;
      }
      stopHeartbeat();
      if (ws) {
        try {
          ws.close();
        } catch {}
      }
      ws = null;
      subscribed = new Set<string>();
    },

    isRunning() {
      return running;
    },
  };
}

function toQuoteCore(row: any, asOf: number, defaults?: { source?: string; transport?: QuoteTransport }): QuoteCore | null {
  if (!row?.symbol) return null;
  const symbol = normalizeSymbol(String(row.symbol));
  const bid = typeof row.bid === "number" ? row.bid : row.bid == null ? null : Number(row.bid);
  const ask = typeof row.ask === "number" ? row.ask : row.ask == null ? null : Number(row.ask);

        : row.price == null

          : Number(row.price);

  const lastUpdatedRaw = row.lastUpdated ?? row.lastApiUpdate ?? row.updatedAt ?? asOf;
  const lastApiRaw = row.lastApiUpdate ?? row.lastUpdated ?? row.updatedAt ?? asOf;
  const lastUpdated = Number.isFinite(Number(lastUpdatedRaw)) ? Number(lastUpdatedRaw) : asOf;
  const lastApiUpdate = Number.isFinite(Number(lastApiRaw)) ? Number(lastApiRaw) : asOf;

  const source = typeof row.source === "string" ? row.source : defaults?.source;
  const transport =
    typeof row.transport === "string" && (row.transport === "rest" || row.transport === "ws" || row.transport === "unknown")

    bid,
    ask,

// Get a fresh database connection for each query to avoid locking issues
// Fallback chain: Hub → Valkey Snapshot → Rolling Buffer → DB → prevClose
export async function getLatestQuoteRow(symbol: string): Promise<any | null> {
  const sym = normalizeSymbol(symbol);

  // 1. Try in-memory hub (fastest)
  const hubQuote = getQuote(sym);
  if (hubQuote) {
    return {
      symbol: sym,
      bid: hubQuote.bid,
      ask: hubQuote.ask,
      price: hubQuote.price,
      lastApiUpdate: hubQuote.lastApiUpdate,
      isStale: hubQuote.isStale,
      prevClose: hubQuote.prevClose,
      source: hubQuote.source,
      transport: hubQuote.transport,
    };
  }

  // 2. Try Valkey per-symbol keys
  const valkeyRows = await getValkeyQuoteRows([sym]);
  if (valkeyRows.length) {
    const cached = valkeyRows[0];
    return {
      symbol: sym,
      bid: cached.bid,
      ask: cached.ask,
      price: cached.price,
      lastApiUpdate: cached.lastApiUpdate,
      isStale: cached.isStale,
      source: cached.source,
      transport: cached.transport,
    };
  }

  // 3. Try rolling buffer (30-second history)
  const rollingQuote = await getFromRollingBuffer(sym);
  if (rollingQuote && rollingQuote.price != null) {
    return {
      symbol: sym,
      bid: rollingQuote.bid,
      ask: rollingQuote.ask,
      price: rollingQuote.price,
      lastApiUpdate: rollingQuote.lastApiUpdate,
      isStale: true,
      source: rollingQuote.source,
      transport: rollingQuote.transport,
    };
  }

  // 4. Try database
  const row = await db.query.quotes.findFirst({
    where: eq(quotes.symbol, sym),
  });

      isPrevCloseFallback: true,
      source: "prev_close",
      transport: "unknown",
    };

export async function getExecutionQuote(symbol: string, side: "BUY" | "SELL", action: "OPEN" | "CLOSE"): Promise<ExecutionQuote> {

  const row = await getLatestQuoteRow(sym);
  if (!row) throw new Error(`QUOTE_NOT_FOUND:${sym}`);

  const bid = toNum(row.bid);
  const ask = toNum(row.ask);
  const px = toNum(row.price);

  const usableBid = bid ?? px;
  const usableAsk = ask ?? px;
  if (usableBid === null || usableAsk === null) throw new Error(`QUOTE_INVALID:${sym}`);

  const mid = (usableBid + usableAsk) / 2;
  const spread = usableAsk - usableBid;

  const lastApiRaw =
    toNum((row as any).lastApiUpdate ?? (row as any).last_api_update) ??
    toNum((row as any).updatedAt ?? (row as any).updated_at);

  const lastApiMs = lastApiRaw === null ? null : (lastApiRaw < 1e12 ? lastApiRaw * 1000 : lastApiRaw);

  const now = Date.now();
  const isStale =
    Boolean((row as any).isStale ?? (row as any).is_stale) ||
    lastApiMs === null ||
    (now - lastApiMs) > STALE_AFTER_MS;

  const marketOpen = isMarketOpenForSymbol(sym, new Date());

  // Institutional: BUY opens at ask, closes at bid. SELL opens at bid, closes at ask.
  const execPrice =
    action === "OPEN"
      ? (side === "BUY" ? usableAsk : usableBid)
      : (side === "BUY" ? usableBid : usableAsk);

  const quoteTs = new Date(lastApiMs ?? now);

  const meta = getQuoteMeta();
  const source = String((row as any).source || meta.source || process.env.QUOTE_SOURCE || "unknown");

  return {
    symbol: sym,
    bid: usableBid,
    ask: usableAsk,
    mid,
    spread,
    quoteTs,
    source,
    isStale,
    marketOpen,
    execPrice,
  };
}

// FX conversion (quoteCurrency -> USD) using available pairs in quotes table
export async function getConversionRate(fromCcy: string, toCcy: string): Promise<number> {
  const from = fromCcy.toUpperCase();
  const to = toCcy.toUpperCase();
  if (from === to) return 1;

  const direct = await getLatestQuoteRow(from + to);
  if (direct) {
    const b = toNum(direct.bid) ?? toNum(direct.price);
    const a = toNum(direct.ask) ?? toNum(direct.price);
    if (b !== null && a !== null) return (a + b) / 2;
  }

  const inverse = await getLatestQuoteRow(to + from);
  if (inverse) {
    const b = toNum(inverse.bid) ?? toNum(inverse.price);
    const a = toNum(inverse.ask) ?? toNum(inverse.price);
    if (b !== null && a !== null) return 1 / ((a + b) / 2);
  }

  // Bridge via USD
  if (from !== "USD" && to !== "USD") {
    return (await getConversionRate(from, "USD")) * (await getConversionRate("USD", to));
  }

  // Fallback rates for common currencies if quote not found
  const fallbackRates: Record<string, number> = {
    JPY: 0.0067,
    EUR: 1.09,
    GBP: 1.27,
    AUD: 0.65,
    USD: 1.0,
  };

  if (to === "USD" && fallbackRates[from]) return fallbackRates[from];
  if (from === "USD" && fallbackRates[to]) return 1 / fallbackRates[to];

  throw new Error(`FX_CONVERSION_MISSING:${from}->${to}`);
}
