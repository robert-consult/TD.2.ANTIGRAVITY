## 5.  Regression Guard (Automated JS Harness)

Optional Node script for continuous verification:

```js
import fetch from "node-fetch";
import assert from "node:assert";

const CK = process.env.TEST_COOKIE;

async function test(endpoint, expectCode) {
  const res = await fetch(`http://localhost:3000${endpoint}`, {
    method: "POST",
    headers: { Cookie: CK },
  });
  const js = await res.json().catch(() => ({}));
  assert.strictEqual(res.status, expectCode, `${endpoint} expected ${expectCode}, got ${res.status}`);
  console.log("✓", endpoint, expectCode, js.deny_code || "");
}

await test("/api/trades/place", 403);
await test("/api/trades/123/cancel", 403);
await test("/api/trades/123/close", 200);
await test("/api/profile/kyc", 403);
```
