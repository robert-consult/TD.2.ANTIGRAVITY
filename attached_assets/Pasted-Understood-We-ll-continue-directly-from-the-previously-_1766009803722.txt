// --- Account State Logic Patch ---

exports.freezeUser = async (req, res) => {
  try {
    const { id, reason } = req.body;
    const user = await prisma.user.update({
      where: { id },
      data: {
        isFrozen: true,
        freezeReason: reason || 'Admin action',
        status: 'Frozen'
      }
    });
    res.json({ success: true, user });
  } catch (e) { res.status(500).json({ error: e.message }); }
};

exports.unfreezeUser = async (req, res) => {
  try {
    const { id } = req.body;
    const user = await prisma.user.findUnique({ where: { id } });
    let status = 'Active';
    if (user.isDisabled) status = 'Disabled';
    const updated = await prisma.user.update({
      where: { id },
      data: { isFrozen: false, freezeReason: null, status }
    });
    res.json({ success: true, user: updated });
  } catch (e) { res.status(500).json({ error: e.message }); }
};

exports.disableUser = async (req, res) => {
  try {
    const { id, reason } = req.body;
    const user = await prisma.user.update({
      where: { id },
      data: {
        isDisabled: true,
        disableReason: reason || 'Admin action',
        status: user.isFrozen ? 'Frozen+Disabled' : 'Disabled'
      }
    });
    res.json({ success: true, user });
  } catch (e) { res.status(500).json({ error: e.message }); }
};

exports.enableUser = async (req, res) => {
  try {
    const { id } = req.body;
    const user = await prisma.user.findUnique({ where: { id } });
    let newStatus = 'Active';
    if (user.isFrozen) newStatus = 'Frozen';
    const updated = await prisma.user.update({
      where: { id },
      data: { isDisabled: false, status: newStatus }
    });
    res.json({ success: true, user: updated });
  } catch (e) { res.status(500).json({ error: e.message }); }
};
```

const { Parser } = require('json2csv');

exports.exportUsersCSV = async (req, res) => {
  try {
    const users = await prisma.user.findMany({
      select: {
        id: true, firstName: true, lastName: true, email: true, username: true,
        phone: true, balance: true, isAdmin: true, isDisabled: true, isFrozen: true,
        freezeReason: true, disableReason: true, leverage: true, maxConcurrent: true,
        maxConcurrentLots: true, minHold: true, maxHold: true, status: true,
        createdAt: true
      }
    });
    const parser = new Parser();
    const csv = parser.parse(users);
    res.header('Content-Type', 'text/csv');
    res.attachment(`users_${Date.now()}.csv`);
    res.send(csv);
  } catch (e) {
    res.status(500).json({ error: e.message });
  }
};
```

  user       User     @relation(fields: [userId], references: [id])
}

### `/backend/controllers/loginHistory.js`

```js
exports.recordLogin = async (req, res) => {
  const ip = req.ip;
  const { userId, device } = req.body;
  await prisma.loginHistory.create({
    data: { userId, ipAddress: ip, device }
  });
  res.json({ success: true });
};

exports.getLoginHistory = async (req, res) => {
  const data = await prisma.loginHistory.findMany({
    include: { user: { select: { email: true, username: true } } },
    orderBy: { createdAt: 'desc' }
  });
  res.json(data);
};
```
