import { db } from "../../db";
import { systemConfig } from "@shared/schema";
import { eq } from "drizzle-orm";

export type ActivityConfig = {
  inactivityThresholdDays: number;
  deletionGraceDays: number;
  botScoreThreshold: number;
  autoQueueInactive: boolean;
  autoQueueBots: boolean;
  autoDeleteDue: boolean;
};

const DEFAULTS: ActivityConfig = {
  inactivityThresholdDays: 90,
  deletionGraceDays: 30,
  botScoreThreshold: 40,
  autoQueueInactive: true,
  autoQueueBots: false,
  autoDeleteDue: false,
};

function toInt(v: unknown, fallback: number): number {
  const n = Number(v);
  return Number.isFinite(n) ? n : fallback;
}

function toBool(v: unknown, fallback: boolean): boolean {
  if (v === undefined || v === null) return fallback;
  return Boolean(v);
}

export async function getActivityConfig(): Promise<ActivityConfig> {
  const row = await db.select().from(systemConfig).where(eq(systemConfig.id, 1)).get();
  if (!row) return { ...DEFAULTS };

  return {
    inactivityThresholdDays: toInt((row as any).activityInactivityThresholdDays, DEFAULTS.inactivityThresholdDays),
    deletionGraceDays: toInt((row as any).activityDeletionGraceDays, DEFAULTS.deletionGraceDays),
    botScoreThreshold: toInt((row as any).activityBotScoreThreshold, DEFAULTS.botScoreThreshold),
    autoQueueInactive: toBool((row as any).activityAutoQueueInactive, DEFAULTS.autoQueueInactive),
    autoQueueBots: toBool((row as any).activityAutoQueueBots, DEFAULTS.autoQueueBots),
    autoDeleteDue: toBool((row as any).activityAutoDeleteDue, DEFAULTS.autoDeleteDue),
  };
}

export async function updateActivityConfig(partial: Partial<ActivityConfig>): Promise<ActivityConfig> {
  const current = await getActivityConfig();
  const next: ActivityConfig = { ...current, ...partial };

  await db
    .update(systemConfig)
    .set({
      activityInactivityThresholdDays: next.inactivityThresholdDays,
      activityDeletionGraceDays: next.deletionGraceDays,
      activityBotScoreThreshold: next.botScoreThreshold,
      activityAutoQueueInactive: next.autoQueueInactive,
      activityAutoQueueBots: next.autoQueueBots,
      activityAutoDeleteDue: next.autoDeleteDue,
      updatedAt: new Date(),
    } as any)
    .where(eq(systemConfig.id, 1));

  return getActivityConfig();
}
6.4 server/bot/botScoring.ts — bot risk scoring engine (complete; reuses Grift)
import type Database from "better-sqlite3";

export type BotScoreReason = {
  code: string;
  points: number;
  detail?: string;
  source: "identity" | "network" | "behavior" | "grift";
};

export type BotScoreResult = {
  userId: number;
  score: number; // 0-100 (higher => more bot-like)
  primarySource: BotScoreReason["source"];
  reasons: BotScoreReason[];
  computedAtSec: number;
};

const UA_AUTOMATION_RE = /(headlesschrome|puppeteer|playwright|selenium|phantomjs|slimerjs|webdriver|bot|spider|crawler|scrapy)/i;
const UA_NONBROWSER_RE = /(curl|wget|python-requests|okhttp|java\s?\/|go-http-client|httpclient|postmanruntime|axios)/i;

const DATACENTER_ORG_RE = /(amazon|aws|google|gcp|microsoft|azure|digitalocean|ovh|hetzner|linode|vultr|oracle|alibaba|tencent|cloudflare|fastly|akamai|choopa|leaseweb)/i;

function clamp(n: number, min: number, max: number) {
  return Math.max(min, Math.min(max, n));
}

function topSource(reasons: BotScoreReason[]): BotScoreReason["source"] {
  const order: BotScoreReason["source"][] = ["identity", "network", "behavior", "grift"];
  const agg = new Map<BotScoreReason["source"], number>();
  for (const r of reasons) agg.set(r.source, (agg.get(r.source) ?? 0) + r.points);

  let best = "identity" as BotScoreReason["source"];
  let bestScore = -1;
  for (const s of order) {
    const v = agg.get(s) ?? 0;
    if (v > bestScore) {
      bestScore = v;
      best = s;
    }
  }
  return best;
}

export function computeBotScoreForUser(db: Database, userId: number, nowSec?: number): BotScoreResult {
  const tNow = nowSec ?? Math.floor(Date.now() / 1000);
  const reasons: BotScoreReason[] = [];

  const latestLogin = db
    .prepare(
      `SELECT user_agent as userAgent, ip, asn, org, created_at as createdAt
       FROM user_login_history
       WHERE user_id = ?
       ORDER BY created_at DESC
       LIMIT 1`
    )
    .get(userId) as any;

  const ua = (latestLogin?.userAgent ?? "") as string;
  const org = (latestLogin?.org ?? "") as string;

  // Network engine
  if (org && DATACENTER_ORG_RE.test(org)) {
    reasons.push({ code: "DATACENTER_ASN_ORG", points: 25, detail: org, source: "network" });
  }

  // Behavior engine: login failures
  const cutoff7d = tNow - 7 * 86400;
  const failures7d = db
    .prepare(
      `SELECT COUNT(1) as c
       FROM user_login_history
       WHERE user_id = ? AND event_type = 'LOGIN_FAILED' AND created_at >= ?`
    )
    .get(userId, cutoff7d) as any;

  const failCount = Number(failures7d?.c ?? 0);
  if (failCount >= 10) {
    reasons.push({ code: "LOGIN_FAILURES_7D_HIGH", points: 15, detail: `failures=${failCount}`, source: "behavior" });
  } else if (failCount >= 5) {
    reasons.push({ code: "LOGIN_FAILURES_7D_MED", points: 8, detail: `failures=${failCount}`, source: "behavior" });
  }

  // Identity engine: shared devices via grift graph
  const sharedDevices = db
    .prepare(
      `SELECT COUNT(1) as c
       FROM (
         SELECT du.device_id
         FROM grift_device_users du
         JOIN grift_device_users other
           ON other.device_id = du.device_id AND other.user_id != du.user_id
         WHERE du.user_id = ?
         GROUP BY du.device_id
       ) t`
    )
    .get(userId) as any;

  const sharedDeviceCount = Number(sharedDevices?.c ?? 0);
  if (sharedDeviceCount >= 5) {
    reasons.push({ code: "SHARED_DEVICES_HIGH", points: 30, detail: `sharedDevices=${sharedDeviceCount}`, source: "identity" });
  } else if (sharedDeviceCount >= 3) {
    reasons.push({ code: "SHARED_DEVICES_MED", points: 20, detail: `sharedDevices=${sharedDeviceCount}`, source: "identity" });
  } else if (sharedDeviceCount >= 1) {
    reasons.push({ code: "SHARED_DEVICES_LOW", points: 10, detail: `sharedDevices=${sharedDeviceCount}`, source: "identity" });
  }

  // Reuse Grift rollups
  const gScore = db
    .prepare(
      `SELECT score_current as scoreCurrent, tier as riskTier, linked_accounts_30d as linked30d
       FROM grift_user_scores
       WHERE user_id = ?`
    )
    .get(userId) as any;

  const linked30d = Number(gScore?.linked30d ?? 0);
  if (linked30d >= 5) {
    reasons.push({ code: "LINKED_ACCOUNTS_30D_HIGH", points: 25, detail: `linked30d=${linked30d}`, source: "identity" });
  } else if (linked30d >= 2) {
    reasons.push({ code: "LINKED_ACCOUNTS_30D_MED", points: 15, detail: `linked30d=${linked30d}`, source: "identity" });
  }

  const riskTier = (gScore?.riskTier ?? "") as string;
  const scoreCurrent = Number(gScore?.scoreCurrent ?? 0);
  if (riskTier === "CRITICAL") {
    reasons.push({ code: "GRIFT_TIER_CRITICAL", points: 35, detail: `score=${scoreCurrent}`, source: "grift" });
  } else if (riskTier === "HIGH") {
    reasons.push({ code: "GRIFT_TIER_HIGH", points: 25, detail: `score=${scoreCurrent}`, source: "grift" });
  } else if (riskTier === "MED") {
    reasons.push({ code: "GRIFT_TIER_MED", points: 15, detail: `score=${scoreCurrent}`, source: "grift" });
  } else if (scoreCurrent >= 10) {
    reasons.push({ code: "GRIFT_SCORE_NONZERO", points: 10, detail: `score=${scoreCurrent}`, source: "grift" });
  }

  // UA engine
  if (!ua || ua.trim().length < 8) {
    reasons.push({ code: "UA_MISSING_OR_SHORT", points: 20, source: "behavior" });
  } else {
    if (UA_AUTOMATION_RE.test(ua)) {
      reasons.push({ code: "UA_AUTOMATION_KEYWORDS", points: 35, detail: ua.slice(0, 120), source: "behavior" });
    } else if (UA_NONBROWSER_RE.test(ua)) {
      reasons.push({ code: "UA_NON_BROWSER_CLIENT", points: 25, detail: ua.slice(0, 120), source: "behavior" });
    }
  }

  // Open automation-like signals (if present)
  const openSignals = db
    .prepare(
      `SELECT COUNT(1) as c
       FROM grift_signals
       WHERE user_id = ? AND status = 'OPEN'
         AND (rule_code LIKE 'UA_%' OR rule_code LIKE 'AUTOMATION_%' OR rule_code LIKE 'BOT_%')`
    )
    .get(userId) as any;

  const openSignalCount = Number(openSignals?.c ?? 0);
  if (openSignalCount >= 1) {
    reasons.push({ code: "OPEN_AUTOMATION_SIGNALS", points: 10, detail: `openSignals=${openSignalCount}`, source: "grift" });
  }

  const raw = reasons.reduce((s, r) => s + r.points, 0);
  const score = clamp(raw, 0, 100);
  const primarySource = topSource(reasons);

  return { userId, score, primarySource, reasons, computedAtSec: tNow };
}

// server/activity/activityScheduler.ts
import * as cron from "node-cron";
import {
  refreshUserActivityRollups,
  refreshBotScores,
  executeDueDeletions,
} from "./activityEngine";

type SchedulerState = {
  rollupsRunning: boolean;
  botRunning: boolean;
  deletesRunning: boolean;
};

const state: SchedulerState = {
  rollupsRunning: false,
  botRunning: false,
  deletesRunning: false,
};

export function startUserActivityScheduler(dbPath: string, sessionsDbPath: string) {
  // Rollups: every 6 hours at minute 7 (staggered from other jobs)
  cron.schedule("7 */6 * * *", () => safeRun("rollups", () => refreshUserActivityRollups(dbPath)));

  // Bot scoring: hourly at minute 17
  cron.schedule("17 * * * *", () => safeRun("bot", () => refreshBotScores(dbPath)));

  // Due deletions: hourly at minute 27
  cron.schedule("27 * * * *", () => safeRun("deletes", () => executeDueDeletions(dbPath, sessionsDbPath)));

  // Optional: run once at startup (lightweight, best-effort)
  safeRun("rollups", () => refreshUserActivityRollups(dbPath));
  safeRun("bot", () => refreshBotScores(dbPath));
  safeRun("deletes", () => executeDueDeletions(dbPath, sessionsDbPath));
}

export async function runImmediateActivityRefresh(dbPath: string) {
  // Run rollups then bot scoring synchronously
  const r1 = refreshUserActivityRollups(dbPath);
  const r2 = refreshBotScores(dbPath);
  return { rollups: r1, botScores: r2 };
}

function safeRun(kind: "rollups" | "bot" | "deletes", fn: () => any) {
  if (kind === "rollups") {
    if (state.rollupsRunning) return;
    state.rollupsRunning = true;
  }
  if (kind === "bot") {
    if (state.botRunning) return;
    state.botRunning = true;
  }
  if (kind === "deletes") {
    if (state.deletesRunning) return;
    state.deletesRunning = true;
  }

  try {
    fn();
  } catch (e) {
    console.warn(`[activityScheduler] ${kind} failed:`, e);
  } finally {
    if (kind === "rollups") state.rollupsRunning = false;
    if (kind === "bot") state.botRunning = false;
    if (kind === "deletes") state.deletesRunning = false;
  }
}

function toBool(v: unknown): boolean {
  return v === true || v === 1 || v === "1" || v === "true";
}

function uniqPositiveInts(input: unknown): number[] {
  if (!Array.isArray(input)) return [];
  const uniq = new Set<number>();
  for (const v of input) {
    const n = Number(v);
    if (Number.isFinite(n) && n > 0) uniq.add(Math.trunc(n));
  }
  return Array.from(uniq);
}

type DeletionSource = "INACTIVITY" | "BOT" | "ADMIN";

function normalizeSource(raw: unknown, fallback: DeletionSource): DeletionSource {
  const s = String(raw ?? "").toUpperCase();
  if (s.includes("BOT")) return "BOT";
  if (s.includes("INACT")) return "INACTIVITY";
  if (s.includes("ADMIN")) return "ADMIN";
  return fallback;
}

function mapBotStatus(row: { status: string; deletedAt: number | null }): "OK" | "SUSPICIOUS" | "WHITELISTED" | "DELETED" {
  if (row.deletedAt != null) return "DELETED";
  const s = String(row.status ?? "OK").toUpperCase();
  if (s === "WHITELISTED") return "WHITELISTED";
  if (s === "DELETED") return "DELETED";
  if (s === "FLAGGED" || s === "SUSPICIOUS") return "SUSPICIOUS";
  return "OK";
}

function featuresFromSignals(signals: BotUserRow["reasons"], score: number): Record<string, any> {
  const codes = new Set((signals ?? []).map((s) => s.code));
  const anyPrefix = (p: string) => Array.from(codes).some((c) => c.startsWith(p));

  const features: Record<string, any> = {
    score,
    signalCount: (signals ?? []).length,
  };

  if (codes.has("UA_MISSING_OR_SHORT")) features.userAgent = "missing/short";
  if (codes.has("UA_HEADLESS")) features.headless = true;
  if (codes.has("UA_AUTOMATION")) features.automation = true;
  if (codes.has("UA_BOT_KEYWORD")) features.uaBotKeyword = true;
  if (codes.has("DEVICE_FINGERPRINT_MISSING")) features.deviceFingerprint = "missing";

  if (anyPrefix("LOGIN_FAILURE")) features.loginFailures = "elevated";
  if (anyPrefix("SESSION_CHURN")) features.sessionChurn = "high";
  if (anyPrefix("SESSION_DURATION")) features.sessionDuration = "short";
  if (anyPrefix("IP_CHURN")) features.ipChurn = "high";
  if (anyPrefix("DEVICE_MULTIACCOUNT")) features.multiAccount = "likely";
  if (anyPrefix("GRIFT_RISK")) features.griftCorrelation = "high";

  const top = (signals ?? []).slice(0, 3).map((s) => s.code);
  if (top.length) features.topSignals = top.join(",");

  return features;
}

function setUserQueuedForDeletionTx(
  db: Database.Database,
  args: {
    userId: number;
    dueAt: number;
    queuedAt: number;
    adminId: number;
    source: string;
    reason?: string;
  }
): boolean {
  const prev = db
    .prepare(`SELECT is_disabled as isDisabled, activity_disable_prev as prevVal FROM users WHERE id = ?`)
    .get(args.userId) as any;

  if (!prev) return false;

  const prevDisabled = Number(prev.isDisabled ?? 0);
  const prevVal = prev.prevVal != null ? Number(prev.prevVal) : null;
  const activityPrev = prevVal != null ? prevVal : prevDisabled;

  const updateRes = db
    .prepare(
      `UPDATE users
       SET activity_disable_prev = ?,
           disabled_by_activity = 1,
           is_disabled = 1,
           deletion_queued_at = ?,
           deletion_due_at = ?,
           deletion_reason = ?,
           deletion_source = ?
       WHERE id = ? AND is_admin = 0 AND deleted_at IS NULL`
    )
    .run(activityPrev, args.queuedAt, args.dueAt, args.reason ?? null, args.source, args.userId);

  if (!updateRes?.changes) return false;

  db.prepare(
    `INSERT INTO user_account_events (user_id, admin_id, event_type, title, description, reason_code, reason_text, metadata, created_at)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`
  ).run(
    args.userId,
    args.adminId,
    "ACCOUNT_DELETION_SCHEDULED",
    "Account deletion scheduled",
    `Deletion scheduled (source=${args.source}) due at ${args.dueAt}.`,
    "DELETION_SOURCE",
    args.source,
    JSON.stringify({ dueAt: args.dueAt, queuedAt: args.queuedAt, source: args.source, reason: args.reason ?? null }),
    args.queuedAt
  );

  return true;
}

export function queueUsersForDeletion(
  dbPath: string,
  userIds: number[],
  adminId: number,
  source: "INACTIVITY" | "BOT" | "ADMIN",
  reason?: string
): { queued: number; dueAt: number } {
  const db = openTradingDb(dbPath);
  try {
    const cfg = readActivityConfig(db);
    const queuedAt = nowSec();
    const dueAt = queuedAt + cfg.deletionGraceDays * 86400;

    const uniq = Array.from(new Set(userIds.map((x) => Number(x)).filter((x) => Number.isFinite(x) && x > 0)));

    let queued = 0;
    const tx = db.transaction(() => {
      for (const uid of uniq) {
        if (uid === adminId) continue;

        const did = setUserQueuedForDeletionTx(db, { userId: uid, dueAt, queuedAt, adminId, source, reason });
        if (did) queued += 1;

        if (source === "BOT") {
          db.prepare(
            `UPDATE bot_user_scores SET status = 'FLAGGED' WHERE user_id = ? AND status != 'WHITELISTED'`
          ).run(uid);
        }
      }
    });

    tx();
    return { queued, dueAt };
  } finally {
    db.close();
  }
}

export function cancelQueuedDeletions(
  dbPath: string,
  userIds: number[],
  adminId: number,
  reason?: string
): { cancelled: number } {
  const db = openTradingDb(dbPath);
  try {
    const now = nowSec();
    const uniq = Array.from(new Set(userIds.map((x) => Number(x)).filter((x) => Number.isFinite(x) && x > 0)));

    let cancelled = 0;

    const tx = db.transaction(() => {
      for (const uid of uniq) {
        if (uid === adminId) continue;

        const row = db
          .prepare(
            `SELECT
               is_disabled as isDisabled,
               disabled_by_activity as disabledByActivity,
               activity_disable_prev as prev,
               deletion_queued_at as deletionQueuedAt
             FROM users
             WHERE id = ? AND is_admin = 0 AND deleted_at IS NULL`
          )
          .get(uid) as any;

        if (!row) continue;

        const queuedAt = row.deletionQueuedAt != null ? Number(row.deletionQueuedAt) : null;
        const disabledByActivity = Number(row.disabledByActivity ?? 0) === 1;

        // Only cancel if there is actually something to cancel
        if (queuedAt == null && !disabledByActivity) continue;

        const prev = row.prev != null ? Number(row.prev) : null;
        const nextDisabled = disabledByActivity && prev != null ? prev : Number(row.isDisabled ?? 0);

        const upd = db.prepare(
          `UPDATE users
           SET is_disabled = ?,
               disabled_by_activity = 0,
               activity_disable_prev = NULL,
               deletion_queued_at = NULL,
               deletion_due_at = NULL,
               deletion_reason = NULL,
               deletion_source = NULL
           WHERE id = ?`
        ).run(nextDisabled, uid);

        if (!upd?.changes) continue;

        db.prepare(
          `INSERT INTO user_account_events (user_id, admin_id, event_type, title, description, reason_code, reason_text, metadata, created_at)
           VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`
        ).run(
          uid,
          adminId,
          "ACCOUNT_DELETION_CANCELLED",
          "Account deletion cancelled",
          reason ? `Cancelled: ${reason}` : "Deletion was cancelled by admin.",
          "CANCEL_REASON",
          reason ?? null,
          JSON.stringify({ reason: reason ?? null }),
          now
        );

        cancelled += 1;
      }
    });

    tx();
    return { cancelled };
  } finally {
    db.close();

                  <TableHead className="w-10">

                            if (!!v) next.add(u.userId);
                            else next.delete(u.userId);
                            setInactiveSelected(next);

                        {u.deletionQueuedAt ? (

                              </div>
                            ) : null}
                          </div>

                          "—"

                      </TableCell>
                    </TableRow>
                  ))
                )}
              </TableBody>
            </Table>
          </div>

      </Card>
    </div>

import type { Request } from "express";
import { extractGriftContext } from "../grift/griftGeo";

type BotScoreDetail = {
  score: number;                 // 1..99 (higher is more human-like)
  reasons: string[];
  components: Record<string, number>; // reason -> delta
  lowScore: boolean;
};

const VERIFIED_BOT_UA = [
  /Googlebot/i,
  /bingbot/i,
  /Baiduspider/i,
  /YandexBot/i,
  /DuckDuckBot/i,
  /Applebot/i,
];

const AUTOMATION_UA = [
  /HeadlessChrome/i,
  /PhantomJS/i,
  /Selenium/i,
  /Playwright/i,
  /Puppeteer/i,
  /python-requests/i,
  /curl\//i,
  /httpclient/i,
  /Go-http-client/i,
  /PostmanRuntime/i,
];

function headerPresent(req: Request, name: string): boolean {
  const v = req.headers[name.toLowerCase()];
  if (Array.isArray(v)) return v.length > 0 && !!String(v[0] || "").trim();
  return !!String(v || "").trim();
}

function clamp(n: number, lo: number, hi: number) {
  return Math.max(lo, Math.min(hi, n));
}

/**
 * Cloudflare-style posture:
 * - Score is a continuous 1..99 where low score is bot-like.
 * - We cannot replicate Cloudflare proprietary internals, but we can replicate the architecture:
 *   multi-signal scoring with identity, header completeness, and automation markers.
 */
export function computeBotScore(req: Request, threshold: number): BotScoreDetail {
  const ctx = extractGriftContext(req);
  const ua = String(ctx.userAgent || "");
  const reasons: string[] = [];
  const components: Record<string, number> = {};

  // Start as human-like, subtract for risk.
  let score = 90;

  // Verified bots (should not be “bad bots” in your account system, but included for completeness)
  if (ua && VERIFIED_BOT_UA.some((r) => r.test(ua))) {
    score = 95;
    reasons.push("verified_bot_user_agent");
    components["verified_bot_user_agent"] = +5;
  }

  // Explicit automation signatures
  if (!ua) {
    score -= 40;
    reasons.push("missing_user_agent");
    components["missing_user_agent"] = -40;
  } else if (AUTOMATION_UA.some((r) => r.test(ua))) {
    score -= 55;
    reasons.push("automation_user_agent");
    components["automation_user_agent"] = -55;
  }

  // Browser header completeness (bots often lack these)
  if (!headerPresent(req, "accept")) {
    score -= 8;
    reasons.push("missing_accept");
    components["missing_accept"] = -8;
  }
  if (!headerPresent(req, "accept-language")) {
    score -= 10;
    reasons.push("missing_accept_language");
    components["missing_accept_language"] = -10;
  }
  if (!headerPresent(req, "sec-fetch-site")) {
    score -= 6;
    reasons.push("missing_sec_fetch_site");
    components["missing_sec_fetch_site"] = -6;
  }
  if (!headerPresent(req, "sec-ch-ua")) {
    score -= 6;
    reasons.push("missing_sec_ch_ua");
    components["missing_sec_ch_ua"] = -6;
  }

  // Your internal identity headers (reused from Grift system)
  if (!ctx.deviceFp) {
    score -= 18;
    reasons.push("missing_device_fp");
    components["missing_device_fp"] = -18;
  }
  if (!ctx.deviceInstallId) {
    score -= 12;
    reasons.push("missing_device_install_id");
    components["missing_device_install_id"] = -12;
  }
  if (!ctx.clientTz) {
    score -= 5;
    reasons.push("missing_client_tz");
    components["missing_client_tz"] = -5;
  }
  if (!ctx.clientLang) {
    score -= 4;
    reasons.push("missing_client_lang");
    components["missing_client_lang"] = -4;
  }

  // Suspicious login/signup posture: missing origin/referer is unusual for browser flows
  const path = (req.path || "").toLowerCase();
  const isAuthFlow = path.includes("/api/auth/login") || path.includes("/api/auth/register");
  if (isAuthFlow) {
    if (!headerPresent(req, "origin") && !headerPresent(req, "referer")) {
      score -= 10;
      reasons.push("auth_missing_origin_referer");
      components["auth_missing_origin_referer"] = -10;
    }
  }

  // If IP is missing (rare), penalize hard
  if (!ctx.ip) {
    score -= 25;
    reasons.push("missing_ip");
    components["missing_ip"] = -25;
  }

  score = clamp(score, 1, 99);
  const lowScore = score <= threshold;

  return { score, reasons, components, lowScore };
}

function softDeleteUser(db: Database.Database, opts: { userId: number; adminId?: number; reason: string }) {
  const ts = nowSec();
  const u = db.prepare(`SELECT id, email, username, is_admin, COALESCE(is_deleted,0) as is_deleted FROM users WHERE id = ?`).get(opts.userId) as any;
  if (!u || u.is_admin || u.is_deleted) return { ok: false, skipped: true };

  const newEmail = `deleted+${opts.userId}.${ts}@example.invalid`;
  const newUsername = `deleted_${opts.userId}_${ts}`;

  const tx = db.transaction(() => {
    db.prepare(`
      UPDATE users
      SET
        is_disabled = 1,
        is_deleted = 1,
        deleted_at = ?,
        deleted_reason_code = ?,
        deleted_reason_text = ?,
        deleted_by = ?,
        email = ?,
        username = ?,
        name = NULL,
        first_name = NULL,
        last_name = NULL,
        display_name = NULL,
        phone = NULL
      WHERE id = ?
    `).run(
      ts,
      opts.reason,
      `Deleted by Activity system: ${opts.reason}`,
      opts.adminId ?? null,
      newEmail,
      newUsername,
      opts.userId
    );

    // Revoke sessions
    db.prepare(`
      UPDATE user_sessions
      SET revoked_at = ?, revoke_reason = 'ACCOUNT_DELETED'
      WHERE user_id = ? AND revoked_at IS NULL
    `).run(ts, opts.userId);

    // Account events
    db.prepare(`
      INSERT INTO user_account_events (user_id, event_type, event_details, ip_address, user_agent, created_at)
      VALUES (?, 'ACCOUNT_DELETED', ?, NULL, NULL, ?)
    `).run(opts.userId, JSON.stringify({ reason: opts.reason, at: ts }), ts);

    // Deletion queue update
    db.prepare(`
      UPDATE user_deletion_queue
      SET status = 'DELETED', deleted_at = ?, deleted_by = ?
      WHERE user_id = ?
    `).run(ts, opts.adminId ?? null, opts.userId);
  });

  tx();
  return { ok: true, skipped: false };
}

export function runActivityScan(dbPath: string) {
  const db = new Database(dbPath);
  try {
    db.pragma("busy_timeout = 5000");
  } catch {}

  const cfg = getCfg(db);
  const ts = nowSec();

  // 1) Rollups (precompute last activity + inactivity)
  const rollupsSql = `
    WITH last_sessions AS (
      SELECT user_id, MAX(last_active_at) AS last_session_at
      FROM user_sessions
      WHERE revoked_at IS NULL
      GROUP BY user_id
    ),
    last_logins AS (
      SELECT user_id, MAX(created_at) AS last_login_at
      FROM user_login_history
      WHERE success = 1 AND user_id IS NOT NULL
      GROUP BY user_id
    ),
    last_trades AS (
      SELECT user_id, MAX(COALESCE(closed_at, opened_at)) AS last_trade_at
      FROM trades
      GROUP BY user_id
    ),
    base AS (
      SELECT
        u.id AS user_id,
        COALESCE(s.last_session_at, 0) AS last_session_at,
        COALESCE(l.last_login_at, 0) AS last_login_at,
        COALESCE(t.last_trade_at, 0) AS last_trade_at,
        max(
          COALESCE(s.last_session_at, 0),
          COALESCE(l.last_login_at, 0),
          COALESCE(t.last_trade_at, 0),
          COALESCE(u.created_at, 0)
        ) AS last_activity_at
      FROM users u
      LEFT JOIN last_sessions s ON s.user_id = u.id
      LEFT JOIN last_logins l ON l.user_id = u.id
      LEFT JOIN last_trades t ON t.user_id = u.id
      WHERE u.is_admin = 0 AND COALESCE(u.is_deleted,0) = 0
    )
    INSERT INTO user_activity_rollups (
      user_id,
      last_session_at,
      last_login_at,
      last_trade_at,
      last_activity_at,
      inactivity_days,
      grace_starts_at,
      scheduled_delete_at,
      computed_at
    )
    SELECT
      user_id,
      NULLIF(last_session_at, 0),
      NULLIF(last_login_at, 0),
      NULLIF(last_trade_at, 0),
      last_activity_at,
      CAST((? - last_activity_at) / 86400 AS INTEGER) AS inactivity_days,
      last_activity_at + (? * 86400) AS grace_starts_at,
      last_activity_at + ((? + ?) * 86400) AS scheduled_delete_at,
      ? AS computed_at
    FROM base
    ON CONFLICT(user_id) DO UPDATE SET
      last_session_at = excluded.last_session_at,
      last_login_at = excluded.last_login_at,
      last_trade_at = excluded.last_trade_at,
      last_activity_at = excluded.last_activity_at,
      inactivity_days = excluded.inactivity_days,
      grace_starts_at = excluded.grace_starts_at,
      scheduled_delete_at = excluded.scheduled_delete_at,
      computed_at = excluded.computed_at
  `;
  db.prepare(rollupsSql).run(ts, cfg.inactivityThresholdDays, cfg.inactivityThresholdDays, cfg.deletionGraceDays, ts);

  // 2) Sync deletion queue for inactivity candidates
  db.prepare(`
    INSERT INTO user_deletion_queue (user_id, status, reason, marked_at, scheduled_delete_at)
    SELECT r.user_id, 'GRACE', 'INACTIVE', ?, r.scheduled_delete_at
    FROM user_activity_rollups r
    JOIN users u ON u.id = r.user_id
    WHERE r.inactivity_days >= ? AND COALESCE(u.is_deleted,0) = 0
    ON CONFLICT(user_id) DO UPDATE SET
      status = CASE
        WHEN user_deletion_queue.status IN ('DELETED','EXEMPT') THEN user_deletion_queue.status
        ELSE 'GRACE'
      END,
      reason = 'INACTIVE',
      scheduled_delete_at = excluded.scheduled_delete_at,
      marked_at = COALESCE(user_deletion_queue.marked_at, excluded.marked_at)
  `).run(ts, cfg.inactivityThresholdDays);

  // Clear candidates that became active again
  db.prepare(`
    UPDATE user_deletion_queue
    SET status = 'CLEARED', cleared_at = ?
    WHERE status = 'GRACE'
      AND user_id IN (
        SELECT user_id FROM user_activity_rollups WHERE inactivity_days < ?
      )
  `).run(ts, cfg.inactivityThresholdDays);

  // 3) Bot aggregates (7d/30d)
  if (cfg.botDetectionEnabled) {
    const cut7 = ts - 7 * 86400;
    const cut30 = ts - 30 * 86400;

    const aggSql = `
      INSERT INTO bot_user_scores (
        user_id, score_7d_min, score_30d_min,
        low_score_events_7d, low_score_events_30d,
        last_observation_at, last_evaluated_at, classification
      )
      SELECT
        user_id,
        COALESCE(MIN(CASE WHEN created_at >= ? THEN score END), 99) AS score_7d_min,
        COALESCE(MIN(CASE WHEN created_at >= ? THEN score END), 99) AS score_30d_min,
        SUM(CASE WHEN created_at >= ? AND score <= ? THEN 1 ELSE 0 END) AS low_score_events_7d,
        SUM(CASE WHEN created_at >= ? AND score <= ? THEN 1 ELSE 0 END) AS low_score_events_30d,
        MAX(created_at) AS last_observation_at,
        ? AS last_evaluated_at,
        CASE
          WHEN (
            SUM(CASE WHEN created_at >= ? AND score <= ? THEN 1 ELSE 0 END) >= ?
            OR COALESCE(MIN(CASE WHEN created_at >= ? THEN score END), 99) <= ?
          )
          THEN 'SUSPECT_BOT'
          ELSE 'HUMAN'
        END AS classification
      FROM bot_observations
      WHERE user_id IS NOT NULL
      GROUP BY user_id
      ON CONFLICT(user_id) DO UPDATE SET
        score_7d_min = excluded.score_7d_min,
        score_30d_min = excluded.score_30d_min,
        low_score_events_7d = excluded.low_score_events_7d,
        low_score_events_30d = excluded.low_score_events_30d,
        last_observation_at = excluded.last_observation_at,
        last_evaluated_at = excluded.last_evaluated_at,
        classification = excluded.classification
    `;
    db.prepare(aggSql).run(
      cut7, cut30,
      cut7, cfg.botScoreThreshold,
      cut30, cfg.botScoreThreshold,
      ts,
      cut7, cfg.botScoreThreshold, cfg.botMinLowScoreEvents7d,
      cut7, cfg.botScoreThreshold
    );
  }

  // 4) Auto-delete due accounts (after grace), if enabled
  let deleted = 0;
  if (cfg.activityAutoDeleteEnabled) {
    const due = db.prepare(`
      SELECT user_id AS userId
      FROM user_deletion_queue
      WHERE status = 'GRACE' AND scheduled_delete_at IS NOT NULL AND scheduled_delete_at <= ?
      LIMIT 200
    `).all(ts) as Array<{ userId: number }>;

    for (const row of due) {
      const res = softDeleteUser(db, { userId: row.userId, reason: "INACTIVE" });
      if (res.ok) deleted++;
    }
  }

  // Retention (keep bot obs bounded)
  db.prepare(`DELETE FROM bot_observations WHERE created_at < ?`).run(ts - 180 * 86400);

  return {
    ok: true,
    config: cfg,
    deletedDueToInactivity: deleted,
    ranAt: ts,
  };
}

export function adminSoftDeleteUsers(dbPath: string, userIds: number[], adminId: number, reason: "INACTIVE" | "BOT" | "MANUAL") {
  const db = new Database(dbPath);
  try {
    db.pragma("busy_timeout = 5000");
  } catch {}
  try {
    let deleted = 0;
    for (const userId of userIds) {
      const res = softDeleteUser(db, { userId, adminId, reason });
      if (res.ok) deleted++;
    }
    return { ok: true, deleted };
  } finally {
    db.close();
  }
}

import * as cron from "node-cron";

let task: ReturnType<typeof cron.schedule> | null = null;

export function startActivityScheduler(dbPath: string) {
  if (task) return;

  // Initial run shortly after boot
  setTimeout(() => {
    try {
      const r = runActivityScan(dbPath);
      console.log("[Activity Scheduler] Initial scan:", r);
    } catch (e) {
      console.error("[Activity Scheduler] Initial scan error:", e);
    }
  }, 5000);

  // Every 6 hours at minute 10 (spreads load away from grift which runs at minute 0 hourly)
  task = cron.schedule("10 */6 * * *", () => {
    try {
      console.log("[Activity Scheduler] Running scan...");
      const r = runActivityScan(dbPath);
      console.log("[Activity Scheduler] Scan result:", r);
    } catch (e) {
      console.error("[Activity Scheduler] Scan error:", e);
    }
  });

  console.log("[Activity Scheduler] Scheduled (every 6 hours)");
}

export function stopActivityScheduler() {
  task?.stop();
  task = null;
}

      score: detail.score,
      thresholdUsed: threshold,
      isLowScore: detail.lowScore,

    } as any).run();
  } catch {
    // do not block auth flows
  }

// server/services/valkey.ts
import Redis from "ioredis";

let client: Redis | null = null;

export function getValkey(): Redis | null {
  const url = process.env.VALKEY_URL;
  if (!url) return null;

  if (client) return client;

  client = new Redis(url, {
    enableReadyCheck: true,
    maxRetriesPerRequest: 1,
    lazyConnect: true,
  });

  client.on("error", () => {
    // keep silent; callers must handle null-ish behavior via try/catch wrappers
  });

  return client;
}

export async function valkeyIncrWithTtl(key: string, ttlSec: number): Promise<number | null> {
  const v = getValkey();
  if (!v) return null;
  try {
    const n = await v.incr(key);
    if (n === 1) await v.expire(key, ttlSec);
    return n;
  } catch {
    return null;
  }
}

export async function valkeySAddWithTtl(key: string, value: string, ttlSec: number): Promise<number | null> {
  const v = getValkey();
  if (!v) return null;
  try {
    const added = await v.sadd(key, value);
    await v.expire(key, ttlSec);
    return added;
  } catch {
    return null;
  }
}

export async function valkeyGetJson<T>(key: string): Promise<T | null> {
  const v = getValkey();
  if (!v) return null;
  try {
    const raw = await v.get(key);
    if (!raw) return null;
    return JSON.parse(raw) as T;
  } catch {
    return null;
  }
}

export async function valkeySetJson(key: string, value: any, ttlSec: number): Promise<boolean> {
  const v = getValkey();
  if (!v) return false;
  try {
    await v.set(key, JSON.stringify(value), "EX", ttlSec);
    return true;
  } catch {
    return false;
  }
}

import { Request } from "express";
import { randomUUID } from "crypto";
import { sha256Hex } from "../services/crypto";
import { valkeyGetJson, valkeySetJson, valkeyIncrWithTtl } from "../services/valkey";

type ChallengeRecord = {
  id: string;
  serverNonce: string;
  difficulty: number; // bits
  exp: number; // unix seconds
  ip: string;
  deviceFp?: string;
  deviceInstallId?: string;
};

export type BotChallengePayload = {
  id: string;
  serverNonce: string;
  difficulty: number;
  expiresAt: number; // unix seconds
};

export type BotProofToken = {
  id: string;
  solutionNonce: number;
  ts: number;
  deviceFp?: string;
  deviceInstallId?: string;
  digest: string; // sha256 hex of proof material
};

const memChallenges = new Map<string, ChallengeRecord>();

function nowSec() {
  return Math.floor(Date.now() / 1000);
}

function getIp(req: Request): string {
  const xf = (req.headers["x-forwarded-for"] as string | undefined) ?? "";
  const ip = xf.split(",")[0]?.trim();
  return ip || req.ip || "0.0.0.0";
}

function getDeviceFp(req: Request) {
  return (req.headers["x-device-fp"] as string | undefined) || undefined;
}
function getDeviceInstallId(req: Request) {
  return (req.headers["x-device-install-id"] as string | undefined) || undefined;
}

function b64urlDecode(s: string): string {
  const pad = s.length % 4 === 0 ? "" : "=".repeat(4 - (s.length % 4));
  const base64 = s.replace(/-/g, "+").replace(/_/g, "/") + pad;
  return Buffer.from(base64, "base64").toString("utf8");
}

export function b64urlEncodeUtf8(s: string): string {
  return Buffer.from(s, "utf8").toString("base64").replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/g, "");
}

export async function issueBotChallenge(req: Request, difficulty: number, ttlSec: number): Promise<BotChallengePayload> {
  const id = randomUUID();
  const serverNonce = randomUUID().replace(/-/g, "");
  const exp = nowSec() + ttlSec;

  const rec: ChallengeRecord = {
    id,
    serverNonce,
    difficulty,
    exp,
    ip: getIp(req),
    deviceFp: getDeviceFp(req),
    deviceInstallId: getDeviceInstallId(req),
  };

  // Prefer Valkey; fallback to memory
  const stored = await valkeySetJson(`bot:ch:${id}`, rec, ttlSec);
  if (!stored) {
    memChallenges.set(id, rec);
    setTimeout(() => memChallenges.delete(id), ttlSec * 1000).unref?.();
  }

  return { id, serverNonce, difficulty, expiresAt: exp };
}

function leadingZeroBitsOfHex(hex: string): number {
  // count leading zero bits in a hex string
  let bits = 0;
  for (let i = 0; i < hex.length; i++) {
    const v = parseInt(hex[i], 16);
    if (v === 0) {
      bits += 4;
      continue;
    }
    // count leading zero bits in this nibble
    if (v < 8) bits += 1;
    if (v < 4) bits += 1;
    if (v < 2) bits += 1;
    return bits;
  }
  return bits;
}

async function loadChallenge(id: string): Promise<ChallengeRecord | null> {
  const fromValkey = await valkeyGetJson<ChallengeRecord>(`bot:ch:${id}`);
  if (fromValkey) return fromValkey;
  return memChallenges.get(id) ?? null;
}

export async function verifyBotProof(req: Request, tokenB64Url: string): Promise<{ ok: boolean; reason?: string }> {
  let tok: BotProofToken;
  try {
    tok = JSON.parse(b64urlDecode(tokenB64Url)) as BotProofToken;
  } catch {
    return { ok: false, reason: "MALFORMED_TOKEN" };
  }

  const ch = await loadChallenge(tok.id);
  if (!ch) return { ok: false, reason: "NO_CHALLENGE" };
  if (nowSec() > ch.exp) return { ok: false, reason: "EXPIRED" };

  // bind to IP + device headers (best-effort; reduces replay across clients)
  const ip = getIp(req);
  if (ip !== ch.ip) return { ok: false, reason: "IP_MISMATCH" };

  const fp = getDeviceFp(req);
  const install = getDeviceInstallId(req);
  if (ch.deviceFp && fp && ch.deviceFp !== fp) return { ok: false, reason: "FP_MISMATCH" };
  if (ch.deviceInstallId && install && ch.deviceInstallId !== install) return { ok: false, reason: "INSTALL_MISMATCH" };

  const material = [
    ch.id,
    ch.serverNonce,
    String(tok.solutionNonce),
    fp || "",
    install || "",
  ].join("|");

  const digest = sha256Hex(material);
  if (digest !== tok.digest) return { ok: false, reason: "DIGEST_MISMATCH" };

  const z = leadingZeroBitsOfHex(digest);
  if (z < ch.difficulty) return { ok: false, reason: "INSUFFICIENT_WORK" };

  // one-time use marker (prevents trivial replay); if valkey absent, accept best-effort
  await valkeyIncrWithTtl(`bot:ch:used:${ch.id}`, 600);

  return { ok: true };
}

async function bumpWindows(req: Request, action: string, email?: string) {

  const fp = getDeviceFp(req);

  await valkeyIncrWithTtl(`bot:w:ip:${ip}:${action}:10m`, 600);

  if (inst) {

  if (fp) {
    await valkeyIncrWithTtl(`bot:w:fp:${fp}:${action}:10m`, 600);

    await valkeySAddWithTtl(`bot:w:ip:${ip}:emails:10m`, email.toLowerCase(), 600);
    if (inst) await valkeySAddWithTtl(`bot:w:inst:${inst}:emails:10m`, email.toLowerCase(), 600);
  }
}

  await bumpWindows(req, opts.action, opts.email);

  const ua = getUa(req);
  const fp = getDeviceFp(req);
  const inst = getInstallId(req);
  const tz = getClientTz(req);

  // Base score: headers + UA
  let score = 0;
  score += uaHeuristicsScore(ua);
  if (!fp) score += 15;
  if (!inst) score += 10;
  if (!tz) score += 5;

  // Optional adaptive lift by action
  if (opts.action === "SIGNUP") score += 10;
  if (opts.action === "TRADE") score += 5;

  // Proof check (adaptive requirement)
  const requiresProof =
    (opts.action === "SIGNUP" && cfg.powEnforceSignup && score >= cfg.powChallengeScore) ||
    (opts.action === "LOGIN" && cfg.powEnforceLogin && score >= cfg.powChallengeScore) ||
    (opts.action === "TRADE" && score >= (cfg.powChallengeScore + 10));

  const proofHdr = (req.headers["x-bot-proof"] as string | undefined) ?? "";
  let proofResult: "OK" | "MISSING" | "INVALID" | "NOT_REQUIRED" = "NOT_REQUIRED";

  if (requiresProof) {
    if (!proofHdr) {
      proofResult = "MISSING";
    } else {
      const ver = await verifyBotProof(req, proofHdr);
      proofResult = ver.ok ? "OK" : "INVALID";
      if (!ver.ok) score += 25; // penalty for bad proof
      if (ver.ok) score = Math.max(0, score - 20); // reward for valid proof
    }
  }

  score = clamp(score, 0, 100);

  // Store assessment if we have a userId
  if (opts.userId) {
    const prev = await db.query.botRiskAssessments.findFirst({ where: eq(botRiskAssessments.userId, opts.userId) });
    const prevScore = prev?.score ?? 0;

    // EWMA smoothing (similar *conceptually* to Cloudflare cookie smoothing, but internal)
    const newScore = Math.round(prevScore * 0.85 + score * 0.15);

    const signals = {
      action: opts.action,
      ip: getIp(req),
      ua,
      deviceFpPresent: Boolean(fp),
      installIdPresent: Boolean(inst),
      tzPresent: Boolean(tz),
      proof: proofResult,
      rawScore: score,
      ewmaPrev: prevScore,
    };

    await db
      .insert(botRiskAssessments)
      .values({
        userId: opts.userId,
        score: newScore,
        label: labelFor(newScore),
        signalsJson: JSON.stringify(signals),
      })
      .onConflictDoUpdate({
        target: botRiskAssessments.userId,
        set: {
          score: newScore,
          label: labelFor(newScore),
          signalsJson: JSON.stringify(signals),
          updatedAt: new Date(),
        } as any,
      });

    score = newScore;
  }

  // If proof is required and missing/invalid, issue a challenge and stop
  if (requiresProof && (proofResult === "MISSING" || proofResult === "INVALID")) {
    const difficulty = clamp(cfg.powBaseDifficulty + Math.floor(score / 30), cfg.powBaseDifficulty, cfg.powMaxDifficulty);
    const ch = await issueBotChallenge(req, difficulty, cfg.powTtlSec);

    return res.status(428).json({
      code: "BOT_CHALLENGE_REQUIRED",
      message: "Proof required before continuing.",
      challenge: ch,
    });
  }

  return { allowed: true as const, score, proof: proofResult };
}

export function listAdminActivity(opts: { days: number; includeDeleted?: boolean; inactiveOnly?: boolean; botsOnly?: boolean }): AdminActivityRow[] {
  const cfg = getActivityConfig();
  const db = getDb();
  const now = nowSec();
  const since = now - opts.days * 86400;

  const includeDeleted = opts.includeDeleted ? 1 : 0;
  const inactiveOnly = opts.inactiveOnly ? 1 : 0;
  const botsOnly = opts.botsOnly ? 1 : 0;

  try {
    const q = db.prepare(`
      WITH
      sess AS (
        SELECT user_id, MAX(last_active_at) AS last_seen_at
        FROM user_sessions
        GROUP BY user_id
      ),
      logins AS (
        SELECT user_id, MAX(created_at) AS last_login_at
        FROM user_login_history
        WHERE success=1
        GROUP BY user_id
      ),
      trades_last AS (
        SELECT user_id, MAX(COALESCE(closed_at, opened_at)) AS last_trade_at
        FROM trades
        GROUP BY user_id
      )
      SELECT
        u.id AS userId,
        u.email AS email,
        u.username AS username,
        u.is_disabled AS isDisabled,
        u.is_deleted AS isDeleted,
        u.created_at AS createdAt,
        max(
          COALESCE(sess.last_seen_at, 0),
          COALESCE(logins.last_login_at, 0),
          COALESCE(trades_last.last_trade_at, 0),
          u.created_at
        ) AS lastActiveAt,
        COALESCE(b.score, 0) AS botScore,
        COALESCE(b.label, 'OK') AS botLabel,
        dq.status AS queueStatus,
        dq.grace_expires_at AS graceExpiresAt
      FROM users u
      LEFT JOIN sess ON sess.user_id = u.id
      LEFT JOIN logins ON logins.user_id = u.id
      LEFT JOIN trades_last ON trades_last.user_id = u.id
      LEFT JOIN bot_risk_assessments b ON b.user_id = u.id
      LEFT JOIN user_deletion_queue dq ON dq.user_id = u.id AND dq.status = 'GRACE'
      WHERE
        u.is_admin = 0
        AND (${includeDeleted}=1 OR u.is_deleted=0)
      ORDER BY lastActiveAt ASC
      LIMIT 2000;
    `);

    const rows = q.all() as any[];

    const out: AdminActivityRow[] = rows.map((r) => {
      const last = Number(r.lastActiveAt || r.createdAt || 0);
      const inactiveDays = Math.floor((now - last) / 86400);
      return {
        userId: Number(r.userId),
        email: String(r.email),
        username: String(r.username),
        isDisabled: Boolean(r.isDisabled),
        isDeleted: Boolean(r.isDeleted),
        createdAt: Number(r.createdAt),
        lastActiveAt: last,
        inactiveDays,
        botScore: Number(r.botScore || 0),
        botLabel: String(r.botLabel || "OK"),
        queueStatus: r.queueStatus ? String(r.queueStatus) : undefined,
        graceExpiresAt: r.graceExpiresAt ? Number(r.graceExpiresAt) : undefined,
      };
    });

    let filtered = out;

    if (inactiveOnly) filtered = filtered.filter((r) => r.inactiveDays >= cfg.inactivityThresholdDays);
    if (botsOnly) filtered = filtered.filter((r) => r.botScore >= cfg.botScoreThreshold);

    return filtered;
  } finally {
    db.close();
  }
}

export function enqueueForDeletion(userIds: number[], reason: "INACTIVE" | "BOT" | "ADMIN", note: string | undefined, actorAdminId: number) {
  const cfg = getActivityConfig();
  const db = getDb();
  const now = nowSec();
  const graceExpiresAt = now + cfg.deletionGraceDays * 86400;

  try {
    const tx = db.transaction(() => {
      const ins = db.prepare(`
        INSERT INTO user_deletion_queue (user_id, status, reason, marked_at, grace_expires_at, note)
        VALUES (?, 'GRACE', ?, ?, ?, ?)
      `);

      const upd = db.prepare(`
        UPDATE user_deletion_queue
        SET status='GRACE', reason=?, marked_at=?, grace_expires_at=?, note=?
        WHERE user_id=? AND status!='EXECUTED_SOFT' AND status!='EXECUTED_HARD'
      `);

      for (const uid of userIds) {
        const existing = db.prepare("SELECT id FROM user_deletion_queue WHERE user_id=?").get(uid) as any;
        if (!existing) ins.run(uid, reason, now, graceExpiresAt, note || null);
        else upd.run(reason, now, graceExpiresAt, note || null, uid);
      }
    });

    tx();
  } finally {
    db.close();
  }
}

function ensureTombstoneUser(db: Database.Database): number {
  const existing = db.prepare("SELECT id FROM users WHERE email=?").get("deleted@system.invalid") as any;
  if (existing?.id) return Number(existing.id);

  const ins = db.prepare(`
    INSERT INTO users (email, username, password_hash, balance, created_at, is_admin)
    VALUES (?, ?, ?, '0.00', ?, 0)
  `);

  const ph = randomBytes(24).toString("hex");
  ins.run("deleted@system.invalid", "deleted_system_user", ph, nowSec());

  const row = db.prepare("SELECT id FROM users WHERE email=?").get("deleted@system.invalid") as any;
  return Number(row.id);
}

export function hardDeleteUsers(userIds: number[], actorAdminId: number, reason: string) {

            email_at_acceptance=NULL,
            ip=NULL,
            user_agent=NULL
        WHERE user_id=?
      `);

      const delSessions = db.prepare("DELETE FROM user_sessions WHERE user_id=?");
      const delSettings = db.prepare("DELETE FROM user_settings WHERE user_id=?");
      const delJournal = db.prepare("DELETE FROM trader_journal WHERE user_id=?");
      const delNotes = db.prepare("DELETE FROM user_admin_notes WHERE user_id=?");
      const delLoginHistory = db.prepare("DELETE FROM user_login_history WHERE user_id=?");
      const delAccountEvents = db.prepare("DELETE FROM user_account_events WHERE user_id=?");
      const delBot = db.prepare("DELETE FROM bot_risk_assessments WHERE user_id=?");

      // Mark queue + delete user
      const markQueue = db.prepare(`
        UPDATE user_deletion_queue
        SET status='EXECUTED_HARD', executed_at=?, executed_by_admin_id=?
        WHERE user_id=?
      `);

      const delUser = db.prepare("DELETE FROM users WHERE id=? AND is_admin=0");

        // Preserve ledger/audit continuity by reassignment
        reassignTrades.run(tombstoneId, uid);
        reassignOrderIntent.run(tombstoneId, uid);
        reassignLegal.run(tombstoneId, uid);

        delSessions.run(uid);
        delSettings.run(uid);
        delJournal.run(uid);
        delNotes.run(uid);
        delLoginHistory.run(uid);
        delAccountEvents.run(uid);
        delBot.run(uid);

        markQueue.run(now, actorAdminId, uid);
        delUser.run(uid);
      }
    });

    tx();
  } finally {
    db.close();
  }
}

export function runInactivitySweep(opts: { dryRun: boolean; actorAdminId: number }) {
  const cfg = getActivityConfig();
  const db = getDb();
  const now = nowSec();
  const thresholdSec = cfg.inactivityThresholdDays * 86400;
  const graceSec = cfg.deletionGraceDays * 86400;

  try {
    const rows = db.prepare(`
      WITH
      sess AS (SELECT user_id, MAX(last_active_at) AS last_seen_at FROM user_sessions GROUP BY user_id),
      logins AS (SELECT user_id, MAX(created_at) AS last_login_at FROM user_login_history WHERE success=1 GROUP BY user_id),
      trades_last AS (SELECT user_id, MAX(COALESCE(closed_at, opened_at)) AS last_trade_at FROM trades GROUP BY user_id)
      SELECT
        u.id AS userId,
        u.is_deleted AS isDeleted,
        max(COALESCE(sess.last_seen_at,0), COALESCE(logins.last_login_at,0), COALESCE(trades_last.last_trade_at,0), u.created_at) AS lastActiveAt
      FROM users u
      LEFT JOIN sess ON sess.user_id=u.id
      LEFT JOIN logins ON logins.user_id=u.id
      LEFT JOIN trades_last ON trades_last.user_id=u.id
      WHERE u.is_admin=0 AND u.is_deleted=0
    `).all() as any[];

    const inactive = rows.filter((r) => (now - Number(r.lastActiveAt)) >= thresholdSec).map((r) => Number(r.userId));

    const due = db.prepare(`
      SELECT user_id AS userId
      FROM user_deletion_queue
      WHERE status='GRACE' AND grace_expires_at <= ?
    `).all(now) as any[];

    const dueIds = due.map((d) => Number(d.userId));

    if (!opts.dryRun) {
      if (cfg.autoQueueInactive) enqueueForDeletion(inactive, "INACTIVE", "auto-sweep", opts.actorAdminId);
      if (cfg.autoSoftDelete && dueIds.length) softDeleteUsers(dueIds, opts.actorAdminId, "auto-grace-expired");
    }

    return {
      inactivityThresholdDays: cfg.inactivityThresholdDays,
      deletionGraceDays: cfg.deletionGraceDays,
      foundInactive: inactive.length,
      foundDue: dueIds.length,
      applied: !opts.dryRun,
      autoQueueInactive: cfg.autoQueueInactive,
      autoSoftDelete: cfg.autoSoftDelete,
    };
  } finally {
    db.close();
  }
}

// client/src/lib/botProof.ts
export type BotChallengePayload = {
  id: string;
  serverNonce: string;
  difficulty: number; // leading zero bits required
  expiresAt: number; // unix seconds
};

function b64urlEncodeUtf8(s: string): string {
  const b64 = btoa(unescape(encodeURIComponent(s)));
  return b64.replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/g, "");
}

async function sha256Hex(s: string): Promise<string> {
  const enc = new TextEncoder();
  const buf = await crypto.subtle.digest("SHA-256", enc.encode(s));
  const bytes = new Uint8Array(buf);
  return Array.from(bytes).map((b) => b.toString(16).padStart(2, "0")).join("");
}

function leadingZeroBits(hex: string): number {
  let bits = 0;
  for (let i = 0; i < hex.length; i++) {
    const v = parseInt(hex[i], 16);
    if (v === 0) { bits += 4; continue; }
    if (v < 8) bits += 1;
    if (v < 4) bits += 1;
    if (v < 2) bits += 1;
    return bits;
  }
  return bits;
}

export async function solveBotChallenge(ch: BotChallengePayload, identity: Record<string, string>): Promise<string> {
  const deviceFp = identity["x-device-fp"] || "";
  const deviceInstallId = identity["x-device-install-id"] || "";

  const start = Date.now();
  let nonce = 0;

  // Guardrails so we don't hang
  const maxMs = 2500;      // tune as needed
  const yieldEvery = 250;  // iterations

  while (Date.now() - start < maxMs) {
    const material = `${ch.id}|${ch.serverNonce}|${nonce}|${deviceFp}|${deviceInstallId}`;
    const digest = await sha256Hex(material);
    if (leadingZeroBits(digest) >= ch.difficulty) {
      const tokenObj = {
        id: ch.id,
        solutionNonce: nonce,
        ts: Math.floor(Date.now() / 1000),
        deviceFp: deviceFp || undefined,
        deviceInstallId: deviceInstallId || undefined,
        digest,
      };
      return b64urlEncodeUtf8(JSON.stringify(tokenObj));
    }
    nonce++;
    if (nonce % yieldEvery === 0) await new Promise((r) => setTimeout(r, 0));
  }

  throw new Error("BOT_PROOF_TIMEOUT");
}

export async function apiRequest(method: string, url: string, body?: any) {
  const headers: Record<string, string> = {
    "Content-Type": "application/json",
    ...getIdentityHeaders(),
  };

  const doFetch = (extra?: Record<string, string>) =>
    fetch(url, {
      method,
      headers: { ...headers, ...(extra ?? {}) },
      body: body ? JSON.stringify(body) : undefined,
      credentials: "include",
    });

  let res = await doFetch();

  // Bot challenge loop
  if (res.status === 428) {
    let data: any = null;
    try { data = await res.json(); } catch {}
    if (data?.code === "BOT_CHALLENGE_REQUIRED" && data?.challenge) {
      const proof = await solveBotChallenge(data.challenge, headers);
      res = await doFetch({ "x-bot-proof": proof });
    } else {
      throw new Error("Precondition required");
    }
  }

  if (!res.ok) {
    const text = await res.text();
    throw new Error(text || `Request failed: ${res.status}`);
  }

  // Preserve existing return behavior (json/text)
  const ct = res.headers.get("content-type") || "";
  if (ct.includes("application/json")) return res.json();
  return res.text();
}
