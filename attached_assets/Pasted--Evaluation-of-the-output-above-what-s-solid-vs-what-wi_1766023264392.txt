   * The prior code only aggregates raw PnL and average holding time; no worst instrument, no win-rate, no hold-time optimization logic, no timeframe filters (last 30d vs all-time), and no timezone handling.

5. **Frontend is not integrated**

```js
const WRITE_METHODS = new Set(["POST", "PUT", "PATCH", "DELETE"]);

export function impersonationWriteBlock(req, res, next) {
  if (req.user?.impersonated === true && WRITE_METHODS.has(req.method)) {
    return res.status(403).json({
      error: "Writes are disabled while in View-As impersonation mode."
    });
  }
  next();
}
```

```js
import express from "express";
import { adminGuard } from "../middleware/adminGuard.js";
import db from "../db.js";

const router = express.Router();

/**
 * GET /admin/users/search?q=...
 * Returns up to 20 matching users by:
 * - numeric id exact match
 * - username ILIKE
 * - email ILIKE
 */
router.get("/users/search", adminGuard, async (req, res) => {
  const qRaw = String(req.query.q || "").trim();
  if (!qRaw) return res.json([]);

  const isNumeric = /^[0-9]+$/.test(qRaw);
  const q = `%${qRaw}%`;

  const result = await db.query(
    `
    SELECT id, username, email, role, status
    FROM users
    WHERE
      (${isNumeric} AND id = $1)
      OR (username ILIKE $2)
      OR (email ILIKE $2)
    ORDER BY id DESC
    LIMIT 20
    `,
    [isNumeric ? Number(qRaw) : -1, q]
  );

  res.json(result.rows);
});

export default router;
```

### `/backend/routes/adminViewAs.js`

```js
import express from "express";
import jwt from "jsonwebtoken";
import { adminGuard } from "../middleware/adminGuard.js";
import db from "../db.js";

const router = express.Router();

function auditMeta(req, extra = {}) {
  return {
    ...extra,
    request_id: req.headers["x-request-id"] || null
  };
}

async function logAdminAction({ adminId, userId, actionType, req, metadata }) {
  await db.query(
    `
    INSERT INTO admin_actions (admin_id, user_id, action_type, metadata, ip, user_agent)
    VALUES ($1, $2, $3, $4, $5, $6)
    `,
    [
      adminId,
      userId,
      actionType,
      metadata ? JSON.stringify(metadata) : null,
      req.ip || null,
      req.headers["user-agent"] || null
    ]
  );
}

/**
 * POST /admin/view-as/start
 * body: { targetUserId }
 */
router.post("/view-as/start", adminGuard, async (req, res) => {
  const { targetUserId } = req.body || {};
  if (!targetUserId) {
    return res.status(400).json({ error: "targetUserId required" });
  }

  const user = await db.query(
    "SELECT id, username, email, role, status FROM users WHERE id = $1",
    [targetUserId]
  );

  if (!user.rows.length) {
    return res.status(404).json({ error: "User not found" });
  }

  const u = user.rows[0];

  // Impersonation token is a user-context token with explicit impersonation claims
  const token = jwt.sign(
    {
      id: u.id,
      role: u.role,
      username: u.username,
      email: u.email,
      impersonated: true,
      impersonated_by: req.user.id
    },
    process.env.JWT_SECRET,
    { expiresIn: "15m" }
  );

  await logAdminAction({
    adminId: req.user.id,
    userId: u.id,
    actionType: "VIEW_AS_START",
    req,
    metadata: auditMeta(req, { target: { id: u.id, username: u.username, email: u.email } })
  });

  return res.json({
    token,
    target: { id: u.id, username: u.username, email: u.email }
  });
});

/**
 * POST /admin/view-as/stop
 * body: { targetUserId }
 * This is purely for audit completeness.
 */
router.post("/view-as/stop", adminGuard, async (req, res) => {
  const { targetUserId } = req.body || {};
  if (!targetUserId) return res.status(400).json({ error: "targetUserId required" });

  await logAdminAction({
    adminId: req.user.id,
    userId: targetUserId,
    actionType: "VIEW_AS_STOP",
    req,
    metadata: auditMeta(req, { note: "Admin exited View-As" })
  });

  res.json({ success: true });
});

export default router;
```

## A7) Trader Insights: computed + cached + timeframe support

### `/backend/services/traderInsights.js`

```js
import db from "../db.js";

function bucketHoldMinutes(mins) {
  if (mins == null || Number.isNaN(mins)) return "unknown";
  if (mins < 1) return "<1m";
  if (mins < 5) return "1-5m";
  if (mins < 15) return "5-15m";
  if (mins < 30) return "15-30m";
  if (mins < 60) return "30-60m";
  if (mins < 180) return "1-3h";
  return "3h+";
}

export async function computeInsights({ userId, sinceDays = 30 }) {
  // Filter window; you can change to 90d or all-time by passing sinceDays=null
  const whereTime = sinceDays ? `AND entry_time >= NOW() - INTERVAL '${sinceDays} days'` : "";

  // Pull only columns needed
  const r = await db.query(
    `
    SELECT instrument, direction, profit,
           EXTRACT(EPOCH FROM (exit_time - entry_time))/60 AS hold_minutes,
           EXTRACT(HOUR FROM entry_time) AS entry_hour
    FROM trades
    WHERE user_id = $1
      AND exit_time IS NOT NULL
      AND entry_time IS NOT NULL
      ${whereTime}
    `,
    [userId]
  );

  const byInstrument = {};
  const byDirection = {};
  const byHour = {};
  const byHoldBucket = {};

  let totalTrades = 0;
  let wins = 0;
  let totalPnl = 0;

  for (const t of r.rows) {
    totalTrades++;
    totalPnl += Number(t.profit || 0);
    if (Number(t.profit || 0) > 0) wins++;

    const inst = t.instrument || "UNKNOWN";
    const dir = t.direction || "UNKNOWN";
    const hour = t.entry_hour != null ? String(t.entry_hour).padStart(2, "0") : "NA";
    const bucket = bucketHoldMinutes(Number(t.hold_minutes));

    byInstrument[inst] ??= { pnl: 0, trades: 0, wins: 0 };
    byInstrument[inst].pnl += Number(t.profit || 0);
    byInstrument[inst].trades++;
    if (Number(t.profit || 0) > 0) byInstrument[inst].wins++;

    byDirection[dir] ??= { pnl: 0, trades: 0, wins: 0 };
    byDirection[dir].pnl += Number(t.profit || 0);
    byDirection[dir].trades++;
    if (Number(t.profit || 0) > 0) byDirection[dir].wins++;

    byHour[hour] ??= { pnl: 0, trades: 0, wins: 0 };
    byHour[hour].pnl += Number(t.profit || 0);
    byHour[hour].trades++;
    if (Number(t.profit || 0) > 0) byHour[hour].wins++;

    byHoldBucket[bucket] ??= { pnl: 0, trades: 0, wins: 0 };
    byHoldBucket[bucket].pnl += Number(t.profit || 0);
    byHoldBucket[bucket].trades++;
    if (Number(t.profit || 0) > 0) byHoldBucket[bucket].wins++;
  }

  function bestWorst(mapObj) {
    const entries = Object.entries(mapObj).map(([k, v]) => ({
      key: k,
      pnl: v.pnl,
      trades: v.trades,
      win_rate: v.trades ? v.wins / v.trades : 0
    }));
    entries.sort((a, b) => b.pnl - a.pnl);
    return {
      best: entries[0] || null,
      worst: entries.length ? entries[entries.length - 1] : null,
      ranked: entries
    };
  }

  const inst = bestWorst(byInstrument);
  const dir = bestWorst(byDirection);
  const hour = bestWorst(byHour);
  const hold = bestWorst(byHoldBucket);

  return {
    window_days: sinceDays,
    totals: {
      trades: totalTrades,
      pnl: totalPnl,
      win_rate: totalTrades ? wins / totalTrades : 0
    },
    instruments: inst,
    directions: dir,
    hours: hour,
    hold_time: hold
  };
}
```

### `/backend/routes/traderInsights.js`

```js
import express from "express";
import db from "../db.js";
import { computeInsights } from "../services/traderInsights.js";

const router = express.Router();

const CACHE_TTL_MINUTES = 12 * 60; // 12 hours

router.get("/insights", async (req, res) => {
  const sinceDays = req.query.since_days ? Number(req.query.since_days) : 30;
  const refresh = String(req.query.refresh || "0") === "1";

  const cached = await db.query(
    `SELECT insights, generated_at FROM trader_insights_cache WHERE user_id = $1`,
    [req.user.id]
  );

  if (!refresh && cached.rows.length) {
    const generatedAt = new Date(cached.rows[0].generated_at).getTime();
    const ageMinutes = (Date.now() - generatedAt) / (1000 * 60);

    // Cache can store multiple windows; simplest is store one blob with window in it.
    // If window mismatch, recompute.
    const blob = cached.rows[0].insights;
    if (ageMinutes < CACHE_TTL_MINUTES && blob?.window_days === sinceDays) {
      return res.json(blob);
    }
  }

  const insights = await computeInsights({ userId: req.user.id, sinceDays });

  await db.query(
    `
    INSERT INTO trader_insights_cache (user_id, insights, generated_at)
    VALUES ($1, $2, NOW())
    ON CONFLICT (user_id)
    DO UPDATE SET insights = $2, generated_at = NOW()
    `,
    [req.user.id, JSON.stringify(insights)]
  );

  res.json(insights);
});

export default router;
```

```jsx
import { useMemo, useState } from "react";
import api from "../utils/api";

export default function ImpersonationBanner() {
  const [busy, setBusy] = useState(false);

  const target = useMemo(() => {
    try {
      return JSON.parse(localStorage.getItem("impersonated_target") || "null");
    } catch {
      return null;
    }
  }, []);

  const isImpersonating = useMemo(() => {
    // Minimal check: presence of stored target
    return Boolean(target);
  }, [target]);

  const exit = async () => {
    const adminToken = localStorage.getItem("admin_token");
    if (!adminToken || !target?.id) {
      // Fallback: hard reload
      localStorage.removeItem("impersonated_target");
      window.location.href = "/admin";
      return;
    }

    setBusy(true);
    try {
      // Temporarily restore admin token so we can log stop-event
      localStorage.setItem("token", adminToken);
      await api.post("/admin/view-as/stop", { targetUserId: target.id });

      // Clear impersonation state
      localStorage.removeItem("impersonated_target");
      localStorage.removeItem("admin_token");

      window.location.href = "/admin?viewas_stopped=1";
    } finally {
      setBusy(false);
    }
  };

  if (!isImpersonating) return null;

  return (
    <div style={{
      position: "sticky",
      top: 0,
      zIndex: 9999,
      padding: "10px 12px",
      borderBottom: "1px solid #333",
      background: "rgba(255, 200, 0, 0.08)"
    }}>
      <strong>VIEW AS ACTIVE</strong>
      <span style={{ marginLeft: 10, opacity: 0.9 }}>
        {target?.username} (id: {target?.id}, {target?.email})
      </span>

      <button
        onClick={exit}
        disabled={busy}
        style={{ float: "right" }}
      >
        {busy ? "Exiting..." : "Exit View As"}
      </button>
    </div>
  );
}
```

```jsx
          placeholder="Write your trade reasoning, emotions, plan vs execution, post-mortem, etc."
          rows={6}
          style={{ width: "100%" }}
        />

        <div style={{ display: "flex", gap: 8, marginTop: 10 }}>
          <input value={mood} onChange={(e) => setMood(e.target.value)} placeholder="Mood (optional)" />
          <input value={tags} onChange={(e) => setTags(e.target.value)} placeholder="Tags (comma separated)" />
          <input value={attachmentUrl} onChange={(e) => setAttachmentUrl(e.target.value)} placeholder="Attachment URL (optional)" />
        </div>

        <button onClick={add} disabled={busy || note.trim().length < 3} style={{ marginTop: 10 }}>
          {busy ? "Saving..." : "Add Entry"}
        </button>
      </div>

      {entries.map((e) => (
        <div key={e.id} className="card" style={{ marginBottom: 10 }}>
          <div style={{ opacity: 0.75 }}>
            {new Date(e.created_at).toLocaleString()}
            {e.updated_at !== e.created_at ? " (edited)" : ""}
          </div>
          <div style={{ marginTop: 8, whiteSpace: "pre-wrap" }}>{e.note}</div>

          <div style={{ marginTop: 8, opacity: 0.8 }}>
            {e.mood ? <span>Mood: {e.mood} </span> : null}
            {Array.isArray(e.tags) && e.tags.length ? <span>Tags: {e.tags.join(", ")} </span> : null}
            {e.attachment_url ? <span>Attachment: {e.attachment_url}</span> : null}
          </div>

          <button onClick={() => remove(e.id)} style={{ marginTop: 10 }}>
            Delete
          </button>
        </div>
      ))}
    </div>
  );
}
```

## B5) Trader Insights card (meets your specified insight categories)

### `/frontend/components/account/TraderInsightsCard.jsx`

```jsx
import { useEffect, useState } from "react";
import api from "../../utils/api";

function fmtPct(x) {
  if (x == null) return "-";
  return `${(x * 100).toFixed(1)}%`;
}

function fmtNum(x) {
  if (x == null) return "-";
  return Number(x).toFixed(2);
}

export default function TraderInsightsCard() {
  const [data, setData] = useState(null);
  const [sinceDays, setSinceDays] = useState(30);

  const load = async (refresh = false) => {
    const res = await api.get(`/insights?since_days=${sinceDays}&refresh=${refresh ? 1 : 0}`);
    setData(res.data);
  };

  useEffect(() => { load(false); }, [sinceDays]);

  if (!data) return null;

  const bestInst = data.instruments?.best;
  const worstInst = data.instruments?.worst;

  const bestDir = data.directions?.best;
  const bestHour = data.hours?.best;
  const bestHold = data.hold_time?.best;

  return (
    <div className="card" style={{ marginTop: 12 }}>
      <div style={{ display: "flex", justifyContent: "space-between", gap: 12 }}>
        <h3 style={{ margin: 0 }}>Trader Insights</h3>

        <div style={{ display: "flex", gap: 8 }}>
          <select value={sinceDays} onChange={(e) => setSinceDays(Number(e.target.value))}>
            <option value={7}>Last 7d</option>
            <option value={30}>Last 30d</option>
            <option value={90}>Last 90d</option>
          </select>
          <button onClick={() => load(true)}>Refresh</button>
        </div>
      </div>

      <div style={{ marginTop: 10, opacity: 0.85 }}>
        Trades: <strong>{data.totals?.trades ?? 0}</strong> | Win rate: <strong>{fmtPct(data.totals?.win_rate)}</strong> | PnL: <strong>{fmtNum(data.totals?.pnl)}</strong>
      </div>

      <div style={{ marginTop: 12 }}>
        <div><strong>Most profitable instrument:</strong> {bestInst ? `${bestInst.key} (PnL ${fmtNum(bestInst.pnl)}, WR ${fmtPct(bestInst.win_rate)})` : "-"}</div>
        <div><strong>Most loss-making instrument:</strong> {worstInst ? `${worstInst.key} (PnL ${fmtNum(worstInst.pnl)}, WR ${fmtPct(worstInst.win_rate)})` : "-"}</div>
        <div><strong>Best direction:</strong> {bestDir ? `${bestDir.key} (PnL ${fmtNum(bestDir.pnl)}, WR ${fmtPct(bestDir.win_rate)})` : "-"}</div>
        <div><strong>Most profitable hour:</strong> {bestHour ? `${bestHour.key}:00 (PnL ${fmtNum(bestHour.pnl)}, WR ${fmtPct(bestHour.win_rate)})` : "-"}</div>
        <div><strong>Optimal holding-time bucket:</strong> {bestHold ? `${bestHold.key} (PnL ${fmtNum(bestHold.pnl)}, WR ${fmtPct(bestHold.win_rate)})` : "-"}</div>
      </div>
    </div>
  );
}
```
