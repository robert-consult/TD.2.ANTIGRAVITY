```tsx
import React, { useEffect, useMemo, useState } from "react";

type AuditRow = {
  id: number;
  userId: number;
  eventType: string;
  correlationId: string;
  actorType: "USER" | "ADMIN" | "SYSTEM";
  actorUserId: number;
  sessionId?: string | null;
  ip?: string | null;
  userAgent?: string | null;
  data: any;
  createdAtMs: number;
};

type FetchParams = {
  userId?: number;
  eventType?: string;
  fromMs?: number;
  toMs?: number;
  q?: string;
  limit?: number;
  offset?: number;
};

async function fetchAudit(params: FetchParams): Promise<{ rows: AuditRow[]; total: number }> {
  const qs = new URLSearchParams();
  Object.entries(params).forEach(([k, v]) => {
    if (v === undefined || v === null || v === "") return;
    qs.set(k, String(v));
  });
  const res = await fetch(`/api/admin/audit?${qs.toString()}`, { headers: { "x-correlation-id": crypto.randomUUID() } });
  if (!res.ok) return { rows: [], total: 0 };
  return res.json();
}

function toDateTime(ms: number) {
  const d = new Date(ms);
  return d.toLocaleString();
}

function downloadCsv(rows: AuditRow[]) {
  const header = ["id","userId","eventType","createdAt","correlationId","actorType","actorUserId","sessionId","ip","userAgent","dataJSON"];
  const lines = rows.map(r => [
    r.id, r.userId, r.eventType, new Date(r.createdAtMs).toISOString(), r.correlationId, r.actorType, r.actorUserId,
    r.sessionId ?? "", r.ip ?? "", (r.userAgent ?? "").replaceAll(",", " "),
    JSON.stringify(r.data ?? {}).replaceAll(",", ";")
  ].join(","));
  const blob = new Blob([header.join(",") + "\n" + lines.join("\n")], { type: "text/csv" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = `user_audit_${Date.now()}.csv`;
  a.click();
  URL.revokeObjectURL(url);
}

export default function AdminAuditTrail() {
  const [rows, setRows] = useState<AuditRow[]>([]);
  const [total, setTotal] = useState(0);
  const [loading, setLoading] = useState(false);

  const [q, setQ] = useState<string>("");
  const [from, setFrom] = useState<string>("");
  const [to, setTo] = useState<string>("");

  const [page, setPage] = useState(0);
  const limit = 50;

  const params = useMemo<FetchParams>(() => ({
    userId: userId ? Number(userId) : undefined,
    eventType: eventType || undefined,
    q: q || undefined,
    fromMs: from ? new Date(from).getTime() : undefined,
    toMs: to ? new Date(to).getTime() : undefined,
    limit,
    offset: page * limit,
  }), [userId, eventType, q, from, to, page]);

  async function load() {
    setLoading(true);
    const data = await fetchAudit(params);
    setRows(data.rows);
    setTotal(data.total);
    setLoading(false);
  }

  useEffect(() => { load(); /* eslint-disable-next-line */ }, [params]);

  return (
    <div className="space-y-3">
      <div className="flex flex-wrap items-end gap-2">
        <div className="flex flex-col">
          <label className="text-xs opacity-70">User ID</label>
          <input className="rounded-md border border-slate-700/60 bg-slate-900/60 px-2 py-1 text-sm" value={userId} onChange={(e)=>setUserId(e.target.value)} placeholder="e.g. 42" />
        </div>
        <div className="flex flex-col">
          <label className="text-xs opacity-70">Event Type</label>
          <input className="rounded-md border border-slate-700/60 bg-slate-900/60 px-2 py-1 text-sm" value={eventType} onChange={(e)=>setEventType(e.target.value)} placeholder="e.g. EMAIL_VERIFIED" />
        </div>
        <div className="flex flex-col">
          <label className="text-xs opacity-70">Query</label>
          <input className="rounded-md border border-slate-700/60 bg-slate-900/60 px-2 py-1 text-sm" value={q} onChange={(e)=>setQ(e.target.value)} placeholder="correlationId, ip, etc." />
        </div>
```

```tsx
  const qs = status ? `?status=${encodeURIComponent(status)}` : "";

async function postAction(userId: number, action: "approve" | "reject" | "request-info") {

    method: "POST",
    headers: { "Content-Type": "application/json", "x-correlation-id": crypto.randomUUID() },
    body: JSON.stringify({}),

}
```

```tsx
type Funnel = {

function pct(a: number, b: number) {
```

### `server/routes/admin.audit.routes.ts`

```ts
import { Router } from "express";
import { db } from "../db";
import { userAudit } from "../../shared/schema";
import { and, desc, eq, gte, lte, like, sql } from "drizzle-orm";

export function buildAdminAuditRoutes() {
  const r = Router();

  r.get("/", async (req, res) => {
    const userId = req.query.userId ? Number(req.query.userId) : undefined;
    const eventType = req.query.eventType ? String(req.query.eventType) : undefined;
    const fromMs = req.query.fromMs ? Number(req.query.fromMs) : undefined;
    const toMs = req.query.toMs ? Number(req.query.toMs) : undefined;
    const q = req.query.q ? String(req.query.q) : undefined;
    const limit = req.query.limit ? Math.min(200, Number(req.query.limit)) : 50;
    const offset = req.query.offset ? Number(req.query.offset) : 0;

    const where: any[] = [];
    if (userId) where.push(eq(userAudit.userId, userId));
    if (eventType) where.push(eq(userAudit.eventType, eventType));
    if (fromMs) where.push(gte(userAudit.createdAtMs, fromMs));
    if (toMs) where.push(lte(userAudit.createdAtMs, toMs));
    if (q) where.push(like(userAudit.data, `%${q}%`)); // naive JSON substring

    const cond = where.length ? and(...where) : undefined;

    const totalRow = await db
      .select({ c: sql<number>`count(1)` })
      .from(userAudit)
      .where(cond as any);

    const rows = await db
      .select()
      .from(userAudit)
      .where(cond as any)
      .orderBy(desc(userAudit.createdAtMs))
      .limit(limit)
      .offset(offset);

    res.json({ rows, total: Number(totalRow[0]?.c ?? rows.length) });
  });

  return r;
}
```

```ts
      })
      .from(users)
      .innerJoin(userPolicyState, eq(userPolicyState.userId, users.id))
      .where(
        and(
          // only tiers relevant to selection pipeline
          // (CANDIDATE_SMS_REQUIRED, VERIFIED_SMS, SELECTED_REAL_CAPITAL)
          // Note: SQLite; emulate IN with OR for clarity
          and(
            and(
              (userPolicyState.contenderTier as any).in?.(["CANDIDATE_SMS_REQUIRED","VERIFIED_SMS","SELECTED_REAL_CAPITAL"]) ??
              // fallback when .in not available
              (userPolicyState.contenderTier as any)
            ),
            status ? eq(userPolicyState.kycStatus, status) : (userPolicyState.kycStatus as any)
          )
        ) as any
      )
      .orderBy(desc(userPolicyState.updatedAtMs));

    res.json(rows);
  });

  // Actions
  r.post("/kyc/:userId/approve", async (req, res) => {
    const userId = Number(req.params.userId);
    await db.update(userPolicyState).set({ kycStatus: "VERIFIED", updatedAtMs: Date.now() } as any).where(eq(userPolicyState.userId, userId));
    res.json({ ok: true });
  });

  r.post("/kyc/:userId/reject", async (req, res) => {
    const userId = Number(req.params.userId);
    await db.update(userPolicyState).set({ kycStatus: "REJECTED", updatedAtMs: Date.now() } as any).where(eq(userPolicyState.userId, userId));
    res.json({ ok: true });
  });

  r.post("/kyc/:userId/request-info", async (req, res) => {
    // implement notification/logging as you wish
    res.json({ ok: true });
  });

  return r;
}
```

### `server/routes/admin.analytics.routes.ts`

```ts
import { Router } from "express";
import { db } from "../db";
import { users, userPolicyState, userAudit } from "../../shared/schema";
import { and, eq, gte, sql } from "drizzle-orm";

const MS_DAY = 86400000;

export function buildAdminAnalyticsRoutes() {
  const r = Router();

  // Funnel
  r.get("/analytics/funnel", async (_req, res) => {
    const now = Date.now();

    // Counts by states (simple, fast approximations)
    const totalUsers = await db.select({ c: sql<number>`count(1)` }).from(users);
    const verified = await db.select({ c: sql<number>`count(1)` }).from(userPolicyState).where(sql`email_verified_at_ms is not null`);
    const contender = await db.select({ c: sql<number>`count(1)` }).from(userPolicyState).where(sql`contender_tier in ('CANDIDATE_SMS_REQUIRED','VERIFIED_SMS','SELECTED_REAL_CAPITAL')`);
    const selected = await db.select({ c: sql<number>`count(1)` }).from(userPolicyState).where(sql`selected_at_ms is not null`);
    const kycSubmitted = await db.select({ c: sql<number>`count(1)` }).from(userPolicyState).where(sql`kyc_status = 'PENDING'`);
    const kycVerified = await db.select({ c: sql<number>`count(1)` }).from(userPolicyState).where(sql`kyc_status = 'VERIFIED'`);

    // Active trader ≈ anyone with an audit "TRADE_*" or a trade row in last 30d; here we use audit heuristic:
    const day30 = now - 30 * MS_DAY;
    const activeTrader = await db.select({ c: sql<number>`count(distinct ${userAudit.userId})` })
      .from(userAudit)
      .where(and(gte(userAudit.createdAtMs, day30), sql`${userAudit.eventType} like 'TRADE_%'`));

    res.json({
      signup: Number(totalUsers[0].c) ?? 0,
      verified: Number(verified[0].c) ?? 0,
      activeTrader: Number(activeTrader[0].c) ?? 0,
      contender: Number(contender[0].c) ?? 0,
      selected: Number(selected[0].c) ?? 0,
      kycSubmitted: Number(kycSubmitted[0].c) ?? 0,
      kycVerified: Number(kycVerified[0].c) ?? 0,
    });
  });

  // Compliance metrics
  r.get("/analytics/compliance", async (_req, res) => {
    const now = Date.now();

    const verifiedWithin14d = await db.select({ c: sql<number>`
      count(1)
    `}).from(userPolicyState).where(sql`
      email_verified_at_ms is not null
      and email_initial_due_at_ms is not null
      and email_verified_at_ms <= email_initial_due_at_ms
    `);

    const overdueReverify = await db.select({ c: sql<number>`count(1)` })
      .from(userPolicyState)
      .where(sql`email_reverify_due_at_ms is not null and (${now} - email_reverify_due_at_ms) > (3 * 86400000)`);

    const lockedForUnverified = await db.select({ c: sql<number>`count(1)` })
      .from(userPolicyState)
      .where(sql`lock_reason = 'EMAIL_UNVERIFIED'`);

    const lockedForReverify = await db.select({ c: sql<number>`count(1)` })
      .from(userPolicyState)
      .where(sql`lock_reason = 'EMAIL_REVERIFY_OVERDUE'`);

    res.json({
      verifiedWithin14d: Number(verifiedWithin14d[0].c) ?? 0,
      overdueReverify: Number(overdueReverify[0].c) ?? 0,
      lockedForUnverified: Number(lockedForUnverified[0].c) ?? 0,
      lockedForReverify: Number(lockedForReverify[0].c) ?? 0,
    });
  });

  // User analytics (DAU/WAU/heatmap) — approximate from userAudit LOGIN/LOGOUT if present
  r.get("/analytics/user", async (_req, res) => {
    const now = Date.now();
    const day7 = now - 7 * MS_DAY;
    const day1 = now - 1 * MS_DAY;

    const dau = await db.select({ c: sql<number>`count(distinct ${userAudit.userId})` })
      .from(userAudit)
      .where(and(gte(userAudit.createdAtMs, day1), eq(userAudit.eventType, "LOGIN")));

    const wau = await db.select({ c: sql<number>`count(distinct ${userAudit.userId})` })
      .from(userAudit)
      .where(and(gte(userAudit.createdAtMs, day7), eq(userAudit.eventType, "LOGIN")));

    const lockedUsers = await db.select({ c: sql<number>`count(1)` })
      .from(userPolicyState)
      .where(sql`lock_reason is not null`);

    const activeTraders = await db.select({ c: sql<number>`count(distinct ${userAudit.userId})` })
      .from(userAudit)
      .where(and(gte(userAudit.createdAtMs, day7), sql`${userAudit.eventType} like 'TRADE_%'`));

    // Heatmap 7x24
    const heat = Array.from({ length: 7 }, () => Array.from({ length: 24 }, () => 0));
    const loginRows = await db.select({
      ts: userAudit.createdAtMs,
    }).from(userAudit).where(and(gte(userAudit.createdAtMs, day7), eq(userAudit.eventType, "LOGIN")));

    for (const r of loginRows) {
      const d = new Date(Number((r as any).ts));
      const day = d.getDay(); // 0..6
      const hour = d.getHours(); // 0..23
      heat[day][hour] += 1;
    }

    res.json({
      dau: Number(dau[0].c) ?? 0,
      wau: Number(wau[0].c) ?? 0,
      activeTraders: Number(activeTraders[0].c) ?? 0,
      lockedUsers: Number(lockedUsers[0].c) ?? 0,
      loginHeatmap: heat,
    });
  });

  return r;
}
```
