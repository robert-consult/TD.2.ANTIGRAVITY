# 1) Backend: return `bestReason` + `bestScore`

## 1.1 Patch `POST /resolve-doc-links` in `server/routes/adminLegalAcceptances.ts`

In the handler you just updated, replace the chooser functions with the versions below and update the `resolved[...]` object to include `reason`/`score`.

```ts
// Inside POST /resolve-doc-links handler (same file)

// helper
const pickWithReason = (docs: any[], scorer: (d: any) => { score: number; reason: string }) => {
  if (!docs || docs.length === 0) return { best: null, bestReason: "NO_MATCH", bestScore: -1 };

  const sorted = docs
    .slice()
    .sort((a, b) => {
      const sa = scorer(a);
      const sb = scorer(b);
      if (sb.score !== sa.score) return sb.score - sa.score;

      const ta = createdAtMs(a);
      const tb = createdAtMs(b);
      if (tb !== ta) return tb - ta;

      return Number(b.id) - Number(a.id);
    });

  const best = sorted[0];
  const bestMeta = scorer(best);
  return { best, bestReason: bestMeta.reason, bestScore: bestMeta.score };
};

// GLOBAL scorer: prefer DEFAULT/GLOBAL
const scoreGlobal = (d: any) => {
  const jt = String(d.jurisdictionType || "");
  const jk = String(d.jurisdictionKey || "");
  if (jt === "DEFAULT" && jk === "GLOBAL") return { score: 3, reason: "DEFAULT/GLOBAL" };
  if (jt === "DEFAULT") return { score: 2, reason: "DEFAULT/*" };
  return { score: 1, reason: "OTHER" };
};

// ADDENDUM scorer: COUNTRY/{ISO2} > REGION/{regionKey} > DEFAULT/ROW
const scoreAddendum = (iso2: string, regionKey: string | null) => {
  const ISO2 = String(iso2 || "").toUpperCase().trim();
  const RK = regionKey ? String(regionKey).trim() : null;

  return (d: any) => {
    const jt = String(d.jurisdictionType || "");
    const jk = String(d.jurisdictionKey || "");

    if (jt === "COUNTRY" && jk === ISO2) return { score: 3, reason: `COUNTRY/${ISO2}` };
    if (RK && jt === "REGION" && jk === RK) return { score: 2, reason: `REGION/${RK}` };
    if (jt === "DEFAULT" && jk === "ROW") return { score: 1, reason: "DEFAULT/ROW" };
    return { score: 0, reason: "OTHER" };
  };
};

// ... inside for-loop over acceptances:
for (const a of acceptances) {
  const gSha = String(a.globalDocSha256 || "");
  const adSha = String(a.addendumSha256 || "");

  const gMatches = globalMap.get(gSha) || [];
  const aMatches = addMap.get(adSha) || [];

  const gPicked = pickWithReason(gMatches, scoreGlobal);
  const aPicked = pickWithReason(
    aMatches,
    scoreAddendum(String(a.countryIso2 || ""), a.regionKey ? String(a.regionKey) : null)
  );

  resolved[Number(a.id)] = {
    global: { best: gPicked.best, matches: gMatches, bestReason: gPicked.bestReason, bestScore: gPicked.bestScore },
    addendum: { best: aPicked.best, matches: aMatches, bestReason: aPicked.bestReason, bestScore: aPicked.bestScore },
  };
}
```

That’s it for backend.

## Result

Admins will see, per acceptance:

* `G#123 / A#456` plus badges like:
