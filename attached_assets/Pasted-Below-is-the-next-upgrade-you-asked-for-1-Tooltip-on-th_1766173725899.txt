```tsx
  const validateTPSL = (takeProfit: number | null, stopLoss: number | null) => {
    const messages: { tp?: string; sl?: string } = {};
    const priceLabel = trade.status === "PENDING" ? "order price" : "current price";

    const tp = typeof takeProfit === "number" && Number.isFinite(takeProfit) ? takeProfit : null;
    const sl = typeof stopLoss === "number" && Number.isFinite(stopLoss) ? stopLoss : null;

    if (trade.type === "BUY") {
      // BUY: TP >= ref + minDistance ; SL <= ref - minDistance
      if (sl !== null && sl > (referencePrice - minDistance)) {
        messages.sl = `Stop Loss must be at least ${isJpyPair ? "0.10" : "0.0010"} (10 points) below ${priceLabel} of ${referencePrice.toFixed(isJpyPair ? 2 : 4)}`;
      }
      if (tp !== null && tp < (referencePrice + minDistance)) {
        messages.tp = `Take Profit must be at least ${isJpyPair ? "0.10" : "0.0010"} (10 points) above ${priceLabel} of ${referencePrice.toFixed(isJpyPair ? 2 : 4)}`;
      }
    } else if (trade.type === "SELL") {
      // SELL: TP <= ref - minDistance ; SL >= ref + minDistance
      if (sl !== null && sl < (referencePrice + minDistance)) {
        messages.sl = `Stop Loss must be at least ${isJpyPair ? "0.10" : "0.0010"} (10 points) above ${priceLabel} of ${referencePrice.toFixed(isJpyPair ? 2 : 4)}`;
      }
      if (tp !== null && tp > (referencePrice - minDistance)) {
        messages.tp = `Take Profit must be at least ${isJpyPair ? "0.10" : "0.0010"} (10 points) below ${priceLabel} of ${referencePrice.toFixed(isJpyPair ? 2 : 4)}`;
      }
    }

    setValidationMessages(messages);
    return Object.keys(messages).length === 0;
  };
```

### 5) Add the one-click Auto-Fix function

Add this function near `handleSubmit`:

```tsx
  const autoFixTargets = () => {
    if (!referencePrice || referencePrice <= 0) return;

    const decimals = isJpyPair ? 2 : 4;

    const recommendedTP =
      trade.type === "BUY"
        ? Number((referencePrice + minDistance).toFixed(decimals))
        : Number((referencePrice - minDistance).toFixed(decimals));

    const recommendedSL =
      trade.type === "BUY"
        ? Number((referencePrice - minDistance).toFixed(decimals))
        : Number((referencePrice + minDistance).toFixed(decimals));

    const currentTPRaw = form.getValues("takeProfit");
    const currentSLRaw = form.getValues("stopLoss");

    const currentTP = typeof currentTPRaw === "number" && Number.isFinite(currentTPRaw) ? currentTPRaw : null;
    const currentSL = typeof currentSLRaw === "number" && Number.isFinite(currentSLRaw) ? currentSLRaw : null;

    // If missing OR invalid, set to recommended; otherwise keep user values.
    let nextTP = currentTP;
    let nextSL = currentSL;

    if (trade.type === "BUY") {
      if (nextTP === null || nextTP < referencePrice + minDistance) nextTP = recommendedTP;
      if (nextSL === null || nextSL > referencePrice - minDistance) nextSL = recommendedSL;
    } else {
      if (nextTP === null || nextTP > referencePrice - minDistance) nextTP = recommendedTP;
      if (nextSL === null || nextSL < referencePrice + minDistance) nextSL = recommendedSL;
    }

    form.setValue("takeProfit", nextTP ?? undefined, { shouldDirty: true, shouldTouch: true });
    form.setValue("stopLoss", nextSL ?? undefined, { shouldDirty: true, shouldTouch: true });

    validateTPSL(nextTP, nextSL);
  };
```
