## B) KycQueue.tsx


tsx
import { useMemo, useState } from "react";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter } from "@/components/ui/dialog";
import { apiRequest } from "@/lib/queryClient";

type KycRow = {
  userId: number;
  email: string;
  username: string;
  status: "NOT_STARTED" | "INVITED" | "SUBMITTED" | "APPROVED" | "REJECTED";
  invitedAt?: number | string | null;
  submittedAt?: number | string | null;
  reviewedAt?: number | string | null;
  reviewerNote?: string | null;
};

export default function KycQueue() {
  const qc = useQueryClient();
  const [search, setSearch] = useState("");
  const [noteOpen, setNoteOpen] = useState(false);
  const [noteUserId, setNoteUserId] = useState<number | null>(null);
  const [noteText, setNoteText] = useState("");

  const { data, isLoading } = useQuery({
    queryKey: ["admin-kyc-queue"],
    queryFn: async () => {
      const res = await apiRequest("GET", `/api/admin/kyc/queue`);
      return (await res.json()) as KycRow[];
    },
  });

  const rows = useMemo(() => {
    const all = data ?? [];
    if (!search.trim()) return all;
    const s = search.toLowerCase();
    return all.filter(
      (r) =>
        r.email.toLowerCase().includes(s) ||
        r.username.toLowerCase().includes(s) ||
        String(r.userId).includes(s) ||
        r.status.toLowerCase().includes(s)
    );
  }, [data, search]);

  const invite = useMutation({
    mutationFn: async (userId: number) => {
      const res = await apiRequest("POST", "/api/admin/kyc/invite", { userId, note: noteText || null });
      return res.json();
    },
    onSuccess: () => {
      setNoteOpen(false);
      setNoteUserId(null);
      setNoteText("");
      qc.invalidateQueries({ queryKey: ["admin-kyc-queue"] });
      qc.invalidateQueries({ queryKey: ["admin-audit-trail"] });
    },
  });

  const review = useMutation({
    mutationFn: async (payload: { userId: number; status: "APPROVED" | "REJECTED"; note?: string | null }) => {
      const res = await apiRequest("POST", "/api/admin/kyc/review", payload);
      return res.json();
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["admin-kyc-queue"] });
      qc.invalidateQueries({ queryKey: ["admin-audit-trail"] });
    },
  });

  const openInvite = (userId: number) => {
    setNoteUserId(userId);
    setNoteText("");
    setNoteOpen(true);
  };

  const fmt = (ts?: number | string | null) => {
    if (!ts) return "—";
    const d = typeof ts === "number" ? new Date(ts > 1e12 ? ts : ts * 1000) : new Date(String(ts));
    return isNaN(d.getTime()) ? String(ts) : d.toLocaleString();
  };

  return (
    <div className="space-y-4">
      <Card className="bg-neutral-850 border-gray-800">
        <CardHeader>
          <CardTitle>KYC Queue</CardTitle>
        </CardHeader>
        <CardContent className="flex flex-col sm:flex-row gap-2 sm:items-center">
          <Input
            placeholder="Search email/username/userId/status..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            className="bg-neutral-900 border-gray-700"
          />
          <div className="text-xs text-gray-400">
            Invite-based KYC. Traders should not see KYC unless invited.
          </div>
        </CardContent>
      </Card>

      <div className="overflow-x-auto">
        {isLoading ? (
          <div className="p-6 text-gray-400">Loading KYC queue...</div>
        ) : (
          <Table>
            <TableHeader>
              <TableRow className="border-b border-gray-700">
                <TableHead className="text-gray-400">User</TableHead>
                <TableHead className="text-gray-400">Status</TableHead>
                <TableHead className="text-gray-400">Invited</TableHead>
                <TableHead className="text-gray-400">Submitted</TableHead>
                <TableHead className="text-gray-400">Reviewed</TableHead>
                <TableHead className="text-gray-400">Actions</TableHead>
              </TableRow>
            </TableHeader>
            <TableBody>
              {rows.length === 0 ? (
                <TableRow>
                  <TableCell colSpan={6} className="py-6 text-center text-gray-400">
                    No results
                  </TableCell>
                </TableRow>
              ) : (
                rows.map((r) => (
                  <TableRow key={r.userId} className="border-b border-gray-800">
                    <TableCell>
                      <div className="font-medium text-gray-200">{r.email}</div>
                      <div className="text-xs text-gray-400">
                        @{r.username} (#{r.userId})
                      </div>
                    </TableCell>
                    <TableCell className="text-xs text-gray-300">{r.status}</TableCell>
                    <TableCell className="text-xs text-gray-400">{fmt(r.invitedAt)}</TableCell>
                    <TableCell className="text-xs text-gray-400">{fmt(r.submittedAt)}</TableCell>
                    <TableCell className="text-xs text-gray-400">{fmt(r.reviewedAt)}</TableCell>
                    <TableCell className="flex gap-2 py-3">
                      <Button
                        size="sm"
                        variant="outline"
                        className="bg-neutral-900 border-gray-700"
                        onClick={() => openInvite(r.userId)}
                        disabled={invite.isPending}
                      >
                        Invite
                      </Button>
                      <Button
                        size="sm"
                        className="bg-green-700 hover:bg-green-800"
                        onClick={() => review.mutate({ userId: r.userId, status: "APPROVED" })}
                        disabled={review.isPending}
                      >
                        Approve
                      </Button>
                      <Button
                        size="sm"
                        className="bg-red-700 hover:bg-red-800"
                        onClick={() => review.mutate({ userId: r.userId, status: "REJECTED" })}
                        disabled={review.isPending}
                      >
                        Reject
                      </Button>
                    </TableCell>
                  </TableRow>
                ))
              )}
            </TableBody>
          </Table>
        )}
      </div>

      <Dialog open={noteOpen} onOpenChange={setNoteOpen}>
        <DialogContent className="bg-neutral-900 border-gray-700">
          <DialogHeader>
            <DialogTitle>Invite Trader to KYC</DialogTitle>
          </DialogHeader>
          <div className="space-y-2">
            <div className="text-sm text-gray-300">
              User ID: <span className="font-mono">{noteUserId ?? "—"}</span>
            </div>
            <Input
              placeholder="Optional internal note..."
              value={noteText}
              onChange={(e) => setNoteText(e.target.value)}
              className="bg-neutral-950 border-gray-700"
            />
            <div className="text-xs text-gray-400">
              KYC is invite-based. This keeps onboarding frictionless for new traders.
            </div>
          </div>
          <DialogFooter>
            <Button variant="outline" className="bg-neutral-900 border-gray-700" onClick={() => setNoteOpen(false)}>
              Cancel
            </Button>
            <Button
              onClick={() => noteUserId && invite.mutate(noteUserId)}
              className="bg-blue-700 hover:bg-blue-800"
              disabled={!noteUserId || invite.isPending}
            >
              Send Invite
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </div>
  );
}

export async function ensureIdentityAndKycSchema(db: any) {

      // drizzle sqlite supports db.run(sql`...`)
      await db.run(stmt);
    } catch (e: any) {
      const msg = String(e?.message ?? e);
      // SQLite duplicate column / table exists, etc.
      if (
        msg.toLowerCase().includes("duplicate column") ||
        msg.toLowerCase().includes("already exists") ||
        msg.toLowerCase().includes("duplicate") ||
        msg.toLowerCase().includes("exists")
      ) {
        return;
      }
      throw e;
    }
  };

  // Users table: add profile fields if not present.
  await safeRun(sql`ALTER TABLE users ADD COLUMN first_name TEXT;`);
  await safeRun(sql`ALTER TABLE users ADD COLUMN last_name TEXT;`);
  await safeRun(sql`ALTER TABLE users ADD COLUMN display_name TEXT;`);
  await safeRun(sql`ALTER TABLE users ADD COLUMN timezone TEXT;`);

function computeGates(u: any, kycStatus: string | null) {
  const createdAt = Number(u?.createdAt ?? u?.created_at ?? 0);
  const ageDays = createdAt > 0 ? (nowMs() - createdAt) / (1000 * 60 * 60 * 24) : 0;

  const balance = Number(u?.balance ?? 0);
  const equity = Number(u?.equity ?? 0);
  const returnPct = balance > 0 ? ((equity - balance) / balance) * 100 : 0;

    kycStatus === "INVITED" ||
    kycStatus === "SUBMITTED" ||
    kycStatus === "APPROVED" ||
    kycStatus === "REJECTED";

    };

    const safeText = (v: any) => {
      if (v === null) return null;
      if (v === undefined) return undefined;
      const s = String(v).trim();
      return s.length ? s.slice(0, 120) : null;
    };

    await db
      .update(users)
      .set({
        ...(body.firstName !== undefined ? { firstName: safeText(body.firstName) } : {}),
        ...(body.lastName !== undefined ? { lastName: safeText(body.lastName) } : {}),
        ...(body.displayName !== undefined ? { displayName: safeText(body.displayName) } : {}),
        ...(body.timezone !== undefined ? { timezone: safeText(body.timezone) } : {}),
        ...(body.locale !== undefined ? { locale: safeText(body.locale) } : {}),
      } as any)
      .where(eq(users.id, userId))
      .run();

    res.json({ ok: true });
  });

  app.use("/api", r);
}
