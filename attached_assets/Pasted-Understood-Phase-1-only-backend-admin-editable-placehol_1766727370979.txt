      multi_account_window_days INTEGER NOT NULL,
      churn_window_hours INTEGER NOT NULL,
      hedge_window_minutes INTEGER NOT NULL,

      ip_unique_threshold INTEGER NOT NULL,
      ua_unique_threshold INTEGER NOT NULL,
      device_unique_threshold INTEGER NOT NULL,

      geo_velocity_kmh_threshold INTEGER NOT NULL,
      geo_velocity_min_distance_km INTEGER NOT NULL,
      geo_velocity_max_hours INTEGER NOT NULL,

      score_multi_account_device INTEGER NOT NULL,
      score_hedge_pair INTEGER NOT NULL,
      score_ip_churn INTEGER NOT NULL,
      score_ua_churn INTEGER NOT NULL,
      score_device_churn INTEGER NOT NULL,
      score_geo_velocity INTEGER NOT NULL,

      tier_med INTEGER NOT NULL,
      tier_high INTEGER NOT NULL,
      tier_critical INTEGER NOT NULL,

      updated_at INTEGER NOT NULL,
      updated_by_admin_id INTEGER
    );
  `);

  // Devices
  sqlite.exec(`
    CREATE TABLE IF NOT EXISTS grift_devices (
      device_id TEXT PRIMARY KEY,
      first_seen_at INTEGER NOT NULL,
      last_seen_at INTEGER NOT NULL,
      first_ip TEXT,
      last_ip TEXT,
      first_user_agent TEXT,
      last_user_agent TEXT,
      first_country TEXT,
      last_country TEXT,
      trust_level TEXT NOT NULL DEFAULT 'NEW'

  `);

  // Device<->User links

      last_seen_at INTEGER NOT NULL,
      seen_count INTEGER NOT NULL DEFAULT 1,
      link_strength INTEGER NOT NULL DEFAULT 0,
      PRIMARY KEY (device_id, user_id)
    );
  `);

  // Signals
  sqlite.exec(`
    CREATE TABLE IF NOT EXISTS grift_signals (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      created_at INTEGER NOT NULL,
      updated_at INTEGER NOT NULL,

      rule_code TEXT NOT NULL,
      severity TEXT NOT NULL,

      primary_user_id INTEGER NOT NULL,

      user_agent TEXT,
      country TEXT,
      region TEXT,
      city TEXT,
      latitude REAL,
      longitude REAL,

      symbol TEXT,
      trade_id INTEGER,

      points INTEGER NOT NULL DEFAULT 0,
      dedupe_key TEXT NOT NULL UNIQUE,
      evidence_json TEXT NOT NULL,

      closed_at INTEGER,
      closed_by_admin_id INTEGER,
      closed_reason TEXT
    );
  `);

  sqlite.exec(`CREATE INDEX IF NOT EXISTS idx_grift_signals_primary ON grift_signals(primary_user_id, status, created_at);`);
  sqlite.exec(`CREATE INDEX IF NOT EXISTS idx_grift_signals_secondary ON grift_signals(secondary_user_id, status, created_at);`);
  sqlite.exec(`CREATE INDEX IF NOT EXISTS idx_grift_device_users_user ON grift_device_users(user_id, last_seen_at);`);

  // Aggregated user scores
  sqlite.exec(`
    CREATE TABLE IF NOT EXISTS grift_user_scores (
      user_id INTEGER PRIMARY KEY,
      score_current INTEGER NOT NULL DEFAULT 0,
      tier TEXT NOT NULL DEFAULT 'LOW',

      score_7d INTEGER NOT NULL DEFAULT 0,
      score_30d INTEGER NOT NULL DEFAULT 0,

      devices_7d INTEGER NOT NULL DEFAULT 0,
      ips_7d INTEGER NOT NULL DEFAULT 0,
      user_agents_7d INTEGER NOT NULL DEFAULT 0,

      open_signals_count INTEGER NOT NULL DEFAULT 0,

  `);

  // Seed config row id=1 if missing
  const row = sqlite.prepare(`SELECT id FROM grift_config WHERE id=1`).get();
  if (!row) {
    const c = DEFAULT_GRIFT_CONFIG;

        tier_med, tier_high, tier_critical,
        updated_at, updated_by_admin_id
      ) VALUES (

        ?, ?, ?,

        ?, ?, ?, ?, ?, ?,
        ?, ?, ?,
        ?, ?
      )
    `).run(

trade submit / order received (not fill)

File: server/grift/griftEngine.ts (new)

import { sqlite } from "@db";
import type { GriftConfig, GriftRuleCode, GriftSeverity } from "./griftTypes";

function clamp(n: number, min: number, max: number) {
  return Math.max(min, Math.min(max, n));
}

function tierFor(score: number, cfg: GriftConfig): "LOW" | "MED" | "HIGH" | "CRITICAL" {
  if (score >= cfg.tierCritical) return "CRITICAL";
  if (score >= cfg.tierHigh) return "HIGH";
  if (score >= cfg.tierMed) return "MED";
  return "LOW";
}

function now() { return Date.now(); }

function getConfig(): GriftConfig {
  const row = sqlite.prepare(`
    SELECT
      id,
      enabled,
      multi_account_window_days as multiAccountWindowDays,
      churn_window_hours as churnWindowHours,
      hedge_window_minutes as hedgeWindowMinutes,
      ip_unique_threshold as ipUniqueThreshold,
      ua_unique_threshold as uaUniqueThreshold,
      device_unique_threshold as deviceUniqueThreshold,
      geo_velocity_kmh_threshold as geoVelocityKmhThreshold,
      geo_velocity_min_distance_km as geoVelocityMinDistanceKm,
      geo_velocity_max_hours as geoVelocityMaxHours,
      score_multi_account_device as scoreMultiAccountDevice,
      score_hedge_pair as scoreHedgePair,
      score_ip_churn as scoreIpChurn,
      score_ua_churn as scoreUaChurn,
      score_device_churn as scoreDeviceChurn,
      score_geo_velocity as scoreGeoVelocity,
      tier_med as tierMed,
      tier_high as tierHigh,
      tier_critical as tierCritical,
      updated_at as updatedAt,
      updated_by_admin_id as updatedByAdminId
    FROM grift_config WHERE id=1
  `).get();

  if (!row) throw new Error("grift_config row id=1 missing. ensureGriftSchema() must run on startup.");
  return row as GriftConfig;
}

function severityFromPoints(points: number): GriftSeverity {
  if (points >= 55) return "CRITICAL";
  if (points >= 35) return "HIGH";
  if (points >= 20) return "MED";
  return "LOW";
}

function makeDedupeKey(rule: GriftRuleCode, primaryUserId: number, secondaryUserId: number | null, bucketMs: number) {
  const bucket = Math.floor(now() / bucketMs);
  return `${rule}:${primaryUserId}:${secondaryUserId ?? 0}:${bucket}`;
}

function upsertDevice(deviceId: string, ip?: string, userAgent?: string, country?: string) {
  const ts = now();
  const existing = sqlite.prepare(`SELECT device_id FROM grift_devices WHERE device_id=?`).get(deviceId);
  if (!existing) {
    sqlite.prepare(`
      INSERT INTO grift_devices (device_id, first_seen_at, last_seen_at, first_ip, last_ip, first_user_agent, last_user_agent, first_country, last_country, trust_level)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, 'NEW')
    `).run(deviceId, ts, ts, ip ?? null, ip ?? null, userAgent ?? null, userAgent ?? null, country ?? null, country ?? null);
  } else {
    sqlite.prepare(`
      UPDATE grift_devices
      SET last_seen_at=?, last_ip=?, last_user_agent=?, last_country=?
      WHERE device_id=?
    `).run(ts, ip ?? null, userAgent ?? null, country ?? null, deviceId);
  }
}

function upsertDeviceUserLink(deviceId: string, userId: number) {
  const ts = now();
  const row = sqlite.prepare(`SELECT seen_count FROM grift_device_users WHERE device_id=? AND user_id=?`).get(deviceId, userId) as any;
  if (!row) {
    sqlite.prepare(`
      INSERT INTO grift_device_users (device_id, user_id, first_seen_at, last_seen_at, seen_count, link_strength)
      VALUES (?, ?, ?, ?, 1, 0)
    `).run(deviceId, userId, ts, ts);
  } else {
    sqlite.prepare(`
      UPDATE grift_device_users
      SET last_seen_at=?, seen_count=seen_count+1
      WHERE device_id=? AND user_id=?
    `).run(ts, deviceId, userId);
  }
}

function insertSignal(args: {
  ruleCode: GriftRuleCode;
  severity: GriftSeverity;
  primaryUserId: number;
  secondaryUserId?: number | null;
  deviceId?: string;
  ip?: string;
  userAgent?: string;
  country?: string;
  region?: string;
  city?: string;
  latitude?: number | null;
  longitude?: number | null;
  symbol?: string;
  tradeId?: number;
  points: number;
  dedupeKey: string;
  evidence: Record<string, any>;
}) {
  const ts = now();
  try {
    sqlite.prepare(`
      INSERT INTO grift_signals (
        created_at, updated_at, rule_code, severity, status,
        primary_user_id, secondary_user_id,
        device_id, ip, user_agent, country, region, city, latitude, longitude,
        symbol, trade_id,
        points, dedupe_key, evidence_json
      ) VALUES (
        ?, ?, ?, ?, 'OPEN',
        ?, ?,
        ?, ?, ?, ?, ?, ?, ?, ?,
        ?, ?,
        ?, ?, ?
      )
    `).run(
      ts, ts, args.ruleCode, args.severity,
      args.primaryUserId, args.secondaryUserId ?? null,
      args.deviceId ?? null, args.ip ?? null, args.userAgent ?? null,
      args.country ?? null, args.region ?? null, args.city ?? null,
      args.latitude ?? null, args.longitude ?? null,
      args.symbol ?? null, args.tradeId ?? null,
      args.points, args.dedupeKey, JSON.stringify(args.evidence)
    );
    return true;
  } catch (e: any) {
    // Unique dedupe_key prevents spamming
    if (String(e?.message ?? "").includes("UNIQUE")) return false;
    throw e;
  }
}

function recomputeUserAggregates(userId: number, cfg: GriftConfig) {
  const ts = now();
  const ms7d = 7 * 24 * 60 * 60 * 1000;
  const ms30d = 30 * 24 * 60 * 60 * 1000;
  const since7d = ts - ms7d;
  const since30d = ts - ms30d;

  const openSignals = sqlite.prepare(`
    SELECT COUNT(*) as c, COALESCE(SUM(points),0) as p
    FROM grift_signals
    WHERE status='OPEN' AND (primary_user_id=? OR secondary_user_id=?)
  `).get(userId, userId) as any;

  const linkedAccounts30d = sqlite.prepare(`
    SELECT COUNT(DISTINCT du.user_id) as c
    FROM grift_device_users du
    WHERE du.device_id IN (
      SELECT device_id FROM grift_device_users WHERE user_id=? AND last_seen_at>=?
    )
    AND du.user_id != ?
    AND du.last_seen_at >= ?
  `).get(userId, since30d, userId, since30d) as any;

  // Approximate churn metrics from signals (Phase 1). Phase 2 can use auth_sessions/user_login_history directly.
  const devices7d = sqlite.prepare(`
    SELECT COUNT(DISTINCT device_id) as c
    FROM grift_signals
    WHERE created_at>=? AND (primary_user_id=? OR secondary_user_id=?) AND device_id IS NOT NULL
  `).get(since7d, userId, userId) as any;

  const ips7d = sqlite.prepare(`
    SELECT COUNT(DISTINCT ip) as c
    FROM grift_signals
    WHERE created_at>=? AND (primary_user_id=? OR secondary_user_id=?) AND ip IS NOT NULL
  `).get(since7d, userId, userId) as any;

  const uas7d = sqlite.prepare(`
    SELECT COUNT(DISTINCT user_agent) as c
    FROM grift_signals
    WHERE created_at>=? AND (primary_user_id=? OR secondary_user_id=?) AND user_agent IS NOT NULL
  `).get(since7d, userId, userId) as any;

  const countries7d = sqlite.prepare(`
    SELECT COUNT(DISTINCT country) as c
    FROM grift_signals
    WHERE created_at>=? AND (primary_user_id=? OR secondary_user_id=?) AND country IS NOT NULL
  `).get(since7d, userId, userId) as any;

  const hedgePairs7d = sqlite.prepare(`
    SELECT COUNT(*) as c
    FROM grift_signals
    WHERE created_at>=? AND rule_code='HEDGE_PAIR' AND status='OPEN'
      AND (primary_user_id=? OR secondary_user_id=?)
  `).get(since7d, userId, userId) as any;

  const pointsOpen = Number(openSignals?.p ?? 0);
  const scoreCurrent = clamp(pointsOpen, 0, 100);
  const tier = tierFor(scoreCurrent, cfg);

  const existing = sqlite.prepare(`SELECT user_id FROM grift_user_scores WHERE user_id=?`).get(userId);
  if (!existing) {
    sqlite.prepare(`
      INSERT INTO grift_user_scores (
        user_id, score_current, tier,
        score_7d, score_30d,
        devices_7d, ips_7d, user_agents_7d, countries_7d,
        linked_accounts_30d, hedge_pairs_7d, open_signals_count,
        last_evaluated_at
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `).run(
      userId, scoreCurrent, tier,
      scoreCurrent, scoreCurrent, // Phase 1: same as current; Phase 2: true rolling
      Number(devices7d?.c ?? 0),
      Number(ips7d?.c ?? 0),
      Number(uas7d?.c ?? 0),
      Number(countries7d?.c ?? 0),
      Number(linkedAccounts30d?.c ?? 0),
      Number(hedgePairs7d?.c ?? 0),
      Number(openSignals?.c ?? 0),
      ts
    );
  } else {
    sqlite.prepare(`
      UPDATE grift_user_scores
      SET
        score_current=?,
        tier=?,
        score_7d=?,
        score_30d=?,
        devices_7d=?,
        ips_7d=?,
        user_agents_7d=?,
        countries_7d=?,
        linked_accounts_30d=?,
        hedge_pairs_7d=?,
        open_signals_count=?,
        last_evaluated_at=?
      WHERE user_id=?
    `).run(
      scoreCurrent, tier,
      scoreCurrent, scoreCurrent,
      Number(devices7d?.c ?? 0),
      Number(ips7d?.c ?? 0),
      Number(uas7d?.c ?? 0),
      Number(countries7d?.c ?? 0),
      Number(linkedAccounts30d?.c ?? 0),
      Number(hedgePairs7d?.c ?? 0),
      Number(openSignals?.c ?? 0),
      ts,
      userId
    );
  }
}

export const griftEngine = {
  getConfig,

  // Call this right after a successful login/session creation.
  onLoginSuccess(input: {
    userId: number;
    deviceId?: string;
    ip?: string;
    userAgent?: string;
    country?: string;
    region?: string;
    city?: string;
    latitude?: number | null;
    longitude?: number | null;
  }) {
    const cfg = getConfig();
    if (!cfg.enabled) return;

    const ts = now();
    const bucketMs = 60 * 60 * 1000; // hourly dedupe bucket for login signals

    if (input.deviceId) {
      upsertDevice(input.deviceId, input.ip, input.userAgent, input.country);
      upsertDeviceUserLink(input.deviceId, input.userId);

      // MULTI_ACCOUNT_DEVICE: device linked to other users in window
      const since = ts - cfg.multiAccountWindowDays * 24 * 60 * 60 * 1000;
      const others = sqlite.prepare(`
        SELECT DISTINCT user_id as userId
        FROM grift_device_users
        WHERE device_id=? AND last_seen_at>=? AND user_id != ?
        LIMIT 10
      `).all(input.deviceId, since, input.userId) as any[];

      if (others.length > 0) {
        for (const o of others) {
          const otherUserId = Number(o.userId);
          const points = clamp(cfg.scoreMultiAccountDevice + 10 * Math.max(0, others.length - 1), 0, 70);
          const dedupeKey = makeDedupeKey("MULTI_ACCOUNT_DEVICE", input.userId, otherUserId, bucketMs);

          insertSignal({
            ruleCode: "MULTI_ACCOUNT_DEVICE",
            severity: severityFromPoints(points),
            primaryUserId: input.userId,
            secondaryUserId: otherUserId,
            deviceId: input.deviceId,
            ip: input.ip,
            userAgent: input.userAgent,
            country: input.country,
            region: input.region,
            city: input.city,
            latitude: input.latitude ?? null,
            longitude: input.longitude ?? null,
            points,
            dedupeKey,
            evidence: {
              deviceId: input.deviceId,
              primaryUserId: input.userId,
              secondaryUserId: otherUserId,
              windowDays: cfg.multiAccountWindowDays
            }
          });
        }
      }
    }

    // NOTE: Phase 1 churn/geo-velocity is signal-driven; Phase 2 can compute from auth_sessions/user_login_history.
    // If you already have login history with lat/lng, you can compute GEO_VELOCITY here by querying last login.
    // We keep this as a backend placeholder now, controlled by cfg weights.

    recomputeUserAggregates(input.userId, cfg);
  },

  // Call this at ORDER_RECEIVED / trade creation (not fill).
  onTradeSubmit(input: {
    userId: number;
    tradeId: number;
    symbol: string;
    side: "BUY" | "SELL";
    deviceId?: string;
    ip?: string;
    userAgent?: string;
    country?: string;
  }) {
    const cfg = getConfig();
    if (!cfg.enabled) return;

    if (!input.symbol || !input.side) return;

    const ts = now();
    const since = ts - cfg.hedgeWindowMinutes * 60 * 1000;
    const bucketMs = 10 * 60 * 1000; // 10-min bucket for hedging dedupe

    // Hedging detection requires a trades table with user_id, symbol, side, created_at.
    // If your trade time column differs, adjust this query once.
    const rows = sqlite.prepare(`
      SELECT id as tradeId, user_id as userId, side as side
      FROM trades
      WHERE symbol = ?
        AND created_at >= ?
        AND user_id != ?
      ORDER BY created_at DESC
      LIMIT 50
    `).all(input.symbol, since, input.userId) as any[];

    if (rows.length === 0) {
      recomputeUserAggregates(input.userId, cfg);
      return;
    }

    const opposite = rows.find(r => String(r.side).toUpperCase() !== input.side);
    if (!opposite) {
      recomputeUserAggregates(input.userId, cfg);
      return;
    }

    // Strengthen signal if same deviceId matches via grift_device_users link.
    let linkedByDevice = false;
    if (input.deviceId) {
      const link = sqlite.prepare(`
        SELECT 1
        FROM grift_device_users
        WHERE device_id=? AND user_id=?
      `).get(input.deviceId, Number(opposite.userId));
      linkedByDevice = !!link;
    }

    // If no device linkage, allow IP-only linkage as weaker evidence (still valuable to flag).
    const points = linkedByDevice ? cfg.scoreHedgePair : Math.max(25, Math.floor(cfg.scoreHedgePair * 0.6));
    const severity: GriftSeverity = linkedByDevice ? "CRITICAL" : "HIGH";

    const dedupeKey = makeDedupeKey("HEDGE_PAIR", input.userId, Number(opposite.userId), bucketMs);

    insertSignal({
      ruleCode: "HEDGE_PAIR",
      severity,
      primaryUserId: input.userId,
      secondaryUserId: Number(opposite.userId),
      deviceId: input.deviceId,
      ip: input.ip,
      userAgent: input.userAgent,
      country: input.country,
      symbol: input.symbol,
      tradeId: input.tradeId,
      points,
      dedupeKey,
      evidence: {
        symbol: input.symbol,
        thisUserId: input.userId,
        otherUserId: Number(opposite.userId),
        linkedByDevice,
        hedgeWindowMinutes: cfg.hedgeWindowMinutes,
        thisTradeId: input.tradeId,
        otherTradeId: Number(opposite.tradeId)
      }
    });

    recomputeUserAggregates(input.userId, cfg);
    recomputeUserAggregates(Number(opposite.userId), cfg);
  }
};

function sanitizeConfigPatch(body: any): Partial<GriftConfig> {
  // Only allow numeric + enabled toggles; clamp ranges to prevent nonsense.
  const out: any = {};
  const num = (v: any) => (typeof v === "number" && Number.isFinite(v) ? v : undefined);

  const setInt = (k: keyof GriftConfig, min: number, max: number) => {
    const v = num(body[k]);
    if (v === undefined) return;
    out[k] = Math.max(min, Math.min(max, Math.round(v)));
  };

  const setBoolInt = (k: keyof GriftConfig) => {
    if (body[k] === 1 || body[k] === 0) out[k] = body[k];
    if (body[k] === true) out[k] = 1;
    if (body[k] === false) out[k] = 0;
  };

  setBoolInt("enabled");

  setInt("multiAccountWindowDays", 1, 365);
  setInt("churnWindowHours", 1, 168);
  setInt("hedgeWindowMinutes", 1, 240);

  setInt("ipUniqueThreshold", 2, 50);
  setInt("uaUniqueThreshold", 2, 50);
  setInt("deviceUniqueThreshold", 2, 20);

  setInt("geoVelocityKmhThreshold", 200, 2000);
  setInt("geoVelocityMinDistanceKm", 50, 20000);
  setInt("geoVelocityMaxHours", 1, 24);

  setInt("scoreMultiAccountDevice", 0, 100);
  setInt("scoreHedgePair", 0, 100);
  setInt("scoreIpChurn", 0, 100);
  setInt("scoreUaChurn", 0, 100);
  setInt("scoreDeviceChurn", 0, 100);
  setInt("scoreGeoVelocity", 0, 100);

  setInt("tierMed", 0, 100);
  setInt("tierHigh", 0, 100);
  setInt("tierCritical", 0, 100);

  return out;
}
