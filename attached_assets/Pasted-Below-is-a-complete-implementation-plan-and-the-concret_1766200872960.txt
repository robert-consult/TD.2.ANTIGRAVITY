### 3.1 NEW FILE: `shared/closeReasons.ts`

```ts
/** 
 * shared/closeReasons.ts
 * Canonical close reason codes + trader labels + legacy normalization
 */

export const CLOSE_REASON_CODES = [
  // Trader-initiated
  "MANUAL_CLOSE",

  // Target-based
  "STOP_LOSS_HIT",
  "TAKE_PROFIT_HIT",
  "TRAILING_STOP_HIT",

  // Time-based
  "MAX_HOLD_TIME",

  // Risk / liquidation
  "MARGIN_STOP_OUT",
  "RISK_LIMIT_CLOSE",

  // Operator / system
  "ADMIN_CLOSE",
  "SYSTEM_CLOSE",

  // Market / instrument
  "MARKET_HALTED",
  "INSTRUMENT_DISABLED",

  // Order lifecycle (typically pending orders)
  "EXPIRED",
  "CANCELED",
  "REJECTED",

  // Fallback
  "UNKNOWN",
] as const;

export type CloseReasonCode = (typeof CLOSE_REASON_CODES)[number];

export const CLOSE_REASON_LABELS: Record<CloseReasonCode, string> = {
  MANUAL_CLOSE: "Manual",
  STOP_LOSS_HIT: "Stop loss hit",
  TAKE_PROFIT_HIT: "Take profit hit",
  TRAILING_STOP_HIT: "Trailing stop hit",
  MAX_HOLD_TIME: "Max hold time",
  MARGIN_STOP_OUT: "Stop out (margin)",
  RISK_LIMIT_CLOSE: "Closed by risk limit",
  ADMIN_CLOSE: "Admin closed",
  SYSTEM_CLOSE: "System closed",
  MARKET_HALTED: "Market halted",
  INSTRUMENT_DISABLED: "Instrument disabled",
  EXPIRED: "Expired",
  CANCELED: "Canceled",
  REJECTED: "Rejected",
  UNKNOWN: "Unknown",
};

export const TRADER_CLOSE_REASON_FILTER_OPTIONS: CloseReasonCode[] = [
  "MANUAL_CLOSE",
  "TAKE_PROFIT_HIT",
  "STOP_LOSS_HIT",
  "TRAILING_STOP_HIT",
  "MAX_HOLD_TIME",
  "MARGIN_STOP_OUT",
  "RISK_LIMIT_CLOSE",
  "ADMIN_CLOSE",
  "SYSTEM_CLOSE",
];

export function normalizeCloseReason(raw: string | null | undefined): CloseReasonCode | null {
  if (!raw) return null;
  const v = String(raw).trim().toUpperCase();

  // Legacy / historical values
  if (v === "MANUAL") return "MANUAL_CLOSE";
  if (v === "SL" || v === "STOPLOSS" || v === "STOP_LOSS") return "STOP_LOSS_HIT";
  if (v === "TP" || v === "TAKEPROFIT" || v === "TAKE_PROFIT") return "TAKE_PROFIT_HIT";
  if (v === "AUTO_TIME_LIMIT" || v === "AUTO_TIME" || v === "TIME_LIMIT") return "MAX_HOLD_TIME";
  if (v === "AUTO") return "SYSTEM_CLOSE";
  if (v === "ADMIN") return "ADMIN_CLOSE";

  // Already-canonical values
  if ((CLOSE_REASON_CODES as readonly string[]).includes(v)) return v as CloseReasonCode;

  return "UNKNOWN";
}

export function closeReasonLabel(raw: string | null | undefined): string | null {
  const code = normalizeCloseReason(raw);
  if (!code) return null;
  return CLOSE_REASON_LABELS[code] || "Unknown";
}

export type CloseReasonTone = "neutral" | "green" | "red" | "yellow" | "orange" | "purple" | "gray";

export function closeReasonTone(code: CloseReasonCode): CloseReasonTone {
  switch (code) {
    case "TAKE_PROFIT_HIT":
      return "green";
    case "STOP_LOSS_HIT":
      return "red";
    case "MAX_HOLD_TIME":
      return "yellow";
    case "MARGIN_STOP_OUT":
    case "RISK_LIMIT_CLOSE":
      return "orange";
    case "ADMIN_CLOSE":
    case "SYSTEM_CLOSE":
      return "purple";
    case "MANUAL_CLOSE":
      return "neutral";
    default:
      return "gray";
  }
}
```
