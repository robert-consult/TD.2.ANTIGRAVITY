You have **two separate issues**:

1. **Sessions are “static”** because you likely **do not bind a `sessionId` to auth** (JWT/cookie) and **do not validate the session row** on each request. Without that, revoking a session in the DB does nothing; the old token continues to work.

2. Timezone labels are not in **UTC/GMT canonical format** because your UI is rendering a label that omits the `"UTC"` prefix and/or doesn’t compute the **current UTC offset (DST-aware)**.

Below is a **drop-in** set of backend + frontend changes that fixes both, plus an implementable **language detection + i18n translation engine** scaffold.

```ts
export async function requireAuth(req: Request, res: Response, next: NextFunction) {

  if (!auth) return res.status(401).json({ error: "unauthorized" });

  // Validate session is still active (this is what makes revoke actually log the user out)
  const row = db
    .select({
      id: userSessions.id,
      userId: userSessions.userId,
      endedAt: userSessions.endedAt,
      revokedAt: userSessions.revokedAt,
    })
    .from(userSessions)
    .where(and(eq(userSessions.id, auth.sessionId), eq(userSessions.userId, auth.userId)))
    .limit(1)
    .all()[0];

  if (!row || row.endedAt !== null || row.revokedAt !== null) {
    return res.status(401).json({ error: "session_expired" });
  }

  // Attach to req context
  (req as any).auth = { userId: auth.userId, sessionId: auth.sessionId };

  // Touch lastSeenAt (non-blocking)
  try {
    touchSession({ sessionId: auth.sessionId });
  } catch {

  }

  next();
}
```

## A2) Backend: ME session endpoints (list + revoke + logout others + logout current)

Create `server/routes/meSessions.ts`:

```ts
// server/routes/meSessions.ts
import type { Express, Request, Response } from "express";
import { db } from "../db";
import { userSessions } from "../../shared/schema";
import { and, desc, eq, isNull, ne } from "drizzle-orm";
import { revokeSession, endSession, getClientIp, getUserAgent, appendSecurityEvent } from "../security/sessionTrail";
import { requireAuth } from "../middleware/auth";

export function registerMeSessionsRoutes(app: Express) {
  // List sessions (active + currentSessionId for UI)
  app.get("/api/me/sessions", requireAuth, (req: Request, res: Response) => {
    const auth = (req as any).auth as { userId: number; sessionId: string };
    const limit = Math.min(50, Math.max(1, Number(req.query.limit || 10)));

    const rows = db
      .select({
        id: userSessions.id,
        createdAt: userSessions.createdAt,
        lastSeenAt: userSessions.lastSeenAt,
        endedAt: userSessions.endedAt,
        ip: userSessions.ip,
        userAgent: userSessions.userAgent,
        countryCode: userSessions.countryCode,
        region: userSessions.region,
        city: userSessions.city,
        inferredTz: userSessions.inferredTz,
        revokedAt: userSessions.revokedAt,
      })
      .from(userSessions)
      .where(and(eq(userSessions.userId, auth.userId), isNull(userSessions.endedAt)))
      .orderBy(desc(userSessions.lastSeenAt))
      .limit(limit)
      .all();

    res.json({ currentSessionId: auth.sessionId, rows });
  });

  // Revoke a specific session (including “other devices”)
  app.post("/api/me/sessions/:sessionId/revoke", requireAuth, (req: Request, res: Response) => {
    const auth = (req as any).auth as { userId: number; sessionId: string };
    const targetSessionId = String(req.params.sessionId);
    const reason = String((req.body as any)?.reason || "User revoked session");

    revokeSession({
      actorUserId: auth.userId,
      targetUserId: auth.userId,
      sessionId: targetSessionId,
      reason,
    });

    res.json({ ok: true });
  });

  // Logout all other devices (revoke all sessions except current)
  app.post("/api/me/sessions/logout-others", requireAuth, (req: Request, res: Response) => {
    const auth = (req as any).auth as { userId: number; sessionId: string };
    const reason = String((req.body as any)?.reason || "User logged out other devices");

    const others = db
      .select({ id: userSessions.id })
      .from(userSessions)
      .where(
        and(
          eq(userSessions.userId, auth.userId),
          isNull(userSessions.endedAt),
          ne(userSessions.id, auth.sessionId)
        )
      )
      .all();

    for (const s of others) {
      revokeSession({
        actorUserId: auth.userId,
        targetUserId: auth.userId,
        sessionId: s.id,
        reason,
      });
    }

    res.json({ ok: true, revokedCount: others.length });
  });

  // Logout current session (end + trail)
  app.post("/api/me/logout", requireAuth, (req: Request, res: Response) => {
    const auth = (req as any).auth as { userId: number; sessionId: string };

    const ip = getClientIp(req as any);
    const userAgent = getUserAgent(req as any);

    // endSession writes LOGOUT trail
    endSession({ userId: auth.userId, sessionId: auth.sessionId, ip, userAgent });

    // IMPORTANT: client must also clear token/cookie; server returns ok only.
    res.json({ ok: true });
  });
}
```

# B) Fix Timezone list: display in UTC/GMT canonical format (DST-aware)

## B1) Backend: return timezones with `(UTC±HH:MM)` prefix + IANA name

Add helper (server-side) to compute current offset minutes for any IANA zone:

```ts
// server/utils/tzOffset.ts
export function getOffsetMinutes(timeZone: string, date = new Date()): number {
  // Computes offset from UTC in minutes for given IANA zone at given date (DST-aware)
  const dtf = new Intl.DateTimeFormat("en-US", {
    timeZone,
    hour12: false,
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
    second: "2-digit",
  });

  const parts = dtf.formatToParts(date);
  const map: any = {};
  for (const p of parts) if (p.type !== "literal") map[p.type] = p.value;

  // Build a UTC timestamp from the "local time in that zone"
  const asUtc = Date.UTC(
    Number(map.year),
    Number(map.month) - 1,
    Number(map.day),
    Number(map.hour),
    Number(map.minute),
    Number(map.second)
  );

  // date.getTime() is actual UTC timestamp; difference gives offset
  return Math.round((asUtc - date.getTime()) / 60000);
}

export function fmtUtcOffset(minutes: number): string {
  const sign = minutes >= 0 ? "+" : "-";
  const abs = Math.abs(minutes);
  const hh = String(Math.floor(abs / 60)).padStart(2, "0");
  const mm = String(abs % 60).padStart(2, "0");
  return `UTC${sign}${hh}:${mm}`;
}
```

Now modify `/api/meta/timezones` so each timezone option contains a **canonical label**:

```ts
// server/routes/meta.ts
import type { Express, Request, Response } from "express";
import { getTimeZones } from "@vvo/tzdb";
import { getOffsetMinutes, fmtUtcOffset } from "../utils/tzOffset";

export function registerMetaRoutes(app: Express) {
  app.get("/api/meta/timezones", (req: Request, res: Response) => {
    const includeUtc = String(req.query.includeUtc || "1") === "1";
    const now = new Date();

    const tzs: any[] = getTimeZones({ includeUtc }) as any[];

    const rows = tzs.map((t) => {
      const name = String(t.name);
      let currentOffsetMinutes = 0;
      try {
        currentOffsetMinutes = getOffsetMinutes(name, now);
      } catch {
        currentOffsetMinutes = Number(t.rawOffsetInMinutes || 0);
      }

      const offsetLabel = fmtUtcOffset(currentOffsetMinutes);

      const city = Array.isArray(t.mainCities) && t.mainCities.length ? t.mainCities.slice(0, 4).join(", ") : "";
      const label = `${offsetLabel} · ${t.alternativeName || name}${city ? ` — ${city}` : ""} (${name})`;

      return {
        name,
        label,
        countryCode: t.countryCode || null,
        countryName: t.countryName || null,
        mainCities: t.mainCities || [],
        abbreviation: t.abbreviation || null,
        rawOffsetInMinutes: t.rawOffsetInMinutes || 0,
        currentOffsetMinutes,
      };
    });

    // Sort by current offset then label
    rows.sort((a, b) => (a.currentOffsetMinutes - b.currentOffsetMinutes) || a.label.localeCompare(b.label));

    res.json({ generatedAt: Date.now(), rows });
  });
}
```

## C2) Language detection for user-generated text (journal notes, feedback, etc.)

Install:


Backend helper:

```ts
// server/utils/langDetect.ts
import { franc } from "franc-min";
import langs from "langs";

export function detectLanguage(text: string): { iso1: string; confidence: "low" | "medium" | "high" } {
  const sample = (text || "").trim();
  if (sample.length < 20) return { iso1: "en", confidence: "low" };

  const iso3 = franc(sample, { minLength: 20 });
  if (iso3 === "und") return { iso1: "en", confidence: "low" };

  const lang = langs.where("3", iso3);
  const iso1 = lang?.["1"] || "en";

  // crude confidence heuristic
  const confidence = sample.length > 200 ? "high" : "medium";
  return { iso1, confidence };
}
```

## C3) Optional “auto-translation engine” (self-host or external)

If you want actual automatic translations without paying a per-call API, use **LibreTranslate** (self-hostable). Implementation pattern:

* Env: `TRANSLATE_URL=http://localhost:5000`
* Endpoint:

```ts
// server/routes/translate.ts
import type { Express, Request, Response } from "express";
import { requireAuth } from "../middleware/auth";

export function registerTranslateRoutes(app: Express) {
  app.post("/api/translate", requireAuth, async (req: Request, res: Response) => {
    const { q, source = "auto", target = "en" } = (req.body || {}) as any;
    if (!q || typeof q !== "string") return res.status(400).json({ error: "missing_q" });

    const base = process.env.TRANSLATE_URL;
    if (!base) return res.status(500).json({ error: "TRANSLATE_URL_not_configured" });

    const r = await fetch(`${base}/translate`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ q, source, target, format: "text" }),
    });

    if (!r.ok) return res.status(502).json({ error: "translate_failed" });

    const data = await r.json();
    res.json({ translatedText: data.translatedText || "" });
  });
}
```

Then the UI can call `/api/translate` when user clicks “Translate”.
