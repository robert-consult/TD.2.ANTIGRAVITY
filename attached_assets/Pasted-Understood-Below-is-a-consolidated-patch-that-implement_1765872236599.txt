# 2) Single source of truth for effective limits

Create a small service to resolve **effective** risk limits given a `userId`. User value wins if not `null`/`undefined`, otherwise global.

```ts
// server/services/settings.ts
import { db } from "../db";
import { globalSettings, userSettings } from "../../shared/schema";
import { eq } from "drizzle-orm";

type RiskLimits = {
  maxPositionSize: number;
  maxTradesPerUser: number;
  maxTradesPerInstrument: number;
  maxConcurrentLots: number;
  defaultLeverage: number;
  allowWeekendTrading: boolean;
  marketOpenTime: string;
  marketCloseTime: string;
  enableAutoClose: boolean;
  autoCloseAfterDays: number;
  autoCloseCheckFrequencyMinutes: number;
  enableLossLimits: boolean;
  dailyLossLimitPct: number;
  lifetimeLossLimitPct: number;
};

const pick = <T>(u: T | null | undefined, g: T): T => (u ?? g);

export async function getEffectiveRiskLimits(userId?: number): Promise<RiskLimits> {
  const g = await db.query.globalSettings.findFirst({ where: eq(globalSettings.id, 1) });
  if (!g) throw new Error("Global settings row missing. Initialize via admin.");

  if (!userId) {
    return {
      maxPositionSize: g.maxPositionSize!,
      maxTradesPerUser: g.maxTradesPerUser!,
      maxTradesPerInstrument: g.maxTradesPerInstrument!,
      maxConcurrentLots: g.maxConcurrentLots!,
      defaultLeverage: g.defaultLeverage!,
      allowWeekendTrading: !!g.allowWeekendTrading,
      marketOpenTime: g.marketOpenTime!,
      marketCloseTime: g.marketCloseTime!,
      enableAutoClose: !!g.enableAutoClose,
      autoCloseAfterDays: g.autoCloseAfterDays!,
      autoCloseCheckFrequencyMinutes: g.autoCloseCheckFrequencyMinutes!,
      enableLossLimits: !!g.enableLossLimits,
      dailyLossLimitPct: g.dailyLossLimitPct!,
      lifetimeLossLimitPct: g.lifetimeLossLimitPct!,
    };
  }

  const u = await db.query.userSettings.findFirst({ where: eq(userSettings.userId, userId) });

  return {
    maxPositionSize: pick(u?.maxPositionSize, g.maxPositionSize!),
    maxTradesPerUser: pick(u?.maxTradesPerUser, g.maxTradesPerUser!),
    maxTradesPerInstrument: pick(u?.maxTradesPerInstrument, g.maxTradesPerInstrument!),
    maxConcurrentLots: pick(u?.maxConcurrentLots, g.maxConcurrentLots!),
    defaultLeverage: pick(u?.defaultLeverage, g.defaultLeverage!),
    allowWeekendTrading: pick(u?.allowWeekendTrading, !!g.allowWeekendTrading),
    marketOpenTime: pick(u?.marketOpenTime, g.marketOpenTime!),
    marketCloseTime: pick(u?.marketCloseTime, g.marketCloseTime!),
    enableAutoClose: pick(u?.enableAutoClose, !!g.enableAutoClose),
    autoCloseAfterDays: pick(u?.autoCloseAfterDays, g.autoCloseAfterDays!),
    autoCloseCheckFrequencyMinutes: pick(u?.autoCloseCheckFrequencyMinutes, g.autoCloseCheckFrequencyMinutes!),
    enableLossLimits: pick(u?.enableLossLimits, !!g.enableLossLimits),
    dailyLossLimitPct: pick(u?.dailyLossLimitPct, g.dailyLossLimitPct!),
    lifetimeLossLimitPct: pick(u?.lifetimeLossLimitPct, g.lifetimeLossLimitPct!),
  };
}
```

# 7) Admin API for per-user settings: keep them nullable, parse safely

Ensure PUT does not coerce empty fields; it should **leave them null** so they inherit global.

```diff
// server/routes/admin.users.ts
app.put("/api/admin/users/:id/settings", requireAdmin, async (req, res) => {
  try {
    const userId = Number(req.params.id);
    const body = req.body ?? {};

    const numOrNull = (v:any) => (v === "" || v === null || v === undefined) ? null : (Number.isFinite(Number(v)) ? Number(v) : null);
    const boolOrNull = (v:any) => (v === "" || v === null || v === undefined) ? null
      : (typeof v === "boolean" ? v
        : (String(v).toLowerCase() === "true" ? true
          : (String(v).toLowerCase() === "false" ? false : null)));
    const timeOrNull = (v:any) => {
      if (v === "" || v === null || v === undefined) return null;
      const s = String(v).trim();
      const m = /^(\d{2}):(\d{2})$/.exec(s);
      if (!m) return null;
      const hh = Number(m[1]), mm = Number(m[2]);
      if (hh < 0 || hh > 23 || mm < 0 || mm > 59) return null;
      return s;
    };

    const next = {
      maxPositionSize: numOrNull(body.maxPositionSize),
      maxTradesPerUser: numOrNull(body.maxTradesPerUser),
      maxTradesPerInstrument: numOrNull(body.maxTradesPerInstrument),
      maxConcurrentLots: numOrNull(body.maxConcurrentLots),
      defaultLeverage: numOrNull(body.defaultLeverage),
      allowWeekendTrading: boolOrNull(body.allowWeekendTrading),
      marketOpenTime: timeOrNull(body.marketOpenTime),
      marketCloseTime: timeOrNull(body.marketCloseTime),
      enableAutoClose: boolOrNull(body.enableAutoClose),
      autoCloseAfterDays: numOrNull(body.autoCloseAfterDays),
      autoCloseCheckFrequencyMinutes: numOrNull(body.autoCloseCheckFrequencyMinutes),
      enableLossLimits: boolOrNull(body.enableLossLimits),
      dailyLossLimitPct: numOrNull(body.dailyLossLimitPct),
      lifetimeLossLimitPct: numOrNull(body.lifetimeLossLimitPct),
      updatedAt: new Date(),
    };

    const existing = await db.query.userSettings.findFirst({ where: eq(userSettings.userId, userId) });
    if (existing) {
      await db.update(userSettings).set(next).where(eq(userSettings.userId, userId));
    } else {
      await db.insert(userSettings).values({ userId, ...next });
    }

    const updated = await db.query.userSettings.findFirst({ where: eq(userSettings.userId, userId) });
    res.json(updated);
  } catch (e) {
    console.error("update user settings error", e);
    res.status(500).json({ message: "Failed to update user settings" });
  }
});
```
