I reviewed the attached signup-flow plan and implementation notes.  The core design is correct (country selection as the “single source of truth” driving both phone validation and legal assembly). The main compliance hardening I’m applying before we proceed is:

1. Assemble terms on the backend (canonical), not the frontend, so the server controls the exact bytes shown and hashed.


2. Issue a signed “terms token” (HMAC) tied to {country, addendumId/version, globalVersion, combinedSha256, issuedAt} and require it on signup.


3. Recompute and verify the hash server-side at signup time, so a client cannot swap text, country, or addendum.


4. Persist a tamper-evident acceptance ledger (hash-chained rows) in legalAcceptances.



Below is a copy/paste-ready implementation for your stack (Express + TypeScript + Drizzle + better-sqlite3 + React/wouter). All code is inline and organized by file path.

1.3 server/legal/cryptoUtils.ts — sha256 + stable stringify + HMAC token

// FILE: /server/legal/cryptoUtils.ts
import crypto from "crypto";

export function sha256Hex(input: string): string {
  return crypto.createHash("sha256").update(input, "utf8").digest("hex");
}

// Deterministic stringify for hashing ledger payloads
export function stableStringify(value: unknown): string {
  if (value === null || typeof value !== "object") return JSON.stringify(value);

  if (Array.isArray(value)) {
    return "[" + value.map((v) => stableStringify(v)).join(",") + "]";
  }

  const obj = value as Record<string, unknown>;
  const keys = Object.keys(obj).sort();
  return (
    "{" +
    keys
      .map((k) => JSON.stringify(k) + ":" + stableStringify(obj[k]))
      .join(",") +
    "}"
  );
}

export function base64UrlEncode(str: string): string {
  return Buffer.from(str, "utf8")
    .toString("base64")
    .replaceAll("+", "-")
    .replaceAll("/", "_")
    .replaceAll("=", "");
}

export function base64UrlDecode(str: string): string {
  const padLen = (4 - (str.length % 4)) % 4;
  const padded = str + "=".repeat(padLen);
  const b64 = padded.replaceAll("-", "+").replaceAll("_", "/");
  return Buffer.from(b64, "base64").toString("utf8");
}

export function hmacSha256Hex(secret: string, payload: string): string {
  return crypto.createHmac("sha256", secret).update(payload, "utf8").digest("hex");
}

1.4 server/legal/termsEngine.ts — resolve addendum, assemble, hash, tokenize

// FILE: /server/legal/termsEngine.ts
import fs from "fs";
import path from "path";
import {
  DOC1_GLOBAL,
  DEFAULT_ADDENDUM,
  COUNTRY_ADDENDA,
  REGION_ADDENDA,
  REGION_RULES_IN_ORDER,
  DocRef,
} from "./doc1Pack";
import { base64UrlEncode, base64UrlDecode, hmacSha256Hex, sha256Hex } from "./cryptoUtils";

type ResolveResult = { addendum: DocRef; regionKey?: string | null };

export type AssembledTerms = {
  global: { id: string; version: string; sha256: string };
  addendum: { id: string; version: string; sha256: string; regionKey?: string | null };
  combined: { sha256: string; text: string };
  meta: { countryIso2: string; regionKey?: string | null; issuedAtMs: number };
  token: string;
};

const CACHE = new Map<string, AssembledTerms>();

function normalizeIso2(iso2: string): string {
  const up = (iso2 || "").trim().toUpperCase();
  if (!/^[A-Z]{2}$/.test(up)) throw new Error("Invalid country ISO2.");
  return up;
}

function resolveRegionKey(iso2: string): string | null {
  for (const rule of REGION_RULES_IN_ORDER) {
    if (rule.iso2.includes(iso2)) return rule.regionKey;
  }
  return null;
}

function resolveAddendum(iso2: string): ResolveResult {
  if (COUNTRY_ADDENDA[iso2]) return { addendum: COUNTRY_ADDENDA[iso2], regionKey: null };

  const regionKey = resolveRegionKey(iso2);
  if (regionKey && REGION_ADDENDA[regionKey]) return { addendum: REGION_ADDENDA[regionKey], regionKey };

  return { addendum: DEFAULT_ADDENDUM, regionKey: null };
}

function readTextFile(relPath: string): string {
  const abs = path.resolve(process.cwd(), relPath);
  return fs.readFileSync(abs, "utf8");
}

function requireSecret(): string {
  const secret = process.env.LEGAL_TERMS_HMAC_SECRET;
  if (!secret || secret.length < 32) {
    throw new Error("LEGAL_TERMS_HMAC_SECRET missing/too short (min 32 chars).");
  }
  return secret;
}

export function assembleDoc1Terms(countryIso2Raw: string): AssembledTerms {
  const countryIso2 = normalizeIso2(countryIso2Raw);
  const { addendum, regionKey } = resolveAddendum(countryIso2);

  const cacheKey = `${countryIso2}|${DOC1_GLOBAL.id}@${DOC1_GLOBAL.version}|${addendum.id}@${addendum.version}`;
  const cached = CACHE.get(cacheKey);
  if (cached) return cached;

  const globalText = readTextFile(DOC1_GLOBAL.path);
  const addendumText = readTextFile(addendum.path);

  const globalSha = sha256Hex(globalText);
  const addendumSha = sha256Hex(addendumText);

  const combinedText =
    globalText +
    "\n\n---\n\n" +
    "AUTO-SELECTED ADDENDUM:\n" +
    `Addendum ID: ${addendum.id}\n` +
    `Addendum Version: ${addendum.version}\n` +
    `Country Selected: ${countryIso2}\n` +
    (regionKey ? `Region Key: ${regionKey}\n` : "") +
    "\n" +
    addendumText;

  const combinedSha = sha256Hex(combinedText);

  const issuedAtMs = Date.now();
  const payloadObj = {
    v: 1,
    countryIso2,
    regionKey: regionKey ?? null,
    global: { id: DOC1_GLOBAL.id, version: DOC1_GLOBAL.version, sha256: globalSha },
    addendum: { id: addendum.id, version: addendum.version, sha256: addendumSha },
    combinedSha256: combinedSha,
    issuedAtMs,
  };

  const payloadJson = JSON.stringify(payloadObj);
  const payloadB64 = base64UrlEncode(payloadJson);
  const sig = hmacSha256Hex(requireSecret(), payloadB64);
  const token = `${payloadB64}.${sig}`;

  const assembled: AssembledTerms = {
    global: { id: DOC1_GLOBAL.id, version: DOC1_GLOBAL.version, sha256: globalSha },
    addendum: { id: addendum.id, version: addendum.version, sha256: addendumSha, regionKey: regionKey ?? null },
    combined: { sha256: combinedSha, text: combinedText },
    meta: { countryIso2, regionKey: regionKey ?? null, issuedAtMs },
    token,
  };

  CACHE.set(cacheKey, assembled);
  return assembled;
}

export type VerifiedToken = {
  ok: true;
  payload: {
    countryIso2: string;
    regionKey: string | null;
    global: { id: string; version: string; sha256: string };
    addendum: { id: string; version: string; sha256: string };
    combinedSha256: string;
    issuedAtMs: number;
  };
} | { ok: false; reason: string };

export function verifyTermsToken(token: string, maxAgeMs = 24 * 60 * 60 * 1000): VerifiedToken {
  try {
    const parts = (token || "").split(".");
    if (parts.length !== 2) return { ok: false, reason: "Bad token format." };
    const [payloadB64, sig] = parts;
    const expected = hmacSha256Hex(requireSecret(), payloadB64);
    if (!cryptoSafeEqualHex(sig, expected)) return { ok: false, reason: "Bad token signature." };

    const payloadJson = base64UrlDecode(payloadB64);
    const payload = JSON.parse(payloadJson);

    if (!payload?.countryIso2 || !payload?.combinedSha256 || !payload?.issuedAtMs) {
      return { ok: false, reason: "Token payload missing fields." };
    }

    const age = Date.now() - Number(payload.issuedAtMs);
    if (age < 0 || age > maxAgeMs) return { ok: false, reason: "Token expired." };

    return { ok: true, payload };
  } catch (e: any) {
    return { ok: false, reason: `Token verify error: ${e?.message || "unknown"}` };
  }
}

function cryptoSafeEqualHex(a: string, b: string): boolean {
  try {
    const ba = Buffer.from(a, "hex");
    const bb = Buffer.from(b, "hex");
    if (ba.length !== bb.length) return false;
    // timingSafeEqual throws if lengths differ (handled above)
    // eslint-disable-next-line @typescript-eslint/no-var-requires
    const crypto = require("crypto");
    return crypto.timingSafeEqual(ba, bb);
  } catch {
    return false;
  }
}

1.6 server/legal/legalAcceptanceService.ts — insert acceptance + hash-chained ledger

// FILE: /server/legal/legalAcceptanceService.ts
import { db } from "../db";
import { legalAcceptances } from "../../shared/schema";
import { desc } from "drizzle-orm";
import { sha256Hex, stableStringify } from "./cryptoUtils";

type InsertAcceptanceArgs = {
  userId: number;
  emailAtAcceptance: string | null;
  countryIso2: string;
  regionKey: string | null;

  global: { id: string; version: string; sha256: string };
  addendum: { id: string; version: string; sha256: string };
  combined: { sha256: string; text: string };

  termsToken: string;

  acceptedAtMs: number;
  ipAddress?: string | null;
  userAgent?: string | null;
  sessionId?: string | null;
};

export function insertLegalAcceptanceWithLedger(args: InsertAcceptanceArgs) {
  return db.transaction((tx) => {
    const last = tx
      .select({
        ledgerSeq: legalAcceptances.ledgerSeq,
        ledgerHash: legalAcceptances.ledgerHash,
      })
      .from(legalAcceptances)
      .orderBy(desc(legalAcceptances.ledgerSeq))
      .limit(1)
      .all();

    const prevSeq = last.length ? Number(last[0].ledgerSeq) : 0;
    const prevHash = last.length ? String(last[0].ledgerHash) : "GENESIS";
    const nextSeq = prevSeq + 1;

    const payloadForLedger = {
      ledgerSeq: nextSeq,
      prevLedgerHash: prevHash,

      userId: args.userId,
      emailAtAcceptance: args.emailAtAcceptance,
      countryIso2: args.countryIso2,
      regionKey: args.regionKey,

      global: args.global,
      addendum: args.addendum,
      combinedSha256: args.combined.sha256,

      acceptedAtMs: args.acceptedAtMs,
      ipAddress: args.ipAddress ?? null,
      userAgent: args.userAgent ?? null,
      sessionId: args.sessionId ?? null,
      termsToken: args.termsToken,
    };

    const ledgerPayloadStr = stableStringify(payloadForLedger);
    const ledgerHash = sha256Hex(`${prevHash}|${ledgerPayloadStr}`);

    tx.insert(legalAcceptances).values({
      userId: args.userId,
      emailAtAcceptance: args.emailAtAcceptance ?? null,

      countryIso2: args.countryIso2,
      regionKey: args.regionKey ?? null,

      globalDocId: args.global.id,
      globalDocVersion: args.global.version,
      globalDocSha256: args.global.sha256,

      addendumId: args.addendum.id,
      addendumVersion: args.addendum.version,
      addendumSha256: args.addendum.sha256,

      combinedSha256: args.combined.sha256,
      combinedText: args.combined.text,

      acceptedAt: new Date(args.acceptedAtMs),
      ipAddress: args.ipAddress ?? null,
      userAgent: args.userAgent ?? null,
      sessionId: args.sessionId ?? null,

      termsToken: args.termsToken,

      ledgerSeq: nextSeq,
      prevLedgerHash: prevHash,
      ledgerHash,
    }).run();

    return { ledgerSeq: nextSeq, prevLedgerHash: prevHash, ledgerHash };
  });
}

 *   email, username, password, countryIso2, phoneE164,
 *   legal: { token, combinedSha256 }
 * }
 */
authSignupRouter.post("/signup", (req, res) => {
  try {
    const { email, username, password, countryIso2, phoneE164, legal } = req.body || {};
    if (!email || !username || !password) return res.status(400).json({ ok: false, error: "Missing required fields." });
    if (!countryIso2) return res.status(400).json({ ok: false, error: "Missing country." });
    if (!legal?.token || !legal?.combinedSha256) return res.status(400).json({ ok: false, error: "Missing legal acceptance." });

    // 1) Verify token signature + expiry
    const vt = verifyTermsToken(String(legal.token));
    if (!vt.ok) return res.status(400).json({ ok: false, error: `Invalid terms token: ${vt.reason}` });

    // 2) Re-assemble canonical terms for the claimed country
    const assembled = assembleDoc1Terms(String(countryIso2));

    // 3) Ensure token payload matches assembled terms
    const tokenCountry = String((vt as any).payload.countryIso2);
    const tokenCombined = String((vt as any).payload.combinedSha256);

    if (tokenCountry !== assembled.meta.countryIso2) {
      return res.status(400).json({ ok: false, error: "Terms token country mismatch." });
    }

    if (tokenCombined !== assembled.combined.sha256) {
      return res.status(400).json({ ok: false, error: "Terms token does not match current terms." });
    }

    // 4) Ensure client-reported hash matches canonical hash
    if (String(legal.combinedSha256) !== assembled.combined.sha256) {
      return res.status(400).json({ ok: false, error: "Client hash mismatch; reload terms and retry." });
    }

    // 5) Create user (simplified: replace with your actual password hashing and user creation logic)
    // IMPORTANT: hash password properly in your existing auth system.
    const created = db
      .insert(users)
      .values({
        email,
        username,
        passwordHash: password, // <-- replace: store a real hash
        countryIso: assembled.meta.countryIso2,
        phone: phoneE164 ?? null,
        createdAt: new Date(),
      } as any)
      .returning({ id: users.id })
      .get();

    const userId = Number(created?.id);
    if (!userId) return res.status(500).json({ ok: false, error: "User creation failed." });

    // 6) Audit: insert legal acceptance snapshot + ledger chain
    const ipAddress = (req.headers["x-forwarded-for"] as string)?.split(",")[0]?.trim() || req.socket.remoteAddress || null;
    const userAgent = String(req.headers["user-agent"] || "");
    const sessionId = String(req.headers["x-session-id"] || ""); // optional if you have one

    const ledger = insertLegalAcceptanceWithLedger({
      userId,
      emailAtAcceptance: email,
      countryIso2: assembled.meta.countryIso2,
      regionKey: assembled.meta.regionKey ?? null,
      global: assembled.global,
      addendum: { id: assembled.addendum.id, version: assembled.addendum.version, sha256: assembled.addendum.sha256 },
      combined: { sha256: assembled.combined.sha256, text: assembled.combined.text },
      termsToken: assembled.token,
      acceptedAtMs: Date.now(),
      ipAddress,
      userAgent,
      sessionId: sessionId || null,
    });

    return res.json({ ok: true, userId, legalLedger: ledger });
  } catch (e: any) {
    return res.status(500).json({ ok: false, error: e?.message || "Signup failed." });
  }
});

function TermsModal(props: {
  open: boolean;
  onClose: () => void;
  title: string;
  text: string;
  footerMeta?: string;
  onScrolledToEnd: () => void;
}) {
  const { open, onClose, title, text, footerMeta, onScrolledToEnd } = props;
  const scrollRef = useRef<HTMLDivElement | null>(null);

  useEffect(() => {
    if (!open) return;
    // reset scroll position on open
    const el = scrollRef.current;
    if (el) el.scrollTop = 0;
  }, [open, text]);

  function onScroll() {
    const el = scrollRef.current;
    if (!el) return;
    const atBottom = el.scrollTop + el.clientHeight >= el.scrollHeight - 8;
    if (atBottom) onScrolledToEnd();
  }

  if (!open) return null;

  return (
    <div style={modalOverlay} onMouseDown={onClose}>
      <div style={modalCard} onMouseDown={(e) => e.stopPropagation()}>
        <div style={{ display: "flex", justifyContent: "space-between", gap: 12, alignItems: "center" }}>
          <div style={{ fontSize: 16, fontWeight: 700 }}>{title}</div>
          <button onClick={onClose} style={buttonStyle}>Close</button>
        </div>

        <div
          ref={scrollRef}
          onScroll={onScroll}
          style={{
            marginTop: 12,
            border: "1px solid rgba(0,0,0,0.15)",
            borderRadius: 8,
            height: 420,
            overflow: "auto",
            padding: 12,
            whiteSpace: "pre-wrap",
            fontSize: 12,
            lineHeight: 1.4,
          }}
        >
          {text || "No terms loaded."}
        </div>

        <div style={{ marginTop: 10, fontSize: 12, opacity: 0.75 }}>
          {footerMeta || ""}
        </div>
      </div>
    </div>
  );
}
