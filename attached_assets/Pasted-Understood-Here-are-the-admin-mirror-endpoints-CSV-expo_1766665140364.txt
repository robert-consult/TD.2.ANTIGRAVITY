Understood. Here are the **admin mirror endpoints + CSV export + timezone “city search” meta endpoint**, all inline and ready to drop in.

For “gold standard” timezones/cities, we are using **`@vvo/tzdb`** (grouped IANA zones + major cities). ([npm][1])
For IP→country/region/city enrichment (login trail), we use **`geoip-lite`** (returns country/region/city when available; geolocation is inherently imperfect). ([npm][2])
For lat/lon → IANA timezone inference, we use **`@photostructure/tz-lookup`**. ([npm][3])
Optional device parsing uses **`ua-parser-js`**. ([npm][4])

## 1) Backend — Admin Security Routes (login trail + sessions + revoke + CSV)

### `server/routes/adminSecurity.ts` (NEW FILE)

```ts
// server/routes/adminSecurity.ts
import type { Express, Request, Response } from "express";
import { and, desc, eq, isNull } from "drizzle-orm";
import { db } from "../db"; // adjust to your db export path
import { securityEvents, userSessions } from "../../shared/schema";
import { revokeSession } from "../security/sessionTrail";

/**
 * IMPORTANT:
 * Replace this with your real admin middleware / auth guard.
 * This stub expects req.user = { id: number, role?: string, isAdmin?: boolean }
 */
function requireAdmin(req: any, res: Response): { adminUserId: number } | null {
  const adminUserId = req.user?.id ?? req.userId;
  const isAdmin = Boolean(req.user?.isAdmin || req.user?.role === "ADMIN");
  if (!adminUserId || !isAdmin) {
    res.status(403).json({ error: "forbidden" });
    return null;
  }
  return { adminUserId: Number(adminUserId) };
}

function toCsv(rows: Array<Record<string, any>>): string {
  if (!rows.length) return "no_data\n";
  const headers = Object.keys(rows[0]);
  const escape = (v: any) => {
    if (v === null || v === undefined) return "";
    const s = String(v);
    if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`;
    return s;
  };
  const lines = [
    headers.join(","),
    ...rows.map((r) => headers.map((h) => escape(r[h])).join(",")),
  ];
  return lines.join("\n") + "\n";
}

export function registerAdminSecurityRoutes(app: Express) {
  /**
   * Login/Security activity trail for a user (append-only securityEvents)
   * GET /api/admin/users/:userId/login-activity?limit=50
   */
  app.get("/api/admin/users/:userId/login-activity", (req: Request, res: Response) => {
    const admin = requireAdmin(req as any, res);
    if (!admin) return;

    const userId = Number(req.params.userId);
    const limit = Math.min(200, Math.max(1, Number(req.query.limit || 50)));

    const rows = db
      .select({
        id: securityEvents.id,
        eventType: securityEvents.eventType,
        eventAt: securityEvents.eventAt,
        sessionId: securityEvents.sessionId,
        ip: securityEvents.ip,
        userAgent: securityEvents.userAgent,
        countryCode: securityEvents.countryCode,
        region: securityEvents.region,
        city: securityEvents.city,
        note: securityEvents.note,
        metaJson: securityEvents.metaJson,
        prevHash: securityEvents.prevHash,
        eventHash: securityEvents.eventHash,
      })
      .from(securityEvents)
      .where(eq(securityEvents.userId, userId))
      .orderBy(desc(securityEvents.eventAt))
      .limit(limit)
      .all();

    res.json({ rows });
  });

  /**
   * CSV export for login activity
   * GET /api/admin/users/:userId/login-activity.csv?limit=500
   */
  app.get("/api/admin/users/:userId/login-activity.csv", (req: Request, res: Response) => {
    const admin = requireAdmin(req as any, res);
    if (!admin) return;

    const userId = Number(req.params.userId);
    const limit = Math.min(2000, Math.max(1, Number(req.query.limit || 500)));

    const rows = db
      .select({
        id: securityEvents.id,
        eventType: securityEvents.eventType,
        eventAt: securityEvents.eventAt,
        sessionId: securityEvents.sessionId,
        ip: securityEvents.ip,
        userAgent: securityEvents.userAgent,
        countryCode: securityEvents.countryCode,
        region: securityEvents.region,
        city: securityEvents.city,
        note: securityEvents.note,
        metaJson: securityEvents.metaJson,
        prevHash: securityEvents.prevHash,
        eventHash: securityEvents.eventHash,
      })
      .from(securityEvents)
      .where(eq(securityEvents.userId, userId))
      .orderBy(desc(securityEvents.eventAt))
      .limit(limit)
      .all();

    res.setHeader("Content-Type", "text/csv; charset=utf-8");
    res.setHeader("Content-Disposition", `attachment; filename="user_${userId}_login_activity.csv"`);
    res.send(toCsv(rows));
  });

  /**
   * Active sessions for a user
   * GET /api/admin/users/:userId/sessions?limit=50&includeEnded=0
   */
  app.get("/api/admin/users/:userId/sessions", (req: Request, res: Response) => {
    const admin = requireAdmin(req as any, res);
    if (!admin) return;

    const userId = Number(req.params.userId);
    const limit = Math.min(200, Math.max(1, Number(req.query.limit || 50)));
    const includeEnded = String(req.query.includeEnded || "0") === "1";

    const where = includeEnded
      ? eq(userSessions.userId, userId)
      : and(eq(userSessions.userId, userId), isNull(userSessions.endedAt));

    const rows = db
      .select({
        id: userSessions.id,
        createdAt: userSessions.createdAt,
        lastSeenAt: userSessions.lastSeenAt,
        endedAt: userSessions.endedAt,
        ip: userSessions.ip,
        userAgent: userSessions.userAgent,
        countryCode: userSessions.countryCode,
        region: userSessions.region,
        city: userSessions.city,
        inferredTz: userSessions.inferredTz,
        revokedAt: userSessions.revokedAt,
        revokedByUserId: userSessions.revokedByUserId,
        revokeReason: userSessions.revokeReason,
      })
      .from(userSessions)
      .where(where)
      .orderBy(desc(userSessions.lastSeenAt))
      .limit(limit)
      .all();

    res.json({ rows });
  });

  /**
   * Admin revoke a specific user session (forces logout) + writes security trail entry
   * POST /api/admin/users/:userId/sessions/:sessionId/revoke
   */
  app.post("/api/admin/users/:userId/sessions/:sessionId/revoke", (req: Request, res: Response) => {
    const admin = requireAdmin(req as any, res);
    if (!admin) return;

    const userId = Number(req.params.userId);
    const sessionId = String(req.params.sessionId);
    const reason = String((req.body as any)?.reason || "Admin revoked session");

    revokeSession({
      actorUserId: admin.adminUserId,
      targetUserId: userId,
      sessionId,
      reason,
    });

    res.json({ ok: true });
  });
}
```

## 2) Backend — Timezone Meta “Search by City” (improves your Profile Settings dropdown)

If your dropdown should support “type Nairobi / Chicago / London” and it returns matching IANA timezones with offsets + main cities, add:

### `server/routes/meta.ts` (ADD THIS ENDPOINT)

```ts
// server/routes/meta.ts
import type { Express, Request, Response } from "express";
import { getTimeZones } from "@vvo/tzdb";

export function registerMetaRoutes(app: Express) {
  // ...keep your existing /api/meta/timezones

  /**
   * Search timezones by IANA name or major city label.
   * GET /api/meta/timezones/search?q=nairobi&limit=30
   */
  app.get("/api/meta/timezones/search", (req: Request, res: Response) => {
    const q = String(req.query.q || "").trim().toLowerCase();
    const limit = Math.min(100, Math.max(1, Number(req.query.limit || 30)));

    if (!q) {
      res.json({ rows: [] });
      return;
    }

    const tzs: any[] = getTimeZones({ includeUtc: true }) as any[];

    const rows = tzs
      .map((t) => {
        const cities: string[] = Array.isArray(t.mainCities) ? t.mainCities : [];
        const hay = [
          String(t.name || ""),
          String(t.alternativeName || ""),
          String(t.countryName || ""),
          String(t.countryCode || ""),
          ...cities,
        ]
          .join(" ")
          .toLowerCase();

        return {
          match: hay.includes(q),
          name: t.name as string,
          countryCode: t.countryCode as string,
          countryName: t.countryName as string,
          alternativeName: t.alternativeName as string,
          mainCities: cities,
          rawOffsetInMinutes: t.rawOffsetInMinutes as number,
          abbreviation: t.abbreviation as string,
          rawFormat: t.rawFormat as string, // already contains offset + name formatting
        };
      })
      .filter((r) => r.match)
      .slice(0, limit)
      .map(({ match, ...rest }) => rest);

    res.json({ rows });
  });
}
```

```ts
// client/src/lib/adminSecurityApi.ts
export type AdminLoginRow = {
  id: number;
  eventType: string;
  eventAt: number;
  sessionId?: string | null;
  ip?: string | null;
  userAgent?: string | null;
  countryCode?: string | null;
  region?: string | null;
  city?: string | null;
  note?: string | null;
  metaJson?: string | null;
  prevHash?: string | null;
  eventHash?: string | null;
};

export type AdminSessionRow = {
  id: string;
  createdAt: number;
  lastSeenAt: number;
  endedAt?: number | null;
  ip: string;
  userAgent: string;
  countryCode?: string | null;
  region?: string | null;
  city?: string | null;
  inferredTz?: string | null;
  revokedAt?: number | null;
  revokedByUserId?: number | null;
  revokeReason?: string | null;
};

async function json<T>(url: string, init?: RequestInit): Promise<T> {
  const res = await fetch(url, {
    ...init,
    headers: { "Content-Type": "application/json", ...(init?.headers || {}) },
  });
  if (!res.ok) throw new Error(`${res.status} ${res.statusText}`);
  return res.json() as Promise<T>;
}

export const adminSecurityApi = {
  getLoginActivity: (userId: number, limit = 50) =>
    json<{ rows: AdminLoginRow[] }>(`/api/admin/users/${userId}/login-activity?limit=${limit}`),

  getSessions: (userId: number, limit = 50, includeEnded = false) =>
    json<{ rows: AdminSessionRow[] }>(
      `/api/admin/users/${userId}/sessions?limit=${limit}&includeEnded=${includeEnded ? "1" : "0"}`
    ),

  revokeSession: (userId: number, sessionId: string, reason?: string) =>
    json<{ ok: boolean }>(`/api/admin/users/${userId}/sessions/${encodeURIComponent(sessionId)}/revoke`, {
      method: "POST",
      body: JSON.stringify({ reason }),
    }),
};
```

## 5) Quick Admin UI panel (vertical timeline markers) (optional)

Drop into `/client/src/pages/` or your admin components and wire into your mini-tab.

### `client/src/pages/AdminUserSecurityPanel.tsx` (NEW FILE)

```tsx
// client/src/pages/AdminUserSecurityPanel.tsx
import React from "react";
import { adminSecurityApi, AdminLoginRow, AdminSessionRow } from "@/lib/adminSecurityApi";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";

function fmt(ts: number) {
  return new Date(ts).toLocaleString();
}
function loc(r: { city?: string | null; region?: string | null; countryCode?: string | null }) {
  const parts = [r.city, r.region, r.countryCode].filter(Boolean);
  return parts.length ? parts.join(", ") : "Unknown";
}

export function AdminUserSecurityPanel({ userId }: { userId: number }) {
  const [trail, setTrail] = React.useState<AdminLoginRow[] | null>(null);
  const [sessions, setSessions] = React.useState<AdminSessionRow[] | null>(null);

  const refresh = React.useCallback(async () => {
    const [t, s] = await Promise.all([
      adminSecurityApi.getLoginActivity(userId, 100).catch(() => ({ rows: [] })),
      adminSecurityApi.getSessions(userId, 50, false).catch(() => ({ rows: [] })),
    ]);
    setTrail(t.rows);
    setSessions(s.rows);
  }, [userId]);

  React.useEffect(() => {
    refresh();
  }, [refresh]);

  return (
    <div className="space-y-6">
      <Card>
        <CardHeader className="flex flex-row items-center justify-between">
          <CardTitle>Security Trail</CardTitle>
          <div className="flex items-center gap-2">
            <Button variant="outline" onClick={refresh}>Refresh</Button>
            <Button
              variant="outline"
              onClick={() => window.open(`/api/admin/users/${userId}/login-activity.csv?limit=2000`, "_blank")}
            >
              Export CSV
            </Button>
          </div>
        </CardHeader>
        <CardContent>
          {!trail ? (
            <div className="text-muted-foreground">Loading…</div>
          ) : trail.length === 0 ? (
            <div className="text-muted-foreground">No security events</div>
          ) : (
            <div className="relative pl-6">
              <div className="absolute left-2 top-0 bottom-0 w-px bg-border" />
              <div className="space-y-5">
                {trail.map((e) => (
                  <div key={e.id} className="relative">
                    <div className="absolute -left-[2px] top-1.5 h-3 w-3 rounded-full border bg-background" />
                    <div className="flex items-start justify-between gap-4">
                      <div className="min-w-0">
                        <div className="font-medium">
                          {e.eventType.replaceAll("_", " ")}
                          {e.ip ? <span className="text-muted-foreground"> · {e.ip}</span> : null}
                        </div>
                        <div className="text-sm text-muted-foreground">
                          {fmt(e.eventAt)} · {loc(e)} · {e.userAgent ? e.userAgent.slice(0, 90) : "Unknown device"}
                        </div>
                        {e.note ? <div className="text-sm mt-1">{e.note}</div> : null}
                        <div className="text-xs text-muted-foreground mt-1 font-mono">
                          hash {String(e.eventHash || "").slice(0, 12)}… prev {String(e.prevHash || "").slice(0, 12)}…
                        </div>
                      </div>
                      {e.sessionId ? (
                        <div className="text-xs text-muted-foreground whitespace-nowrap font-mono">
                          SID {e.sessionId.slice(0, 8)}…
                        </div>
                      ) : null}
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}
        </CardContent>
      </Card>

      <Card>
        <CardHeader>
          <CardTitle>Active Sessions</CardTitle>
        </CardHeader>
        <CardContent>
          {!sessions ? (
            <div className="text-muted-foreground">Loading…</div>
          ) : sessions.length === 0 ? (
            <div className="text-muted-foreground">No active sessions</div>
          ) : (
            <div className="space-y-3">
              {sessions.map((s) => (
                <div key={s.id} className="flex items-start justify-between gap-4 border-b pb-3 last:border-b-0 last:pb-0">
                  <div className="min-w-0">
                    <div className="font-medium">
                      {s.ip} <span className="text-muted-foreground">· last seen {fmt(s.lastSeenAt)}</span>
                    </div>
                    <div className="text-sm text-muted-foreground">
                      Started {fmt(s.createdAt)} · {loc(s)} · {s.inferredTz ? `inferred ${s.inferredTz}` : "no inferred tz"}
                    </div>
                    <div className="text-xs text-muted-foreground font-mono">
                      {s.userAgent.slice(0, 110)}
                    </div>
                  </div>
                  <Button
                    variant="destructive"
                    size="sm"
                    onClick={async () => {
                      await adminSecurityApi.revokeSession(userId, s.id, "Admin revoked session");
                      await refresh();
                    }}
                  >
                    Revoke
                  </Button>
                </div>
              ))}
            </div>
          )}
        </CardContent>
      </Card>
    </div>
  );
}
```
