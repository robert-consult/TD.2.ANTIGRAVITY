## 2) Critical failures / drift risks (Replit audit missed these)

### A) PolicyDecision does not enforce SL/TP modify + pending cancel when email-locked

Your context matrix says locked accounts should be **reduce-only** (and typically **no pending cancel, no SL/TP edits** unless explicitly permitted). In current code:

* `shared/policyDecision.ts` **does not implement** `TRADE_MODIFY_SLTP` at all inside locked-state logic, so it falls through to **ALLOW**.
* `TRADE_CANCEL_PENDING` is treated as reduce-only and is **allowed when locked**, but your config field `allowCancelPendingWhenLocked` is **never used**.

Evidence:

* `TRADE_MODIFY_SLTP` exists only in the action union but is not handled in the locked blocks (falls through). See `shared/policyDecision.ts` around the locked checks.
* Reduce-only action set includes cancel-pending, so cancel gets allowed when locked. See `shared/policyDecision.ts` lines ~187–214 (in your ZIP).

Impact:

* A trader with an **email-locked** account can still:

  * modify SL/TP (risk controls drift),

Evidence:

Impact:

// shared/policyDecision.ts
// Single-source-of-truth: UI + backend must call this (directly or via requirePolicy)
// Returns allow + denyCode so both layers stay aligned.

export type PolicyAction =
  | "TRADE_OPEN_OR_INCREASE"
  | "TRADE_CLOSE_OR_REDUCE"
  | "TRADE_CANCEL_PENDING"
  | "TRADE_MODIFY_SLTP"
  | "EMAIL_SEND_VERIFY"
  | "EMAIL_RESEND_VERIFICATION"
  | "PHONE_VERIFY_START"
  | "KYC_VIEW"
  | "PAYOUT_PROFILE_VIEW"
  | "PAYOUT_SET_CURRENCY";

export type DenyCode =
  | "OK"
  | "AUTH_REQUIRED"
  | "EMAIL_NOT_VERIFIED"
  | "EMAIL_REVERIFY_OVERDUE"
  | "EMAIL_REVERIFY_OVERDUE_GRACE"
  | "THROTTLED_EMAIL_SEND"
  | "SMS_DAILY_LIMIT_REACHED"
  | "SMS_COOLDOWN"
  | "EMAIL_DAILY_LIMIT_REACHED"
  | "EMAIL_COOLDOWN"
  | "KYC_NOT_INVITED"
  | "KYC_NOT_APPROVED"
  | "NOT_ELIGIBLE"
  | "TRADE_LOCKED_EMAIL_NOT_VERIFIED"
  | "TRADE_LOCKED_EMAIL_REVERIFY_OVERDUE"
  | "TRADE_CANCEL_NOT_ALLOWED_WHEN_LOCKED"
  | "TRADE_TARGETS_NOT_ALLOWED_WHEN_LOCKED"
  | "SMS_ALREADY_VERIFIED";

export type AccountState =
  | "ACTIVE"
  | "WARN_EMAIL_REVERIFY_DUE_SOON"
  | "WARN_EMAIL_REVERIFY_OVERDUE_GRACE"
  | "LOCKED_EMAIL_NOT_VERIFIED"
  | "LOCKED_EMAIL_REVERIFY_OVERDUE";

export type ContenderTier =
  | "NONE"
  | "CANDIDATE_EMAIL_ONLY"
  | "CANDIDATE_SMS_REQUIRED"
  | "VERIFIED_SMS"
  | "SELECTED_REAL_CAPITAL";

export type UserTier = "CANDIDATE" | "PERFORMER" | "SELECTED";

export type KycStatus =
  | "NOT_STARTED"
  | "INVITED"
  | "SUBMITTED"
  | "UNDER_REVIEW"
  | "PENDING_DOCS"
  | "APPROVED"
  | "REJECTED";

export type DecisionContext = {
  now: number;

  user: {
    id: number;

    // Email
    emailVerifiedAt: number | null;
    userCreatedAt: number;
    emailInitialDueAt: number | null; // builder should provide; fallback to createdAt+initialGrace
    emailReverifyDueAtMs: number | null;

    // Rate limits
    emailSendCountDay: number;
    emailLastSentAtMs: number | null;

    smsSendCountDay: number;
    smsLastSentAt: number | null;
    phoneVerifiedAt: number | null;

    // Tiering
    userTier: UserTier;
    contenderTier: ContenderTier;

    // Eligibility computed by builder
    contenderEligible: boolean;

    // KYC
    kycStatus: KycStatus;
  };

  // Optional: allow future “risk-reducing only” logic
  tradeChange?: {
    riskReducing?: boolean;
  };

  config?: Partial<PolicyConfig>;
};

export type PolicyConfig = {
  // Email policy
  emailInitialGraceDays: number;      // e.g., 14
  emailReverifyPeriodDays: number;    // e.g., 30
  emailOverdueGraceDays: number;      // e.g., 7
  emailDueSoonDays: number;           // e.g., 7

  emailDailyLimit: number;            // e.g., 3
  emailCooldownMinutes: number;       // e.g., 2

  // SMS policy
  smsDailyLimit: number;              // e.g., 3
  smsCooldownMinutes: number;         // e.g., 2

  // Locked account behavior
  allowReduceOnlyWhenLocked: boolean;                 // close/reduce always true
  allowCancelPendingWhenLocked: boolean;              // typically false
  allowRiskReducingSltpChangeWhenLocked: boolean;     // typically false unless you pass tradeChange.riskReducing
};

const DEFAULT_CONFIG: PolicyConfig = {
  emailInitialGraceDays: 14,
  emailReverifyPeriodDays: 30,
  emailOverdueGraceDays: 7,
  emailDueSoonDays: 7,

  emailDailyLimit: 3,
  emailCooldownMinutes: 2,

  smsDailyLimit: 3,
  smsCooldownMinutes: 2,

  allowReduceOnlyWhenLocked: true,
  allowCancelPendingWhenLocked: false,
  allowRiskReducingSltpChangeWhenLocked: false,
};

const MS_DAY = 24 * 60 * 60 * 1000;

function getConfig(ctx: DecisionContext): PolicyConfig {
  return { ...DEFAULT_CONFIG, ...(ctx.config ?? {}) };
}

function resolveEmailInitialDueAt(ctx: DecisionContext, cfg: PolicyConfig): number {
  if (ctx.user.emailInitialDueAt && ctx.user.emailInitialDueAt > 0) return ctx.user.emailInitialDueAt;
  return ctx.user.userCreatedAt + cfg.emailInitialGraceDays * MS_DAY;
}

export function resolveAccountState(ctx: DecisionContext): AccountState {
  const cfg = getConfig(ctx);
  const now = ctx.now;

  const emailVerifiedAt = ctx.user.emailVerifiedAt;
  const initialDueAt = resolveEmailInitialDueAt(ctx, cfg);

  // Not verified: grace then lock
  if (!emailVerifiedAt) {
    if (now <= initialDueAt) return "ACTIVE";
    return "LOCKED_EMAIL_NOT_VERIFIED";
  }

  // Verified: rolling reverify schedule
  const dueAt =
    ctx.user.emailReverifyDueAtMs && ctx.user.emailReverifyDueAtMs > 0
      ? ctx.user.emailReverifyDueAtMs
      : emailVerifiedAt + cfg.emailReverifyPeriodDays * MS_DAY;

  const overdueGraceEndsAt = dueAt + cfg.emailOverdueGraceDays * MS_DAY;

  // Due soon warning (before due)
  const daysLeft = Math.floor((dueAt - now) / MS_DAY);
  if (daysLeft >= 0 && daysLeft <= cfg.emailDueSoonDays) {
    return "WARN_EMAIL_REVERIFY_DUE_SOON";
  }

  // Overdue but within grace: warn, not locked
  if (now > dueAt && now <= overdueGraceEndsAt) {
    return "WARN_EMAIL_REVERIFY_OVERDUE_GRACE";
  }

  // Overdue beyond grace: lock
  if (now > overdueGraceEndsAt) {
    return "LOCKED_EMAIL_REVERIFY_OVERDUE";
  }

  return "ACTIVE";
}

function isLocked(state: AccountState): boolean {
  return state === "LOCKED_EMAIL_NOT_VERIFIED" || state === "LOCKED_EMAIL_REVERIFY_OVERDUE";
}

function emailThrottleDecision(ctx: DecisionContext, cfg: PolicyConfig): { allow: boolean; denyCode: DenyCode } {
  if (ctx.user.emailSendCountDay >= cfg.emailDailyLimit) {
    return { allow: false, denyCode: "EMAIL_DAILY_LIMIT_REACHED" };
  }
  if (ctx.user.emailLastSentAtMs) {
    const cooldownMs = cfg.emailCooldownMinutes * 60 * 1000;
    if (ctx.now - ctx.user.emailLastSentAtMs < cooldownMs) {
      return { allow: false, denyCode: "EMAIL_COOLDOWN" };
    }
  }
  return { allow: true, denyCode: "OK" };
}

function smsThrottleDecision(ctx: DecisionContext, cfg: PolicyConfig): { allow: boolean; denyCode: DenyCode } {
  if (ctx.user.smsSendCountDay >= cfg.smsDailyLimit) {
    return { allow: false, denyCode: "SMS_DAILY_LIMIT_REACHED" };
  }
  if (ctx.user.smsLastSentAt) {
    const cooldownMs = cfg.smsCooldownMinutes * 60 * 1000;
    if (ctx.now - ctx.user.smsLastSentAt < cooldownMs) {
      return { allow: false, denyCode: "SMS_COOLDOWN" };
    }
  }
  return { allow: true, denyCode: "OK" };
}

export function decidePolicy(
  ctx: DecisionContext,
  action: PolicyAction
): { allow: boolean; denyCode: DenyCode; accountState: AccountState } {
  const cfg = getConfig(ctx);
  const accountState = resolveAccountState(ctx);
  const locked = isLocked(accountState);

  // Trade actions
  if (action === "TRADE_OPEN_OR_INCREASE") {
    if (accountState === "LOCKED_EMAIL_NOT_VERIFIED") {
      return { allow: false, denyCode: "TRADE_LOCKED_EMAIL_NOT_VERIFIED", accountState };
    }
    if (accountState === "LOCKED_EMAIL_REVERIFY_OVERDUE") {
      return { allow: false, denyCode: "TRADE_LOCKED_EMAIL_REVERIFY_OVERDUE", accountState };
    }
    return { allow: true, denyCode: "OK", accountState };
  }

  if (action === "TRADE_CLOSE_OR_REDUCE") {
    // Always allow close/reduce
    return { allow: true, denyCode: "OK", accountState };
  }

  if (action === "TRADE_CANCEL_PENDING") {
    if (!locked) return { allow: true, denyCode: "OK", accountState };
    if (cfg.allowCancelPendingWhenLocked) return { allow: true, denyCode: "OK", accountState };
    return { allow: false, denyCode: "TRADE_CANCEL_NOT_ALLOWED_WHEN_LOCKED", accountState };
  }

  if (action === "TRADE_MODIFY_SLTP") {
    if (!locked) return { allow: true, denyCode: "OK", accountState };

    if (cfg.allowRiskReducingSltpChangeWhenLocked && ctx.tradeChange?.riskReducing) {
      return { allow: true, denyCode: "OK", accountState };
    }

    return { allow: false, denyCode: "TRADE_TARGETS_NOT_ALLOWED_WHEN_LOCKED", accountState };
  }

  // Email verification sends (allow even if trading is locked; this is how user unlocks)
  if (action === "EMAIL_SEND_VERIFY" || action === "EMAIL_RESEND_VERIFICATION") {
    const t = emailThrottleDecision(ctx, cfg);
    return { allow: t.allow, denyCode: t.allow ? "OK" : "THROTTLED_EMAIL_SEND", accountState };
  }

  // SMS verification start: gate by eligibility + email verified + throttles
  if (action === "PHONE_VERIFY_START") {
    if (ctx.user.phoneVerifiedAt) {
      return { allow: false, denyCode: "SMS_ALREADY_VERIFIED", accountState };
    }
    if (!ctx.user.emailVerifiedAt) {
      return { allow: false, denyCode: "EMAIL_NOT_VERIFIED", accountState };
    }
    if (locked) {
      return {
        allow: false,
        denyCode: accountState === "LOCKED_EMAIL_NOT_VERIFIED" ? "EMAIL_NOT_VERIFIED" : "EMAIL_REVERIFY_OVERDUE",
        accountState,
      };
    }
    if (!ctx.user.contenderEligible) {
      return { allow: false, denyCode: "NOT_ELIGIBLE", accountState };
    }

    const t = smsThrottleDecision(ctx, cfg);
    return { allow: t.allow, denyCode: t.allow ? "OK" : t.denyCode, accountState };
  }

  // KYC view: invite-based (or selected), but always require email OK
  if (action === "KYC_VIEW") {
    if (!ctx.user.emailVerifiedAt) return { allow: false, denyCode: "EMAIL_NOT_VERIFIED", accountState };
    if (locked) return { allow: false, denyCode: "EMAIL_REVERIFY_OVERDUE", accountState };

    const status = ctx.user.kycStatus;
    const invitedLike =
      status === "INVITED" ||
      status === "SUBMITTED" ||
      status === "UNDER_REVIEW" ||
      status === "PENDING_DOCS" ||
      status === "APPROVED";

    if (ctx.user.userTier === "SELECTED" || invitedLike) {
      return { allow: true, denyCode: "OK", accountState };
    }

    return { allow: false, denyCode: "KYC_NOT_INVITED", accountState };
  }

  // Payout profile view: same gating as KYC view
  if (action === "PAYOUT_PROFILE_VIEW") {
    const d = decidePolicy(ctx, "KYC_VIEW");
    if (!d.allow) return { allow: false, denyCode: d.denyCode, accountState };
    return { allow: true, denyCode: "OK", accountState };
  }

  // Setting payout currency: require KYC approved
  if (action === "PAYOUT_SET_CURRENCY") {
    const d = decidePolicy(ctx, "KYC_VIEW");
    if (!d.allow) return { allow: false, denyCode: d.denyCode, accountState };
    if (ctx.user.kycStatus !== "APPROVED") return { allow: false, denyCode: "KYC_NOT_APPROVED", accountState };
    return { allow: true, denyCode: "OK", accountState };
  }

  return { allow: true, denyCode: "OK", accountState };
}

export function featureGates(ctx: DecisionContext) {
  const tradeOpen = decidePolicy(ctx, "TRADE_OPEN_OR_INCREASE");
  const tradeClose = decidePolicy(ctx, "TRADE_CLOSE_OR_REDUCE");
  const tradeCancel = decidePolicy(ctx, "TRADE_CANCEL_PENDING");
  const tradeTargets = decidePolicy(ctx, "TRADE_MODIFY_SLTP");
  const canStartSms = decidePolicy(ctx, "PHONE_VERIFY_START");
  const canViewKyc = decidePolicy(ctx, "KYC_VIEW");
  const canViewPayout = decidePolicy(ctx, "PAYOUT_PROFILE_VIEW");
  const canSetCurrency = decidePolicy(ctx, "PAYOUT_SET_CURRENCY");

  return {
    accountState: tradeOpen.accountState,
    denyCodes: {
      tradeOpen: tradeOpen.denyCode,
      tradeClose: tradeClose.denyCode,
      tradeCancel: tradeCancel.denyCode,
      tradeTargets: tradeTargets.denyCode,
      smsStart: canStartSms.denyCode,
      kycView: canViewKyc.denyCode,
      payoutView: canViewPayout.denyCode,
      payoutSetCurrency: canSetCurrency.denyCode,
    },
    canTradeOpenOrIncrease: tradeOpen.allow,
    canTradeCloseOrReduce: tradeClose.allow,
    canCancelPending: tradeCancel.allow,
    canModifySltp: tradeTargets.allow,
    canStartSms: canStartSms.allow,
    canViewKyc: canViewKyc.allow,
    canViewPayoutProfile: canViewPayout.allow,
    canSetPreferredPaymentCurrency: canSetCurrency.allow,
  };
}
