Below is the **next step**: a **System Config minitab** (“Legal Docs”) that lets admins **load, preview, replace, and roll back** the **Global Master** and **Country/Region Addenda**—and a backend that:

* Stores documents **in DB** (so you can replace without redeploying files)
* Selects the correct **active doc** per country/region at signup
* Preserves **hashing + HMAC tokenization + acceptance ledger**
* Adds a **separate tamper-evident audit chain** for **legal document changes** (who changed what, when, from/to hashes)

Everything below is **copy/paste** file-by-file.

---

## 0) What changes conceptually

* **Docs move from filesystem-only to DB-first** (with optional bootstrap import from your existing text files).
* Signup terms are assembled from **active pointers**:

  * `DOC1 / GLOBAL_MASTER / DEFAULT / GLOBAL`
  * `DOC1 / ADDENDUM / COUNTRY / {ISO2}` else `REGION / {REGION_KEY}` else `DEFAULT / ROW`
* Admin minitab can:

  * View active doc + all prior versions
  * Create a new version and **activate** it in one action
  * Roll back by switching the pointer
  * Preview what a country will receive (“assemble preview”)

```ts
export function requireAdmin(req: Request, res: Response, next: NextFunction) {
  // Option A: header token (useful until you wire real auth roles)
  const token = String(req.headers["x-admin-token"] || "");
  const expected = process.env.ADMIN_API_TOKEN;

  if (expected && token && token === expected) return next();

  const user = anyReq.user;

  return res.status(403).json({ ok: false, error: "Admin access required." });
}
```

```ts
// FILE: /server/legal/cryptoUtils.ts
import crypto from "crypto";

export function sha256Hex(input: string): string {
  return crypto.createHash("sha256").update(input, "utf8").digest("hex");
}

export function stableStringify(value: unknown): string {
  if (value === null || typeof value !== "object") return JSON.stringify(value);
  if (Array.isArray(value)) return "[" + value.map((v) => stableStringify(v)).join(",") + "]";
  const obj = value as Record<string, unknown>;
  const keys = Object.keys(obj).sort();
  return "{" + keys.map((k) => JSON.stringify(k) + ":" + stableStringify(obj[k])).join(",") + "}";
}

export function base64UrlEncode(str: string): string {
  return Buffer.from(str, "utf8")
    .toString("base64")
    .replaceAll("+", "-")
    .replaceAll("/", "_")
    .replaceAll("=", "");
}

export function base64UrlDecode(str: string): string {
  const padLen = (4 - (str.length % 4)) % 4;
  const padded = str + "=".repeat(padLen);
  const b64 = padded.replaceAll("-", "+").replaceAll("_", "/");
  return Buffer.from(b64, "base64").toString("utf8");
}

export function hmacSha256Hex(secret: string, payload: string): string {
  return crypto.createHmac("sha256", secret).update(payload, "utf8").digest("hex");
}

export function timingSafeEqualHex(a: string, b: string): boolean {
  try {
    const ba = Buffer.from(a, "hex");
    const bb = Buffer.from(b, "hex");
    if (ba.length !== bb.length) return false;
    return crypto.timingSafeEqual(ba, bb);
  } catch {
    return false;
  }
}
```

## 2.4 `/server/legal/legalDocsRepo.ts` — DB helpers for docs + pointers

```ts
// FILE: /server/legal/legalDocsRepo.ts
import { and, desc, eq } from "drizzle-orm";
import { db } from "../db";
import { legalDocuments, legalDocPointers } from "../../shared/schema";
import { sha256Hex } from "./cryptoUtils";

export type DocTarget = {
  docSet: string; // "DOC1"
  docType: "GLOBAL_MASTER" | "ADDENDUM";
  jurisdictionType: "DEFAULT" | "COUNTRY" | "REGION";
  jurisdictionKey: string; // "GLOBAL" | "ROW" | "US" | "WEST_EUROPE"
};

export type LegalDocumentRow = {
  id: number;
  docSet: string;
  docType: string;
  jurisdictionType: string;
  jurisdictionKey: string;
  version: string;
  sha256: string;
  content: string;
  notes: string | null;
  createdAt: number;
  createdByAdminUserId: number | null;
};

export function computeDocSha(content: string): string {
  return sha256Hex(content);
}

export function getPointer(target: DocTarget) {
  return db
    .select()
    .from(legalDocPointers)
    .where(
      and(
        eq(legalDocPointers.docSet, target.docSet),
        eq(legalDocPointers.docType, target.docType),
        eq(legalDocPointers.jurisdictionType, target.jurisdictionType),
        eq(legalDocPointers.jurisdictionKey, target.jurisdictionKey)
      )
    )
    .get();
}

export function getActiveDoc(target: DocTarget) {
  const pointer = getPointer(target);
  if (!pointer) return { pointer: null, doc: null };

  const doc = db
    .select()
    .from(legalDocuments)
    .where(eq(legalDocuments.id, pointer.activeDocumentId))
    .get();

  return { pointer, doc };
}

export function listVersions(target: DocTarget) {
  return db
    .select()
    .from(legalDocuments)
    .where(
      and(
        eq(legalDocuments.docSet, target.docSet),
        eq(legalDocuments.docType, target.docType),
        eq(legalDocuments.jurisdictionType, target.jurisdictionType),
        eq(legalDocuments.jurisdictionKey, target.jurisdictionKey)
      )
    )
    .orderBy(desc(legalDocuments.createdAt))
    .all();
}

export function createDocumentVersion(args: {
  target: DocTarget;
  version: string;
  content: string;
  notes?: string | null;
  adminUserId?: number | null;
}) {
  const sha = computeDocSha(args.content);
  const now = Date.now();

  db.insert(legalDocuments)
    .values({
      docSet: args.target.docSet,
      docType: args.target.docType,
      jurisdictionType: args.target.jurisdictionType,
      jurisdictionKey: args.target.jurisdictionKey,
      version: args.version,
      sha256: sha,
      content: args.content,
      notes: args.notes ?? null,
      createdAt: new Date(now),
      createdByAdminUserId: args.adminUserId ?? null,
    })
    .run();

  // Fetch inserted by (target + sha)
  const row = db
    .select()
    .from(legalDocuments)
    .where(
      and(
        eq(legalDocuments.docSet, args.target.docSet),
        eq(legalDocuments.docType, args.target.docType),
        eq(legalDocuments.jurisdictionType, args.target.jurisdictionType),
        eq(legalDocuments.jurisdictionKey, args.target.jurisdictionKey),
        eq(legalDocuments.sha256, sha)
      )
    )
    .orderBy(desc(legalDocuments.createdAt))
    .limit(1)
    .get();

  if (!row) throw new Error("Failed to fetch inserted legal document.");
  return row;
}

export function upsertPointer(args: {
  target: DocTarget;
  activeDocumentId: number;
  adminUserId?: number | null;
}) {
  const now = Date.now();

  const existing = getPointer(args.target);
  if (!existing) {
    db.insert(legalDocPointers)
      .values({
        docSet: args.target.docSet,
        docType: args.target.docType,
        jurisdictionType: args.target.jurisdictionType,
        jurisdictionKey: args.target.jurisdictionKey,
        activeDocumentId: args.activeDocumentId,
        updatedAt: new Date(now),
        updatedByAdminUserId: args.adminUserId ?? null,
      })
      .run();
    return;
  }

  db.update(legalDocPointers)
    .set({
      activeDocumentId: args.activeDocumentId,
      updatedAt: new Date(now),
      updatedByAdminUserId: args.adminUserId ?? null,
    })
    .where(eq(legalDocPointers.id, existing.id))
    .run();
}
```

## 2.5 `/server/legal/legalDocChangeAuditService.ts` — tamper-evident audit chain for doc changes

```ts
// FILE: /server/legal/legalDocChangeAuditService.ts
import { desc, eq } from "drizzle-orm";
import { db } from "../db";
import { legalDocChangeAudit } from "../../shared/schema";
import { sha256Hex, stableStringify } from "./cryptoUtils";

export function appendLegalDocChangeAudit(args: {
  adminUserId?: number | null;
  action: "CREATE_VERSION" | "SET_ACTIVE" | "REPLACE_ACTIVE" | "ROLLBACK";
  docSet: string;
  docType: string;
  jurisdictionType: string;
  jurisdictionKey: string;
  oldActiveDocumentId?: number | null;
  newActiveDocumentId?: number | null;
  note?: string | null;
}) {
  return db.transaction((tx) => {
    const last = tx
      .select({ seq: legalDocChangeAudit.seq, eventHash: legalDocChangeAudit.eventHash })
      .from(legalDocChangeAudit)
      .orderBy(desc(legalDocChangeAudit.seq))
      .limit(1)
      .all();

    const prevSeq = last.length ? Number(last[0].seq) : 0;
    const prevHash = last.length ? String(last[0].eventHash) : "GENESIS";
    const nextSeq = prevSeq + 1;

    const payload = {
      seq: nextSeq,
      prevHash,

      adminUserId: args.adminUserId ?? null,
      action: args.action,

      docSet: args.docSet,
      docType: args.docType,
      jurisdictionType: args.jurisdictionType,
      jurisdictionKey: args.jurisdictionKey,

      oldActiveDocumentId: args.oldActiveDocumentId ?? null,
      newActiveDocumentId: args.newActiveDocumentId ?? null,

      note: args.note ?? null,
      createdAtMs: Date.now(),
    };

    const payloadStr = stableStringify(payload);
    const eventHash = sha256Hex(`${prevHash}|${payloadStr}`);

    tx.insert(legalDocChangeAudit)
      .values({
        seq: nextSeq,
        prevHash,
        eventHash,
        adminUserId: args.adminUserId ?? null,
        action: args.action,
        docSet: args.docSet,
        docType: args.docType,
        jurisdictionType: args.jurisdictionType,
        jurisdictionKey: args.jurisdictionKey,
        oldActiveDocumentId: args.oldActiveDocumentId ?? null,
        newActiveDocumentId: args.newActiveDocumentId ?? null,
        note: args.note ?? null,
        createdAt: new Date(payload.createdAtMs),
      })
      .run();

    return { seq: nextSeq, prevHash, eventHash };
  });
}
```

## 2.6 `/server/legal/termsEngineDb.ts` — assemble DOC1 using active pointers (DB-first)

```ts
// FILE: /server/legal/termsEngineDb.ts
import { REGION_RULES_IN_ORDER } from "./regionRules";
import { base64UrlEncode, base64UrlDecode, hmacSha256Hex, sha256Hex, timingSafeEqualHex } from "./cryptoUtils";
import { getActiveDoc } from "./legalDocsRepo";

export type AssembledTerms = {
  global: { id: string; version: string; sha256: string; documentId: number };
  addendum: { id: string; version: string; sha256: string; documentId: number; regionKey: string | null };
  combined: { sha256: string; text: string };
  meta: { docSet: "DOC1"; countryIso2: string; regionKey: string | null; issuedAtMs: number };
  token: string;
};

function normalizeIso2(iso2: string): string {
  const up = (iso2 || "").trim().toUpperCase();
  if (!/^[A-Z]{2}$/.test(up)) throw new Error("Invalid country ISO2.");
  return up;
}

function resolveRegionKey(iso2: string): string | null {
  for (const rule of REGION_RULES_IN_ORDER) {
    if (rule.iso2.includes(iso2)) return rule.regionKey;
  }
  return null;
}

function requireSecret(): string {
  const secret = process.env.LEGAL_TERMS_HMAC_SECRET;
  if (!secret || secret.length < 32) throw new Error("LEGAL_TERMS_HMAC_SECRET missing/too short (min 32 chars).");
  return secret;
}

export function assembleDoc1Terms(countryIso2Raw: string): AssembledTerms {
  const countryIso2 = normalizeIso2(countryIso2Raw);
  const regionKey = resolveRegionKey(countryIso2);

  // GLOBAL MASTER (DEFAULT/GLOBAL)
  const globalTarget = {
    docSet: "DOC1",
    docType: "GLOBAL_MASTER" as const,
    jurisdictionType: "DEFAULT" as const,
    jurisdictionKey: "GLOBAL",
  };

  const globalActive = getActiveDoc(globalTarget);
  if (!globalActive.doc) throw new Error("Missing active DOC1 global master (DEFAULT/GLOBAL).");

  // ADDENDUM selection priority:
  // 1) COUNTRY/{ISO2}
  // 2) REGION/{regionKey}
  // 3) DEFAULT/ROW
  const addCountryTarget = {
    docSet: "DOC1",
    docType: "ADDENDUM" as const,
    jurisdictionType: "COUNTRY" as const,
    jurisdictionKey: countryIso2,
  };
  const addRegionTarget = regionKey
    ? {
        docSet: "DOC1",
        docType: "ADDENDUM" as const,
        jurisdictionType: "REGION" as const,
        jurisdictionKey: regionKey,
      }
    : null;

  const addDefaultTarget = {
    docSet: "DOC1",
    docType: "ADDENDUM" as const,
    jurisdictionType: "DEFAULT" as const,
    jurisdictionKey: "ROW",
  };

  const addCountry = getActiveDoc(addCountryTarget);
  const addRegion = addRegionTarget ? getActiveDoc(addRegionTarget) : { doc: null as any, pointer: null as any };
  const addDefault = getActiveDoc(addDefaultTarget);

  const addDoc = addCountry.doc ?? addRegion.doc ?? addDefault.doc;
  const addRegionUsed = addCountry.doc ? null : addRegion.doc ? regionKey : null;

  if (!addDoc) throw new Error("Missing active DOC1 addendum (COUNTRY/REGION/DEFAULT).");

  const globalText = String(globalActive.doc.content);
  const addendumText = String(addDoc.content);

  // Strong invariant: stored sha must match actual content
  const globalSha = sha256Hex(globalText);
  const addSha = sha256Hex(addendumText);
  if (globalSha !== String(globalActive.doc.sha256)) throw new Error("Global master sha mismatch (DB content vs sha).");
  if (addSha !== String(addDoc.sha256)) throw new Error("Addendum sha mismatch (DB content vs sha).");

  const combinedText =
    globalText +
    "\n\n---\n\n" +
    "AUTO-SELECTED ADDENDUM:\n" +
    `Addendum ID: ${addDoc.jurisdictionType === "COUNTRY" ? `ADD-${countryIso2}` : String(addDoc.jurisdictionKey)}\n` +
    `Addendum Version: ${String(addDoc.version)}\n` +
    `Country Selected: ${countryIso2}\n` +
    (addRegionUsed ? `Region Key: ${addRegionUsed}\n` : "") +
    "\n" +
    addendumText;

  const combinedSha = sha256Hex(combinedText);
  const issuedAtMs = Date.now();

  const payloadObj = {
    v: 1,
    docSet: "DOC1",
    countryIso2,
    regionKey: addRegionUsed,
    global: { id: "DOC1_GLOBAL", version: String(globalActive.doc.version), sha256: globalSha, documentId: Number(globalActive.doc.id) },
    addendum: { id: "DOC1_ADDENDUM", version: String(addDoc.version), sha256: addSha, documentId: Number(addDoc.id) },
    combinedSha256: combinedSha,
    issuedAtMs,
  };

  const payloadJson = JSON.stringify(payloadObj);
  const payloadB64 = base64UrlEncode(payloadJson);
  const sig = hmacSha256Hex(requireSecret(), payloadB64);
  const token = `${payloadB64}.${sig}`;

  return {
    global: { id: "DOC1_GLOBAL", version: String(globalActive.doc.version), sha256: globalSha, documentId: Number(globalActive.doc.id) },
    addendum: { id: "DOC1_ADDENDUM", version: String(addDoc.version), sha256: addSha, documentId: Number(addDoc.id), regionKey: addRegionUsed },
    combined: { sha256: combinedSha, text: combinedText },
    meta: { docSet: "DOC1", countryIso2, regionKey: addRegionUsed, issuedAtMs },
    token,
  };
}

export function verifyTermsToken(token: string, maxAgeMs = 24 * 60 * 60 * 1000) {
  try {
    const parts = String(token || "").split(".");
    if (parts.length !== 2) return { ok: false as const, reason: "Bad token format." };
    const [payloadB64, sig] = parts;

    const expected = hmacSha256Hex(requireSecret(), payloadB64);
    if (!timingSafeEqualHex(sig, expected)) return { ok: false as const, reason: "Bad token signature." };

    const payloadJson = base64UrlDecode(payloadB64);
    const payload = JSON.parse(payloadJson);

    const age = Date.now() - Number(payload?.issuedAtMs || 0);
    if (age < 0 || age > maxAgeMs) return { ok: false as const, reason: "Token expired." };

    return { ok: true as const, payload };
  } catch (e: any) {
    return { ok: false as const, reason: e?.message || "Token verify error." };
  }
}
```

# 3) Bootstrap: import your existing text files into DB once (optional but recommended)

If you already have your word-for-word text pack in the repo, this seeds `legalDocuments` + pointers so the system runs immediately.

## 3.1 `/server/legal/bootstrapDoc1FromFiles.ts`

```ts
// FILE: /server/legal/bootstrapDoc1FromFiles.ts
import fs from "fs";
import path from "path";
import { db } from "../db";
import { legalDocuments, legalDocPointers } from "../../shared/schema";
import { and, eq, desc } from "drizzle-orm";
import { sha256Hex } from "./cryptoUtils";

function read(rel: string): string {
  return fs.readFileSync(path.resolve(process.cwd(), rel), "utf8");
}

async function ensureDocAndPointer(args: {
  target: { docSet: string; docType: string; jurisdictionType: string; jurisdictionKey: string };
  version: string;
  filePath: string;
}) {
  const content = read(args.filePath);
  const sha = sha256Hex(content);
  const now = Date.now();

  const existing = db
    .select()
    .from(legalDocuments)
    .where(
      and(
        eq(legalDocuments.docSet, args.target.docSet),
        eq(legalDocuments.docType, args.target.docType),
        eq(legalDocuments.jurisdictionType, args.target.jurisdictionType),
        eq(legalDocuments.jurisdictionKey, args.target.jurisdictionKey),
        eq(legalDocuments.sha256, sha)
      )
    )
    .get();

  let docId: number;
  if (existing) {
    docId = Number(existing.id);
  } else {
    db.insert(legalDocuments)
      .values({
        docSet: args.target.docSet,
        docType: args.target.docType,
        jurisdictionType: args.target.jurisdictionType,
        jurisdictionKey: args.target.jurisdictionKey,
        version: args.version,
        sha256: sha,
        content,
        notes: "BOOTSTRAP_IMPORT",
        createdAt: new Date(now),
        createdByAdminUserId: null,
      })
      .run();

    const inserted = db
      .select()
      .from(legalDocuments)
      .where(
        and(
          eq(legalDocuments.docSet, args.target.docSet),
          eq(legalDocuments.docType, args.target.docType),
          eq(legalDocuments.jurisdictionType, args.target.jurisdictionType),
          eq(legalDocuments.jurisdictionKey, args.target.jurisdictionKey),
          eq(legalDocuments.sha256, sha)
        )
      )
      .orderBy(desc(legalDocuments.createdAt))
      .limit(1)
      .get();

    if (!inserted) throw new Error("Bootstrap insert failed.");
    docId = Number(inserted.id);
  }

  const pointer = db
    .select()
    .from(legalDocPointers)
    .where(
      and(
        eq(legalDocPointers.docSet, args.target.docSet),
        eq(legalDocPointers.docType, args.target.docType),
        eq(legalDocPointers.jurisdictionType, args.target.jurisdictionType),
        eq(legalDocPointers.jurisdictionKey, args.target.jurisdictionKey)
      )
    )
    .get();

  if (!pointer) {
    db.insert(legalDocPointers)
      .values({
        docSet: args.target.docSet,
        docType: args.target.docType,
        jurisdictionType: args.target.jurisdictionType,
        jurisdictionKey: args.target.jurisdictionKey,
        activeDocumentId: docId,
        updatedAt: new Date(now),
        updatedByAdminUserId: null,
      })
      .run();
  }
}

export async function bootstrapDoc1FromFilesIfEmpty() {
  const any = db.select().from(legalDocuments).limit(1).get();
  if (any) return;

  // Adjust these file paths to match where you store the text pack in repo:
  await ensureDocAndPointer({
    target: { docSet: "DOC1", docType: "GLOBAL_MASTER", jurisdictionType: "DEFAULT", jurisdictionKey: "GLOBAL" },
    version: "1.0",
    filePath: "server/legal/doc1/DOC1_GLOBAL_v1.0.txt",
  });

  // Default addendum ROW
  await ensureDocAndPointer({
    target: { docSet: "DOC1", docType: "ADDENDUM", jurisdictionType: "DEFAULT", jurisdictionKey: "ROW" },
    version: "1.0",
    filePath: "server/legal/doc1/addenda/ADD-ROW_v1.0.txt",
  });

  // Countries (repeat as needed)
  const countries = ["US","AE","IN","KE","NG","BD","TH","PH","SG","HK","CN","BR","AR","CO","MX","ET","GH","AO","ZA","UG","TZ","RU","UA","GB"];
  for (const iso2 of countries) {
    const fp = `server/legal/doc1/addenda/ADD-${iso2}_v1.0.txt`;
    if (fs.existsSync(path.resolve(process.cwd(), fp))) {
      await ensureDocAndPointer({
        target: { docSet: "DOC1", docType: "ADDENDUM", jurisdictionType: "COUNTRY", jurisdictionKey: iso2 },
        version: "1.0",
        filePath: fp,
      });
    }
  }

  // Regions (repeat as needed)
  const regions = ["WEST_EUROPE","EAST_EUROPE","SE_ASIA","MIDDLE_EAST","NORTH_AFRICA","WEST_AFRICA","EAST_AFRICA","SOUTHERN_AFRICA","CENTRAL_AFRICA","SOUTH_AMERICA","CENTRAL_AMERICA","NORTH_AMERICA","CENTRAL_ASIA","JAPAN","SOUTH_KOREA","PAKISTAN","AUSTRALASIA","PACIFIC_ISLANDS","INDIAN_OCEAN_ISLANDS"];
  for (const rk of regions) {
    const fp = `server/legal/doc1/addenda/ADD-${rk}_v1.0.txt`;
    if (fs.existsSync(path.resolve(process.cwd(), fp))) {
      await ensureDocAndPointer({
        target: { docSet: "DOC1", docType: "ADDENDUM", jurisdictionType: "REGION", jurisdictionKey: rk },
        version: "1.0",
        filePath: fp,
      });
    }
  }
}
```

```tsx
            <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 10 }}>

                <input value={editorVersion} onChange={(e) => setEditorVersion(e.target.value)} style={input} />
              </Field>
              <Field label="Note (optional)">
                <input value={editorNote} onChange={(e) => setEditorNote(e.target.value)} style={input} />
              </Field>
            </div>

            <Field label="New Content">
              <textarea value={editorContent} onChange={(e) => setEditorContent(e.target.value)} style={{ ...textarea, height: 220 }} />
            </Field>

            <div style={{ display: "flex", gap: 10 }}>
              <button style={primaryBtn} onClick={onReplaceActive}>
                Replace Active
              </button>
            </div>
          </div>
        </div>
      </div>

      {/* Assemble preview */}
      <div style={card}>
        <div style={{ fontWeight: 700, marginBottom: 8 }}>Assemble Preview (QA what signup will load)</div>
        <div style={{ display: "flex", gap: 10, alignItems: "center" }}>
          <Field label="Country ISO2">
            <input value={previewCountry} onChange={(e) => setPreviewCountry(e.target.value.toUpperCase())} style={input} />
          </Field>
          <button style={primaryBtn} onClick={onPreviewAssemble}>Preview Assemble</button>
        </div>

        {preview?.ok && (
          <div style={{ marginTop: 10, display: "grid", gap: 6, fontSize: 12 }}>
            <div>Combined SHA256: <span style={{ fontFamily: "monospace" }}>{preview.combinedSha256}</span></div>
            <div>Global Doc ID: {preview.global?.documentId} (v{preview.global?.version})</div>
            <div>Addendum Doc ID: {preview.addendum?.documentId} (v{preview.addendum?.version})</div>
            <div>Region Used: {preview.regionKey ?? "none"}</div>
            <details style={{ marginTop: 8 }}>
              <summary style={{ cursor: "pointer" }}>Show Rendered Text</summary>
              <pre style={{ whiteSpace: "pre-wrap", marginTop: 8, border: "1px solid rgba(0,0,0,0.12)", padding: 10, borderRadius: 10 }}>
                {preview.text}
              </pre>
            </details>
          </div>
        )}
      </div>
    </div>
  );
}

function Field(props: { label: string; children: React.ReactNode }) {
```
