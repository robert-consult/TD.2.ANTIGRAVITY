import db from '../db.js';

export async function generateTraderInsights(userId) {
  const trades = await db.query(
    `
    SELECT instrument, direction,
           profit,
           EXTRACT(EPOCH FROM (exit_time - entry_time))/60 AS hold_minutes,
           EXTRACT(HOUR FROM entry_time) AS hour
    FROM trades
    WHERE user_id = $1
    `,
    [userId]
  );

  const insights = {
    byInstrument: {},
    byDirection: {},
    byHour: {},
    holdTimes: []
  };

  trades.rows.forEach(t => {
    insights.byInstrument[t.instrument] ??= { pnl: 0, count: 0 };
    insights.byInstrument[t.instrument].pnl += t.profit;
    insights.byInstrument[t.instrument].count++;

    insights.byDirection[t.direction] ??= { pnl: 0, count: 0 };
    insights.byDirection[t.direction].pnl += t.profit;
    insights.byDirection[t.direction].count++;

    insights.byHour[t.hour] ??= { pnl: 0, count: 0 };
    insights.byHour[t.hour].pnl += t.profit;
    insights.byHour[t.hour].count++;

    insights.holdTimes.push(t.hold_minutes);
  });

  return insights;
}

/**
 * GET /admin/users/search?q=...
 * Returns up to 20 matching users by:
 * - numeric id exact match
 * - username ILIKE
 * - email ILIKE
 */
router.get("/users/search", adminGuard, async (req, res) => {
  const qRaw = String(req.query.q || "").trim();
  if (!qRaw) return res.json([]);

  const isNumeric = /^[0-9]+$/.test(qRaw);
  const q = `%${qRaw}%`;

  const result = await db.query(
    `
    SELECT id, username, email, role, status
    FROM users
    WHERE
      (${isNumeric} AND id = $1)
      OR (username ILIKE $2)
      OR (email ILIKE $2)
    ORDER BY id DESC
    LIMIT 20
    `,
    [isNumeric ? Number(qRaw) : -1, q]
  );

  res.json(result.rows);
});

import db from "../db.js";

function bucketHoldMinutes(mins) {
  if (mins == null || Number.isNaN(mins)) return "unknown";
  if (mins < 1) return "<1m";
  if (mins < 5) return "1-5m";
  if (mins < 15) return "5-15m";
  if (mins < 30) return "15-30m";
  if (mins < 60) return "30-60m";
  if (mins < 180) return "1-3h";
  return "3h+";
}

export async function computeInsights({ userId, sinceDays = 30 }) {
  // Filter window; you can change to 90d or all-time by passing sinceDays=null
  const whereTime = sinceDays ? `AND entry_time >= NOW() - INTERVAL '${sinceDays} days'` : "";

  // Pull only columns needed
  const r = await db.query(
    `
    SELECT instrument, direction, profit,
           EXTRACT(EPOCH FROM (exit_time - entry_time))/60 AS hold_minutes,
           EXTRACT(HOUR FROM entry_time) AS entry_hour
    FROM trades
    WHERE user_id = $1
      AND exit_time IS NOT NULL
      AND entry_time IS NOT NULL
      ${whereTime}
    `,
    [userId]
  );

  const byInstrument = {};
  const byDirection = {};
  const byHour = {};
  const byHoldBucket = {};

  let totalTrades = 0;
  let wins = 0;
  let totalPnl = 0;

  for (const t of r.rows) {
    totalTrades++;
    totalPnl += Number(t.profit || 0);
    if (Number(t.profit || 0) > 0) wins++;

    const inst = t.instrument || "UNKNOWN";
    const dir = t.direction || "UNKNOWN";
    const hour = t.entry_hour != null ? String(t.entry_hour).padStart(2, "0") : "NA";
    const bucket = bucketHoldMinutes(Number(t.hold_minutes));

    byInstrument[inst] ??= { pnl: 0, trades: 0, wins: 0 };
    byInstrument[inst].pnl += Number(t.profit || 0);
    byInstrument[inst].trades++;
    if (Number(t.profit || 0) > 0) byInstrument[inst].wins++;

    byDirection[dir] ??= { pnl: 0, trades: 0, wins: 0 };
    byDirection[dir].pnl += Number(t.profit || 0);
    byDirection[dir].trades++;
    if (Number(t.profit || 0) > 0) byDirection[dir].wins++;

    byHour[hour] ??= { pnl: 0, trades: 0, wins: 0 };
    byHour[hour].pnl += Number(t.profit || 0);
    byHour[hour].trades++;
    if (Number(t.profit || 0) > 0) byHour[hour].wins++;

    byHoldBucket[bucket] ??= { pnl: 0, trades: 0, wins: 0 };
    byHoldBucket[bucket].pnl += Number(t.profit || 0);
    byHoldBucket[bucket].trades++;
    if (Number(t.profit || 0) > 0) byHoldBucket[bucket].wins++;
  }

  function bestWorst(mapObj) {
    const entries = Object.entries(mapObj).map(([k, v]) => ({
      key: k,
      pnl: v.pnl,
      trades: v.trades,
      win_rate: v.trades ? v.wins / v.trades : 0
    }));
    entries.sort((a, b) => b.pnl - a.pnl);
    return {
      best: entries[0] || null,
      worst: entries.length ? entries[entries.length - 1] : null,
      ranked: entries
    };
  }

  const inst = bestWorst(byInstrument);
  const dir = bestWorst(byDirection);
  const hour = bestWorst(byHour);
  const hold = bestWorst(byHoldBucket);

  return {
    window_days: sinceDays,
    totals: {
      trades: totalTrades,
      pnl: totalPnl,
      win_rate: totalTrades ? wins / totalTrades : 0
    },
    instruments: inst,
    directions: dir,
    hours: hour,
    hold_time: hold
  };
}

const CACHE_TTL_MINUTES = 12 * 60; // 12 hours

router.get("/insights", async (req, res) => {
  const sinceDays = req.query.since_days ? Number(req.query.since_days) : 30;
  const refresh = String(req.query.refresh || "0") === "1";

  const cached = await db.query(
    `SELECT insights, generated_at FROM trader_insights_cache WHERE user_id = $1`,
    [req.user.id]
  );

  if (!refresh && cached.rows.length) {
    const generatedAt = new Date(cached.rows[0].generated_at).getTime();
    const ageMinutes = (Date.now() - generatedAt) / (1000 * 60);

    // Cache can store multiple windows; simplest is store one blob with window in it.
    // If window mismatch, recompute.
    const blob = cached.rows[0].insights;
    if (ageMinutes < CACHE_TTL_MINUTES && blob?.window_days === sinceDays) {
      return res.json(blob);
    }
  }

  const insights = await computeInsights({ userId: req.user.id, sinceDays });

  await db.query(
    `
    INSERT INTO trader_insights_cache (user_id, insights, generated_at)
    VALUES ($1, $2, NOW())
    ON CONFLICT (user_id)
    DO UPDATE SET insights = $2, generated_at = NOW()
    `,
    [req.user.id, JSON.stringify(insights)]
  );

  res.json(insights);
});

    `

    FROM users
    WHERE (${isNum} AND id = $1)
       OR (username ILIKE $2)
       OR (email ILIKE $2)
    ORDER BY id DESC
    LIMIT 20
    `,
    [isNum ? Number(qRaw) : -1, q]
  );

  res.json(rows);
});

import db from "../db";

function bucketHoldMinutes(mins?: number | null): string {
  const m = typeof mins === "number" ? mins : NaN;
  if (Number.isNaN(m)) return "unknown";
  if (m < 1) return "<1m";
  if (m < 5) return "1-5m";
  if (m < 15) return "5-15m";
  if (m < 30) return "15-30m";
  if (m < 60) return "30-60m";
  if (m < 180) return "1-3h";
  return "3h+";
}

type Agg = { pnl: number; trades: number; wins: number };
type Ranked = { key: string; pnl: number; trades: number; win_rate: number };

function bestWorst(map: Record<string, Agg>) {
  const arr: Ranked[] = Object.entries(map).map(([k, v]) => ({
    key: k,
    pnl: v.pnl,
    trades: v.trades,
    win_rate: v.trades ? v.wins / v.trades : 0
  }));
  arr.sort((a, b) => b.pnl - a.pnl);
  return { best: arr[0] || null, worst: arr.length ? arr[arr.length - 1] : null, ranked: arr };
}

export async function computeInsights(userId: number, sinceDays = 30) {
  const whereTime = sinceDays ? `AND entry_time >= NOW() - INTERVAL '${sinceDays} days'` : "";
  const q = await db.query(
    `
    SELECT instrument, direction, profit,
           EXTRACT(EPOCH FROM (exit_time - entry_time))/60 AS hold_minutes,
           EXTRACT(HOUR FROM entry_time) AS entry_hour
    FROM trades
    WHERE user_id=$1
      AND exit_time IS NOT NULL
      AND entry_time IS NOT NULL
      ${whereTime}
    `,
    [userId]
  );

  const byInstrument: Record<string, Agg> = {};
  const byDirection: Record<string, Agg> = {};
  const byHour: Record<string, Agg> = {};
  const byHold: Record<string, Agg> = {};
  let trades = 0, wins = 0, pnl = 0;

  for (const t of q.rows as any[]) {
    const inst = t.instrument || "UNKNOWN";
    const dir = t.direction || "UNKNOWN";
    const hour = t.entry_hour != null ? String(t.entry_hour).padStart(2, "0") : "NA";
    const bucket = bucketHoldMinutes(typeof t.hold_minutes === "number" ? t.hold_minutes : Number(t.hold_minutes));

    const p = Number(t.profit || 0);
    trades++; pnl += p; if (p > 0) wins++;

    byInstrument[inst] ??= { pnl: 0, trades: 0, wins: 0 };
    byInstrument[inst].pnl += p; byInstrument[inst].trades++; if (p > 0) byInstrument[inst].wins++;

    byDirection[dir] ??= { pnl: 0, trades: 0, wins: 0 };
    byDirection[dir].pnl += p; byDirection[dir].trades++; if (p > 0) byDirection[dir].wins++;

    byHour[hour] ??= { pnl: 0, trades: 0, wins: 0 };
    byHour[hour].pnl += p; byHour[hour].trades++; if (p > 0) byHour[hour].wins++;

    byHold[bucket] ??= { pnl: 0, trades: 0, wins: 0 };
    byHold[bucket].pnl += p; byHold[bucket].trades++; if (p > 0) byHold[bucket].wins++;
  }

  return {
    window_days: sinceDays,
    totals: { trades, pnl, win_rate: trades ? wins / trades : 0 },
    instruments: bestWorst(byInstrument),
    directions: bestWorst(byDirection),
    hours: bestWorst(byHour),
    hold_time: bestWorst(byHold)
  };
}

export type CsvHeader = { key: string; label: string };

function escapeCsvValue(v: unknown): string {
  if (v === null || v === undefined) return "";
  const s = typeof v === "string" ? v : JSON.stringify(v);
  // Escape quotes by doubling; wrap if contains delimiter/newline/quote
  const mustWrap = /[",\n\r]/.test(s);
  const escaped = s.replace(/"/g, '""');
  return mustWrap ? `"${escaped}"` : escaped;
}

export function toCsv(headers: CsvHeader[], rows: Record<string, any>[]): string {
  const headLine = headers.map(h => escapeCsvValue(h.label)).join(",");
  const lines = rows.map(r => headers.map(h => escapeCsvValue(r[h.key])).join(","));
  return [headLine, ...lines].join("\n") + "\n";
}

export function sendCsv(res: any, filename: string, csv: string) {
  res.setHeader("Content-Type", "text/csv; charset=utf-8");
  res.setHeader("Content-Disposition", `attachment; filename="${filename}"`);
  res.status(200).send(csv);
}

    return await db.query<T>(sql, params);

  await logAdminAction({

import api from "./api";

export async function downloadWithAuth(url: string, filename: string) {
  const res = await api.get(url, { responseType: "blob" });
  const blob = new Blob([res.data], { type: "text/csv;charset=utf-8" });

  const a = document.createElement("a");
  const objectUrl = window.URL.createObjectURL(blob);
  a.href = objectUrl;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  window.URL.revokeObjectURL(objectUrl);
}

This implementation:

Attempts to read sessions from several likely tables (user_sessions, sessions, auth_sessions)

If those don’t exist, it falls back to pairing LOGIN/LOGOUT events from common “events” tables (best-effort)

Normalizes output into a single shape:

login_time, logout_time, login_ip, user_agent, session_length_seconds

1) Backend — New Service: Session Fetch + Normalize
server/services/sessions.ts
import { tryQuery } from "../utils/tryQuery";

export type NormalizedSession = {
  session_id: string | number;
  login_time: string | null;
  logout_time: string | null;
  login_ip: string | null;
  user_agent: string | null;
  session_length_seconds: number | null;
};

function toIso(v: any): string | null {
  if (!v) return null;
  const d = new Date(v);
  return Number.isNaN(d.getTime()) ? null : d.toISOString();
}

function lenSeconds(login: any, logout: any): number | null {
  const a = login ? new Date(login).getTime() : NaN;
  const b = logout ? new Date(logout).getTime() : NaN;
  if (Number.isNaN(a) || Number.isNaN(b) || b < a) return null;
  return Math.floor((b - a) / 1000);
}

/**
 * Attempts to load session history from:
 *  1) user_sessions
 *  2) sessions
 *  3) auth_sessions
 *  4) (fallback) event pairing from auth/user events tables
 *
 * Returns { source, sessions }
 */
export async function getUserSessions(userId: number, sinceDays = 30): Promise<{
  source: string;
  sessions: NormalizedSession[];
}> {
  const whereSince = sinceDays ? `NOW() - INTERVAL '${sinceDays} days'` : null;

  // 1) user_sessions table (common naming)
  {
    const r = await tryQuery<any>(
      `
      SELECT
        id,
        user_id,
        login_time,
        logout_time,
        login_ip,
        user_agent
      FROM user_sessions
      WHERE user_id = $1
        ${whereSince ? `AND login_time >= ${whereSince}` : ""}
      ORDER BY login_time DESC
      LIMIT 500
      `,
      [userId]
    );
    if (r?.rows?.length) {
      return {
        source: "user_sessions",
        sessions: r.rows.map((x) => ({
          session_id: x.id,
          login_time: toIso(x.login_time),
          logout_time: toIso(x.logout_time),
          login_ip: x.login_ip ?? null,
          user_agent: x.user_agent ?? null,
          session_length_seconds: lenSeconds(x.login_time, x.logout_time)
        }))
      };
    }
  }

  // 2) sessions table (common naming)
  {
    const r = await tryQuery<any>(
      `
      SELECT
        id,
        user_id,
        created_at,
        ended_at,
        ip,
        user_agent
      FROM sessions
      WHERE user_id = $1
        ${whereSince ? `AND created_at >= ${whereSince}` : ""}
      ORDER BY created_at DESC
      LIMIT 500
      `,
      [userId]
    );
    if (r?.rows?.length) {
      return {
        source: "sessions",
        sessions: r.rows.map((x) => ({
          session_id: x.id,
          login_time: toIso(x.created_at),
          logout_time: toIso(x.ended_at),
          login_ip: x.ip ?? null,
          user_agent: x.user_agent ?? null,
          session_length_seconds: lenSeconds(x.created_at, x.ended_at)
        }))
      };
    }
  }

  // 3) auth_sessions table (alternate naming)
  {
    const r = await tryQuery<any>(
      `
      SELECT
        session_id,
        user_id,
        started_at,
        ended_at,
        ip_address,
        user_agent
      FROM auth_sessions
      WHERE user_id = $1
        ${whereSince ? `AND started_at >= ${whereSince}` : ""}
      ORDER BY started_at DESC
      LIMIT 500
      `,
      [userId]
    );
    if (r?.rows?.length) {
      return {
        source: "auth_sessions",
        sessions: r.rows.map((x) => ({
          session_id: x.session_id,
          login_time: toIso(x.started_at),
          logout_time: toIso(x.ended_at),
          login_ip: x.ip_address ?? null,
          user_agent: x.user_agent ?? null,
          session_length_seconds: lenSeconds(x.started_at, x.ended_at)
        }))
      };
    }
  }

  // 4) Fallback: pair login/logout events from common event tables
  // Try a few likely schemas. If found, pair sequentially (LOGIN -> LOGOUT).
  const eventCandidates: Array<{
    table: string;
    timeCol: string;
    typeCol: string;
    ipCol?: string;
    uaCol?: string;
    loginValues: string[];
    logoutValues: string[];
  }> = [
    {
      table: "auth_events",
      timeCol: "created_at",
      typeCol: "event_type",
      ipCol: "ip",
      uaCol: "user_agent",
      loginValues: ["LOGIN", "login"],
      logoutValues: ["LOGOUT", "logout"]
    },
    {
      table: "user_events",
      timeCol: "created_at",
      typeCol: "event_type",
      ipCol: "ip",
      uaCol: "user_agent",
      loginValues: ["LOGIN", "login"],
      logoutValues: ["LOGOUT", "logout"]
    },
    {
      table: "user_activity",
      timeCol: "created_at",
      typeCol: "action",
      ipCol: "ip",
      uaCol: "user_agent",
      loginValues: ["LOGIN", "login"],
      logoutValues: ["LOGOUT", "logout"]
    }
  ];

  for (const c of eventCandidates) {
    const r = await tryQuery<any>(
      `
      SELECT
        id,
        user_id,
        ${c.timeCol} AS t,
        ${c.typeCol} AS typ
        ${c.ipCol ? `, ${c.ipCol} AS ip` : ""}
        ${c.uaCol ? `, ${c.uaCol} AS ua` : ""}
      FROM ${c.table}
      WHERE user_id = $1
        ${whereSince ? `AND ${c.timeCol} >= ${whereSince}` : ""}
      ORDER BY ${c.timeCol} ASC
      LIMIT 2000
      `,
      [userId]
    );

    if (r?.rows?.length) {
      const events = r.rows.map((e: any) => ({
        id: e.id,
        t: e.t,
        typ: String(e.typ || ""),
        ip: e.ip ?? null,
        ua: e.ua ?? null
      }));

      // Pair: take the last seen LOGIN, close it on next LOGOUT
      const sessions: NormalizedSession[] = [];
      let open: { login_time: any; login_ip: any; user_agent: any; event_id: any } | null = null;

      const isLogin = (typ: string) => c.loginValues.includes(typ);
      const isLogout = (typ: string) => c.logoutValues.includes(typ);

      for (const e of events) {
        if (isLogin(e.typ)) {
          // If already open, close it as "unknown logout" and start new
          if (open) {
            sessions.push({
              session_id: `eventpair_${open.event_id}_abandoned`,
              login_time: toIso(open.login_time),
              logout_time: null,
              login_ip: open.login_ip ?? null,
              user_agent: open.user_agent ?? null,
              session_length_seconds: null
            });
          }
          open = { login_time: e.t, login_ip: e.ip, user_agent: e.ua, event_id: e.id };
          continue;
        }
        if (isLogout(e.typ) && open) {
          sessions.push({
            session_id: `eventpair_${open.event_id}_${e.id}`,
            login_time: toIso(open.login_time),
            logout_time: toIso(e.t),
            login_ip: open.login_ip ?? null,
            user_agent: open.user_agent ?? null,
            session_length_seconds: lenSeconds(open.login_time, e.t)
          });
          open = null;
        }
      }

      // leftover open login
      if (open) {
        sessions.push({
          session_id: `eventpair_${open.event_id}_open`,
          login_time: toIso(open.login_time),
          logout_time: null,
          login_ip: open.login_ip ?? null,
          user_agent: open.user_agent ?? null,
          session_length_seconds: null
        });
      }

      sessions.sort((a, b) => {
        const ta = a.login_time ? new Date(a.login_time).getTime() : 0;
        const tb = b.login_time ? new Date(b.login_time).getTime() : 0;
        return tb - ta;
      });

      return { source: `events:${c.table}`, sessions: sessions.slice(0, 500) };
    }
  }

  return { source: "none", sessions: [] };
}

export type CsvHeader = { key: string; label: string };

function preventCsvFormulaInjection(s: string): string {
  // Excel/Sheets formula injection protection
  // If begins with = + - @, prefix with apostrophe
  return /^[=+\-@]/.test(s) ? `'${s}` : s;
}

function escapeCsvValue(v: unknown): string {
  if (v === null || v === undefined) return "";
  let s = typeof v === "string" ? v : JSON.stringify(v);

  s = preventCsvFormulaInjection(s);

  const mustWrap = /[",\n\r]/.test(s);
  const escaped = s.replace(/"/g, '""');
  return mustWrap ? `"${escaped}"` : escaped;
}

export function toCsv(headers: CsvHeader[], rows: Record<string, any>[]): string {
  const headLine = headers.map(h => escapeCsvValue(h.label)).join(",");
  const lines = rows.map(r => headers.map(h => escapeCsvValue(r[h.key])).join(","));
  return [headLine, ...lines].join("\n") + "\n";
}

export function sendCsv(res: any, filename: string, csv: string) {
  res.setHeader("Content-Type", "text/csv; charset=utf-8");
  res.setHeader("Content-Disposition", `attachment; filename="${filename}"`);
  res.status(200).send(csv);
}
