File: client/src/lib/perfProfile.ts
Purpose:

export type NetEffectiveType = "slow-2g" | "2g" | "3g" | "4g" | "unknown";

export type PerfProfile = {
  effectiveType: NetEffectiveType;
  saveData: boolean;
  deviceMemoryGB: number | null;
  hardwareConcurrency: number | null;
  rttMs: number | null;
  downlinkMbps: number | null;
  isConstrained: boolean;
};

function numOrNull(value: unknown): number | null {
  const n = Number(value);
  return Number.isFinite(n) ? n : null;
}

function clamp(n: number, lo: number, hi: number) {
  return Math.max(lo, Math.min(hi, n));
}

export function getPerfProfile(): PerfProfile {
  const nav: any = typeof navigator !== "undefined" ? navigator : undefined;
  const conn: any = nav?.connection || nav?.mozConnection || nav?.webkitConnection;

  const effectiveType = (conn?.effectiveType as NetEffectiveType) || "unknown";
  const saveData = Boolean(conn?.saveData);
  const rttMs = numOrNull(conn?.rtt);
  const downlinkMbps = numOrNull(conn?.downlink);

  const deviceMemoryGB = numOrNull((nav as any)?.deviceMemory);
  const hardwareConcurrency = numOrNull((nav as any)?.hardwareConcurrency);

  const networkConstrained =
    saveData ||
    effectiveType === "slow-2g" ||
    effectiveType === "2g" ||
    effectiveType === "3g" ||
    (effectiveType === "4g" &&
      ((rttMs != null && rttMs > 300) || (downlinkMbps != null && downlinkMbps < 1.5)));

  const deviceConstrained =
    (deviceMemoryGB != null && deviceMemoryGB <= 4) ||
    (hardwareConcurrency != null && hardwareConcurrency <= 4);

  return {
    effectiveType,
    saveData,
    deviceMemoryGB,
    hardwareConcurrency,
    rttMs,
    downlinkMbps,
    isConstrained: networkConstrained || deviceConstrained,
  };
}

export function recommendedPollIntervalMs(baseMs: number, profile: PerfProfile = getPerfProfile()): number {
  const base = Math.max(250, baseMs);
  if (!profile.isConstrained) return base;

  if (profile.effectiveType === "4g") return clamp(Math.round(base * 2), base, 30000);
  return clamp(Math.round(base * 3), base, 60000);
}

export function recommendedWsReconnectIntervalMs(baseMs: number, profile: PerfProfile = getPerfProfile()): number {
  const base = Math.max(500, baseMs);
  if (!profile.isConstrained) return base;

  if (profile.effectiveType === "4g") return clamp(Math.round(base * 2), base, 20000);
  return clamp(Math.round(base * 3), base, 30000);
}

import { useQuery } from "@tanstack/react-query";
import { useLiveUpdates } from "@/live/LiveUpdatesProvider";
import { getPerfProfile, recommendedPollIntervalMs } from "@/lib/perfProfile";

export interface Quote {
  symbol: string;
  name: string;
  price: number;
  change: number;
  time: number;
  bid?: number;
  ask?: number;
  spread?: number;
  prevClose?: number;
  changePct?: number;
  percent_change?: number;
  isStale?: boolean;
  lastApiUpdate?: number;
  dataAge?: number;
}

interface SymbolConfig {
  id: number;
  symbol: string;
  name: string;
  enabled?: boolean;
}

const FALLBACK_POLL_INTERVAL_BASE = 5000;

function toNumber(value: any): number | null {
  if (value === null || value === undefined) return null;
  const n = Number(value);
  return Number.isFinite(n) ? n : null;
}

function calculatePctChange(current: number | null, previous: number | null): number {
  if (!current || !previous || previous === 0) return 0;
  return ((current - previous) / previous) * 100;
}

export function useQuotes(options?: { enabled?: boolean; throttleMs?: number }) {
  const enabled = options?.enabled ?? true;

  const [quotes, setQuotes] = useState<Quote[]>([]);
  const [isConnected, setIsConnected] = useState(false);
  const quotesRef = useRef<Map<string, Quote>>(new Map());
  const symbolNameMapRef = useRef<Map<string, string>>(new Map());
  const prevSymbolsRef = useRef<string[]>([]);

  const perfProfile = useMemo(() => getPerfProfile(), []);
  const fallbackInterval = useMemo(
    () => recommendedPollIntervalMs(FALLBACK_POLL_INTERVAL_BASE, perfProfile),
    [perfProfile],
  );

  const throttleMs = options?.throttleMs ?? (perfProfile.isConstrained ? 300 : 0);
  const lastPublishRef = useRef(0);
  const publishTimerRef = useRef<number | null>(null);

  const { isConnected: isWsConnected, sendMessage, subscribe } = useLiveUpdates();

  const { data: symbols = [] } = useQuery<SymbolConfig[]>({
    queryKey: ["/api/config/symbols"],
    enabled,
  });

  useEffect(() => {
    const map = new Map<string, string>();
    (symbols || []).forEach((s) => {
      if (s.enabled !== false) map.set(String(s.symbol), s.name);
    });
    symbolNameMapRef.current = map;
  }, [symbols]);

  const requestedSymbols = useMemo(() => {
    if (!enabled) return [];
    return (symbols || [])
      .filter((s) => s.enabled !== false)
      .map((s) => String(s.symbol).toUpperCase());
  }, [enabled, symbols]);

  useEffect(() => {
    if (enabled) return;
    prevSymbolsRef.current = [];
    quotesRef.current = new Map();
    setQuotes([]);
    setIsConnected(false);
  }, [enabled]);

  useEffect(() => {
    if (!enabled || !isWsConnected) {
      prevSymbolsRef.current = [];
      return;
    }

    if (!requestedSymbols.length) {
      if (prevSymbolsRef.current.length) {
        sendMessage({ type: "quotes:unsubscribe", symbols: prevSymbolsRef.current });
        prevSymbolsRef.current = [];
      }
      return;
    }

    const prev = new Set(prevSymbolsRef.current);
    const next = new Set(requestedSymbols);

    const subscribeSymbols = [...next].filter((s) => !prev.has(s));
    const unsubscribeSymbols = [...prev].filter((s) => !next.has(s));

    if (subscribeSymbols.length) sendMessage({ type: "quotes:subscribe", symbols: subscribeSymbols });
    if (unsubscribeSymbols.length) sendMessage({ type: "quotes:unsubscribe", symbols: unsubscribeSymbols });

    prevSymbolsRef.current = requestedSymbols;

    return () => {
      if (requestedSymbols.length && isWsConnected) {
        sendMessage({ type: "quotes:unsubscribe", symbols: requestedSymbols });
      }
    };
  }, [enabled, isWsConnected, requestedSymbols.join("|"), sendMessage]);

  const schedulePublish = useCallback(() => {
    if (throttleMs <= 0) {
      setQuotes(Array.from(quotesRef.current.values()));
      return;
    }

    const now = Date.now();
    const elapsed = now - lastPublishRef.current;
    if (elapsed >= throttleMs) {
      lastPublishRef.current = now;
      setQuotes(Array.from(quotesRef.current.values()));
      return;
    }

    if (publishTimerRef.current != null) return;

    publishTimerRef.current = window.setTimeout(() => {
      publishTimerRef.current = null;
      lastPublishRef.current = Date.now();
      setQuotes(Array.from(quotesRef.current.values()));
    }, Math.max(0, throttleMs - elapsed));
  }, [throttleMs]);

  useEffect(() => {
    return () => {
      if (publishTimerRef.current != null) {
        window.clearTimeout(publishTimerRef.current);
        publishTimerRef.current = null;
      }
    };
  }, []);

  const applyQuoteRows = useCallback(
    (rows: any[], replace: boolean) => {
      if (!enabled || !Array.isArray(rows) || rows.length === 0) return;
      const nextMap = replace ? new Map<string, Quote>() : new Map(quotesRef.current);
      const nameMap = symbolNameMapRef.current;

      for (const row of rows) {
        if (!row?.symbol) continue;
        const symbol = String(row.symbol).toUpperCase();
        if (nameMap.size > 0 && !nameMap.has(symbol)) continue;

        const existing = nextMap.get(symbol);
        const bid = toNumber(row.bid);
        const ask = toNumber(row.ask);
        const price =
          toNumber(row.price) ??
          (bid != null && ask != null ? (bid + ask) / 2 : null) ??
          existing?.price ??
          0;

        const prevClose = toNumber(row.prevClose) ?? existing?.prevClose ?? price ?? 0;
        const pctChange = toNumber(row.pctChange) ?? calculatePctChange(price, prevClose);
        const change = toNumber(row.change) ?? (price != null && prevClose != null ? price - prevClose : 0);
        const spread = toNumber(row.spread) ?? (bid != null && ask != null ? Math.abs(ask - bid) : undefined);
        const lastApiUpdate = toNumber(row.lastApiUpdate) ?? existing?.lastApiUpdate;
        const dataAge = toNumber(row.dataAge) ?? (lastApiUpdate ? Date.now() - lastApiUpdate : undefined);
        const isStale = Boolean(row.isStale ?? existing?.isStale ?? false);

        nextMap.set(symbol, {
          symbol,
          name: nameMap.get(symbol) || symbol,
          price: price ?? 0,
          change: change ?? 0,
          time: row.timestamp || Date.now(),
          bid: bid ?? undefined,
          ask: ask ?? undefined,
          spread,
          prevClose: prevClose ?? price ?? 0,
          changePct: pctChange,
          percent_change: pctChange,
          isStale,
          lastApiUpdate: lastApiUpdate ?? undefined,
          dataAge: dataAge ?? undefined,
        });
      }

      quotesRef.current = nextMap;
      schedulePublish();
    },
    [enabled, schedulePublish],
  );

  useEffect(() => {
    if (!enabled) return;
    return subscribe((message) => {
      if (!message || typeof message !== "object") return;
      if (message.type === "quotes:snapshot") applyQuoteRows(message.rows ?? [], true);
      if (message.type === "quotes:update") applyQuoteRows(message.rows ?? [], false);
    });
  }, [applyQuoteRows, enabled, subscribe]);

  const symbolsQuery = requestedSymbols.join(",");
  const quotesUrl = symbolsQuery
    ? `/api/quotes/latest?symbols=${encodeURIComponent(symbolsQuery)}`
    : "/api/quotes/latest";

  const { data: latestQuotesData, isLoading, isError } = useQuery({
    queryKey: [quotesUrl],
    enabled,
    refetchInterval: enabled && isWsConnected ? false : fallbackInterval,
    refetchOnMount: "always",
    refetchOnWindowFocus: false,
    staleTime: 0,
  });

  useEffect(() => {
    if (!enabled) return;
    if (Array.isArray(latestQuotesData) && latestQuotesData.length > 0) {
      applyQuoteRows(latestQuotesData, true);
    }
  }, [enabled, latestQuotesData, applyQuoteRows]);

  useEffect(() => {
    if (!enabled) {
      setIsConnected(false);
      return;
    }
    setIsConnected(isWsConnected || (!isError && !!latestQuotesData));
  }, [enabled, isWsConnected, latestQuotesData, isError]);

  const hasStaleData = quotes.some((q) => q.isStale);

  return {
    quotes,
    isConnected,
    isLoading: enabled && isLoading && quotes.length === 0,
    hasStaleData,
    pollIntervalMs: fallbackInterval,
  };
}

type LiveUpdateMessage = Record<string, any>;
type LiveUpdateListener = (message: LiveUpdateMessage) => void;

type LiveUpdatesClient = {
  isConnected: boolean;
  sendMessage: (message: LiveUpdateMessage) => boolean;
  subscribe: (listener: LiveUpdateListener) => () => void;

const LiveUpdatesContext = createContext<LiveUpdatesClient | null>(null);

export function LiveUpdatesProvider({ children }: { children: ReactNode }) {

  const wsUrl = getWsUrl();

    reconnectAttempts: perf.isConstrained ? 6 : 10,
    onMessage: (message) => {

      for (const listener of listenersRef.current) listener(message);

  });

    return { isConnected, sendMessage, subscribe };

}

export function useLiveUpdates() {

function applyStoredLocale(user: User | null): User | null {
  if (!user || typeof window === "undefined") return user;
  try {
    const stored = localStorage.getItem("i18n.locale");
    if (!stored) return user;
    return { ...user, language: stored };
  } catch {
    return user;
  }

    setUser((prev) => (prev ? { ...prev, ...patch } : prev));

    const res = await apiRequest("GET", "/api/auth/current-user");

      setUser(null);

      setLoading(false);
    }

    if (!user) return;
    try {
      const next = await fetchCurrentUser();
      if (next) setUser(next);

      // ignore
    }

    checkAuth();
  }, [checkAuth]);

  const refreshMaybe = useCallback(() => {
    if (!user) return;
    const now = Date.now();
    if (now - lastRefreshAtRef.current < 30000) return;
    lastRefreshAtRef.current = now;
    refreshUser();
  }, [refreshUser, user]);

    if (!user) return;

    const handleVisibilityOrFocus = () => {
      if (typeof document !== "undefined" && document.visibilityState !== "visible") return;
      refreshMaybe();
    };

    window.addEventListener("focus", handleVisibilityOrFocus);
    window.addEventListener("online", handleVisibilityOrFocus);
    document.addEventListener("visibilitychange", handleVisibilityOrFocus);

    return () => {
      window.removeEventListener("focus", handleVisibilityOrFocus);
      window.removeEventListener("online", handleVisibilityOrFocus);
      document.removeEventListener("visibilitychange", handleVisibilityOrFocus);
    };
  }, [refreshMaybe, user]);

  return (
    <AuthContext.Provider
      value={{
        user,
        isAuthenticated: !!user,
        loading,
        login: async () => {},
        register: async () => {},
        logout: async () => {},
        checkAuth,
        refreshUser,
        updateUser,
        stopImpersonating: async () => {},
      }}
    >
      {children}
    </AuthContext.Provider>
  );

export type PerfHints = {
  constrained: boolean;
  saveData: boolean;
  effectiveType?: string;
  downlinkMbps?: number;
  rttMs?: number;
  deviceMemoryGB?: number;
  hardwareConcurrency?: number;

function computePerfHints(): PerfHints {
  if (typeof navigator === "undefined") {
    return { constrained: false, saveData: false };
  }

  const navAny = navigator as any;
  const conn = navAny.connection || navAny.mozConnection || navAny.webkitConnection;

  const effectiveType = typeof conn?.effectiveType === "string" ? String(conn.effectiveType) : undefined;
  const saveData = Boolean(conn?.saveData);

  const downlinkMbps = typeof conn?.downlink === "number" ? conn.downlink : undefined;
  const rttMs = typeof conn?.rtt === "number" ? conn.rtt : undefined;

  const deviceMemoryGB = typeof navAny.deviceMemory === "number" ? navAny.deviceMemory : undefined;
  const hardwareConcurrency = typeof navAny.hardwareConcurrency === "number" ? navAny.hardwareConcurrency : undefined;

  const networkConstrained =
    saveData ||
    (effectiveType != null && effectiveType !== "4g") ||
    (typeof rttMs === "number" && rttMs > 250) ||
    (typeof downlinkMbps === "number" && downlinkMbps < 1.5);

  const deviceConstrained =
    (typeof deviceMemoryGB === "number" && deviceMemoryGB <= 4) ||
    (typeof hardwareConcurrency === "number" && hardwareConcurrency <= 4);

  const envForced =
    typeof import.meta !== "undefined" &&
    (import.meta as any).env &&
    (String((import.meta as any).env.VITE_FORCE_CONSTRAINED ?? "").toLowerCase() === "1" ||
      String((import.meta as any).env.VITE_FORCE_CONSTRAINED ?? "").toLowerCase() === "true");

  return {
    constrained: envForced || networkConstrained || deviceConstrained,
    saveData,
    effectiveType,
    downlinkMbps,
    rttMs,
    deviceMemoryGB,
    hardwareConcurrency,
  };
}

    if (!isLoading && !user) {

import React, { createContext, useContext, useEffect, useMemo, useState } from "react";
import { useWebSocket, WebSocketMessage } from "@/hooks/use-websocket";
import { useAuth } from "@/hooks/use-auth";
import { getPollingIntervals } from "@/lib/perfHints";
import { useQueryClient } from "@tanstack/react-query";

interface LiveUpdatesContextType {
  connected: boolean;
  send: (message: WebSocketMessage) => void;
  subscribe: (eventType: string, callback: (message: WebSocketMessage) => void) => () => void;
}

const LiveUpdatesContext = createContext<LiveUpdatesContextType | null>(null);

export function LiveUpdatesProvider({ children }: { children: React.ReactNode }) {
  const { user, loading } = useAuth();
  const queryClient = useQueryClient();

  const [eventListeners, setEventListeners] = useState<Map<string, Set<(message: WebSocketMessage) => void>>>(new Map());

  const perfIntervals = getPollingIntervals();
  const wsEnabled = !!user && !loading;

  const { connected, sendMessage } = useWebSocket({
    url: "/api/ws",
    enabled: wsEnabled,
    autoReconnect: true,
    reconnectDelayMin: perfIntervals.wsReconnectMinMs,
    reconnectDelayMax: perfIntervals.wsReconnectMaxMs,
    onOpen: () => {
      if (import.meta.env.DEV) console.debug("WebSocket connected");
    },
    onClose: () => {
      if (import.meta.env.DEV) console.debug("WebSocket disconnected");
    },
    onMessage: (message: WebSocketMessage) => {
      const listeners = eventListeners.get(message.type);
      if (listeners) listeners.forEach((callback) => callback(message));

      if (message.type === "trades:updated") {
        queryClient.invalidateQueries({ queryKey: ["/api/trades"] });
        queryClient.invalidateQueries({ queryKey: ["/api/trades/pending"] });
        queryClient.invalidateQueries({ queryKey: ["/api/trades/history"] });
      }

      if (message.type === "account:updated" && (message as any)?.payload?.summary) {
        queryClient.setQueryData(["/api/account/summary"], (message as any).payload.summary);
      }
    },
    onError: (error) => {
      if (import.meta.env.DEV) console.error("WebSocket error:", error);
    },
  });

  useEffect(() => {
    if (!connected) setEventListeners(new Map());
  }, [connected]);

  const contextValue = useMemo(
    () => ({
      connected,
      send: sendMessage,
      subscribe: (eventType: string, callback: (message: WebSocketMessage) => void) => {
        setEventListeners((prev) => {
          const next = new Map(prev);
          if (!next.has(eventType)) next.set(eventType, new Set());
          next.get(eventType)!.add(callback);
          return next;
        });

        return () => {
          setEventListeners((prev) => {
            const next = new Map(prev);
            const listeners = next.get(eventType);
            if (listeners) {
              listeners.delete(callback);
              if (listeners.size === 0) next.delete(eventType);
            }
            return next;
          });
        };
      },
    }),
    [connected, sendMessage]
  );

  return <LiveUpdatesContext.Provider value={contextValue}>{children}</LiveUpdatesContext.Provider>;
}

export function useLiveUpdates() {
  const context = useContext(LiveUpdatesContext);
  if (!context) throw new Error("useLiveUpdates must be used within a LiveUpdatesProvider");
  return context;
}

  usedMargin: number;
  equity: number;
  freeMargin: number;
  floatingPnl: number;
  marginLevel: number | null;
  openPositions: number;

export function useAccountSummary(opts?: { enabled?: boolean }) {
  const enabled = opts?.enabled ?? true;
  const { connected, send, subscribe } = useLiveUpdates();
  const queryClient = useQueryClient();
  const intervals = getPollingIntervals();

  const query = useQuery<AccountSummary>({

    queryFn: async () => fetchWithIdentity<AccountSummary>("/api/account/summary"),
    enabled,
    staleTime: connected ? Infinity : 5_000,
    refetchOnWindowFocus: false,
    refetchInterval: enabled && !connected ? intervals.accountSummaryMs : false,
  });

  useEffect(() => {
    if (!enabled || !connected) return;

    send({ type: "account:subscribe" });

    const unsub = subscribe("account:updated", (message) => {

    });

    return () => {
      try {
        send({ type: "account:unsubscribe" });
      } catch {}
      unsub();
    };
  }, [connected, enabled, send, subscribe, queryClient]);

  return { summary: query.data, isLoading: query.isLoading, error: query.error, refetch: query.refetch };
