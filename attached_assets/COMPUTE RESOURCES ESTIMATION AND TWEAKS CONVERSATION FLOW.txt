export type QuoteWire = {
  symbol: string;
  name?: string;
  price?: number;
  bid?: number;
  ask?: number;
  spread?: number;
  change?: number;
  prevClose?: number;

  time?: number;

type LiveEvent = {
  type: string;
  userId?: number;
  payload?: any;
  ts?: number;
  [k: string]: any;
};

type Listener = (event: LiveEvent) => void;

type SubscriptionHandle = { unsubscribe: () => void };

type LiveUpdatesContextValue = {
  isConnected: boolean;
  lastMessageAt: number | null;
  send: (data: any) => boolean;

  getQuote: (symbol: string) => QuoteWire | undefined;
  getAllQuotes: () => QuoteWire[];
  quotesVersion: number;

  // Quote subscriptions (symbol-level and all-symbol)
  subscribeQuotes: (symbols: string[]) => SubscriptionHandle;
  subscribeAllQuotes: () => SubscriptionHandle;

  // Generic event listeners
  onEvent: (type: string, listener: Listener) => SubscriptionHandle;
};

function buildWsUrl(): string {
  const proto = window.location.protocol === "https:" ? "wss://" : "ws://";
  return `${proto}${window.location.host}/ws`;
}

function normalizeQuoteWire(q: any): QuoteWire | null {
  if (!q || typeof q !== "object") return null;
  const symbol = String(q.symbol ?? "").trim();
  if (!symbol) return null;

  const toNum = (v: any): number | undefined => {
    if (v === null || v === undefined || v === "") return undefined;
    const n = typeof v === "number" ? v : Number(v);
    return Number.isFinite(n) ? n : undefined;
  };

  const price = toNum(q.price);
  const bid = toNum(q.bid);
  const ask = toNum(q.ask);
  const spread =
    toNum(q.spread) ??
    (bid !== undefined && ask !== undefined ? Math.abs(ask - bid) : undefined);
  const prevClose = toNum(q.prevClose);
  const pctChange =
    toNum(q.pctChange) ?? toNum(q.changePct) ?? toNum(q.percent_change);

  return {
    symbol,
    name: typeof q.name === "string" ? q.name : undefined,
    price,
    bid,
    ask,
    spread,
    change: toNum(q.change),
    prevClose,
    pctChange,
    changePct: pctChange,
    percent_change: pctChange,
    isStale: !!q.isStale,
    lastApiUpdate: toNum(q.lastApiUpdate),
    dataAge: toNum(q.dataAge),
    timestamp: toNum(q.timestamp) ?? toNum(q.time),
    time: toNum(q.time) ?? toNum(q.timestamp),
  };
}

export function LiveUpdatesProvider({ children }: { children: React.ReactNode }) {
  const { user } = useAuth();
  const wsRef = useRef<WebSocket | null>(null);
  const reconnectTimerRef = useRef<number | null>(null);
  const shouldRunRef = useRef(true);

  const [isConnected, setIsConnected] = useState(false);
  const [lastMessageAt, setLastMessageAt] = useState<number | null>(null);

  // Quotes cache (in-memory)
  const quotesBySymbolRef = useRef<Map<string, QuoteWire>>(new Map());
  const [quotesVersion, setQuotesVersion] = useState(0);

  // Subscription state
  const subscribeAllCountRef = useRef(0);
  const symbolRefCountRef = useRef<Map<string, number>>(new Map());
  const lastSubscribeSentRef = useRef<string>("");

  // Generic listeners
  const listenersByTypeRef = useRef<Map<string, Set<Listener>>>(new Map());

  const send = useCallback((data: any): boolean => {
    const ws = wsRef.current;
    if (!ws || ws.readyState !== WebSocket.OPEN) return false;
    try {
      ws.send(JSON.stringify(data));
      return true;
    } catch {
      return false;
    }
  }, []);

  const emitToListeners = useCallback((event: LiveEvent) => {
    const byType = listenersByTypeRef.current;
    const invoke = (set?: Set<Listener>) => {
      if (!set) return;
      for (const fn of set) {
        try {
          fn(event);
        } catch (e) {
          console.error("LiveUpdates listener error", e);
        }
      }
    };
    invoke(byType.get(event.type));
    invoke(byType.get("*"));
  }, []);

  const computeDesiredQuoteSubscription = useCallback((): {
    all: boolean;
    symbols: string[];
  } => {
    const all = subscribeAllCountRef.current > 0;
    if (all) return { all: true, symbols: [] };
    const symbols = Array.from(symbolRefCountRef.current.keys()).sort();
    return { all: false, symbols };
  }, []);

  const sendQuoteSubscriptionIfChanged = useCallback(() => {
    const desired = computeDesiredQuoteSubscription();
    const signature = desired.all ? "ALL" : desired.symbols.join(",");
    if (signature === lastSubscribeSentRef.current) return;
    lastSubscribeSentRef.current = signature;

    if (desired.all) {
      send({ type: "quotes:subscribeAll" });
      return;
    }

    // Resubscribe full set (idempotent)
    send({ type: "quotes:subscribe", symbols: desired.symbols });
  }, [computeDesiredQuoteSubscription, send]);

  const mergeQuotes = useCallback((quotes: any[]) => {
    let changed = false;
    const map = quotesBySymbolRef.current;

    for (const raw of quotes) {
      const q = normalizeQuoteWire(raw);
      if (!q) continue;

      const prev = map.get(q.symbol);
      const next: QuoteWire = { ...(prev ?? {}), ...q };

      if (!next.timestamp) next.timestamp = prev?.timestamp ?? Date.now();
      if (!next.time) next.time = next.timestamp;

      const same =
        prev &&
        prev.price === next.price &&
        prev.bid === next.bid &&
        prev.ask === next.ask &&
        prev.spread === next.spread &&
        prev.prevClose === next.prevClose &&
        prev.pctChange === next.pctChange &&
        prev.isStale === next.isStale &&
        prev.timestamp === next.timestamp;

      if (!same) {
        map.set(q.symbol, next);
        changed = true;
      }
    }

    if (changed) setQuotesVersion((v) => v + 1);
  }, []);

  const handleMessage = useCallback(
    (raw: MessageEvent) => {
      setLastMessageAt(Date.now());
      let msg: any;
      try {
        msg = JSON.parse(raw.data);
      } catch {
        return;
      }
      if (!msg || typeof msg !== "object") return;

      const ev: LiveEvent = {
        ...msg,
        ts: typeof msg.ts === "number" ? msg.ts : Date.now(),
      };
      emitToListeners(ev);

      // New refactor: quotes streaming payloads
      if (ev.type === "quotes:snapshot" || ev.type === "quotes:delta") {
        const quotes = Array.isArray(ev.payload?.quotes)
          ? ev.payload.quotes
          : Array.isArray((ev as any).quotes)
          ? (ev as any).quotes
          : null;
        if (quotes) mergeQuotes(quotes);
      }

      // Legacy: quotes:updated has no payload -> fallback polling handles it.
    },
    [emitToListeners, mergeQuotes]
  );

  const connect = useCallback(() => {
    if (!shouldRunRef.current) return;

    const url = buildWsUrl();
    const ws = new WebSocket(url);
    wsRef.current = ws;

    ws.onopen = () => {
      setIsConnected(true);

      // Authenticate for user-scoped events
      if (user?.id) {
        try {
          ws.send(JSON.stringify({ type: "auth", userId: user.id }));
        } catch {
          // ignore
        }
      }

      // Apply pending quote subscriptions
      sendQuoteSubscriptionIfChanged();
    };

    ws.onmessage = handleMessage;

    ws.onerror = (e) => {
      console.error("WS error", e);
    };

    ws.onclose = () => {
      setIsConnected(false);
      wsRef.current = null;
      if (!shouldRunRef.current) return;

      // Simple backoff (3s)
      if (reconnectTimerRef.current)
        window.clearTimeout(reconnectTimerRef.current);
      reconnectTimerRef.current = window.setTimeout(() => {
        connect();
      }, 3000);
    };
  }, [handleMessage, sendQuoteSubscriptionIfChanged, user?.id]);

  useEffect(() => {
    shouldRunRef.current = true;
    connect();
    return () => {
      shouldRunRef.current = false;
      if (reconnectTimerRef.current)
        window.clearTimeout(reconnectTimerRef.current);
      reconnectTimerRef.current = null;
      const ws = wsRef.current;
      wsRef.current = null;
      try {
        ws?.close();
      } catch {
        // ignore
      }
    };
  }, [connect]);

  useEffect(() => {
    if (!user?.id) return;
    send({ type: "auth", userId: user.id });
  }, [send, user?.id]);

  // Fallback snapshot polling (slow safety net; should go idle when server streams deltas)
  useEffect(() => {
    const fallbackMs = Number(
      (import.meta as any).env?.VITE_QUOTES_FALLBACK_POLL_MS ?? 5000
    );
    if (!Number.isFinite(fallbackMs) || fallbackMs <= 0) return;

    const interval = window.setInterval(async () => {
      const desired = computeDesiredQuoteSubscription();
      const hasAnySubscription = desired.all || desired.symbols.length > 0;
      if (!hasAnySubscription) return;

      if (
        lastMessageAt &&
        Date.now() - lastMessageAt < Math.max(1500, fallbackMs)
      )
        return;

      try {
        let url = "/api/quotes/latest";
        if (!desired.all && desired.symbols.length > 0) {
          url += `?symbols=${encodeURIComponent(desired.symbols.join(","))}`;
        }
        const res = await fetch(url, { credentials: "include" });
        if (!res.ok) return;
        const data = await res.json();
        if (Array.isArray(data)) mergeQuotes(data);
      } catch {
        // ignore
      }
    }, fallbackMs);

    return () => window.clearInterval(interval);
  }, [computeDesiredQuoteSubscription, lastMessageAt, mergeQuotes]);

  const subscribeQuotes = useCallback(
    (symbols: string[]): SubscriptionHandle => {
      const cleaned = Array.from(
        new Set(
          (symbols || [])
            .map((s) => String(s ?? "").trim().toUpperCase())
            .filter(Boolean)
        )
      );

      if (cleaned.length === 0) return { unsubscribe: () => {} };

      const refCounts = symbolRefCountRef.current;
      for (const sym of cleaned) {
        refCounts.set(sym, (refCounts.get(sym) ?? 0) + 1);
      }
      sendQuoteSubscriptionIfChanged();

      return {
        unsubscribe: () => {
          const refCounts = symbolRefCountRef.current;
          for (const sym of cleaned) {
            const next = (refCounts.get(sym) ?? 0) - 1;
            if (next <= 0) refCounts.delete(sym);
            else refCounts.set(sym, next);
          }
          sendQuoteSubscriptionIfChanged();
        },
      };
    },
    [sendQuoteSubscriptionIfChanged]
  );

  const subscribeAllQuotes = useCallback((): SubscriptionHandle => {
    subscribeAllCountRef.current += 1;
    sendQuoteSubscriptionIfChanged();
    return {
      unsubscribe: () => {
        subscribeAllCountRef.current = Math.max(
          0,
          subscribeAllCountRef.current - 1
        );
        sendQuoteSubscriptionIfChanged();
      },
    };
  }, [sendQuoteSubscriptionIfChanged]);

  const onEvent = useCallback(
    (type: string, listener: Listener): SubscriptionHandle => {
      const key = String(type || "*");
      const map = listenersByTypeRef.current;
      const set = map.get(key) ?? new Set<Listener>();
      set.add(listener);
      map.set(key, set);

      return {
        unsubscribe: () => {
          const set = map.get(key);
          if (!set) return;
          set.delete(listener);
          if (set.size === 0) map.delete(key);
        },
      };
    },
    []
  );

  const getQuote = useCallback((symbol: string) => {
    const sym = String(symbol ?? "").trim().toUpperCase();
    if (!sym) return undefined;
    return quotesBySymbolRef.current.get(sym);
  }, []);

  const getAllQuotes = useCallback(() => {
    return Array.from(quotesBySymbolRef.current.values());
  }, []);

  const value: LiveUpdatesContextValue = useMemo(
    () => ({
      isConnected,
      lastMessageAt,
      send,
      getQuote,
      getAllQuotes,
      quotesVersion,
      subscribeQuotes,
      subscribeAllQuotes,
      onEvent,
    }),
    [
      getAllQuotes,
      getQuote,
      isConnected,
      lastMessageAt,
      onEvent,
      quotesVersion,
      send,
      subscribeAllQuotes,
      subscribeQuotes,
    ]
  );

  return (
    <LiveUpdatesContext.Provider value={value}>
      {children}
    </LiveUpdatesContext.Provider>
  );
}

    for (const s of symbols) {
      if (s.enabled === false) continue;
      map[String(s.symbol).toUpperCase()] = s.name;
    }
    return map;
  }, [symbols]);

  const quotes: Quote[] = useMemo(() => {
    const all = live.getAllQuotes();

    const enabledSyms = new Set(Object.keys(symbolNameMap));
    const out: Quote[] = [];

    for (const q of all) {
      const sym = String(q.symbol ?? "").toUpperCase();
      if (!enabledSyms.has(sym)) continue;
      const price = typeof q.price === "number" ? q.price : 0;
      const change = typeof q.change === "number" ? q.change : 0;
      const time =
        (typeof q.time === "number"
          ? q.time
          : typeof q.timestamp === "number"
          ? q.timestamp
          : Date.now()) as number;

      const pct =
        typeof q.pctChange === "number"
          ? q.pctChange
          : typeof q.changePct === "number"
          ? q.changePct
          : typeof q.percent_change === "number"
          ? q.percent_change
          : undefined;

      out.push({
        ...q,
        symbol: sym,
        name: symbolNameMap[sym] || sym,
        price,
        change,
        time,
        changePct: pct,
        percent_change: pct,
      });
    }

    out.sort((a, b) => a.symbol.localeCompare(b.symbol));
    return out;
  }, [live, live.quotesVersion, symbolNameMap]);

  const hasStaleData = quotes.some((q) => !!q.isStale);
  const isLoading = subscribed && quotes.length === 0;

  return {
    quotes,
    isConnected: live.isConnected,
    isLoading,
    hasStaleData,
  };
}

      // errors are counted by checks below

      if (Date.now() - lastMsgAt > 45_000) {
        socket.close();
        break;
      }

      sleep(1);
    }

    socket.close();
  });

  check(res, { "ws status is 101": (r) => r && r.status === 101 });

  sleep(1);

function sanitizeRoute(path: string): string {
  if (!path) return "/";

  // Replace numeric IDs
  let out = path.replace(/\/\d+(?=\/|$)/g, "/:id");

  // Replace UUIDs
  out = out.replace(
    /\/[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[1-5][0-9a-fA-F]{3}-[89abAB][0-9a-fA-F]{3}-[0-9a-fA-F]{12}(?=\/|$)/g,
    "/:uuid"
  );

  // Replace long hex tokens
  out = out.replace(/\/[0-9a-fA-F]{24,}(?=\/|$)/g, "/:token");

  return out;
}

function resolveRouteLabel(req: Request): string {
  // Prefer Express route patterns when available
  const anyReq: any = req as any;

  const routePath = anyReq?.route?.path;
  const baseUrl = anyReq?.baseUrl;

  if (routePath) {
    const combined = `${baseUrl || ""}${routePath}`;
    return sanitizeRoute(combined);
  }

  return sanitizeRoute(req.path);
}

export function installHttpMetrics(app: Express) {
  app.use((req: Request, res: Response, next: NextFunction) => {
    httpRequestsInFlight.inc();

    const start = process.hrtime.bigint();

    res.on("finish", () => {
      const end = process.hrtime.bigint();
      const seconds = Number(end - start) / 1e9;

      httpRequestsInFlight.dec();

      const route = resolveRouteLabel(req);
      const labels = {
        method: req.method,
        route,
        status_code: String(res.statusCode),
      };

      httpRequestsTotal.inc(labels, 1);
      httpRequestDurationSeconds.observe(labels, seconds);
    });

    next();
  });
}

  });
}

      wsBytesSentTotal.inc({ type: t }, bytes);
    },

      const t = normalizeType(type);
      const bytes =
        typeof payload === "string" ? Buffer.byteLength(payload, "utf8") : payload.length;

      wsBytesReceivedTotal.inc({ type: t }, bytes);
    },

    lastPongAt?: number;
  };

              sock.ping();
            } catch {
              // ignore
            }
          }

6) Prometheus alerts (PrometheusRule)
Create: prometheus-rules-tradecentral.yaml

This focuses on the failure modes that will hurt you at scale: disconnect storms, event loop lag, 5xx surges, DB pool saturation, DB deadlocks, Redis memory pressure.

apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: tradecentral-rules
  labels:
    release: kube-prometheus-stack
spec:
  groups:
    - name: tradecentral.ws
      rules:
        - alert: TradeCentralWsActiveConnectionsDropFast
          expr: |
            (sum(ws_active_connections{namespace="tradecentral"})
             -
             sum(ws_active_connections{namespace="tradecentral"} offset 2m)) < -2000
          for: 1m
          labels:
            severity: warning
          annotations:
            summary: "WS connections dropped sharply (possible disconnect storm)"
            description: "Active WS connections dropped by >2000 within 2 minutes."

        - alert: TradeCentralWsThroughputHigh
          expr: |
            sum(rate(ws_bytes_sent_total{namespace="tradecentral"}[5m])) > 80000000
          for: 5m
          labels:
            severity: warning
          annotations:
            summary: "WS egress high"
            description: "WS egress > ~80MB/s sustained. Check quote subscriptions and batching."

    - name: tradecentral.node
      rules:
        - alert: TradeCentralEventLoopLagHigh
          expr: |
            max(tradecentral_nodejs_eventloop_lag_seconds{namespace="tradecentral"}) > 0.2
          for: 2m
          labels:
            severity: warning
          annotations:
            summary: "Node event loop lag high"
            description: "Event loop lag > 200ms for 2 minutes. Reduce synchronous work; check CPU/memory."

        - alert: TradeCentralProcessRssHigh
          expr: |
            max(tradecentral_process_resident_memory_bytes{namespace="tradecentral"}) > 5.5e+09
          for: 5m
          labels:
            severity: warning
          annotations:
            summary: "Node RSS high"
            description: "Process RSS > ~5.5GB; risk of OOM in 6Gi pods. Consider more pods or memory optimization."

    - name: tradecentral.http
      rules:
        - alert: TradeCentralHttp5xxRateHigh
          expr: |
            (
              sum(rate(http_requests_total{namespace="tradecentral",status_code=~"5.."}[5m]))
              /
              clamp_min(sum(rate(http_requests_total{namespace="tradecentral"}[5m])), 1)
            ) > 0.02
          for: 5m
          labels:
            severity: critical
          annotations:
            summary: "HTTP 5xx rate high"
            description: "HTTP 5xx rate > 2% for 5 minutes."

        - alert: TradeCentralHttpLatencyP95High
          expr: |
            histogram_quantile(
              0.95,
              sum by (le) (rate(http_request_duration_seconds_bucket{namespace="tradecentral"}[5m]))
            ) > 0.75
          for: 5m
          labels:
            severity: warning
          annotations:
            summary: "HTTP p95 latency high"
            description: "Overall HTTP p95 latency > 750ms for 5 minutes."

    - name: tradecentral.db
      rules:
        - alert: TradeCentralPgBouncerClientWaiting
          expr: |
            sum(pgbouncer_pools_client_waiting{namespace="tradecentral"}) > 0
          for: 2m
          labels:
            severity: warning
          annotations:
            summary: "PgBouncer clients waiting (pool saturation)"
            description: "Clients are waiting for a DB connection. Increase pool size or reduce DB concurrency."

        - alert: TradeCentralPostgresDeadlocks
          expr: |
            increase(pg_stat_database_deadlocks{namespace="tradecentral"}[10m]) > 0
          for: 1m
          labels:
            severity: critical
          annotations:
            summary: "Postgres deadlocks detected"
            description: "Deadlocks occurred in the last 10 minutes. Review transaction ordering and indexes."

        - alert: TradeCentralPostgresConnectionsHigh
          expr: |
            sum(pg_stat_activity_count{namespace="tradecentral"}) > 300
          for: 5m
          labels:
            severity: warning
          annotations:
            summary: "Postgres connections high"
            description: "Active connections > 300. Ensure PgBouncer is in use and app pool settings are sane."

    - name: tradecentral.redis
      rules:
        - alert: TradeCentralRedisMemoryHigh
          expr: |
            sum(redis_memory_used_bytes{namespace="tradecentral"}) > 2.5e+09
          for: 10m
          labels:
            severity: warning
          annotations:
            summary: "Redis memory high"
            description: "Redis memory > 2.5GB sustained. Check quote snapshot caching and pub/sub usage."

        - alert: TradeCentralRedisRejectedConnections
          expr: |
            increase(redis_rejected_connections_total{namespace="tradecentral"}[10m]) > 0
          for: 1m
          labels:
            severity: critical
          annotations:
            summary: "Redis rejected connections"
            description: "Redis is rejecting connections. Increase limits or investigate resource pressure."
