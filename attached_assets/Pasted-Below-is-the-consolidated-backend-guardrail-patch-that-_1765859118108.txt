Below is the **consolidated backend guardrail patch** that **adds/enforces the “50 lots max concurrently” rule** (across **any number of trades**) **on top of** the existing limits (**max 5 active trades total, max 3 active per instrument**), and makes it **bug-resistant** by enforcing at:

* **placement time** (`riskMiddleware` → blocks the request)
* **fill time** (`orderEngine` → cancels pending orders if limits are violated after admin changes)

It counts **OPEN + PENDING** as “concurrent/active”.

## 1) `server/risk.ts` — REPLACE ENTIRE FILE WITH THIS

```ts
import { storage } from "./storage";
import { Request, Response, NextFunction } from "express";
import dayjs from "dayjs";
import { Trade, globalSettings } from "@shared/schema";
import { db } from "@db";
import { eq } from "drizzle-orm";

/**
 * Risk management middleware
 * Enforces:
 * - Max active trades per user (OPEN + PENDING)  [default 5]
 * - Max active trades per instrument (OPEN + PENDING) [default 3]
 * - Max concurrent lots per user across ALL active trades (OPEN + PENDING) [default 50]
 * - Daily loss limit (hardcoded 10% of initial)   [unchanged]
 * - Lifetime loss limit (hardcoded 20% of initial; disables account) [unchanged]
 *
 * NOTE: Global settings act as a HARD CAP. User settings can only be stricter.
 */

const INITIAL_BALANCE_USD = 1_000_000;
const DEFAULTS = {
  maxTradesPerUser: 5,
  maxTradesPerInstrument: 3,
  maxConcurrentLots: 50,
  dailyLossLimitPct: 10,
  lifetimeLossLimitPct: 20,
};

function toInt(v: any): number | null {
  if (v === null || v === undefined) return null;
  if (typeof v === "number" && Number.isFinite(v)) return Math.trunc(v);
  if (typeof v === "string" && v.trim() !== "") {
    const x = Number(v);
    return Number.isFinite(x) ? Math.trunc(x) : null;
  }
  return null;
}

async function getEffectiveLimits(userId: number) {
  const gs =
    (await db.query.globalSettings.findFirst({
      where: eq(globalSettings.id, 1),
    })) ?? null;

  const us = (await storage.getUserSettingsById(userId)) ?? null;

  const gMaxTradesPerUser = Number(gs?.maxTradesPerUser ?? DEFAULTS.maxTradesPerUser);
  const gMaxTradesPerInstrument = Number(gs?.maxTradesPerInstrument ?? DEFAULTS.maxTradesPerInstrument);
  const gMaxConcurrentLots = Number(gs?.maxConcurrentLots ?? DEFAULTS.maxConcurrentLots);

  // User overrides are allowed but cannot exceed the global cap (global is hard guardrail)
  const effectiveMaxTradesPerUser = Math.min(Number(us?.maxConcurrent ?? gMaxTradesPerUser), gMaxTradesPerUser);
  const effectiveMaxConcurrentLots = Math.min(Number(us?.maxConcurrentLots ?? gMaxConcurrentLots), gMaxConcurrentLots);

  return {
    maxTradesPerInstrument: gMaxTradesPerInstrument,
    maxTradesPerUser: effectiveMaxTradesPerUser,
    maxConcurrentLots: effectiveMaxConcurrentLots,
    dailyLossLimitPct: DEFAULTS.dailyLossLimitPct,
    lifetimeLossLimitPct: DEFAULTS.lifetimeLossLimitPct,
  };
}

export async function riskMiddleware(req: Request, res: Response, next: NextFunction) {
  const userId = (req.session as any).userId;
  if (!userId) return res.status(401).json({ message: "Unauthorized" });

  const user = await storage.getUserById(userId);
  if (!user) return res.status(404).json({ message: "User not found" });

  const limits = await getEffectiveLimits(userId);

  // Treat OPEN + PENDING as "active/concurrent"
  const openTrades = await storage.getOpenTradesByUserId(userId);
  const pendingTrades = await storage.getPendingTradesByUserId(userId);
  const activeTrades = [...openTrades, ...pendingTrades];

  // 1) Max active trades per user
  if (activeTrades.length >= limits.maxTradesPerUser) {
    return res.status(400).json({
      message: `Maximum ${limits.maxTradesPerUser} concurrent trades allowed (OPEN + PENDING).`,
      activeTrades: activeTrades.length,
      limit: limits.maxTradesPerUser,
    });
  }

  // 2) Max active trades per instrument
  const symbolId = toInt((req.body as any)?.symbolId);
  if (symbolId !== null) {
    const activePerSymbol = activeTrades.filter((t) => Number(t.symbolId) === symbolId).length;
    if (activePerSymbol >= limits.maxTradesPerInstrument) {
      return res.status(400).json({
        message: `Maximum ${limits.maxTradesPerInstrument} concurrent trades allowed per instrument (OPEN + PENDING).`,
        symbolId,
        activePerSymbol,
        limit: limits.maxTradesPerInstrument,
      });
    }
  }

  // 3) Max concurrent lots across ALL active trades (OPEN + PENDING)
  // Request uses either `lots` or `size` (your routes unify size->lots).
  const requestedLots = toInt((req.body as any)?.lots ?? (req.body as any)?.size);
  if (requestedLots !== null && requestedLots > 0) {
    const currentLots = activeTrades.reduce((sum, t) => sum + Number((t as any).lots ?? 0), 0);

    // Core requirement: 1 trade of 50 blocks new trades; 30+30 blocks 3rd, etc.
    if (currentLots + requestedLots > limits.maxConcurrentLots) {
      return res.status(409).json({
        message: `Maximum concurrent lots exceeded (OPEN + PENDING). Current=${currentLots}, Requested=${requestedLots}, Limit=${limits.maxConcurrentLots}.`,
        currentLots,
        requestedLots,
        limit: limits.maxConcurrentLots,
      });
    }
  }

  // 4) Daily loss (unchanged logic; still uses fixed initial balance baseline)
  const todayStart = dayjs().startOf("day").unix();
  const dailyTrades = await storage.getClosedTradesByDateRange(userId, todayStart);

  let dailyPnL = 0;
  dailyTrades.forEach((trade: Trade) => {
    if (trade.profit) dailyPnL += parseFloat(trade.profit);
  });

  const dailyLossPercent = dailyPnL < 0 ? (Math.abs(dailyPnL) / INITIAL_BALANCE_USD) * 100 : 0;
  if (dailyLossPercent >= limits.dailyLossLimitPct) {
    return res.status(403).json({
      message: `Daily loss limit of ${limits.dailyLossLimitPct}% reached. Try again tomorrow.`,
      dailyLossPercent,
      limit: limits.dailyLossLimitPct,
    });
  }

  // 5) Lifetime loss (unchanged)
  const currentBalance = parseFloat(String(user.balance));
  const lifetimeLossPercent = ((INITIAL_BALANCE_USD - currentBalance) / INITIAL_BALANCE_USD) * 100;

  if (lifetimeLossPercent >= limits.lifetimeLossLimitPct) {
    await storage.disableUserAccount(userId);
    return res.status(403).json({
      message: `Lifetime loss limit of ${limits.lifetimeLossLimitPct}% reached. Account has been disabled.`,
      lifetimeLossPercent,
      limit: limits.lifetimeLossLimitPct,
    });
  }

  return next();
}
```

## What this gives you (behavioral proof)

* If user has **1 OPEN trade of 50 lots**, then:

  * placing any new order with **requestedLots ≥ 1** fails: `currentLots(50) + requestedLots > 50`.
* If user has **2 trades of 30 lots + 30 lots**, then:

  * third trade is blocked because `60 + x > 50`.
* It counts **PENDING** too, so users cannot bypass with limit/stop orders.
* If an admin later tightens limits and the user is already over, the engine will **cancel pending orders** (reducing risk) instead of filling them.
