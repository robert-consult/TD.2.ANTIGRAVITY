import { db } from "@db";
import { sql } from "drizzle-orm";
import { getDb } from "./sqlite";

export function ensureUserAccountControlColumns() {
  const db = getDb();
  try {
    const cols = db.prepare(`PRAGMA table_info(users)`).all().map((c: any) => c.name);
    const add = (col: string, ddl: string) => { if (!cols.includes(col)) db.exec(ddl); };

    add("is_frozen", "ALTER TABLE users ADD COLUMN is_frozen INTEGER NOT NULL DEFAULT 0");
    add("freeze_reason_code", "ALTER TABLE users ADD COLUMN freeze_reason_code TEXT");
    add("freeze_reason_text", "ALTER TABLE users ADD COLUMN freeze_reason_text TEXT");
    add("frozen_at", "ALTER TABLE users ADD COLUMN frozen_at INTEGER");
    add("frozen_until", "ALTER TABLE users ADD COLUMN frozen_until INTEGER");
    add("frozen_by_admin_id", "ALTER TABLE users ADD COLUMN frozen_by_admin_id INTEGER");
  } catch (err) {
    console.error("ensureUserAccountControlColumns failed:", err);
  } finally {
    db.close();
  }
}

export function ensureUserAuditTables() {
  const db = getDb();
  try {
    db.exec(`
      CREATE TABLE IF NOT EXISTS login_events (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        user_id INTEGER,
        username TEXT,
        ip TEXT NOT NULL,
        user_agent TEXT,
        success INTEGER NOT NULL DEFAULT 0,
        failure_reason TEXT,
        created_at INTEGER NOT NULL DEFAULT (strftime('%s','now'))
      );
      CREATE INDEX IF NOT EXISTS idx_login_events_user_id ON login_events(user_id);
      CREATE INDEX IF NOT EXISTS idx_login_events_created_at ON login_events(created_at);

      CREATE TABLE IF NOT EXISTS user_admin_notes (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        user_id INTEGER NOT NULL,
        admin_user_id INTEGER,
        type TEXT NOT NULL DEFAULT 'NOTE',
        severity TEXT NOT NULL DEFAULT 'INFO',
        flag_code TEXT,
        content TEXT NOT NULL,

        resolved_by_admin_id INTEGER
      );
      CREATE INDEX IF NOT EXISTS idx_user_admin_notes_user_id ON user_admin_notes(user_id);
      CREATE INDEX IF NOT EXISTS idx_user_admin_notes_created_at ON user_admin_notes(created_at);

      CREATE TABLE IF NOT EXISTS user_balance_events (
        id INTEGER PRIMARY KEY AUTOINCREMENT,

### B) Admin balance adjustment that creates deposit/withdrawal history

```ts
async adjustUserBalanceAdmin(params: {
  userId: number;
  newBalance: string;
  adminUserId: number | null;
  type: "DEPOSIT" | "WITHDRAWAL" | "ADJUSTMENT";
  note?: string;
}): Promise<User> {
  const user = await this.getUser(params.userId);
  if (!user) throw new Error("User not found");

  const prev = parseFloat(user.balance || "0");
  const next = parseFloat(params.newBalance || "0");
  const amount = next - prev;

  const [updatedUser] = await db
    .update(users)
    .set({ balance: params.newBalance })
    .where(eq(users.id, params.userId))
    .returning();

  await db.insert(userBalanceEvents).values({
    userId: params.userId,
    adminUserId: params.adminUserId,
    type: params.type,
    amount,
    previousBalance: prev,
    newBalance: next,
    note: params.note || null,
  });

  return updatedUser;
},
```
