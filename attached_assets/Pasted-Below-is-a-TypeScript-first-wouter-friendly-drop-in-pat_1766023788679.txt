```ts
import { Request, Response, NextFunction } from "express";

export function adminGuard(req: Request, res: Response, next: NextFunction) {
  if (!req.user || req.user.role !== "admin") {
    return res.status(403).json({ error: "Admin access required" });
  }
  next();
}
```

`server/middleware/impersonationWriteBlock.ts`

```ts
import { Request, Response, NextFunction } from "express";

const WRITE = new Set(["POST", "PUT", "PATCH", "DELETE"]);

export function impersonationWriteBlock(req: Request, res: Response, next: NextFunction) {
  if (req.user?.impersonated && WRITE.has(req.method)) {
    return res.status(403).json({ error: "Writes disabled while impersonating (View-As mode)" });
  }
  next();
}
```

## 1.3 Admin: user search + view-as start/stop

Assumptions:

* `server/db.ts` exports `query<T=any>(sql: string, params?: any[]): Promise<{ rows: T[] }>`
* `server/auth.ts` has already mounted a JWT auth middleware that populates `req.user`
* `process.env.JWT_SECRET` is set

`server/routes/admin.ts`

```ts
import { Router } from "express";
import jwt from "jsonwebtoken";
import { adminGuard } from "../middleware/adminGuard";
import db from "../db";

const admin = Router();

type UserRow = { id: number; username: string; email: string; role: string; status?: string };

function auditMeta(req: any, extra: Record<string, any> = {}) {
  return {
    ...extra,
    request_id: req.headers?.["x-request-id"] || null
  };
}

async function logAdminAction(opts: {
  adminId: number;
  userId: number;
  actionType: string;
  req: any;
  metadata?: any;
}) {
  const { adminId, userId, actionType, req, metadata } = opts;
  await db.query(
    `INSERT INTO admin_actions (admin_id, user_id, action_type, metadata, ip, user_agent)
     VALUES ($1,$2,$3,$4,$5,$6)`,
    [
      adminId,
      userId,
      actionType,
      metadata ? JSON.stringify(metadata) : null,
      req.ip || null,
      req.headers?.["user-agent"] || null
    ]
  );
}

/**
 * GET /admin/users/search?q=...
 * match by numeric id OR ILIKE username/email
 */
admin.get("/users/search", adminGuard, async (req, res) => {
  const qRaw = String(req.query.q || "").trim();
  if (!qRaw) return res.json([]);

  const isNum = /^[0-9]+$/.test(qRaw);
  const q = `%${qRaw}%`;
  const { rows } = await db.query<UserRow>(
    `
    SELECT id, username, email, role, status
    FROM users
    WHERE (${isNum} AND id = $1)
       OR (username ILIKE $2)
       OR (email ILIKE $2)
    ORDER BY id DESC
    LIMIT 20
    `,
    [isNum ? Number(qRaw) : -1, q]
  );

  res.json(rows);
});

/**
 * POST /admin/view-as/start
 * body: { targetUserId: number }
 */
admin.post("/view-as/start", adminGuard, async (req, res) => {
  const { targetUserId } = req.body || {};
  if (!targetUserId) return res.status(400).json({ error: "targetUserId required" });

  const userQ = await db.query<UserRow>(
    "SELECT id, username, email, role FROM users WHERE id = $1",
    [targetUserId]
  );
  if (!userQ.rows.length) return res.status(404).json({ error: "User not found" });
  const u = userQ.rows[0];

  const token = jwt.sign(
    {
      id: u.id,
      role: u.role,
      username: u.username,
      email: u.email,
      impersonated: true,
      impersonated_by: req.user!.id
    },
    process.env.JWT_SECRET!,
    { expiresIn: "15m" }
  );

  await logAdminAction({
    adminId: req.user!.id,
    userId: u.id,
    actionType: "VIEW_AS_START",
    req,
    metadata: auditMeta(req, { target: { id: u.id, username: u.username, email: u.email } })
  });

  res.json({ token, target: { id: u.id, username: u.username, email: u.email } });
});

/**
 * POST /admin/view-as/stop
 * body: { targetUserId: number }
 */
admin.post("/view-as/stop", adminGuard, async (req, res) => {
  const { targetUserId } = req.body || {};
  if (!targetUserId) return res.status(400).json({ error: "targetUserId required" });

  await logAdminAction({
    adminId: req.user!.id,
    userId: Number(targetUserId),
    actionType: "VIEW_AS_STOP",
    req,
    metadata: auditMeta(req, { note: "Admin exited View-As" })
  });

  res.json({ success: true });
});

export default admin;
```

```ts
import { Router } from "express";
import db from "../db";

const journal = Router();

function cleanTags(tags: unknown): string[] {
  if (!Array.isArray(tags)) return [];
  return tags.map(t => String(t || "").trim()).filter(Boolean).slice(0, 20);
}

// GET /journal
journal.get("/journal", async (req, res) => {
  const { rows } = await db.query(
    `
    SELECT id, user_id, trade_id, note, mood, tags, attachment_url, created_at, updated_at
    FROM trader_journal
    WHERE user_id = $1
    ORDER BY created_at DESC
    LIMIT 200
    `,
    [req.user!.id]
  );
  res.json(rows);
});

// POST /journal
journal.post("/journal", async (req, res) => {
  const { trade_id = null, note, mood = null, tags = [], attachment_url = null } = req.body || {};
  const noteClean = String(note || "").trim();
  if (!noteClean || noteClean.length < 3) return res.status(400).json({ error: "note min length 3" });
  if (noteClean.length > 10000) return res.status(400).json({ error: "note too long (max 10,000)" });

  const { rows } = await db.query(
    `
    INSERT INTO trader_journal (user_id, trade_id, note, mood, tags, attachment_url)
    VALUES ($1,$2,$3,$4,$5,$6) RETURNING *
    `,
    [req.user!.id, trade_id, noteClean, mood, cleanTags(tags), attachment_url]
  );
  res.json(rows[0]);
});

// PUT /journal/:id
journal.put("/journal/:id", async (req, res) => {
  const { id } = req.params;
  const { note, mood = null, tags = [], attachment_url = null } = req.body || {};
  const noteClean = String(note || "").trim();
  if (!noteClean || noteClean.length < 3) return res.status(400).json({ error: "note min length 3" });

  const { rows } = await db.query(
    `
    UPDATE trader_journal
    SET note=$1, mood=$2, tags=$3, attachment_url=$4, updated_at=NOW()
    WHERE id=$5 AND user_id=$6
    RETURNING *
    `,
    [noteClean, mood, cleanTags(tags), attachment_url, id, req.user!.id]
  );
  if (!rows.length) return res.status(404).json({ error: "not found" });
  res.json(rows[0]);
});

// DELETE /journal/:id
journal.delete("/journal/:id", async (req, res) => {
  const { id } = req.params;
  const del = await db.query(`DELETE FROM trader_journal WHERE id=$1 AND user_id=$2 RETURNING id`, [
    id,
    req.user!.id
  ]);
  if (!del.rows.length) return res.status(404).json({ error: "not found" });
  res.json({ success: true });
});

export default journal;
```

## 1.5 Trader Insights (compute + cache + timeframe)

`server/services/insights.ts`

```ts
import db from "../db";

function bucketHoldMinutes(mins?: number | null): string {
  const m = typeof mins === "number" ? mins : NaN;
  if (Number.isNaN(m)) return "unknown";
  if (m < 1) return "<1m";
  if (m < 5) return "1-5m";
  if (m < 15) return "5-15m";
  if (m < 30) return "15-30m";
  if (m < 60) return "30-60m";
  if (m < 180) return "1-3h";
  return "3h+";
}

type Agg = { pnl: number; trades: number; wins: number };
type Ranked = { key: string; pnl: number; trades: number; win_rate: number };

function bestWorst(map: Record<string, Agg>) {
  const arr: Ranked[] = Object.entries(map).map(([k, v]) => ({
    key: k,
    pnl: v.pnl,
    trades: v.trades,
    win_rate: v.trades ? v.wins / v.trades : 0
  }));
  arr.sort((a, b) => b.pnl - a.pnl);
  return { best: arr[0] || null, worst: arr.length ? arr[arr.length - 1] : null, ranked: arr };
}

export async function computeInsights(userId: number, sinceDays = 30) {
  const whereTime = sinceDays ? `AND entry_time >= NOW() - INTERVAL '${sinceDays} days'` : "";
  const q = await db.query(
    `
    SELECT instrument, direction, profit,
           EXTRACT(EPOCH FROM (exit_time - entry_time))/60 AS hold_minutes,
           EXTRACT(HOUR FROM entry_time) AS entry_hour
    FROM trades
    WHERE user_id=$1
      AND exit_time IS NOT NULL
      AND entry_time IS NOT NULL
      ${whereTime}
    `,
    [userId]
  );

  const byInstrument: Record<string, Agg> = {};
  const byDirection: Record<string, Agg> = {};
  const byHour: Record<string, Agg> = {};
  const byHold: Record<string, Agg> = {};
  let trades = 0, wins = 0, pnl = 0;

  for (const t of q.rows as any[]) {
    const inst = t.instrument || "UNKNOWN";
    const dir = t.direction || "UNKNOWN";
    const hour = t.entry_hour != null ? String(t.entry_hour).padStart(2, "0") : "NA";
    const bucket = bucketHoldMinutes(typeof t.hold_minutes === "number" ? t.hold_minutes : Number(t.hold_minutes));

    const p = Number(t.profit || 0);
    trades++; pnl += p; if (p > 0) wins++;

    byInstrument[inst] ??= { pnl: 0, trades: 0, wins: 0 };
    byInstrument[inst].pnl += p; byInstrument[inst].trades++; if (p > 0) byInstrument[inst].wins++;

    byDirection[dir] ??= { pnl: 0, trades: 0, wins: 0 };
    byDirection[dir].pnl += p; byDirection[dir].trades++; if (p > 0) byDirection[dir].wins++;

    byHour[hour] ??= { pnl: 0, trades: 0, wins: 0 };
    byHour[hour].pnl += p; byHour[hour].trades++; if (p > 0) byHour[hour].wins++;

    byHold[bucket] ??= { pnl: 0, trades: 0, wins: 0 };
    byHold[bucket].pnl += p; byHold[bucket].trades++; if (p > 0) byHold[bucket].wins++;
  }

  return {
    window_days: sinceDays,
    totals: { trades, pnl, win_rate: trades ? wins / trades : 0 },
    instruments: bestWorst(byInstrument),
    directions: bestWorst(byDirection),
    hours: bestWorst(byHour),
    hold_time: bestWorst(byHold)
  };
}
```

`server/routes/insights.ts`

```ts
import { Router } from "express";
import db from "../db";
import { computeInsights } from "../services/insights";

const insights = Router();
const CACHE_TTL_MIN = 12 * 60;

insights.get("/insights", async (req, res) => {
  const sinceDays = req.query.since_days ? Number(req.query.since_days) : 30;
  const refresh = String(req.query.refresh || "0") === "1";

  const cached = await db.query<{ insights: any; generated_at: Date }>(
    `SELECT insights, generated_at FROM trader_insights_cache WHERE user_id=$1`,
    [req.user!.id]
  );

  if (!refresh && cached.rows.length) {
    const blob = cached.rows[0].insights as any;
    const ageMin = (Date.now() - new Date(cached.rows[0].generated_at).getTime()) / (1000 * 60);
    if (blob?.window_days === sinceDays && ageMin < CACHE_TTL_MIN) {
      return res.json(blob);
    }
  }

  const out = await computeInsights(req.user!.id, sinceDays);
  await db.query(
    `
    INSERT INTO trader_insights_cache (user_id, insights, generated_at)
    VALUES ($1,$2,NOW())
    ON CONFLICT (user_id) DO UPDATE SET insights=$2, generated_at=NOW()
    `,
    [req.user!.id, out]
  );
  res.json(out);
});

export default insights;
```

```tsx
import { useMemo, useState } from "react";
import api from "../lib/api";

export default function ImpersonationBanner() {
  const [busy, setBusy] = useState(false);

  const target = useMemo(() => {
    try { return JSON.parse(localStorage.getItem("impersonated_target") || "null"); }
    catch { return null; }
  }, []);

  const isImpersonating = !!target;

  const exit = async () => {
    const adminToken = localStorage.getItem("admin_token");
    if (!adminToken || !target?.id) {
      localStorage.removeItem("impersonated_target");
      return (window.location.href = "/admin");
    }
    setBusy(true);
    try {
      // temporarily set admin token to log stop-event
      localStorage.setItem("token", adminToken);
      await api.post("/admin/view-as/stop", { targetUserId: target.id });
      localStorage.removeItem("impersonated_target");
      localStorage.removeItem("admin_token");
      window.location.href = "/admin";
    } finally {
      setBusy(false);
    }
  };

  if (!isImpersonating) return null;

  return (
    <div style={{
      position: "sticky", top: 0, zIndex: 9999, padding: "10px 12px",
      borderBottom: "1px solid #333", background: "rgba(255,200,0,0.08)"
    }}>
      <strong>VIEW AS ACTIVE</strong>
      <span style={{ marginLeft: 10, opacity: 0.9 }}>
        {target?.username} (id: {target?.id}, {target?.email})
      </span>
      <button style={{ float: "right" }} onClick={exit} disabled={busy}>
        {busy ? "Exiting..." : "Exit View As"}
      </button>
    </div>
  );
}
```

```tsx
import { useEffect, useMemo, useState } from "react";
import api from "../../lib/api";

type UserLite = { id: number; username: string; email: string; role: string; status?: string };

export default function ViewAsTab() {
  const [q, setQ] = useState("");
  const [results, setResults] = useState<UserLite[]>([]);
  const [selected, setSelected] = useState<UserLite | null>(null);
  const [busy, setBusy] = useState(false);

  const canSearch = useMemo(() => q.trim().length >= 2, [q]);

  useEffect(() => {
    let dead = false;
    const run = async () => {
      if (!canSearch) { setResults([]); return; }
      const res = await api.get(`/admin/users/search?q=${encodeURIComponent(q.trim())}`);
      if (!dead) setResults(res.data || []);
    };
    run();
    return () => { dead = true; };
  }, [q, canSearch]);

  const startViewAs = async () => {
    if (!selected?.id) return;
    setBusy(true);
    try {
      const adminToken = localStorage.getItem("token");
      if (adminToken) localStorage.setItem("admin_token", adminToken);

      const res = await api.post("/admin/view-as/start", { targetUserId: selected.id });
      localStorage.setItem("token", res.data.token);
      localStorage.setItem("impersonated_target", JSON.stringify(res.data.target));
      window.location.href = "/dashboard?impersonated=1";
    } finally {
      setBusy(false);
    }
  };

  return (
    <div className="card">
      <h3>View As</h3>
      <p>Search by ID, username, or email. Impersonation expires in 15 minutes.</p>
      <input
        value={q}
        onChange={(e) => setQ(e.target.value)}
        placeholder="Search user..."
        style={{ width: "100%", marginBottom: 8 }}
      />
      <div style={{ maxHeight: 280, overflow: "auto" }}>
        {results.map(u => (
          <div key={u.id}
               onClick={() => setSelected(u)}
               style={{
                 padding: 8, border: "1px solid #333", marginBottom: 6, cursor: "pointer",
                 background: selected?.id === u.id ? "rgba(255,255,255,0.06)" : "transparent"
               }}>
            <div><strong>{u.username}</strong> (id: {u.id})</div>
            <div style={{ opacity: .8 }}>{u.email}</div>
            <div style={{ opacity: .6 }}>role: {u.role} | status: {u.status ?? "-"}</div>
          </div>
        ))}
      </div>
      <button disabled={!selected || busy} onClick={startViewAs} style={{ marginTop: 8 }}>
        {busy ? "Starting..." : "View As Selected User"}
      </button>
    </div>
  );
}
```

```tsx
import { useEffect, useState } from "react";
import api from "../lib/api";
import { Link } from "wouter";

type Entry = {
  id: number; user_id: number; trade_id?: number | null;
  note: string; mood?: string | null; tags?: string[]; attachment_url?: string | null;
  created_at: string; updated_at: string;
};

export default function TraderJournal() {
  const [entries, setEntries] = useState<Entry[]>([]);
  const [note, setNote] = useState("");
  const [mood, setMood] = useState("");
  const [tags, setTags] = useState("");
  const [attachmentUrl, setAttachmentUrl] = useState("");
  const [busy, setBusy] = useState(false);

  const load = async () => {
    const res = await api.get("/journal");
    setEntries(res.data || []);
  };

  useEffect(() => { load(); }, []);

  const add = async () => {
    if (note.trim().length < 3) return;

      await api.post("/journal", {
        note: note.trim(),
        mood: mood || null,
        tags: tags.split(",").map(s => s.trim()).filter(Boolean),
        attachment_url: attachmentUrl || null
      });
      setNote(""); setMood(""); setTags(""); setAttachmentUrl("");
      await load();
    } finally { setBusy(false); }
  };

  const remove = async (id: number) => {
    await api.delete(`/journal/${id}`);
    await load();
  };

  return (
    <div style={{ padding: 16 }}>
      <h2>Trader Journal</h2>

      <div className="card" style={{ marginBottom: 14 }}>
        <textarea
          value={note}
          onChange={(e) => setNote(e.target.value)}
          rows={6}
          placeholder="Reasoning, emotions, plan vs execution, post-mortem..."
          style={{ width: "100%" }}
        />
        <div style={{ display: "flex", gap: 8, marginTop: 8 }}>
          <input value={mood} onChange={e => setMood(e.target.value)} placeholder="Mood (optional)" />
          <input value={tags} onChange={e => setTags(e.target.value)} placeholder="Tags (comma separated)" />
          <input value={attachmentUrl} onChange={e => setAttachmentUrl(e.target.value)} placeholder="Attachment URL" />
        </div>
        <button onClick={add} disabled={busy || note.trim().length < 3} style={{ marginTop: 8 }}>
          {busy ? "Saving..." : "Add Entry"}
        </button>
      </div>

      {entries.map(e => (
        <div key={e.id} className="card" style={{ marginBottom: 10 }}>
          <div style={{ opacity: .75 }}>
            {new Date(e.created_at).toLocaleString()}
            {e.updated_at !== e.created_at ? " (edited)" : ""}
          </div>
          <div style={{ marginTop: 8, whiteSpace: "pre-wrap" }}>{e.note}</div>
          <div style={{ marginTop: 8, opacity: .85 }}>
            {e.mood ? <span>Mood: {e.mood} </span> : null}
            {e.tags?.length ? <span>Tags: {e.tags.join(", ")} </span> : null}
            {e.attachment_url ? <span>Attachment: {e.attachment_url}</span> : null}
          </div>
          <button onClick={() => remove(e.id)} style={{ marginTop: 8 }}>Delete</button>
        </div>
      ))}

      <div style={{ marginTop: 24 }}>
        <Link href="/account">← Back to Account</Link>
      </div>
    </div>
  );
}
```

## 2.4 Trader Insights card

`client/src/components/account/TraderInsightsCard.tsx`

```tsx
import { useEffect, useState } from "react";
import api from "../../lib/api";

type Ranked = { key: string; pnl: number; trades: number; win_rate: number };
type Section = { best: Ranked | null; worst: Ranked | null; ranked: Ranked[] };

type Insights = {
  window_days: number;
  totals: { trades: number; pnl: number; win_rate: number };
  instruments: Section;
  directions: Section;
  hours: Section;
  hold_time: Section;
};

function pct(x?: number) { return x == null ? "-" : `${(x * 100).toFixed(1)}%`; }
function num(x?: number) { return x == null ? "-" : Number(x).toFixed(2); }

export default function TraderInsightsCard() {
  const [data, setData] = useState<Insights | null>(null);
  const [sinceDays, setSinceDays] = useState(30);

  const load = async (refresh = false) => {
    const res = await api.get(`/insights?since_days=${sinceDays}&refresh=${refresh ? 1 : 0}`);
    setData(res.data);
  };

  useEffect(() => { load(false); }, [sinceDays]);

  if (!data) return null;
  const bestInst = data.instruments?.best;
  const worstInst = data.instruments?.worst;
  const bestDir = data.directions?.best;
  const bestHour = data.hours?.best;
  const bestHold = data.hold_time?.best;

  return (
    <div className="card" style={{ marginTop: 12 }}>
      <div style={{ display: "flex", justifyContent: "space-between", gap: 12 }}>
        <h3 style={{ margin: 0 }}>Trader Insights</h3>
        <div style={{ display: "flex", gap: 8 }}>
          <select value={sinceDays} onChange={e => setSinceDays(Number(e.target.value))}>
            <option value={7}>Last 7d</option>
            <option value={30}>Last 30d</option>
            <option value={90}>Last 90d</option>
          </select>
          <button onClick={() => load(true)}>Refresh</button>
        </div>
      </div>

      <div style={{ marginTop: 10, opacity: .9 }}>
        Trades: <strong>{data.totals.trades}</strong> |
        Win rate: <strong>{pct(data.totals.win_rate)}</strong> |
        PnL: <strong>{num(data.totals.pnl)}</strong>
      </div>

      <div style={{ marginTop: 12 }}>
        <div><strong>Most profitable instrument:</strong> {bestInst ? `${bestInst.key} (PnL ${num(bestInst.pnl)}, WR ${pct(bestInst.win_rate)})` : "-"}</div>
        <div><strong>Most loss-making instrument:</strong> {worstInst ? `${worstInst.key} (PnL ${num(worstInst.pnl)}, WR ${pct(worstInst.win_rate)})` : "-"}</div>
        <div><strong>Best direction:</strong> {bestDir ? `${bestDir.key} (PnL ${num(bestDir.pnl)}, WR ${pct(bestDir.win_rate)})` : "-"}</div>
        <div><strong>Most profitable hour:</strong> {bestHour ? `${bestHour.key}:00 (PnL ${num(bestHour.pnl)}, WR ${pct(bestHour.win_rate)})` : "-"}</div>
        <div><strong>Optimal hold-time bucket:</strong> {bestHold ? `${bestHold.key} (PnL ${num(bestHold.pnl)}, WR ${pct(bestHold.win_rate)})` : "-"}</div>
      </div>
    </div>
  );
}
```

Place this **below your Trading Statistics card** on the Account page component.

# 3) Notes on integration

* **User Management mini-tab**: wherever you render user-admin tabs, add a new tab labeled “View As” and place `<ViewAsTab/>` in that panel.
* **Impersonation safety**: The global middleware blocks **writes** while impersonating. If you require narrow exceptions (e.g., allow “support-notes/create”), wrap those endpoints before the middleware or add an allow-list check inside `impersonationWriteBlock`.
* **Caching**: Insights cache TTL is 12h. Users can force refresh via the card. You can parametrize TTL via env if desired.
* **Audit**: Start/Stop events log admin id, target id, IP, user-agent, and request id (if sent by your proxy). Extend `metadata` with more fields as needed.
