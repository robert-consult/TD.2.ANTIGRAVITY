import React, { useState } from 'react';
import { ShieldAlert, CheckCircle2, AlertTriangle, Info, Zap } from 'lucide-react';

const AlertProvider = ({ response }) => {
  const [showDetails, setShowDetails] = useState(false);

  // 1. Logic Mapping: Converting technical codes to UI States
  const getAlertConfig = (code) => {
    switch (code) {
      case 'IMPERSONATION_WRITE_BLOCK':
        return {
          title: 'Action Restricted',
          message: 'You are in "View Only" mode. Exit to place trades.',
          type: 'error',
          icon: <ShieldAlert className="w-6 h-6 text-red-500" />,
          glow: 'shadow-[0_0_15px_rgba(239,68,68,0.3)]',
          bg: 'bg-red-500/10',
          border: 'border-red-500/20'
        };
      case 'TRADE_SUCCESS':
        return {
          title: 'Order Executed',
          message: 'Your market order has been filled successfully.',
          type: 'success',
          icon: <CheckCircle2 className="w-6 h-6 text-green-500" />,
          glow: 'shadow-[0_0_15px_rgba(34,197,94,0.3)]',
          bg: 'bg-green-500/10',
          border: 'border-green-500/20'
        };
      default:
        return {
          title: 'System Notification',
          message: 'An unexpected update occurred.',
          type: 'info',
          icon: <Info className="w-6 h-6 text-blue-500" />,
          glow: 'shadow-[0_0_15px_rgba(59,130,246,0.3)]',
          bg: 'bg-blue-500/10',
          border: 'border-blue-500/20'
        };
    }
  };

  const config = getAlertConfig(response.code);

  return (
    <div className={`max-w-md w-full glass-card rounded-2xl p-4 border ${config.border} ${config.glow} transition-all duration-300`}>
      <div className="flex items-start">
        {/* Icon with Dynamic Glow */}
        <div className={`p-3 rounded-xl ${config.bg} flex-shrink-0`}>
          {config.icon}
        </div>

        <div className="ml-4 flex-1">
          <div className="flex justify-between items-center">
            <h3 className="text-white font-bold text-sm tracking-wide uppercase italic">
              {config.title}
            </h3>
            <button className="text-gray-500 hover:text-white transition-colors">
              <span className="text-lg">×</span>
            </button>
          </div>
          
          <p className="text-gray-400 text-sm mt-1 leading-relaxed">
            {config.message}
          </p>

          {/* Technical Details Toggle (For Devs/Power Users) */}
          <div className="mt-4">
            <button 
              onClick={() => setShowDetails(!showDetails)}
              className="text-[10px] text-gray-600 hover:text-gray-400 font-mono tracking-tighter uppercase"
            >
              {showDetails ? '[-] Hide Logs' : '[+] View Raw Payload'}
            </button>
            
            {showDetails && (
              <pre className="mt-2 p-2 bg-black/50 rounded-lg text-[10px] text-gray-500 font-mono overflow-x-auto border border-white/5">
                {JSON.stringify(response, null, 2)}
              </pre>
            )}
          </div>
        </div>
      </div>
      
      {/* Action Button: Only shows for specific errors */}
      {response.code === 'IMPERSONATION_WRITE_BLOCK' && (
        <button className="w-full mt-4 py-2 bg-red-500 hover:bg-red-600 text-white text-xs font-bold rounded-lg transition-all shadow-lg shadow-red-500/20">
          EXIT VIEW MODE

1. The "Spring & Slide" Animation Logic
Copy this component code. It wraps your alert and handles the entry/exit physics.

JavaScript
import { motion, AnimatePresence } from 'framer-motion';

const PopupWrapper = ({ isVisible, children }) => {
  return (
    <AnimatePresence>
      {isVisible && (
        <motion.div
          // 1. Starting Position (Off-screen & Invisible)
          initial={{ opacity: 0, y: -20, scale: 0.95 }}
          
          // 2. Entry Animation (The "Spring" Pop)
          animate={{ 
            opacity: 1, 
            y: 0, 
            scale: 1,
            transition: { 
              type: "spring", 
              stiffness: 300, 
              damping: 20 
            } 
          }}
          
          // 3. Exit Animation (Glide out)
          exit={{ 
            opacity: 0, 
            scale: 0.9, 
            transition: { duration: 0.2 } 
          }}
          
          // 4. Interaction (Subtle lift on hover)
          whileHover={{ y: -2 }}
          className="fixed top-6 right-6 z-[9999] cursor-pointer"
        >
          {children}
        </motion.div>
      )}
    </AnimatePresence>
  );
};
