## 2.1 Add `/verify-email` page and route (fix Gap A)

### **Create** `client/src/pages/VerifyEmail.tsx`

```tsx
import { useEffect, useMemo, useState } from "react";
import { Link, useLocation } from "wouter";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";

function getTokenFromUrl(): string | null {
  const url = new URL(window.location.href);
  return url.searchParams.get("token");
}

export default function VerifyEmail() {
  const [, setLocation] = useLocation();
  const token = useMemo(() => getTokenFromUrl(), []);
  const [status, setStatus] = useState<"idle" | "verifying" | "success" | "error">("idle");
  const [message, setMessage] = useState<string>("");

  useEffect(() => {
    const run = async () => {
      if (!token) {
        setStatus("error");
        setMessage("Missing verification token.");
        return;
      }

      setStatus("verifying");
      try {
        const res = await fetch("/api/verification/email/verify", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ token }),
          credentials: "include",
        });

        const data = await res.json().catch(() => ({}));
        if (!res.ok) {
          setStatus("error");
          setMessage(data?.message || "Verification failed. The token may be expired or already used.");
          return;
        }

        setStatus("success");
        setMessage("Your email has been verified successfully.");
      } catch (e: any) {
        setStatus("error");
        setMessage(e?.message || "Verification failed due to a network error.");
      }
    };

    run();
  }, [token]);

  return (
    <div className="min-h-screen flex items-center justify-center bg-neutral-900 p-4">
      <Card className="w-full max-w-md bg-neutral-800 border-gray-700">
        <CardHeader>
          <CardTitle>Email Verification</CardTitle>
        </CardHeader>
        <CardContent className="space-y-4">
          {status === "verifying" && (
            <div className="text-sm text-gray-300">Verifying your email…</div>
          )}

          {status === "success" && (
            <>
              <div className="text-sm text-green-400">{message}</div>
              <div className="flex gap-2">
                <Button onClick={() => setLocation("/")}>Go to Dashboard</Button>
                <Link href="/profile" className="text-sm text-gray-300 underline self-center">
                  Profile Settings
                </Link>
              </div>
            </>
          )}

          {status === "error" && (
            <>
              <div className="text-sm text-red-400">{message}</div>
              <div className="flex gap-2">
                <Button variant="outline" onClick={() => setLocation("/profile")}>
                  Open Profile Settings
                </Button>
                <Button variant="ghost" onClick={() => setLocation("/")}>
                  Home
                </Button>
              </div>
            </>
          )}

          {status === "idle" && (
            <div className="text-sm text-gray-300">Waiting for token…</div>
          )}
        </CardContent>
      </Card>
    </div>
  );
}
```

export type PolicyDecision = {
  action: PolicyAction;
  allowed: boolean;
  deny_code?: DenyCode;

  accountState: AccountState;
  contenderEligible: boolean;
  contenderPath: "PATH1" | "PATH2" | null;

  // IMPORTANT: top-level fields (required by your server middleware + verification routes)
  showLockedBanner?: boolean;
  redirectTo?: "/verify-email" | "/profile" | "/support" | "/kyc";
};

export function resolveAccountState(ctx: DecisionContext, cfg: PolicyConfig): AccountState {
  if (ctx.user.suspendedAt) return "SUSPENDED_ADMIN";

  if (!ctx.user.emailVerifiedAt) {
    return (ctx.now < ctx.user.emailInitialDueAt)
      ? "ACTIVE_UNVERIFIED_GRACE"
      : "LOCKED_EMAIL_UNVERIFIED";
  }

  if (ctx.user.emailReverifyDueAt) {
    const overdueAt = new Date(ctx.user.emailReverifyDueAt.getTime() + cfg.emailReverifyOverdueGraceDays * 86400000);
    if (ctx.now >= overdueAt) return "LOCKED_EMAIL_REVERIFY_OVERDUE";
  }

  return "ACTIVE_VERIFIED";
}

export function computeContenderEligibility(ctx: DecisionContext, cfg: PolicyConfig): {
  eligible: boolean;
  path: "PATH1" | "PATH2" | null;
} {
  const state = resolveAccountState(ctx, cfg);
  if (state !== "ACTIVE_VERIFIED") return { eligible: false, path: null };

  const m = ctx.metrics;

  const path1 =
    m.accountAgeDays >= cfg.contenderPath1.minAccountAgeDays &&
    m.balancePctOfStart >= cfg.contenderPath1.minBalancePctOfStart &&
    m.tradesLifetime >= cfg.contenderPath1.minTradesLifetime;

  const path2 =
    m.accountAgeDays >= cfg.contenderPath2.minAccountAgeDays &&
    (m.lastTradeDaysAgo ?? 999999) <= cfg.contenderPath2.lastTradeWithinDays &&
    m.returnLast90d >= cfg.contenderPath2.minReturnLast90d &&
    m.tradesLast90d >= cfg.contenderPath2.minTradesLast90d;

  if (path1) return { eligible: true, path: "PATH1" };
  if (path2) return { eligible: true, path: "PATH2" };
  return { eligible: false, path: null };
}

function isExposureIncreasingTradingAction(action: PolicyAction): boolean {
  return action === "TRADE_OPEN_OR_INCREASE" || action === "TRADE_PLACE_PENDING";
}

function isReduceOnlyTradingAction(action: PolicyAction): boolean {
  return action === "TRADE_CLOSE_OR_REDUCE" || action === "TRADE_CANCEL_PENDING";
}

export function decidePolicy(
  action: PolicyAction,
  ctx: DecisionContext,
  cfg: PolicyConfig = DEFAULT_POLICY_CONFIG
): PolicyDecision {
  const accountState = resolveAccountState(ctx, cfg);
  const contender = computeContenderEligibility(ctx, cfg);

  const deny = (deny_code: DenyCode, redirectTo?: PolicyDecision["redirectTo"], showLockedBanner?: boolean): PolicyDecision => ({
    action,
    allowed: false,
    deny_code,
    accountState,
    contenderEligible: contender.eligible,
    contenderPath: contender.path,
    redirectTo,
    showLockedBanner,
  });

  const allow = (): PolicyDecision => ({
    action,
    allowed: true,
    accountState,
    contenderEligible: contender.eligible,
    contenderPath: contender.path,
  });

  if (accountState === "SUSPENDED_ADMIN") {
    return deny("ACCOUNT_SUSPENDED", "/support", true);
  }

  if (accountState === "LOCKED_EMAIL_UNVERIFIED") {
    if (isExposureIncreasingTradingAction(action) || action === "TRADE_MODIFY_SLTP") {
      return deny("EMAIL_NOT_VERIFIED", "/verify-email", true);
    }
    if (isReduceOnlyTradingAction(action)) {
      if (action === "TRADE_CANCEL_PENDING" && !cfg.allowCancelPendingWhenLocked) {
        return deny("ACTION_NOT_ALLOWED_IN_STATE", "/verify-email", true);
      }
      return cfg.allowReduceOnlyWhenLocked ? allow() : deny("ACTION_NOT_ALLOWED_IN_STATE", "/verify-email", true);
    }
  }

  if (accountState === "LOCKED_EMAIL_REVERIFY_OVERDUE") {
    if (isExposureIncreasingTradingAction(action) || action === "TRADE_MODIFY_SLTP") {
      return deny("EMAIL_REVERIFY_OVERDUE", "/verify-email", true);
    }
    if (isReduceOnlyTradingAction(action)) {
      if (action === "TRADE_CANCEL_PENDING" && !cfg.allowCancelPendingWhenLocked) {
        return deny("ACTION_NOT_ALLOWED_IN_STATE", "/verify-email", true);
      }
      return cfg.allowReduceOnlyWhenLocked ? allow() : deny("ACTION_NOT_ALLOWED_IN_STATE", "/verify-email", true);
    }
  }

  if (action === "EMAIL_RESEND_VERIFICATION") {
    const nowMs = ctx.now.getTime();
    const last = ctx.throttle.emailLastSentAtMs ?? 0;
    const cooldownMs = cfg.emailResendCooldownSec * 1000;

    if (ctx.throttle.emailSendCountDay >= cfg.emailDailySendCap) return deny("EMAIL_SEND_RATE_LIMIT");
    if (nowMs - last < cooldownMs) return deny("EMAIL_SEND_COOLDOWN");
    return allow();
  }

  if (action === "PHONE_VERIFY_START" || action === "PHONE_VERIFY_CONFIRM") {
    if (accountState !== "ACTIVE_VERIFIED") return deny("SMS_NOT_ELIGIBLE_TIER", "/verify-email", true);

    const eligible =
      contender.eligible ||
      ctx.user.contenderTier === "CANDIDATE_SMS_REQUIRED" ||
      ctx.user.contenderTier === "VERIFIED_SMS" ||
      ctx.user.contenderTier === "SELECTED_REAL_CAPITAL" ||
      ctx.user.userTier === "PERFORMER" ||
      ctx.user.userTier === "SELECTED";

    if (!eligible) return deny("SMS_NOT_ELIGIBLE_TIER");

    if (action === "PHONE_VERIFY_START") {
      const nowMs = ctx.now.getTime();
      const last = ctx.throttle.smsLastSentAtMs ?? 0;
      if (ctx.throttle.smsSendCountDay >= cfg.smsDailySendCap) return deny("SMS_RATE_LIMIT");
      if (nowMs - last < 60_000) return deny("SMS_RATE_LIMIT");
    }

    if (action === "PHONE_VERIFY_CONFIRM") {
      if (ctx.throttle.otpFailCount >= cfg.otpMaxAttempts) return deny("SMS_OTP_TOO_MANY_ATTEMPTS");
    }

    return allow();
  }

  if (action === "KYC_VIEW" || action === "KYC_SUBMIT" || action === "PREFERRED_PAYMENT_CURRENCY_SET") {
    const selected =
      ctx.user.userTier === "SELECTED" ||
      ctx.user.contenderTier === "SELECTED_REAL_CAPITAL" ||
      !!ctx.user.selectedAt ||
      ["INVITED", "SUBMITTED", "APPROVED"].includes(ctx.kyc.status);

    if (!selected) return deny("KYC_NOT_SELECTED", "/profile");

    if (action === "KYC_SUBMIT") {
      if (!(ctx.kyc.status === "INVITED" || ctx.kyc.status === "REJECTED")) {
        return deny("KYC_STATE_INVALID");
      }
    }

    return allow();
  }

  // Default trade permissions
  if (action === "TRADE_OPEN_OR_INCREASE" || action === "TRADE_PLACE_PENDING") return allow();
  if (action === "TRADE_CLOSE_OR_REDUCE" || action === "TRADE_CANCEL_PENDING") return allow();
  if (action === "TRADE_MODIFY_SLTP") return allow(); // active states; locked handled above

  return allow();
}

export function featureGates(ctx: DecisionContext, cfg: PolicyConfig = DEFAULT_POLICY_CONFIG) {
  const st = resolveAccountState(ctx, cfg);
  const contender = computeContenderEligibility(ctx, cfg);

  const selected =
    ctx.user.userTier === "SELECTED" ||
    ctx.user.contenderTier === "SELECTED_REAL_CAPITAL" ||
    !!ctx.user.selectedAt ||
    ["INVITED", "SUBMITTED", "APPROVED"].includes(ctx.kyc.status);

  const canRequestPayout = selected && ctx.kyc.status === "APPROVED";

  return {
    accountState: st,
    contenderEligible: contender.eligible,
    canTradeOpenOrIncrease: decidePolicy("TRADE_OPEN_OR_INCREASE", ctx, cfg).allowed,
    canTradeCloseOrReduce: decidePolicy("TRADE_CLOSE_OR_REDUCE", ctx, cfg).allowed,
    canStartSms: decidePolicy("PHONE_VERIFY_START", ctx, cfg).allowed,
    canViewKyc: selected && decidePolicy("KYC_VIEW", ctx, cfg).allowed,
    canSetPreferredPaymentCurrency: selected && decidePolicy("PREFERRED_PAYMENT_CURRENCY_SET", ctx, cfg).allowed,
    canRequestPayout,
  };
}
