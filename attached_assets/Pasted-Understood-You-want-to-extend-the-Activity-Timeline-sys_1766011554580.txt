#### `/backend/routes/activity.js`

Add login/logout tracking + timeline fetch/export.

```js
const express = require('express');
const router = express.Router();
const ActivityLog = require('../models/ActivityLog');
const json2csv = require('json2csv').parse;

// Log account or login events
router.post('/log', async (req, res) => {
  try {
    const { userEmail, type, title, description, ipAddress, loginTime, logoutTime } = req.body;

    let sessionLength = null;
    if (loginTime && logoutTime) {
      sessionLength = Math.round((new Date(logoutTime) - new Date(loginTime)) / 1000);
    }

    const log = new ActivityLog({
      userEmail,
      type,
      title,
      description,
      ipAddress,
      loginTime,
      logoutTime,
      sessionLength
    });

    await log.save();
    res.status(200).json({ message: 'Activity logged successfully' });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// Fetch timeline for a user
router.get('/timeline/:email', async (req, res) => {
  const email = req.params.email;
  const logs = await ActivityLog.find({ userEmail: email }).sort({ timestamp: -1 });
  res.json(logs);
});

// Export CSV for timeline
router.get('/timeline/:email/export', async (req, res) => {
  const email = req.params.email;
  const logs = await ActivityLog.find({ userEmail: email }).sort({ timestamp: -1 });
  const fields = ['timestamp', 'type', 'title', 'description', 'loginTime', 'logoutTime', 'sessionLength', 'ipAddress'];
  const csv = json2csv(logs, { fields });
  res.header('Content-Type', 'text/csv');
  res.attachment(`${email}-activity.csv`);
  res.send(csv);
});

module.exports = router;
```

### ✅ **FRONTEND ENHANCEMENTS**

#### `/frontend/src/components/ActivityTimeline.jsx`

Update timeline UI to include **login/logout events** and **session durations**, with enhanced export.

```jsx
import React, { useEffect, useState } from 'react';
import axios from 'axios';
import { Button } from '@/components/ui/button';
import { format } from 'date-fns';

export default function ActivityTimeline({ userEmail, onClose }) {
  const [logs, setLogs] = useState([]);

  useEffect(() => {
    axios.get(`/api/activity/timeline/${userEmail}`).then(res => {
      setLogs(res.data);
    });
  }, [userEmail]);

  const handleExport = () => {
    window.open(`/api/activity/timeline/${userEmail}/export`, '_blank');
  };

  const renderIcon = (type) => {
    switch (type) {
      case 'LOGIN':
        return <span className="text-green-400">LOGIN</span>;
      case 'LOGOUT':
        return <span className="text-yellow-400">LOGOUT</span>;
      case 'FREEZE':
        return <span className="text-orange-500">FREEZE</span>;
      case 'UNFREEZE':
        return <span className="text-cyan-500">UNFREEZE</span>;
      default:
        return <span className="text-purple-400">{type}</span>;
    }
  };

  return (
    <div className="bg-gray-900 p-4 rounded-xl shadow-lg text-white w-full max-w-2xl">
      <h2 className="text-lg font-bold mb-4">Activity Timeline: {userEmail}</h2>

      <div className="relative border-l border-gray-700 pl-6 space-y-3 max-h-[60vh] overflow-y-auto">
        {logs.map((log, idx) => (
          <div key={idx} className="relative">
            <div className="absolute -left-2 top-2 w-3 h-3 bg-purple-500 rounded-full"></div>
            <div className="p-3 bg-gray-800 rounded-lg hover:bg-gray-700 transition">
              <div className="flex justify-between">
                <div className="flex items-center gap-2">
                  {renderIcon(log.type)}
                  <span className="font-semibold">{log.title}</span>
                </div>
                <span className="text-xs text-gray-400">
                  {format(new Date(log.timestamp), 'MM/dd/yyyy, h:mm:ss a')}
                </span>
              </div>
              <p className="text-sm text-gray-400">{log.description}</p>

              {log.type === 'LOGIN' && (
                <p className="text-xs text-gray-500 mt-1">
                  Login IP: {log.ipAddress || 'N/A'}
                </p>
              )}

              {log.type === 'LOGOUT' && (
                <p className="text-xs text-gray-500 mt-1">
                  Session Length: {log.sessionLength
                    ? `${Math.floor(log.sessionLength / 60)}m ${log.sessionLength % 60}s`
                    : 'Unknown'}
                </p>
              )}
            </div>
          </div>
        ))}
      </div>

      <div className="mt-4 flex justify-end gap-2">
        <Button onClick={handleExport}>Export CSV</Button>
        <Button variant="outline" onClick={onClose}>Close</Button>
      </div>
    </div>
  );
}
```

### ✅ **SESSION TRACKING HOOK**

Track user login/logout automatically.

#### `/frontend/src/hooks/useSessionTracker.js`

```js
import { useEffect } from 'react';
import axios from 'axios';

export default function useSessionTracker(userEmail) {
  useEffect(() => {
    if (!userEmail) return;
    const loginTime = new Date().toISOString();
    const ip = window.localStorage.getItem('userIP') || 'unknown';

    axios.post('/api/activity/log', {
      userEmail,
      type: 'LOGIN',
      title: 'User login',
      description: 'User session started',
      ipAddress: ip,
      loginTime
    });

    const handleBeforeUnload = () => {
      const logoutTime = new Date().toISOString();
      axios.post('/api/activity/log', {
        userEmail,
        type: 'LOGOUT',
        title: 'User logout',
        description: 'User session ended',
        ipAddress: ip,
        loginTime,
        logoutTime
      });
    };

    window.addEventListener('beforeunload', handleBeforeUnload);
    return () => window.removeEventListener('beforeunload', handleBeforeUnload);
  }, [userEmail]);
}
```

Usage in your main dashboard or layout:

```jsx
import useSessionTracker from '@/hooks/useSessionTracker';

function Dashboard({ user }) {
  useSessionTracker(user.email);
  return <MainDashboard />;
}
```
