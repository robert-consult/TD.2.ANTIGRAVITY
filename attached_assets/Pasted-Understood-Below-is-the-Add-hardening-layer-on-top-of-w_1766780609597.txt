```ts
function ensureColumn(table: string, column: string, ddl: string) {
  const cols = sqlite.prepare(`PRAGMA table_info(${table})`).all() as any[];
  const exists = cols.some((c) => String(c.name).toLowerCase() === column.toLowerCase());
  if (!exists) sqlite.exec(`ALTER TABLE ${table} ADD COLUMN ${ddl}`);
}
```

## 1.2 `server/grift/griftAdminAudit.ts` (NEW)

Hash-chained admin actions for tamper evidence.

```ts
// server/grift/griftAdminAudit.ts
import { sqlite } from "@db";
import crypto from "crypto";

function sha256Hex(s: string) {
  return crypto.createHash("sha256").update(s).digest("hex");
}

export function logGriftAdminAction(args: {
  adminId: number | null;
  action: string;      // GRIFT_*
  entityType: "SIGNAL" | "CASE" | "CONFIG" | "USER";
  entityId: string;
  payload: any;
}) {
  const ts = Date.now();
  const payloadJson = JSON.stringify(args.payload ?? {});

  const prev = sqlite.prepare(`
    SELECT event_hash as h
    FROM grift_admin_actions
    ORDER BY id DESC
    LIMIT 1
  `).get() as any;

  const prevHash = prev?.h ?? null;

  const material = [
    prevHash ?? "",
    ts,
    args.adminId ?? "",
    args.action,
    args.entityType,
    args.entityId,
    payloadJson,
  ].join("|");

  const eventHash = sha256Hex(material);

  sqlite.prepare(`
    INSERT INTO grift_admin_actions (
      ts, admin_id, action, entity_type, entity_id, payload_json, prev_hash, event_hash
    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?)
  `).run(
    ts,
    args.adminId ?? null,
    args.action,
    args.entityType,
    args.entityId,
    payloadJson,
    prevHash,
    eventHash
  );

  return { ts, prevHash, eventHash };
}

export function listGriftAdminActions(limit = 200) {
  const lim = Math.max(10, Math.min(2000, limit));
  return sqlite.prepare(`
    SELECT *
    FROM grift_admin_actions
    ORDER BY ts DESC
    LIMIT ?
  `).all(lim);
}
```

## 1.4 `server/grift/griftScheduler.ts` (NEW)

Batch recompute periodically (safe and lightweight: only active users with recent observations).

```ts
// server/grift/griftScheduler.ts
import { sqlite } from "@db";
import { griftEngine } from "./griftEngine";
import { logGriftAdminAction } from "./griftAdminAudit";

export function startGriftScheduler(opts?: { intervalMinutes?: number; lookbackHours?: number }) {
  const intervalMinutes = Math.max(2, Math.min(60, opts?.intervalMinutes ?? 10));
  const lookbackHours = Math.max(1, Math.min(168, opts?.lookbackHours ?? 24));

  const runOnce = () => {
    const since = Date.now() - lookbackHours * 60 * 60 * 1000;

    const users = sqlite.prepare(`
      SELECT DISTINCT user_id as userId
      FROM grift_observations
      WHERE ts >= ?
      LIMIT 5000
    `).all(since) as any[];

    let ok = 0;
    for (const u of users) {
      try {
        griftEngine.recomputeUser(Number(u.userId));
        ok++;
      } catch {
        // swallow to avoid scheduler death
      }
    }

    // Record scheduler run as audit event (adminId null)
    logGriftAdminAction({
      adminId: null,
      action: "GRIFT_SCHEDULER_RECOMPUTE",
      entityType: "CONFIG",
      entityId: "grift",
      payload: { intervalMinutes, lookbackHours, usersConsidered: users.length, usersRecomputed: ok },
    });
  };

  // Run immediately, then interval
  runOnce();
  const handle = setInterval(runOnce, intervalMinutes * 60 * 1000);

  return () => clearInterval(handle);
}
```

```ts
function csvEscape(v: any) {
  if (v === null || v === undefined) return "";
  const s = typeof v === "string" ? v : JSON.stringify(v);
  if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`;
  return s;
}

function sendCsv(res: any, filename: string, rows: any[]) {
  const cols = Array.from(
    rows.reduce((set: Set<string>, r: any) => {
      Object.keys(r || {}).forEach((k) => set.add(k));
      return set;
    }, new Set<string>())
  );

  const header = cols.join(",");
  const lines = rows.map((r) => cols.map((c) => csvEscape(r[c])).join(","));
  const csv = [header, ...lines].join("\n");

  res.setHeader("Content-Type", "text/csv");
  res.setHeader("Content-Disposition", `attachment; filename="${filename}"`);
  res.send(csv);
}

    if (action === "FREEZE") {
      if (hasCol(userCols, "isFrozen")) {
        sqlite.prepare(`UPDATE users SET isFrozen=1 WHERE id=?`).run(userId);
        applied = true;
        note = "Updated users.isFrozen=1";
      } else if (hasCol(userCols, "frozen")) {
        sqlite.prepare(`UPDATE users SET frozen=1 WHERE id=?`).run(userId);
        applied = true;
        note = "Updated users.frozen=1";
      }
      // optional reason columns
      if (applied && hasCol(userCols, "frozenReason")) {
        sqlite.prepare(`UPDATE users SET frozenReason=? WHERE id=?`).run(reason, userId);
      }
      if (applied && hasCol(userCols, "frozenAt")) {
        sqlite.prepare(`UPDATE users SET frozenAt=? WHERE id=?`).run(ts, userId);
      }
    } else if (action === "DISABLE") {
      if (hasCol(userCols, "isDisabled")) {
        sqlite.prepare(`UPDATE users SET isDisabled=1 WHERE id=?`).run(userId);
        applied = true;
        note = "Updated users.isDisabled=1";
      } else if (hasCol(userCols, "disabled")) {
        sqlite.prepare(`UPDATE users SET disabled=1 WHERE id=?`).run(userId);
        applied = true;
        note = "Updated users.disabled=1";
      }
      if (applied && hasCol(userCols, "disabledReason")) {
        sqlite.prepare(`UPDATE users SET disabledReason=? WHERE id=?`).run(reason, userId);
      }
      if (applied && hasCol(userCols, "disabledAt")) {
        sqlite.prepare(`UPDATE users SET disabledAt=? WHERE id=?`).run(ts, userId);
      }
    }
  } catch {
    applied = false;
    note = "Attempted update failed; logged only.";
  }

  logGriftAdminAction({
    adminId,
    action: "GRIFT_ENFORCEMENT",
    entityType: "USER",
    entityId: String(userId),
    payload: { action, reason, applied, note },
  });

  res.json({ ok: true, applied, note });
});
```

```tsx
async function downloadCsvFromUrl(url: string, filenameFallback: string) {
  const res = await fetch(url, {
    method: "GET",
    credentials: "include",
    headers: {
      "X-Device-Id": getDeviceId(),
    },
  });

  if (!res.ok) {
    const t = await res.text().catch(() => "");
    throw new Error(`Export failed: ${res.status} ${t}`);
  }

  const blob = await res.blob();
  const cd = res.headers.get("content-disposition") || "";
  const match = /filename="([^"]+)"/.exec(cd);
  const filename = match?.[1] || filenameFallback;

  const href = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = href;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(href);
}

  const [signalsLoading, setSignalsLoading] = useState(false);

  // User drilldown
  const [selectedUserId, setSelectedUserId] = useState<number | null>(null);
  const [detail, setDetail] = useState<UserDetail | null>(null);
  const [detailLoading, setDetailLoading] = useState(false);

  // Cases
  const [caseStatusFilter, setCaseStatusFilter] = useState<string>("");
  const [cases, setCases] = useState<GriftCase[]>([]);
  const [casesLoading, setCasesLoading] = useState(false);

  const [selectedCaseId, setSelectedCaseId] = useState<number | null>(null);
  const [caseDetail, setCaseDetail] = useState<CaseDetail | null>(null);
  const [caseDetailLoading, setCaseDetailLoading] = useState(false);

  // Create case modal
  const [newCaseOpen, setNewCaseOpen] = useState(false);
  const [newCasePrimaryUserId, setNewCasePrimaryUserId] = useState<string>("");
  const [newCaseTitle, setNewCaseTitle] = useState<string>("Grift Review");
  const [newCaseSummary, setNewCaseSummary] = useState<string>("");
  const [newCasePriority, setNewCasePriority] = useState<string>("MED");
  const [newCaseSignalId, setNewCaseSignalId] = useState<string>("");

  // Audit
  const [auditRows, setAuditRows] = useState<AuditRow[]>([]);
  const [auditLoading, setAuditLoading] = useState(false);
  const [auditLimit, setAuditLimit] = useState<number>(200);

  const breakdownMatrix = useMemo(() => {
    const m: Record<string, Record<string, number>> = {};
    (breakdown?.rows ?? []).forEach((r) => {
      const rule = r.ruleCode;
      const key = `${String(r.severity).toUpperCase()}|${String(r.status).toUpperCase()}`;
      m[rule] = m[rule] || {};
      m[rule][key] = Number(r.c ?? 0);
    });
    return m;
  }, [breakdown]);

  const tierCounts = useMemo(() => {
    const map: Record<string, number> = { LOW: 0, MED: 0, HIGH: 0, CRITICAL: 0 };
    (summary?.tiers ?? []).forEach((t) => {
      const key = String(t.tier || "LOW").toUpperCase();
      map[key] = Number(t.c ?? 0);
    });
    return map;
  }, [summary]);

  const severityCounts = useMemo(() => {
    const map: Record<string, number> = { LOW: 0, MED: 0, HIGH: 0, CRITICAL: 0 };
    (summary?.bySeverity ?? []).forEach((s) => {
      const key = String(s.severity || "").toUpperCase();
      if (map[key] !== undefined) map[key] = Number(s.c ?? 0);
    });
    return map;
  }, [summary]);

  async function refreshSummary() {
    const [s, b, c] = await Promise.all([
      apiFetch<GriftSummary>("/api/admin/grift/summary"),
      apiFetch<GriftBreakdown>(`/api/admin/grift/breakdown?sinceHours=${sigSinceHours}`),
      apiFetch<GriftConfig>("/api/admin/grift/config"),
    ]);
    setSummary(s);
    setBreakdown(b);
    setConfig(c);
    setConfigEdit({});
  }

  async function refreshUsers() {
    const u = await apiFetch<FlaggedUserRow[]>(`/api/admin/grift/users?minScore=${minScore}`);
    setUsers(u);
  }

  async function refreshSignals() {
    setSignalsLoading(true);
    try {
      const userIdParam = sigUserId.trim() ? `&userId=${encodeURIComponent(sigUserId.trim())}` : "";
      const url =
        `/api/admin/grift/signals?status=${encodeURIComponent(sigStatus)}` +
        `&severity=${encodeURIComponent(sigSeverity)}` +
        `&ruleCode=${encodeURIComponent(sigRule)}` +
        `&sinceHours=${encodeURIComponent(String(sigSinceHours))}` +
        `&limit=300` +
        userIdParam;

      const r = await apiFetch<SignalsResponse>(url);
      setSignals(r.rows || []);
    } finally {
      setSignalsLoading(false);
    }
  }

  async function refreshCases() {
    setCasesLoading(true);
    try {
      const st = caseStatusFilter ? `?status=${encodeURIComponent(caseStatusFilter)}` : "";
      const r = await apiFetch<GriftCase[]>(`/api/admin/grift/cases${st}`);
      setCases(r || []);
    } finally {
      setCasesLoading(false);
    }
  }

  async function refreshAudit() {
    setAuditLoading(true);
    try {
      const r = await apiFetch<AuditRow[]>(`/api/admin/grift/audit?limit=${auditLimit}`);
      setAuditRows(r || []);
    } finally {
      setAuditLoading(false);
    }
  }

  async function refreshAll() {
    setErr(null);
    setLoading(true);
    try {
      await refreshSummary();
      await refreshUsers();
      await refreshSignals();
      await refreshCases();
      await refreshAudit();
    } catch (e: any) {
      setErr(e?.message ?? "Failed to load grift data");
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => { refreshAll(); /* eslint-disable-next-line */ }, []);
  useEffect(() => { refreshUsers().catch(() => {}); /* eslint-disable-next-line */ }, [minScore]);

  async function openUser(userId: number) {
    setSelectedUserId(userId);
    setDetail(null);
    setDetailLoading(true);
    setErr(null);
    try {
      const d = await apiFetch<UserDetail>(`/api/admin/grift/users/${userId}`);
      setDetail(d);
    } catch (e: any) {
      setErr(e?.message ?? "Failed to load user detail");
    } finally {
      setDetailLoading(false);
    }
  }

  async function setSignalStatus(signalId: number, status: "OPEN" | "IN_REVIEW" | "CLOSED" | "IGNORED") {
    const reason =
      status === "CLOSED" || status === "IGNORED"
        ? (prompt(`${status} reason (optional):`) ?? "")
        : "";
    setErr(null);
    await apiFetch(`/api/admin/grift/signals/${signalId}/status`, {
      method: "POST",
      body: { status, reason: reason || null },
    });
    await refreshSummary();
    await refreshUsers();
    await refreshSignals();
    await refreshCases();
    await refreshAudit();
    if (selectedUserId) await openUser(selectedUserId);
    if (selectedCaseId) await openCase(selectedCaseId);
  }

  async function saveConfig() {
    if (!config) return;
    setSavingConfig(true);
    setErr(null);
    try {
      const updated = await apiFetch<GriftConfig>("/api/admin/grift/config", { method: "PUT", body: configEdit });
      setConfig(updated);
      setConfigEdit({});
      await refreshAll();
    } catch (e: any) {
      setErr(e?.message ?? "Failed to save config");
    } finally {
      setSavingConfig(false);
    }
  }

  async function createCaseFromUI(primaryUserId: number, signalId?: number) {
    setErr(null);
    const title = newCaseTitle || "Grift Review";
    const priority = newCasePriority || "MED";
    const summary = newCaseSummary || "";

    const created = await apiFetch<{ id: number }>("/api/admin/grift/cases", {
      method: "POST",
      body: {
        primaryUserId,
        title,
        summary,
        priority,
        signalId: signalId ?? (newCaseSignalId ? Number(newCaseSignalId) : null),
      },
    });

    setNewCaseOpen(false);
    setNewCasePrimaryUserId("");
    setNewCaseSignalId("");
    setNewCaseSummary("");
    await refreshCases();
    await refreshAudit();
    if (created?.id) await openCase(created.id);
    setTab("Cases");
  }

  async function quickCreateCaseFromSignal(signal: any) {
    setNewCaseTitle(`Signal Review: ${signal.rule_code}`);
    setNewCasePriority(signal.severity === "CRITICAL" ? "CRITICAL" : signal.severity === "HIGH" ? "HIGH" : "MED");
    setNewCaseSummary(`Auto-created from Signal #${signal.id}\n\nEvidence:\n${signal.evidence_json}`);
    setNewCasePrimaryUserId(String(signal.primary_user_id));
    setNewCaseSignalId(String(signal.id));
    setNewCaseOpen(true);
  }

  async function quickCreateCaseFromUser(user: FlaggedUserRow) {
    setNewCaseTitle(`User Review: ${user.email || `User #${user.userId}`}`);
    setNewCasePriority(user.tier === "CRITICAL" ? "CRITICAL" : user.tier === "HIGH" ? "HIGH" : "MED");
    setNewCaseSummary(`Auto-created from user queue.\nScore=${user.scoreCurrent}, Tier=${user.tier}`);
    setNewCasePrimaryUserId(String(user.userId));
    setNewCaseSignalId("");
    setNewCaseOpen(true);
  }

  async function openCase(caseId: number) {
    setSelectedCaseId(caseId);
    setCaseDetail(null);
    setCaseDetailLoading(true);
    setErr(null);
    try {
      const d = await apiFetch<CaseDetail>(`/api/admin/grift/cases/${caseId}`);
      setCaseDetail(d);
    } catch (e: any) {
      setErr(e?.message ?? "Failed to load case detail");
    } finally {
      setCaseDetailLoading(false);
    }
  }

  async function caseAddNote(caseId: number) {
    const note = prompt("Case note:") ?? "";
    if (!note.trim()) return;
    await apiFetch(`/api/admin/grift/cases/${caseId}/note`, { method: "POST", body: { note } });
    await openCase(caseId);
    await refreshAudit();
    await refreshCases();
  }

  async function caseAssign(caseId: number) {
    const s = prompt("Assign to adminId (number) or blank to unassign:") ?? "";
    const assignedToAdminId = s.trim() ? Number(s.trim()) : null;
    await apiFetch(`/api/admin/grift/cases/${caseId}/assign`, { method: "POST", body: { assignedToAdminId } });
    await openCase(caseId);
    await refreshAudit();
    await refreshCases();
  }

  async function caseSetStatus(caseId: number, status: "OPEN" | "IN_REVIEW" | "CLOSED") {
    const reason = status === "CLOSED" ? (prompt("Close reason (optional):") ?? "") : "";
    await apiFetch(`/api/admin/grift/cases/${caseId}/status`, { method: "POST", body: { status, reason: reason || null } });
    await openCase(caseId);
    await refreshAudit();
    await refreshCases();
  }

  async function caseLinkSignal(caseId: number) {
    const s = prompt("SignalId to link:") ?? "";
    const signalId = Number(s.trim());
    if (!Number.isFinite(signalId)) return;
    await apiFetch(`/api/admin/grift/cases/${caseId}/link-signal`, { method: "POST", body: { signalId } });
    await openCase(caseId);
    await refreshAudit();
  }

  async function enforceOnUser(userId: number, action: "FREEZE" | "DISABLE") {
    const reason = prompt(`${action} reason (optional):`) ?? "";
    await apiFetch(`/api/admin/grift/users/${userId}/enforce`, {
      method: "POST",
      body: { action, reason: reason || "Grift enforcement" },
    });
    await refreshAudit();
  }

  async function exportUsersCsvServer() {
    await downloadCsvFromUrl(`/api/admin/grift/exports/users.csv?minScore=${minScore}`, `grift_users_${Date.now()}.csv`);
  }

  async function exportSignalsCsvServer() {
    await downloadCsvFromUrl(
      `/api/admin/grift/exports/signals.csv?status=${encodeURIComponent(sigStatus)}&sinceHours=${sigSinceHours}`,
      `grift_signals_${Date.now()}.csv`
    );
  }

  async function exportCasesCsvServer() {
    const st = caseStatusFilter ? `?status=${encodeURIComponent(caseStatusFilter)}` : "";
    await downloadCsvFromUrl(`/api/admin/grift/exports/cases.csv${st}`, `grift_cases_${Date.now()}.csv`);
  }

  return (
    <div className="p-4 space-y-4">
      <div className="flex items-center justify-between gap-3">
        <div>
          <div className="text-lg font-semibold">Grift Detection</div>
          <div className="text-sm text-muted-foreground">
            Detection + triage + case management + exports + tamper-evident admin audit.
          </div>
        </div>

        <div className="flex gap-2">
          <button className="px-3 py-2 rounded border hover:bg-muted" onClick={refreshAll} disabled={loading}>
            Refresh
          </button>
        </div>
      </div>

      {err && (
        <div className="border border-red-300 bg-red-50 text-red-900 rounded p-3 text-sm">
          {err}
        </div>
      )}

      {/* Sub-tabs */}
      <div className="flex flex-wrap gap-2">
        <TabBtn active={tab === "Dashboard"} onClick={() => setTab("Dashboard")}>Dashboard</TabBtn>
        <TabBtn active={tab === "Signals"} onClick={() => setTab("Signals")}>Signals</TabBtn>
        <TabBtn active={tab === "Users"} onClick={() => setTab("Users")}>Users</TabBtn>
        <TabBtn active={tab === "Cases"} onClick={() => setTab("Cases")}>Cases</TabBtn>
        <TabBtn active={tab === "Exports"} onClick={() => setTab("Exports")}>Exports</TabBtn>
        <TabBtn active={tab === "Config"} onClick={() => setTab("Config")}>Config</TabBtn>
        <TabBtn active={tab === "Audit"} onClick={() => setTab("Audit")}>Audit</TabBtn>
      </div>

      {/* DASHBOARD */}
      {tab === "Dashboard" && (
        <div className="space-y-4">
          <div className="grid grid-cols-1 md:grid-cols-6 gap-3">
            <div className="rounded border p-3">
              <div className="text-sm text-muted-foreground">Open Signals</div>
              <div className="text-2xl font-semibold">{summary?.open?.openTotal ?? 0}</div>
            </div>
            <div className="rounded border p-3">
              <div className="text-sm text-muted-foreground">Open High/Critical</div>
              <div className="text-2xl font-semibold">{summary?.open?.openHigh ?? 0}</div>
            </div>
            <div className="rounded border p-3">
              <div className="text-sm text-muted-foreground">Open MED</div>
              <div className="text-2xl font-semibold">{severityCounts["MED"] ?? 0}</div>
            </div>
            <div className="rounded border p-3">
              <div className="text-sm text-muted-foreground">Open HIGH</div>
              <div className="text-2xl font-semibold">{severityCounts["HIGH"] ?? 0}</div>
            </div>
            <div className="rounded border p-3">
              <div className="text-sm text-muted-foreground">Users HIGH</div>
              <div className="text-2xl font-semibold">{tierCounts["HIGH"] ?? 0}</div>
            </div>
            <div className="rounded border p-3">
              <div className="text-sm text-muted-foreground">Users CRITICAL</div>
              <div className="text-2xl font-semibold">{tierCounts["CRITICAL"] ?? 0}</div>
            </div>
          </div>

          <div className="grid grid-cols-1 lg:grid-cols-3 gap-3">
            <div className="rounded border p-3 space-y-2">
              <div className="font-semibold">Top Open Rule Codes</div>
              {(summary?.topRules ?? []).length ? (
                <div className="space-y-1">
                  {summary!.topRules.map((r) => (
                    <div key={r.ruleCode} className="flex items-center justify-between text-sm">
                      <span className="font-mono text-xs">{r.ruleCode}</span>
                      <span className="font-semibold">{r.c}</span>
                    </div>
                  ))}
                </div>
              ) : (
                <div className="text-sm text-muted-foreground">No open signals.</div>
              )}
            </div>

            <div className="lg:col-span-2 rounded border p-3 space-y-2 overflow-auto">
              <div className="flex items-center justify-between">
                <div className="font-semibold">Breakdown (since {Math.round(sigSinceHours)}h)</div>
                <div className="text-xs text-muted-foreground">Data since: {fmtTs(breakdown?.since)}</div>
              </div>

              <table className="w-full text-sm min-w-[900px]">
                <thead className="bg-muted">
                  <tr>
                    <th className="text-left p-2">Rule</th>
                    {["LOW", "MED", "HIGH", "CRITICAL"].map((sev) =>
                      STATUSES.map((st) => (
                        <th key={`${sev}|${st}`} className="text-left p-2">
                          {sev}/{st}
                        </th>
                      ))
                    )}
                  </tr>
                </thead>
                <tbody>
                  {RULES.map((rule) => {
                    const row = breakdownMatrix[rule] || {};
                    return (
                      <tr key={rule} className="border-t">
                        <td className="p-2 font-mono text-xs">{rule}</td>
                        {["LOW", "MED", "HIGH", "CRITICAL"].map((sev) =>
                          STATUSES.map((st) => {
                            const key = `${sev}|${st}`;
                            return (
                              <td key={`${rule}-${key}`} className="p-2">
                                {row[key] ?? 0}
                              </td>
                            );
                          })
                        )}
                      </tr>
                    );
                  })}
                </tbody>
              </table>

              <div className="pt-2 text-xs text-muted-foreground">
                Tip: use the Signals tab to filter by Rule/Severity/Status and action signals directly (IN_REVIEW / CLOSED / IGNORED).
              </div>
            </div>
          </div>
        </div>
      )}

      {/* SIGNALS */}
      {tab === "Signals" && (
        <div className="rounded border overflow-hidden">
          <div className="p-3 border-b flex items-center justify-between">
            <div className="font-semibold">Signals Queue</div>
            <div className="text-xs text-muted-foreground">
              Status workflow: OPEN → IN_REVIEW → CLOSED/IGNORED
            </div>
          </div>

          <div className="p-3 grid grid-cols-1 md:grid-cols-6 gap-2 border-b">
            <label className="text-xs">
              Status
              <select className="mt-1 w-full border rounded px-2 py-2" value={sigStatus} onChange={(e) => setSigStatus(e.target.value)}>
                {STATUSES.map((s) => <option key={s} value={s}>{s}</option>)}
              </select>
            </label>

            <label className="text-xs">
              Severity
              <select className="mt-1 w-full border rounded px-2 py-2" value={sigSeverity} onChange={(e) => setSigSeverity(e.target.value)}>
                {SEVERITIES.map((s) => <option key={s} value={s}>{s || "ALL"}</option>)}
              </select>
            </label>

            <label className="text-xs">
              Rule
              <select className="mt-1 w-full border rounded px-2 py-2" value={sigRule} onChange={(e) => setSigRule(e.target.value)}>
                <option value="">ALL</option>
                {RULES.map((r) => <option key={r} value={r}>{r}</option>)}
              </select>
            </label>

            <label className="text-xs">
              User ID (optional)
              <input className="mt-1 w-full border rounded px-2 py-2" value={sigUserId} onChange={(e) => setSigUserId(e.target.value)} />
            </label>

            <label className="text-xs">
              Lookback (hours)
              <input className="mt-1 w-full border rounded px-2 py-2" type="number" value={sigSinceHours} onChange={(e) => setSigSinceHours(Number(e.target.value))} />
            </label>

            <div className="flex items-end gap-2">
              <button className="px-3 py-2 rounded border hover:bg-muted" onClick={refreshSignals} disabled={signalsLoading}>
                {signalsLoading ? "Loading…" : "Apply"}
              </button>
              <button className="px-3 py-2 rounded border hover:bg-muted" onClick={() => exportSignalsCsvServer()}>
                Export CSV
              </button>
            </div>
          </div>

          {signalsLoading ? (
            <div className="p-3 text-sm text-muted-foreground">Loading signals…</div>
          ) : signals.length === 0 ? (
            <div className="p-3 text-sm text-muted-foreground">No signals for current filters.</div>
          ) : (
            <table className="w-full text-sm">
              <thead className="bg-muted">
                <tr>
                  <th className="text-left p-2">Time</th>
                  <th className="text-left p-2">Rule</th>
                  <th className="text-left p-2">Severity</th>
                  <th className="text-left p-2">Users</th>
                  <th className="text-left p-2">Symbol</th>
                  <th className="text-left p-2">Device/IP</th>
                  <th className="text-left p-2">Points</th>
                  <th className="text-left p-2">Actions</th>
                </tr>
              </thead>
              <tbody>
                {signals.map((s: any) => (
                  <tr key={s.id} className="border-t align-top">
                    <td className="p-2">{fmtTs(s.created_at)}</td>
                    <td className="p-2 font-mono text-xs">{s.rule_code}</td>
                    <td className="p-2"><TierBadge tier={s.severity} /></td>
                    <td className="p-2">
                      <button className="underline" onClick={() => openUser(Number(s.primary_user_id))}>
                        #{s.primary_user_id}
                      </button>
                      {s.secondary_user_id ? (
                        <>
                          <span className="mx-2">↔</span>
                          <button className="underline" onClick={() => openUser(Number(s.secondary_user_id))}>
                            #{s.secondary_user_id}
                          </button>
                        </>
                      ) : null}
                    </td>
                    <td className="p-2">{s.symbol ?? "—"}</td>
                    <td className="p-2">
                      <div className="font-mono text-xs break-all">{s.device_id ?? "—"}</div>
                      <div className="text-xs text-muted-foreground">{s.ip ?? "—"}</div>
                    </td>
                    <td className="p-2 font-semibold">{s.points}</td>
                    <td className="p-2 space-y-2">
                      <div className="flex flex-wrap gap-2">
                        {s.status !== "IN_REVIEW" && (
                          <button className="px-3 py-1 rounded border hover:bg-muted" onClick={() => setSignalStatus(Number(s.id), "IN_REVIEW")}>
                            Mark IN_REVIEW
                          </button>
                        )}
                        {s.status !== "CLOSED" && (
                          <button className="px-3 py-1 rounded border hover:bg-muted" onClick={() => setSignalStatus(Number(s.id), "CLOSED")}>
                            Close
                          </button>
                        )}
                        {s.status !== "IGNORED" && (
                          <button className="px-3 py-1 rounded border hover:bg-muted" onClick={() => setSignalStatus(Number(s.id), "IGNORED")}>
                            Ignore
                          </button>
                        )}
                        <button className="px-3 py-1 rounded border hover:bg-muted" onClick={() => quickCreateCaseFromSignal(s)}>
                          Create Case
                        </button>
                      </div>

                      <details>
                        <summary className="cursor-pointer underline text-xs">evidence</summary>
                        <pre className="text-xs whitespace-pre-wrap break-words mt-2">{s.evidence_json}</pre>
                      </details>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}
        </div>
      )}

      {/* USERS */}
      {tab === "Users" && (
        <div className="rounded border overflow-hidden">
          <div className="p-3 border-b flex items-center justify-between">
            <div className="font-semibold">Flagged Users</div>
            <div className="flex items-center gap-2 text-xs text-muted-foreground">
              <span>Min score:</span>
              <input className="border rounded px-2 py-1 w-20" type="number" min={0} max={100} value={minScore} onChange={(e) => setMinScore(Number(e.target.value))} />
              <button className="px-3 py-2 rounded border hover:bg-muted" onClick={() => exportUsersCsvServer()}>
                Export CSV
              </button>
            </div>
          </div>

          {users.length === 0 ? (
            <div className="p-3 text-sm text-muted-foreground">No users at this threshold.</div>
          ) : (
            <table className="w-full text-sm">
              <thead className="bg-muted">
                <tr>
                  <th className="text-left p-2">User</th>
                  <th className="text-left p-2">Tier</th>
                  <th className="text-left p-2">Score</th>
                  <th className="text-left p-2">Devices(7d)</th>
                  <th className="text-left p-2">IPs(7d)</th>
                  <th className="text-left p-2">ASNs(7d)</th>
                  <th className="text-left p-2">Linked(30d)</th>
                  <th className="text-left p-2">Hedge(7d)</th>
                  <th className="text-left p-2">Open Signals</th>
                  <th className="text-left p-2">Actions</th>
                </tr>
              </thead>
              <tbody>
                {users.map((u) => (
                  <tr key={u.userId} className="border-t">
                    <td className="p-2">
                      <div className="font-medium">{u.name || "User #" + u.userId}</div>
                      <div className="text-muted-foreground">{u.email}</div>
                    </td>
                    <td className="p-2"><TierBadge tier={u.tier} /></td>
                    <td className="p-2 font-semibold">{u.scoreCurrent}</td>
                    <td className="p-2">{u.devices7d}</td>
```

# 5) What “Add” now includes (definition of done)

* **Signals**: OPEN/IN_REVIEW/CLOSED/IGNORED, with admin reasons and audit logging
* **Cases**: create/assign/note/link-signal/status lifecycle
* **Audit**: hash-chained `grift_admin_actions` surfaced in UI
* **Exports**: server-side CSV for users/signals/cases
* **Enforcement**: safe placeholder freeze/disable endpoint that only updates user columns if they exist; otherwise logs only
* **Scheduler**: periodic recompute + audit trail of scheduler runs

If you want the next increment after this, the only meaningful “bigger” addition is **case assignment to named admins** (resolve admin IDs to emails/names) and **auto-case creation rules** (e.g., auto-create case for any CRITICAL hedge-pair signal).
