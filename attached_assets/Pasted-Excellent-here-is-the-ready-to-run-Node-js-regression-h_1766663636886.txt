## File: `tests/policyRegression.js`

```js
/**
 * Hedge-Fund Grade Policy Regression Harness
 * -------------------------------------------
 * Validates backend enforcement and deny_code alignment.
 * Requires:
 *   - Backend running on localhost:3000
 *   - Authenticated cookie in TEST_COOKIE env
 *
 * Usage:
 *   TEST_COOKIE="connect.sid=..." npm run test:policy
 */

import fetch from "node-fetch";
import assert from "node:assert";
import fs from "fs";

const CK = process.env.TEST_COOKIE;
const BASE = process.env.BASE_URL || "http://localhost:3000";
const HEADERS = { Cookie: CK, "Content-Type": "application/json" };
const results = [];

/* ---------- helpers ---------- */

async function call(method, path, body, expectCodes) {
  const url = `${BASE}${path}`;
  const res = await fetch(url, {
    method,
    headers: HEADERS,
    body: body ? JSON.stringify(body) : undefined,
  });
  const txt = await res.text();
  let js = null;
  try { js = JSON.parse(txt); } catch {}
  const ok = expectCodes.includes(res.status);
  results.push({
    path,
    method,
    status: res.status,
    ok,
    deny_code: js?.deny_code || js?.denyCode || null,
    body: js,
  });
  console.log(`${ok ? "✓" : "✗"} [${res.status}] ${method} ${path}`,
    js?.deny_code ? `(${js.deny_code})` : "");
  return { res, js };
}

function section(title) {
  console.log(`\n=== ${title} ===`);
}

/* ---------- main test suite ---------- */

async function run() {
  if (!CK) {
    console.error("Missing TEST_COOKIE env. Run with TEST_COOKIE='connect.sid=...' npm run test:policy");
    process.exit(1);
  }

  section("1. Locked-state trading checks");

  await call("POST", "/api/trades/place", {}, [403]);
  await call("PUT", "/api/trades/123/targets", { sl: 1.23, tp: 2.34 }, [403]);
  await call("POST", "/api/trades/123/cancel", {}, [403]);
  await call("POST", "/api/trades/123/close", {}, [200]);

  section("2. Email resend throttles");
  for (let i = 0; i < 5; i++) {
    await call("POST", "/api/verification/email/send", {}, [200, 403, 429]);
  }

  section("3. SMS verify throttles");
  await call("POST", "/api/sms/verify/start", { phone: "+15555550123" }, [200, 403, 429]);
  await call("POST", "/api/sms/verify/confirm", { phone: "+15555550123", code: "000000" }, [200, 403, 429]);

  section("4. KYC & payout gating");
  await call("GET", "/api/profile/kyc", null, [200, 403]);
  await call("PUT", "/api/profile/payout/currency", { currency: "USD" }, [200, 403]);
  await call("POST", "/api/profile/payout/request", {}, [200, 403]);

  section("5. Policy summary endpoint");
  const { js: vs } = await call("GET", "/api/verification/status", null, [200]);
  assert(vs && typeof vs.canTradeOpenOrIncrease === "boolean",
    "verification/status missing canTradeOpenOrIncrease");

  /* ---------- output report ---------- */
  const report = {
    runAt: new Date().toISOString(),
    summary: {
      total: results.length,
      failed: results.filter(r => !r.ok).length,
    },
    results,
  };

  fs.writeFileSync("policy-regression-report.json", JSON.stringify(report, null, 2));
  console.log("\nReport written to policy-regression-report.json");

  if (report.summary.failed > 0) process.exitCode = 1;
}

run().catch(e => {
  console.error("Fatal error:", e);
  process.exit(1);
});
```

## Add to `package.json`

```json
{
  "scripts": {
    "test:policy": "node tests/policyRegression.js"
  },
  "devDependencies": {
    "node-fetch": "^3.3.2"
  }
}
```
