# 1) Schema: add `legalDocTargets`

## 1.1 `shared/schema.ts` (ADD TABLE)

```ts
// FILE: /shared/schema.ts
// ADD these imports near the top with other drizzle sqlite imports
import { sqliteTable, text, integer, index, uniqueIndex } from "drizzle-orm/sqlite-core";

// Ensure legalDocuments is already defined above.
// Then ADD this table definition (place near legalDocuments for clarity).

export const legalDocTargets = sqliteTable(
  "legal_doc_targets",
  {
    id: integer("id").primaryKey({ autoIncrement: true }),

    // Which doc family this target belongs to (DOC1 for your signup terms)
    docSet: text("doc_set").notNull(), // "DOC1"
    docType: text("doc_type").notNull(), // "GLOBAL_MASTER" | "ADDENDUM"
    jurisdictionType: text("jurisdiction_type").notNull(), // "DEFAULT" | "COUNTRY" | "REGION"
    jurisdictionKey: text("jurisdiction_key").notNull(), // e.g., "GLOBAL" | "ROW" | "KE" | "SE_ASIA"

    // The explicitly active legalDocuments.id for this target
    activeDocId: integer("active_doc_id").notNull(),

    // Operational metadata
    updatedAt: integer("updated_at", { mode: "timestamp_ms" }).notNull(),
    updatedByAdminUserId: integer("updated_by_admin_user_id"), // nullable if you don't have admin user ids
    note: text("note"), // optional: "FTC update", "Kenya update", etc.
  },
  (t) => ({
    uqTarget: uniqueIndex("uq_legal_doc_target").on(t.docSet, t.docType, t.jurisdictionType, t.jurisdictionKey),
    idxActive: index("idx_legal_doc_target_active").on(t.activeDocId),
  })
);
```

# 2) Backend: Targets router

## 2.1 `server/routes/adminLegalDocTargets.ts` (NEW FILE)

```ts
// FILE: /server/routes/adminLegalDocTargets.ts
import { Router } from "express";
import { and, desc, eq } from "drizzle-orm";
import { db } from "../db";
import { requireAdmin } from "../middleware/requireAdmin";
import { legalDocTargets, legalDocuments } from "../../shared/schema";

export const adminLegalDocTargetsRouter = Router();
adminLegalDocTargetsRouter.use(requireAdmin);

function normText(v: any) {
  return String(v || "").trim();
}

function validateTarget(body: any) {
  const docSet = normText(body.docSet);
  const docType = normText(body.docType);
  const jurisdictionType = normText(body.jurisdictionType);
  const jurisdictionKey = normText(body.jurisdictionKey);

  if (!docSet || !docType || !jurisdictionType || !jurisdictionKey) {
    throw new Error("docSet, docType, jurisdictionType, jurisdictionKey are required.");
  }
  return { docSet, docType, jurisdictionType, jurisdictionKey };
}

// GET /api/admin/legal-docs/targets/get?docSet=DOC1&docType=ADDENDUM&jurisdictionType=COUNTRY&jurisdictionKey=KE
adminLegalDocTargetsRouter.get("/get", (req, res) => {
  try {
    const docSet = normText(req.query.docSet);
    const docType = normText(req.query.docType);
    const jurisdictionType = normText(req.query.jurisdictionType);
    const jurisdictionKey = normText(req.query.jurisdictionKey);

    if (!docSet || !docType || !jurisdictionType || !jurisdictionKey) {
      return res.status(400).json({ ok: false, error: "Missing target parameters." });
    }

    const row = db
      .select()
      .from(legalDocTargets)
      .where(
        and(
          eq(legalDocTargets.docSet, docSet),
          eq(legalDocTargets.docType, docType),
          eq(legalDocTargets.jurisdictionType, jurisdictionType),
          eq(legalDocTargets.jurisdictionKey, jurisdictionKey)
        ) as any
      )
      .get();

    return res.json({ ok: true, target: row || null });
  } catch (e: any) {
    return res.status(400).json({ ok: false, error: e?.message || "Failed to get target." });
  }
});

// GET /api/admin/legal-docs/targets/list?docSet=DOC1&docType=ADDENDUM
adminLegalDocTargetsRouter.get("/list", (req, res) => {
  try {
    const docSet = normText(req.query.docSet);
    const docType = normText(req.query.docType);

    const where: any[] = [];
    if (docSet) where.push(eq(legalDocTargets.docSet, docSet));
    if (docType) where.push(eq(legalDocTargets.docType, docType));

    const rows = db
      .select()
      .from(legalDocTargets)
      .where(where.length ? (and(...(where as any)) as any) : undefined)
      .orderBy(desc(legalDocTargets.updatedAt))
      .limit(500)
      .all();

    return res.json({ ok: true, rows });
  } catch (e: any) {
    return res.status(400).json({ ok: false, error: e?.message || "Failed to list targets." });
  }
});

// POST /api/admin/legal-docs/targets/set-active
// body: { docSet, docType, jurisdictionType, jurisdictionKey, activeDocId, note? }
adminLegalDocTargetsRouter.post("/set-active", (req, res) => {
  try {
    const body = req.body || {};
    const { docSet, docType, jurisdictionType, jurisdictionKey } = validateTarget(body);

    const activeDocId = Number(body.activeDocId);
    if (!Number.isFinite(activeDocId) || activeDocId <= 0) throw new Error("activeDocId must be a positive number.");

    // Ensure the document exists and matches target
    const doc = db
      .select({
        id: legalDocuments.id,
        docSet: legalDocuments.docSet,
        docType: legalDocuments.docType,
        jurisdictionType: legalDocuments.jurisdictionType,
        jurisdictionKey: legalDocuments.jurisdictionKey,
      })
      .from(legalDocuments)
      .where(eq(legalDocuments.id, activeDocId))
      .get();

    if (!doc) throw new Error("activeDocId does not exist in legalDocuments.");
    if (
      doc.docSet !== docSet ||
      doc.docType !== docType ||
      doc.jurisdictionType !== jurisdictionType ||
      doc.jurisdictionKey !== jurisdictionKey
    ) {
      throw new Error("activeDocId document does not match the specified target.");
    }

    const now = Date.now();
    const note = body.note != null ? String(body.note) : null;

    // If your auth middleware exposes admin identity, you can set this.
    // Otherwise it remains null.
    const adminUserId = (req as any)?.user?.id ? Number((req as any).user.id) : null;

    // Upsert into legal_doc_targets
    db.insert(legalDocTargets)
      .values({
        docSet,
        docType,
        jurisdictionType,
        jurisdictionKey,
        activeDocId,
        updatedAt: new Date(now),
        updatedByAdminUserId: adminUserId,
        note,
      } as any)
      .onConflictDoUpdate({
        target: [legalDocTargets.docSet, legalDocTargets.docType, legalDocTargets.jurisdictionType, legalDocTargets.jurisdictionKey],
        set: {
          activeDocId,
          updatedAt: new Date(now) as any,
          updatedByAdminUserId: adminUserId,
          note,
        },
      })
      .run();

    const saved = db
      .select()
      .from(legalDocTargets)
      .where(
        and(
          eq(legalDocTargets.docSet, docSet),
          eq(legalDocTargets.docType, docType),
          eq(legalDocTargets.jurisdictionType, jurisdictionType),
          eq(legalDocTargets.jurisdictionKey, jurisdictionKey)
        ) as any
      )
      .get();

    return res.json({ ok: true, target: saved });
  } catch (e: any) {
    return res.status(400).json({ ok: false, error: e?.message || "Failed to set active target." });
  }
});
```

```ts
const getActiveDocByTarget = (target: {
  docSet: string;
  docType: string;
  jurisdictionType: string;
  jurisdictionKey: string;
}) => {
  const t = db
    .select()
    .from(legalDocTargets)
    .where(
      and(
        eq(legalDocTargets.docSet, target.docSet),
        eq(legalDocTargets.docType, target.docType),
        eq(legalDocTargets.jurisdictionType, target.jurisdictionType),
        eq(legalDocTargets.jurisdictionKey, target.jurisdictionKey)
      ) as any
    )
    .get();

  if (!t) return { target: null, doc: null, mode: "MISSING_TARGET" as const };

  const doc = db
    .select({
      id: legalDocuments.id,
      docSet: legalDocuments.docSet,
      docType: legalDocuments.docType,
      jurisdictionType: legalDocuments.jurisdictionType,
      jurisdictionKey: legalDocuments.jurisdictionKey,
      version: legalDocuments.version,
      sha256: legalDocuments.sha256,
      content: legalDocuments.content,
      createdAt: legalDocuments.createdAt,
    })
    .from(legalDocuments)
    .where(eq(legalDocuments.id, t.activeDocId))
    .get();

  if (!doc) return { target: t, doc: null, mode: "BROKEN_TARGET" as const };

  return { target: t, doc, mode: "ACTIVE_TARGET" as const };
```

### 4.1 Add state + loader for the current target

Add these state hooks:

```tsx
  const [activeDocId, setActiveDocId] = useState<number | null>(null);
  const [activeLoading, setActiveLoading] = useState(false);
  const [setActiveLoadingId, setSetActiveLoadingId] = useState<number | null>(null);
```

Add helper to load the target’s active doc:

```tsx
  async function loadActiveTarget() {
    setActiveLoading(true);
    try {
      const qs = new URLSearchParams({
        docSet,
        docType,
        jurisdictionType,
        jurisdictionKey,
      });
      const res = await fetch(`/api/admin/legal-docs/targets/get?${qs.toString()}`);
      const data = await res.json();
      if (!data?.ok) throw new Error(data?.error || "Failed to load active target.");
      setActiveDocId(data.target?.activeDocId ? Number(data.target.activeDocId) : null);
    } catch {
      setActiveDocId(null);
    } finally {
      setActiveLoading(false);
    }
  }
```

Call `loadActiveTarget()` whenever the selector changes (same place you load versions):

```tsx
  useEffect(() => {
    loadActiveTarget().catch(() => {});
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [docSet, docType, jurisdictionType, jurisdictionKey]);
```

Add setter helper:

```tsx
  async function setActive(docId: number) {

        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          docSet,
          docType,
          jurisdictionType,
          jurisdictionKey,
          activeDocId: docId,
          note: "Set via AdminLegalDocs",

      setActiveDocId(Number(data.target.activeDocId));

      alert(e?.message || "Failed to set active doc.");

      setSetActiveLoadingId(null);

  }
```
