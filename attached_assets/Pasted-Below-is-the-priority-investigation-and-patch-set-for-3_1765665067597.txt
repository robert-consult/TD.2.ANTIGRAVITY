# Priority: Fix the order lifecycle to professional semantics (and close the integrity holes)

## What is broken today (from your code)

1. **Pending orders exist in schema** (`orderType`, `limitPrice`, `stopPrice`, `status=PENDING`) and you can fetch/cancel them, but there is **no server-side matching/execution engine** that watches quotes and triggers them.
2. **Order placement logic does not use orderType** in `server/routes.ts` (it always pulls a quote and uses it as `openPrice`), while `storage.createTrade()` sets `PENDING` only based on `orderType`. This creates inconsistent state: *a pending order has an “openPrice” derived from market at placement time* rather than the requested limit/stop price.
3. **Integrity risk**: execution can still be influenced by simulated quotes (if the quote endpoint/DB is stale) and (separately) by client-supplied prices in other flows. For professional semantics, **execution must be server-authoritative and quote-gated**.

## Target behavior (professional minimum)

* **Market**: fill immediately at **ask (BUY)** / **bid (SELL)**.
* **Limit**:

  * BUY limit triggers when **ask ≤ limit**
  * SELL limit triggers when **bid ≥ limit**
  * fill at **current ask/bid** (which is “limit or better” when triggered).
* **Stop**:

  * BUY stop triggers when **ask ≥ stop**
  * SELL stop triggers when **bid ≤ stop**
  * fill at **current ask/bid** (slippage possible).
* **Open positions SL/TP** auto-close:

  * BUY closes at **bid** (SL if bid ≤ SL, TP if bid ≥ TP)
  * SELL closes at **ask** (SL if ask ≥ SL, TP if ask ≤ TP)
* **Quote gating**:

  * If quote is **stale** or market **closed**, pending orders **do not trigger** and SL/TP **do not fire** (defer).
* **Audit trail (backend-only)**:

  * requested_price, trigger_price, fill_price, slippage
  * fill_quote_ts, fill_source, fill_spread (+ bid/ask/mid)
* **Trader UI**: only show **close_reason** (e.g., `TP`, `SL`, `MANUAL`, `AUTO_TIME_LIMIT (LIMIT TIME REACHED)`, etc.).

```ts
      requested_price REAL,
```

## Step 2 — Add a server-side Order Engine that consumes bid/ask and triggers PENDING + SL/TP

Create `server/engine/orderEngine.ts`:

```ts
import { db } from "@db";
import { and, eq, desc } from "drizzle-orm";
import { trades, users, userSettings, symbolConfigs, tradeAudit } from "@shared/schema";
import { requiredMargin } from "../lib/margin";
import { recalcAccount } from "../recalcAccount";

/**
 * IMPORTANT:
 * - This engine must ONLY act on non-stale quotes.
 * - It must not use client-supplied prices.
 */

type Quote = {
  symbol: string;
  price?: number;
  bid?: number | null;
  ask?: number | null;
  isStale?: boolean;
  lastUpdated?: number; // ms
};

let running = false;

function n(v: any): number | null {
  const x = Number(v);
  return Number.isFinite(x) ? x : null;
}

function getBidAsk(q: Quote): { bid: number; ask: number; mid: number; spread: number } | null {
  const bid = n(q.bid ?? q.price);
  const ask = n(q.ask ?? q.price);
  if (bid === null || ask === null) return null;
  const mid = (bid + ask) / 2;
  const spread = ask - bid;
  return { bid, ask, mid, spread };
}

// positive = adverse slippage; negative = improvement
function adverseSlippage(side: "BUY" | "SELL", requested: number, fill: number): number {
  return side === "BUY" ? (fill - requested) : (requested - fill);
}

async function audit(params: {
  tradeId: number;
  eventType: string;
  requestedPrice?: number | null;
  triggerPrice?: number | null;
  fillPrice?: number | null;
  slippage?: number | null;
  quoteTs?: Date | null;
  quoteSource?: string | null;
  quoteBid?: number | null;
  quoteAsk?: number | null;
  quoteMid?: number | null;
  quoteSpread?: number | null;
  note?: string | null;
}) {
  await db.insert(tradeAudit).values({
    tradeId: params.tradeId,
    eventType: params.eventType,
    eventAt: new Date(),
    requestedPrice: params.requestedPrice ?? null,
    triggerPrice: params.triggerPrice ?? null,
    fillPrice: params.fillPrice ?? null,
    slippage: params.slippage ?? null,
    quoteTs: params.quoteTs ?? null,
    quoteSource: params.quoteSource ?? null,
    quoteBid: params.quoteBid ?? null,
    quoteAsk: params.quoteAsk ?? null,
    quoteMid: params.quoteMid ?? null,
    quoteSpread: params.quoteSpread ?? null,
    note: params.note ?? null,
  });
}

async function processPendingForSymbol(symbol: string, q: Quote) {
  const ba = getBidAsk(q);
  if (!ba) return;

  const rows = await db
    .select({
      t: trades,
      u: users,
      s: userSettings,
      sym: symbolConfigs,
    })
    .from(trades)
    .leftJoin(users, eq(trades.userId, users.id))
    .leftJoin(userSettings, eq(userSettings.userId, users.id))
    .leftJoin(symbolConfigs, eq(trades.symbolId, symbolConfigs.id))
    .where(and(eq(trades.status, "PENDING"), eq(symbolConfigs.symbol, symbol)))
    .orderBy(desc(trades.openedAt));

  for (const r of rows) {
    const t = r.t;
    const u = r.u;
    if (!u) continue;

    const side = (t.type as "BUY" | "SELL");
    const orderType = String(t.orderType ?? "Market").toLowerCase();

    // Requested price
    const requested =
      orderType.includes("limit") ? n(t.limitPrice) :
      orderType.includes("stop")  ? n(t.stopPrice)  :
      null;

    if (requested === null) {
      // Reject malformed pending order (no requested price)
      await db.update(trades)
        .set({ status: "CANCELED", closeReason: "REJECTED", closedAt: new Date() })
        .where(and(eq(trades.id, t.id), eq(trades.status, "PENDING")));
      await audit({ tradeId: t.id, eventType: "ORDER_REJECTED", note: "Missing limitPrice/stopPrice" });
      continue;
    }

    // Trigger rules
    let triggered = false;
    let triggerPrice: number;

    if (orderType.includes("limit")) {
      if (side === "BUY") { triggered = ba.ask <= requested; triggerPrice = ba.ask; }
      else { triggered = ba.bid >= requested; triggerPrice = ba.bid; }
    } else if (orderType.includes("stop")) {
      if (side === "BUY") { triggered = ba.ask >= requested; triggerPrice = ba.ask; }
      else { triggered = ba.bid <= requested; triggerPrice = ba.bid; }
    } else {
      // Unknown order type in PENDING -> reject
      await db.update(trades)
        .set({ status: "CANCELED", closeReason: "REJECTED", closedAt: new Date() })
        .where(and(eq(trades.id, t.id), eq(trades.status, "PENDING")));
      await audit({ tradeId: t.id, eventType: "ORDER_REJECTED", note: `Unknown orderType=${t.orderType}` });
      continue;
    }

    if (!triggered) continue;

    // Fill price (professional: bid/ask side)
    const fillPrice =
      side === "BUY" ? ba.ask : ba.bid;

    // Re-check risk at execution time
    const lots = Number(t.lots ?? 1);
    const leverage = Number(u.leverage ?? 5);
    const neededMargin = requiredMargin(symbol, lots, fillPrice, leverage);

    // open count checks
    const openCount = await db.select({ c: trades.id })
      .from(trades)
      .where(and(eq(trades.userId, u.id), eq(trades.status, "OPEN")));

    const maxConcurrent = Number(r.s?.maxConcurrent ?? 5);
    if (openCount.length >= maxConcurrent) {
      await db.update(trades)
        .set({ status: "CANCELED", closeReason: "REJECTED", closedAt: new Date() })
        .where(and(eq(trades.id, t.id), eq(trades.status, "PENDING")));
      await audit({ tradeId: t.id, eventType: "ORDER_REJECTED", note: "Max concurrent open trades exceeded" });
      continue;
    }

    // per-instrument open cap (3)
    const openPerSymbol = await db.select({ c: trades.id })
      .from(trades)
      .leftJoin(symbolConfigs, eq(trades.symbolId, symbolConfigs.id))
      .where(and(eq(trades.userId, u.id), eq(trades.status, "OPEN"), eq(symbolConfigs.symbol, symbol)));

    if (openPerSymbol.length >= 3) {
      await db.update(trades)
        .set({ status: "CANCELED", closeReason: "REJECTED", closedAt: new Date() })
        .where(and(eq(trades.id, t.id), eq(trades.status, "PENDING")));
      await audit({ tradeId: t.id, eventType: "ORDER_REJECTED", note: "Max open trades per instrument exceeded" });
      continue;
    }

    // margin availability (use user's current freeMargin)
    const freeMargin = Number(u.freeMargin ?? 0);
    if (freeMargin < neededMargin) {
      await db.update(trades)
        .set({ status: "CANCELED", closeReason: "REJECTED", closedAt: new Date() })
        .where(and(eq(trades.id, t.id), eq(trades.status, "PENDING")));
      await audit({ tradeId: t.id, eventType: "ORDER_REJECTED", note: "Insufficient free margin at execution" });
      continue;
    }

    // Atomically flip PENDING -> OPEN (idempotent condition on status)
    const updated = await db.update(trades)
      .set({
        status: "OPEN",
        executedAt: new Date(),
        openPrice: fillPrice,
      })
      .where(and(eq(trades.id, t.id), eq(trades.status, "PENDING")))
      .returning();

    if (updated.length === 0) continue;

    await audit({
      tradeId: t.id,
      eventType: "ORDER_FILLED",
      requestedPrice: requested,
      triggerPrice,
      fillPrice,
      slippage: adverseSlippage(side, requested, fillPrice),
      quoteTs: q.lastUpdated ? new Date(q.lastUpdated) : new Date(),
      quoteSource: "quotes_db",
      quoteBid: ba.bid,
      quoteAsk: ba.ask,
      quoteMid: ba.mid,
      quoteSpread: ba.spread,
      note: `orderType=${t.orderType}`,
    });

    await recalcAccount(u.id);
  }
}

async function processStopsForOpenTrades(symbol: string, q: Quote) {
  const ba = getBidAsk(q);
  if (!ba) return;

  const openRows = await db
    .select({ t: trades, u: users })
    .from(trades)
    .leftJoin(users, eq(trades.userId, users.id))
    .leftJoin(symbolConfigs, eq(trades.symbolId, symbolConfigs.id))
    .where(and(eq(trades.status, "OPEN"), eq(symbolConfigs.symbol, symbol)));

  for (const r of openRows) {
    const t = r.t;
    const u = r.u;
    if (!u) continue;

    const side = (t.type as "BUY" | "SELL");
    const sl = n(t.stopLoss);
    const tp = n(t.takeProfit);

    // Close price uses correct side:
    // BUY closes at bid, SELL closes at ask
    const closePx = side === "BUY" ? ba.bid : ba.ask;

    let reason: "SL" | "TP" | null = null;

    if (side === "BUY") {
      if (sl !== null && ba.bid <= sl) reason = "SL";
      else if (tp !== null && ba.bid >= tp) reason = "TP";
    } else {
      if (sl !== null && ba.ask >= sl) reason = "SL";
      else if (tp !== null && ba.ask <= tp) reason = "TP";
    }

    if (!reason) continue;

    // NOTE: your current profit model elsewhere is inconsistent; keep this minimal here:
    // profit = (close-open) * contract * lots, converted in your margin lib via FX_RATES.
    // You should unify all realized P/L later (second priority plan).
    const lots = Number(t.lots ?? 1);
    const openPx = Number(t.openPrice);
    const diff = side === "BUY" ? (closePx - openPx) : (openPx - closePx);
    const profit = (diff * 100000 * lots).toFixed(2); // placeholder: unify in step-2 plan

    const closed = await db.update(trades)
      .set({
        status: "CLOSED",
        closePrice: closePx,
        profit,
        closeReason: reason,
        closedAt: new Date(),
      })
      .where(and(eq(trades.id, t.id), eq(trades.status, "OPEN")))
      .returning();

    if (closed.length === 0) continue;

    // Update balance
    const newBalance = (Number(u.balance) + Number(profit)).toFixed(2);
    await db.update(users).set({ balance: newBalance }).where(eq(users.id, u.id));

    await audit({
      tradeId: t.id,
      eventType: "POSITION_CLOSED",
      fillPrice: closePx,
      quoteTs: q.lastUpdated ? new Date(q.lastUpdated) : new Date(),
      quoteSource: "quotes_db",
      quoteBid: ba.bid,
      quoteAsk: ba.ask,
      quoteMid: ba.mid,
      quoteSpread: ba.spread,
      note: `closeReason=${reason}`,
    });

    await recalcAccount(u.id);
  }
}

export async function onQuotesUpdated(quotes: Quote[]) {
  // single-flight to avoid overlapping ticks
  if (running) return;
  running = true;
  try {
    for (const q of quotes) {
      if (!q?.symbol) continue;
      if (q.isStale) continue; // quote gating

      const symbol = String(q.symbol).toUpperCase();
      await processPendingForSymbol(symbol, q);
      await processStopsForOpenTrades(symbol, q);
    }
  } finally {
    running = false;
  }
}
```
