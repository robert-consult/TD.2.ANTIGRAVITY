## 1️⃣  **Backend — risk management logic**

Find your existing backend risk check, likely in
`server/risk.ts` or `server/controllers/tradeController.js`
and **replace / expand it** with this logic block:

```ts
// server/risk.ts
import { getUserOpenTrades } from "./storage";
import { getTradeSettingsForUser } from "./settings";

export async function verifyRiskLimits(userId: string, instrument: string, requestedLots: number) {
  const openTrades = await getUserOpenTrades(userId);
  const tradeSettings = await getTradeSettingsForUser(userId);

  const maxConcurrentTrades = tradeSettings.maxConcurrentTrades ?? 5;
  const maxConcurrentPerInstrument = tradeSettings.maxConcurrentPerInstrument ?? 3;
  const maxConcurrentLots = tradeSettings.maxConcurrentLots ?? 50;

  // 1. Total concurrent trade limit
  if (openTrades.length >= maxConcurrentTrades) {
    throw new Error(`Trade limit reached: Max ${maxConcurrentTrades} open trades allowed`);
  }

  // 2. Per instrument concurrent trade limit
  const perInstrument = openTrades.filter(t => t.instrument === instrument);
  if (perInstrument.length >= maxConcurrentPerInstrument) {
    throw new Error(`Trade limit reached: Max ${maxConcurrentPerInstrument} open trades for ${instrument}`);
  }

  // 3. New — total concurrent lots limit
  const totalLots = openTrades.reduce((sum, t) => sum + t.lots, 0);
  if (totalLots + requestedLots > maxConcurrentLots) {
    throw new Error(`Lot limit reached: Total ${totalLots} lots open. Cannot exceed ${maxConcurrentLots} lots.`);
  }

  // If all checks pass
  return true;
}
```

Call this function from wherever trades are opened (usually `openTrade()` or `placeOrder()`):

```ts
// inside trade opening endpoint
await verifyRiskLimits(userId, instrument, requestedLots);
// ... proceed to create trade if passes
```

## 2️⃣ **Backend — helper: trade settings fetch**

In `server/settings.ts` (create if doesn’t exist):

```ts
import db from "../db";

export async function getTradeSettingsForUser(userId: string) {
  const userSetting = await db.user_settings.findUnique({ where: { user_id: userId } });
  const platformDefaults = await db.trade_settings.findFirst();

  return {
    maxConcurrentTrades: userSetting?.max_concurrent_trades ?? platformDefaults?.default_max_concurrent ?? 5,
    maxConcurrentPerInstrument: userSetting?.max_concurrent_per_instrument ?? platformDefaults?.max_concurrent_per_instrument ?? 3,
    maxConcurrentLots: userSetting?.max_concurrent_lots ?? platformDefaults?.default_max_concurrent_lots ?? 50,
  };
}
```
