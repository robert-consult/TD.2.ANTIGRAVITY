Your backend already implements a hash-chain across chunks to guarantee:

Chain mechanism (already in server/migration/migrationService.ts)
For chunk i:

chunkSha256 = sha256(chunkBytes)

On import, it recomputes the same linkHash chain and fails if any part is missing or modified.

#!/usr/bin/env bash
set -euo pipefail

# Usage:
#   ./migration_download_and_verify.sh \
#     --base "https://YOUR_DOMAIN" \
#     --job "MIGEXP_XXXXXXXX" \
#     --out "./exports/MIGEXP_XXXXXXXX" \
#     [--cookie "./cookie.txt"] \
#     [--insecure]
#
# Notes:
# - Uses /api/admin/migration/export-jobs/:jobId/manifest
# - Uses /api/admin/migration/export-jobs/:jobId/chunks/:index
# - Resumes downloads: curl -C -
# - Verifies SHA256 for each chunk and verifies the link-hash chain.

BASE=""
JOB=""
OUT=""
COOKIE_FILE=""
INSECURE="false"

while [[ $# -gt 0 ]]; do
  case "$1" in
    --base) BASE="${2:-}"; shift 2;;
    --job) JOB="${2:-}"; shift 2;;
    --out) OUT="${2:-}"; shift 2;;
    --cookie) COOKIE_FILE="${2:-}"; shift 2;;
    --insecure) INSECURE="true"; shift 1;;
    *) echo "Unknown arg: $1" >&2; exit 1;;
  esac
done

if [[ -z "$BASE" || -z "$JOB" || -z "$OUT" ]]; then
  echo "Missing required args. See usage in script header." >&2
  exit 1
fi

MANIFEST_URL="${BASE%/}/api/admin/migration/export-jobs/${JOB}/manifest"
CHUNK_URL_PREFIX="${BASE%/}/api/admin/migration/export-jobs/${JOB}/chunks"

mkdir -p "$OUT"

CURL_OPTS=(
  "-L"
  "--fail"
  "--retry" "8"
  "--retry-delay" "2"
  "--retry-connrefused"
  "--connect-timeout" "20"
  "--max-time" "0"
)

if [[ -n "$COOKIE_FILE" ]]; then
  # cookie jar or Netscape cookie file. If you use session cookies, export them into this file.
  CURL_OPTS+=("-b" "$COOKIE_FILE")
fi

if [[ "$INSECURE" == "true" ]]; then
  CURL_OPTS+=("-k")
fi

echo "==> Downloading manifest: $MANIFEST_URL"
curl "${CURL_OPTS[@]}" -o "${OUT}/manifest.json" "$MANIFEST_URL"

# Parse manifest
MANIFEST="${OUT}/manifest.json"

# Determine if chunked
CHUNK_ENABLED="$(jq -r '.chunking.enabled // false' "$MANIFEST")"
CHUNKS_LEN="$(jq -r '.chunks | length' "$MANIFEST")"

if [[ "$CHUNK_ENABLED" != "true" || "$CHUNKS_LEN" -le 1 ]]; then
  echo "==> Manifest indicates unchunked export (single file)."
  echo "==> Downloading data as a single file (or part 0 endpoint if implemented that way)."

  # Prefer chunk endpoint 0 if server always uses it; otherwise use /data
  DATA_URL="${BASE%/}/api/admin/migration/export-jobs/${JOB}/data"
  echo "==> Downloading data: $DATA_URL"
  curl "${CURL_OPTS[@]}" -C - -o "${OUT}/data.ndjson" "$DATA_URL"

  # Optional verify overall hash if present
  DATA_SHA="$(jq -r '.dataSha256 // empty' "$MANIFEST")"
  if [[ -n "$DATA_SHA" ]]; then
    echo "==> Verifying overall data SHA256..."
    ACTUAL="$(sha256sum "${OUT}/data.ndjson" | awk '{print $1}')"
    if [[ "$ACTUAL" != "$DATA_SHA" ]]; then
      echo "ERROR: data SHA mismatch. expected=$DATA_SHA actual=$ACTUAL" >&2
      exit 2
    fi
    echo "OK: overall data hash matches."
  else
    echo "==> No dataSha256 in manifest; skipping overall hash verify."
  fi

  echo "DONE."
  exit 0
fi

echo "==> Chunked export detected: ${CHUNKS_LEN} parts"

# Download all chunks with resume
echo "==> Downloading chunks to: $OUT"
for ((i=0; i<CHUNKS_LEN; i++)); do
