## 2) Frontend: generic component location

### Create file (generic path)

Use one of these common patterns—agent picks whichever matches your repo:

* `client/src/components/legal/SignupAvailabilityGate.tsx`
* OR `client/src/components/SignupAvailabilityGate.tsx`

**File content (use exactly):**

```tsx
// FILE: client/src/components/legal/SignupAvailabilityGate.tsx
import React, { useEffect, useMemo, useState } from "react";

type AvailabilityResp = {
  ok: boolean;
  available: boolean;
  enforce: boolean;
  code?: string;
  reason?: string | null;
  selected?: {
    globalTarget: { docSet: string; docType: string; jurisdictionType: string; jurisdictionKey: string };
    addendumTarget: { docSet: string; docType: string; jurisdictionType: string; jurisdictionKey: string };
  };
  modes?: { global: string; addendum: string };
  warnings?: string[];
};

export function SignupAvailabilityGate(props: {
  countryIso2: string;
  regionKey?: string | null;
  onAvailabilityChange?: (available: boolean, enforce: boolean, payload?: AvailabilityResp | null) => void;
}) {
  const countryIso2 = useMemo(() => String(props.countryIso2 || "").trim().toUpperCase(), [props.countryIso2]);
  const regionKey = useMemo(() => (props.regionKey ? String(props.regionKey).trim() : ""), [props.regionKey]);

  const [loading, setLoading] = useState(false);
  const [data, setData] = useState<AvailabilityResp | null>(null);

  useEffect(() => {
    const ISO2 = countryIso2;
    if (!ISO2 || ISO2.length !== 2) {
      setData(null);
      props.onAvailabilityChange?.(false, false, null);
      return;
    }

    let cancelled = false;

    (async () => {
      setLoading(true);
      try {
        const qs = new URLSearchParams({ countryIso2: ISO2 });
        if (regionKey) qs.set("regionKey", regionKey);

        const res = await fetch(`/api/legal/doc1/availability?${qs.toString()}`);
        const j = (await res.json()) as AvailabilityResp;
        if (cancelled) return;

        if (!j?.ok) throw new Error((j as any)?.error || "Availability check failed.");
        setData(j);
        props.onAvailabilityChange?.(Boolean(j.available), Boolean(j.enforce), j);
      } catch {
        if (cancelled) return;
        setData({ ok: true, available: false, enforce: false, reason: "CHECK_FAILED" } as any);
        props.onAvailabilityChange?.(false, false, null);
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();

    return () => {
      cancelled = true;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [countryIso2, regionKey]);

  if (!countryIso2 || countryIso2.length !== 2) {
    return (
      <div style={boxMuted}>
        <div style={{ fontWeight: 700 }}>Availability</div>
        <div style={{ fontSize: 12, opacity: 0.8 }}>Select a country to check availability.</div>
      </div>
    );
  }

  const available = Boolean(data?.available);
  const enforce = Boolean(data?.enforce);

  return (
    <div style={available ? boxOk : boxWarn}>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "baseline", gap: 10 }}>
        <div style={{ fontWeight: 800 }}>
          Availability: {loading ? "Checking…" : available ? "Available" : enforce ? "Not Available" : "Limited Availability"}
        </div>
        <div style={{ fontSize: 12, opacity: 0.8 }}>
          Gate: <strong>{loading ? "…" : enforce ? "ENFORCED" : "OFF"}</strong>
        </div>
      </div>

      {data?.selected && (
        <div style={{ marginTop: 8, fontSize: 12, opacity: 0.85 }}>
          <div style={{ fontFamily: "monospace" }}>
            Global: {data.selected.globalTarget.jurisdictionType}/{data.selected.globalTarget.jurisdictionKey} ({data.modes?.global || "—"})
          </div>
          <div style={{ fontFamily: "monospace" }}>
            Addendum: {data.selected.addendumTarget.jurisdictionType}/{data.selected.addendumTarget.jurisdictionKey} ({data.modes?.addendum || "—"})
          </div>
        </div>
      )}

      {!loading && !available && enforce && (
        <div style={{ marginTop: 8, fontSize: 12 }}>
          This jurisdiction is not yet supported. Please check back later.
        </div>
      )}

      {!loading && !available && !enforce && (
        <div style={{ marginTop: 8, fontSize: 12, opacity: 0.9 }}>
          Coverage is not fully configured for this jurisdiction; the platform may be running in fallback mode.
        </div>
      )}

      {!loading && Array.isArray(data?.warnings) && data!.warnings!.length > 0 && (
        <div style={{ marginTop: 8, fontSize: 12, opacity: 0.85 }}>
          <div style={{ fontWeight: 700 }}>Warnings</div>
          <ul style={{ margin: "6px 0 0 18px" }}>
            {data!.warnings!.slice(0, 6).map((w, i) => (
              <li key={i}>{w}</li>
            ))}
          </ul>
        </div>
      )}
    </div>
  );
}

const boxBase: React.CSSProperties = {
  border: "1px solid rgba(0,0,0,0.15)",
  borderRadius: 12,
  padding: 12,
  background: "white",
};

const boxOk: React.CSSProperties = {
  ...boxBase,
};

const boxWarn: React.CSSProperties = {
  ...boxBase,
};

const boxMuted: React.CSSProperties = {
  ...boxBase,
  opacity: 0.85,
};
```

### Minimal integration snippet (copy/paste)

```tsx
import React, { useMemo, useState } from "react";
import { SignupAvailabilityGate } from "../components/legal/SignupAvailabilityGate";

export default function SignupPage() {
  const [countryIso2, setCountryIso2] = useState<string>("US");
  const [regionKey, setRegionKey] = useState<string | null>(null);

  // You can choose strict behavior:
  // Recommended: canProceed only when available=true (even if gate off).
  const [canProceed, setCanProceed] = useState(false);
  const [gateEnforced, setGateEnforced] = useState(false);

  const allowProceed = useMemo(() => {
    // strict policy:
    return canProceed;

    // loose policy alternative (NOT recommended):
    // return gateEnforced ? canProceed : true;
  }, [canProceed]);

  return (
    <div style={{ display: "grid", gap: 12 }}>
      {/* country selector */}
      <label style={{ display: "grid", gap: 6 }}>
        <div style={{ fontSize: 12, opacity: 0.75 }}>Country</div>
        <input
          value={countryIso2}
          onChange={(e) => setCountryIso2(e.target.value)}
          placeholder="US"
          style={{ padding: "10px 12px", borderRadius: 10, border: "1px solid rgba(0,0,0,0.25)" }}
        />
      </label>

      {/* optional region selector */}
      <label style={{ display: "grid", gap: 6 }}>
        <div style={{ fontSize: 12, opacity: 0.75 }}>Region (optional)</div>
        <input
          value={regionKey || ""}
          onChange={(e) => setRegionKey(e.target.value ? e.target.value : null)}
          placeholder="SE_ASIA"
          style={{ padding: "10px 12px", borderRadius: 10, border: "1px solid rgba(0,0,0,0.25)" }}
        />
      </label>

      {/* availability gate */}
      <SignupAvailabilityGate
        countryIso2={countryIso2}
        regionKey={regionKey}
        onAvailabilityChange={(available, enforce) => {
          setCanProceed(available);
          setGateEnforced(enforce);
        }}
      />

      {/* continue */}
      <button
        disabled={!allowProceed}
        style={{
          padding: "12px 14px",
          borderRadius: 12,
          border: "1px solid rgba(0,0,0,0.25)",
          background: allowProceed ? "white" : "rgba(0,0,0,0.05)",
          cursor: allowProceed ? "pointer" : "not-allowed",
          fontWeight: 800,
        }}
        onClick={() => {
          // Next step will call /api/legal/doc1/resolve and render the doc + acceptance checkbox
          // For now: proceed to “Terms Preview” step
        }}
      >
        Continue
      </button>
    </div>
  );
}
```
