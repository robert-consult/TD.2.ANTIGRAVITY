export type AccountState =
  | "ACTIVE_VERIFIED"
  | "ACTIVE_UNVERIFIED_GRACE"
  | "LOCKED_EMAIL_NOT_VERIFIED"
  | "LOCKED_EMAIL_REVERIFY_OVERDUE"
  | "SUSPENDED";

export function resolveAccountState(ctx: DecisionContext, cfg: PolicyConfig = DEFAULT_POLICY_CONFIG): AccountState {
  if (ctx.user.suspendedAt) return "SUSPENDED";

  const now = ctx.now.getTime();
  const createdAt = ctx.user.createdAt.getTime();
  const gracePeriodMs = cfg.gracePeriodDays * 24 * 3600 * 1000;

  if (ctx.user.emailVerifiedAt) {
    if (ctx.user.emailReverifyDueAt && now > ctx.user.emailReverifyDueAt.getTime()) {
      return "LOCKED_EMAIL_REVERIFY_OVERDUE";
    }
    return "ACTIVE_VERIFIED";
  }

  if (now - createdAt < gracePeriodMs) {
    return "ACTIVE_UNVERIFIED_GRACE";
  }

  return "LOCKED_EMAIL_NOT_VERIFIED";
}

export function computeContenderEligibility(
  ctx: DecisionContext,
  cfg: PolicyConfig = DEFAULT_POLICY_CONFIG
): {
  eligible: boolean;
  path1: boolean;
  path2: boolean;
} {
  const path1 =
    ctx.metrics.accountAgeDays >= cfg.contenderMinAgeDays &&
    ctx.metrics.balancePctOfStart >= cfg.contenderMinBalancePct &&
    ctx.metrics.tradesLifetime >= cfg.contenderMinTradesLifetime;

  const path2 =
    ctx.metrics.accountAgeDays >= cfg.contenderPath2MinAgeDays &&
    (ctx.metrics.lastTradeDaysAgo ?? Infinity) <= cfg.contenderPath2MaxDaysSinceLastTrade &&
    ctx.metrics.returnLast90d >= cfg.contenderPath2MinReturnLast90 &&
    ctx.metrics.tradesLast90d >= cfg.contenderPath2MinTradesLast90;

  return { eligible: path1 || path2, path1, path2 };
}

function isExposureIncreasingTradingAction(action: PolicyAction): boolean {
  return action === "TRADE_OPEN_OR_INCREASE" || action === "TRADE_PLACE_PENDING";
}

function isCloseOrReduceTradingAction(action: PolicyAction): boolean {
  return action === "TRADE_CLOSE_OR_REDUCE";

function isCancelPendingTradingAction(action: PolicyAction): boolean {
  return action === "TRADE_CANCEL_PENDING";

function isModifyTargetsTradingAction(action: PolicyAction): boolean {
  return action === "TRADE_MODIFY_SLTP";

function isRiskReducingStopChange(intent?: DecisionContext["tradeIntent"]): boolean {
  if (!intent) return false;
  if (intent.kind !== "SLTP_UPDATE") return false;

  const before = intent.slBefore;
  const after = intent.slAfter;
  const side = (intent.side ?? "").toUpperCase();

  // Setting an SL when none existed is risk-reducing.
  if ((before === null || before === undefined) && after !== null && after !== undefined) return true;

  // Removing an SL is never risk-reducing.
  if (before !== null && before !== undefined && (after === null || after === undefined)) return false;

  // If we still don't have both, can't decide safely.
  if (before === null || before === undefined) return false;
  if (after === null || after === undefined) return false;

  if (side === "BUY" || side === "LONG") {
    return after >= before;
  }
  if (side === "SELL" || side === "SHORT") {
    return after <= before;
  }

  // Unknown side: safest default.
  return false;
}

function isSelectedForKyc(ctx: DecisionContext): boolean {
  return (
    ctx.user.userTier === "SELECTED" ||
    ctx.user.contenderTier === "SELECTED_REAL_CAPITAL" ||
    !!ctx.user.selectedAt ||
    ["INVITED", "SUBMITTED", "APPROVED"].includes(ctx.kyc.status)
  );
}

export function decidePolicy(
  action: PolicyAction,
  ctx: DecisionContext,
  cfg: PolicyConfig = DEFAULT_POLICY_CONFIG
): PolicyDecision {
  const accountState = resolveAccountState(ctx, cfg);
  const contender = computeContenderEligibility(ctx, cfg);

  if (accountState === "SUSPENDED") {
    return deny("ACCOUNT_SUSPENDED", { accountState, showLockedBanner: true });
  }

  // Email lock states: allow close/reduce (optional), disallow exposure increase and order/target manipulation.
  if (accountState === "LOCKED_EMAIL_NOT_VERIFIED" || accountState === "LOCKED_EMAIL_REVERIFY_OVERDUE") {
    const code = accountState === "LOCKED_EMAIL_NOT_VERIFIED" ? "EMAIL_NOT_VERIFIED" : "EMAIL_REVERIFY_OVERDUE";

    if (isExposureIncreasingTradingAction(action)) {
      return deny(code, { accountState, showLockedBanner: true, redirectTo: "/verify-email" });
    }

    if (isCloseOrReduceTradingAction(action)) {
      return cfg.allowReduceOnlyWhenLocked
        ? allow({ accountState, showLockedBanner: true })
        : deny("ACTION_NOT_ALLOWED_IN_STATE", { accountState, showLockedBanner: true });
    }

    if (isCancelPendingTradingAction(action)) {
      return cfg.allowCancelPendingWhenLocked
        ? allow({ accountState, showLockedBanner: true })
        : deny("TRADE_CANCEL_PENDING_NOT_ALLOWED_WHEN_LOCKED", { accountState, showLockedBanner: true });
    }

    if (isModifyTargetsTradingAction(action)) {
      if (cfg.allowRiskReducingSltpChangeWhenLocked && isRiskReducingStopChange(ctx.tradeIntent)) {
        return allow({ accountState, showLockedBanner: true });
      }
      return deny("TRADE_MODIFY_TARGETS_NOT_ALLOWED_WHEN_LOCKED", {
        accountState,
        showLockedBanner: true,
      });
    }

    // Non-trading actions below may still be permitted (e.g., resend email).
  }

  if (action === "EMAIL_RESEND_VERIFICATION") {
    const nowMs = ctx.now.getTime();
    const last = ctx.throttle.emailLastSentAtMs ?? 0;
    const cooldownMs = cfg.emailResendCooldownSec * 1000;

    if (ctx.throttle.emailSendCountDay >= cfg.emailDailySendCap) {
      return deny("EMAIL_SEND_RATE_LIMIT");
    }
    if (nowMs - last < cooldownMs) {
      return deny("EMAIL_SEND_COOLDOWN");
    }
    return allow();
  }

  if (action === "PHONE_VERIFY_START" || action === "PHONE_VERIFY_CONFIRM") {
    if (accountState !== "ACTIVE_VERIFIED") {
      return deny("SMS_NOT_ELIGIBLE", { accountState, showLockedBanner: true, redirectTo: "/verify-email" });
    }

    if (!contender.eligible) {
      return deny("SMS_NOT_ELIGIBLE_CONTENDER", { accountState, showLockedBanner: false });
    }

    if (action === "PHONE_VERIFY_START") {
      const nowMs = ctx.now.getTime();
      const last = ctx.throttle.smsLastSentAtMs ?? 0;

      if (ctx.throttle.smsSendCountDay >= cfg.smsDailySendCap) return deny("SMS_RATE_LIMIT");
      if (nowMs - last < 60_000) return deny("SMS_RATE_LIMIT");
    }

    if (action === "PHONE_VERIFY_CONFIRM") {
      if (ctx.throttle.otpFailCount >= cfg.otpMaxAttempts) return deny("SMS_OTP_TOO_MANY_ATTEMPTS");
    }

    return allow();
  }

  if (action === "KYC_VIEW" || action === "KYC_SUBMIT" || action === "PREFERRED_PAYMENT_CURRENCY_SET" || action === "PAYOUT_REQUEST") {
    if (accountState !== "ACTIVE_VERIFIED") {
      const code = accountState === "LOCKED_EMAIL_REVERIFY_OVERDUE" ? "EMAIL_REVERIFY_OVERDUE" : "EMAIL_NOT_VERIFIED";
      return deny(code, { accountState, showLockedBanner: true, redirectTo: "/verify-email" });
    }

    const selected = isSelectedForKyc(ctx);
    if (!selected) return deny("KYC_NOT_SELECTED", { redirectTo: "/profile" });

    if (action === "KYC_SUBMIT") {
      if (!(ctx.kyc.status === "INVITED" || ctx.kyc.status === "REJECTED")) {
        return deny("KYC_STATE_INVALID");
      }
      return allow();
    }

    if (action === "PREFERRED_PAYMENT_CURRENCY_SET") {
      // Currency is a payout preference: only meaningful once KYC is approved (or the user has been moved to SELECTED).
      if (!(ctx.kyc.status === "APPROVED" || ctx.user.userTier === "SELECTED")) {
        return deny("PAYOUT_PROFILE_NOT_READY");
      }
      return allow();
    }

    if (action === "PAYOUT_REQUEST") {
      if (ctx.user.userTier !== "SELECTED") return deny("PAYOUT_NOT_SELECTED");
      if (ctx.kyc.status !== "APPROVED") return deny("PAYOUT_KYC_NOT_APPROVED");
      if (!ctx.user.phoneVerifiedAt) return deny("PAYOUT_PHONE_NOT_VERIFIED");
      return allow();
    }

    // KYC_VIEW
    return allow();
  }

  // Active/unlocked states (and unverified grace): trade actions are allowed; backend risk guardrails still apply elsewhere.
  if (action === "TRADE_OPEN_OR_INCREASE" || action === "TRADE_PLACE_PENDING") {
    return allow({ accountState });
  }
  if (action === "TRADE_CLOSE_OR_REDUCE") {
    return allow({ accountState });
  }
  if (action === "TRADE_CANCEL_PENDING") {
    return allow({ accountState });
  }
  if (action === "TRADE_MODIFY_SLTP") {
    return allow({ accountState });
  }

  return allow({ accountState });
}

export type PolicySummary = {
  accountState: AccountState;
  contender: { eligible: boolean; path1: boolean; path2: boolean };
  trade: {
    openOrIncrease: Pick<PolicyDecision, "allowed" | "deny_code" | "redirectTo">;
    closeOrReduce: Pick<PolicyDecision, "allowed" | "deny_code" | "redirectTo">;
    cancelPending: Pick<PolicyDecision, "allowed" | "deny_code" | "redirectTo">;
    modifyTargets: Pick<PolicyDecision, "allowed" | "deny_code" | "redirectTo">;
  };
  sms: {
    start: Pick<PolicyDecision, "allowed" | "deny_code" | "redirectTo">;
  };
  kyc: {
    view: Pick<PolicyDecision, "allowed" | "deny_code" | "redirectTo">;
  };
};

export function policySummary(ctx: DecisionContext, cfg: PolicyConfig = DEFAULT_POLICY_CONFIG): PolicySummary {
  const contender = computeContenderEligibility(ctx, cfg);

  const openOrIncrease = decidePolicy("TRADE_OPEN_OR_INCREASE", ctx, cfg);
  const closeOrReduce = decidePolicy("TRADE_CLOSE_OR_REDUCE", ctx, cfg);
  const cancelPending = decidePolicy("TRADE_CANCEL_PENDING", ctx, cfg);
  const modifyTargets = decidePolicy("TRADE_MODIFY_SLTP", ctx, cfg);

  const smsStart = decidePolicy("PHONE_VERIFY_START", ctx, cfg);
  const kycView = decidePolicy("KYC_VIEW", ctx, cfg);

  return {
    accountState: resolveAccountState(ctx, cfg),
    contender,
    trade: {
      openOrIncrease: { allowed: openOrIncrease.allowed, deny_code: openOrIncrease.deny_code, redirectTo: openOrIncrease.redirectTo },
      closeOrReduce: { allowed: closeOrReduce.allowed, deny_code: closeOrReduce.deny_code, redirectTo: closeOrReduce.redirectTo },
      cancelPending: { allowed: cancelPending.allowed, deny_code: cancelPending.deny_code, redirectTo: cancelPending.redirectTo },
      modifyTargets: { allowed: modifyTargets.allowed, deny_code: modifyTargets.deny_code, redirectTo: modifyTargets.redirectTo },
    },
    sms: {
      start: { allowed: smsStart.allowed, deny_code: smsStart.deny_code, redirectTo: smsStart.redirectTo },
    },
    kyc: {
      view: { allowed: kycView.allowed, deny_code: kycView.deny_code, redirectTo: kycView.redirectTo },
    },
  };
}

export function featureGates(ctx: DecisionContext, cfg: PolicyConfig = DEFAULT_POLICY_CONFIG) {
  const st = resolveAccountState(ctx, cfg);
  const contender = computeContenderEligibility(ctx, cfg);
  const selected = isSelectedForKyc(ctx);

  const dOpen = decidePolicy("TRADE_OPEN_OR_INCREASE", ctx, cfg);
  const dClose = decidePolicy("TRADE_CLOSE_OR_REDUCE", ctx, cfg);
  const dCancel = decidePolicy("TRADE_CANCEL_PENDING", ctx, cfg);
  const dModify = decidePolicy("TRADE_MODIFY_SLTP", ctx, cfg);
  const dSms = decidePolicy("PHONE_VERIFY_START", ctx, cfg);
  const dKyc = decidePolicy("KYC_VIEW", ctx, cfg);

  return {
    accountState: st,
    contenderEligible: contender.eligible,

    // Trade actions
    canTradeOpenOrIncrease: dOpen.allowed,
    canTradeCloseOrReduce: dClose.allowed,
    canTradeCancelPending: dCancel.allowed,
    canTradeModifyTargets: dModify.allowed,

    // Verification
    canStartSms: dSms.allowed,

    // KYC/Payout
    canViewKyc: selected && dKyc.allowed,
    canSetPreferredPaymentCurrency: selected && decidePolicy("PREFERRED_PAYMENT_CURRENCY_SET", ctx, cfg).allowed,
    canRequestPayout: decidePolicy("PAYOUT_REQUEST", ctx, cfg).allowed,

    // Deny codes (for UI messaging)
    denyCodes: {
      tradeOpenOrIncrease: dOpen.deny_code ?? null,
      tradeCloseOrReduce: dClose.deny_code ?? null,
      tradeCancelPending: dCancel.deny_code ?? null,
      tradeModifyTargets: dModify.deny_code ?? null,
      smsStart: dSms.deny_code ?? null,
      kycView: dKyc.deny_code ?? null,
    },
  };
}
