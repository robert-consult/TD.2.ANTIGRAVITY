function allow(extra?: Partial<PolicyDecision>): PolicyDecision {
  return { allowed: true, ...(extra ?? {}) };
}

function deny(code: DenyCode, extra?: Partial<PolicyDecision>): PolicyDecision {
  return { allowed: false, deny_code: code, ...(extra ?? {}) };

export function resolveAccountState(ctx: DecisionContext, cfg: PolicyConfig = DEFAULT_POLICY_CONFIG): AccountState {
  const now = ctx.now.getTime();

  if (ctx.user.suspendedAt) return "SUSPENDED";

  // verified case + rolling reverify
  if (ctx.user.emailVerifiedAt) {
    const due = ctx.user.emailReverifyDueAt?.getTime() ?? null;
    if (due) {
      const lockAt = due + cfg.emailReverifyOverdueGraceDays * 86400000;
      if (now > lockAt) return "LOCKED_EMAIL_REVERIFY_OVERDUE";
    }
    return "ACTIVE_VERIFIED";
  }

  // not verified case + initial grace
  const initialDue = ctx.user.emailInitialDueAt?.getTime() ?? null;
  if (initialDue && now <= initialDue) return "ACTIVE_UNVERIFIED_GRACE";

  return "LOCKED_EMAIL_NOT_VERIFIED";
}

export function computeContenderEligibility(ctx: DecisionContext, cfg: PolicyConfig = DEFAULT_POLICY_CONFIG) {
  const path1 =
    ctx.metrics.accountAgeDays >= cfg.contenderPath1MinAgeDays &&
    ctx.metrics.tradesLifetime >= cfg.contenderPath1MinTradesLifetime &&
    ctx.metrics.balancePctOfStart >= cfg.contenderPath1MinBalancePctOfStart;

  const path2 =
    ctx.metrics.accountAgeDays >= cfg.contenderPath2MinAgeDays &&
    (ctx.metrics.lastTradeDaysAgo ?? Infinity) <= cfg.contenderPath2MaxDaysSinceLastTrade &&
    ctx.metrics.returnLast90d >= cfg.contenderPath2MinReturnLast90 &&
    ctx.metrics.tradesLast90d >= cfg.contenderPath2MinTradesLast90;

  return { eligible: path1 || path2, path1, path2 };
}

function isTradeAction(action: PolicyAction): boolean {
  return action.startsWith("TRADE_");
}

function isExposureIncreasingTradingAction(action: PolicyAction): boolean {
  return action === "TRADE_OPEN_OR_INCREASE" || action === "TRADE_PLACE_PENDING";
}

function isReduceOnlyTradingAction(action: PolicyAction): boolean {
  return action === "TRADE_CLOSE_OR_REDUCE";
}

function isCancelPendingTradingAction(action: PolicyAction): boolean {
  return action === "TRADE_CANCEL_PENDING";
}

function isSltpModifyAction(action: PolicyAction): boolean {
  return action === "TRADE_MODIFY_SLTP";
}

export function decidePolicy(
  action: PolicyAction,
  ctx: DecisionContext,
  cfg: PolicyConfig = DEFAULT_POLICY_CONFIG
): PolicyDecision {
  const accountState = resolveAccountState(ctx, cfg);
  const contender = computeContenderEligibility(ctx, cfg);

  // Admin override
  if (action === "ADMIN_ANY") return allow({ accountState });

  // Suspended: hard deny everything except admin
  if (accountState === "SUSPENDED") {
    return deny("ACCOUNT_SUSPENDED", { accountState, showLockedBanner: true });
  }

  // EMAIL RESEND: allowed even when locked, but throttled
  if (action === "EMAIL_RESEND_VERIFICATION") {
    const nowMs = ctx.now.getTime();
    const last = ctx.throttle.emailLastSentAtMs ?? 0;
    const cooldownMs = cfg.emailResendCooldownSec * 1000;

    if (ctx.throttle.emailSendCountDay >= cfg.emailDailySendCap) {
      return deny("EMAIL_SEND_RATE_LIMIT", { retryAfterSeconds: 3600 });
    }
    if (nowMs - last < cooldownMs) {
      const retryAfterSeconds = Math.max(1, Math.ceil((cooldownMs - (nowMs - last)) / 1000));
      return deny("EMAIL_SEND_COOLDOWN", { retryAfterSeconds });
    }
    return allow({ accountState });
  }

  // LOCKED states: allow only reduce-only + (optionally) cancel pending; block SL/TP modify by default.
  if (accountState === "LOCKED_EMAIL_NOT_VERIFIED" || accountState === "LOCKED_EMAIL_REVERIFY_OVERDUE") {
    const denyCode: DenyCode = accountState === "LOCKED_EMAIL_NOT_VERIFIED" ? "EMAIL_NOT_VERIFIED" : "EMAIL_REVERIFY_OVERDUE";

    if (isTradeAction(action)) {
      if (isReduceOnlyTradingAction(action)) {
        return cfg.allowReduceOnlyWhenLocked
          ? allow({ accountState, showLockedBanner: true })
          : deny("EMAIL_LOCKED", { accountState, showLockedBanner: true, redirectTo: "/profile" });
      }

      if (isCancelPendingTradingAction(action)) {
        return cfg.allowCancelPendingWhenLocked
          ? allow({ accountState, showLockedBanner: true })
          : deny("EMAIL_LOCKED", { accountState, showLockedBanner: true, redirectTo: "/profile" });
      }

      if (isSltpModifyAction(action)) {
        if (cfg.allowRiskReducingSltpChangeWhenLocked && ctx.tradeIntent?.isRiskReducing === true) {
          return allow({ accountState, showLockedBanner: true });
        }
        // Per your spec, SL/TP modify should be blocked while locked
        return deny(denyCode, { accountState, showLockedBanner: true, redirectTo: "/profile" });
      }

      // Any other trade action while locked
      return deny(denyCode, { accountState, showLockedBanner: true, redirectTo: "/profile" });
    }

    // Non-trade actions while locked: deny by default
    return deny(denyCode, { accountState, showLockedBanner: true, redirectTo: "/profile" });
  }

  // SMS Verification: must be ACTIVE_VERIFIED + contender eligible (per your tiered access spec)
  if (action === "PHONE_VERIFY_START" || action === "PHONE_VERIFY_CONFIRM") {
    if (accountState !== "ACTIVE_VERIFIED") {
      return deny("SMS_NOT_ELIGIBLE", { accountState, showLockedBanner: true, redirectTo: "/profile" });
    }
    if (!contender.eligible) {
      return deny("SMS_NOT_ELIGIBLE_NOT_CONTENDER", { accountState, redirectTo: "/profile" });
    }

    if (action === "PHONE_VERIFY_START") {
      const nowMs = ctx.now.getTime();
      const last = ctx.throttle.smsLastSentAtMs ?? 0;

      if (ctx.throttle.smsSendCountDay >= cfg.smsDailySendCap) return deny("SMS_RATE_LIMIT", { retryAfterSeconds: 3600 });
      if (nowMs - last < 60000) return deny("SMS_RATE_LIMIT", { retryAfterSeconds: Math.ceil((60000 - (nowMs - last)) / 1000) });
    }

    if (action === "PHONE_VERIFY_CONFIRM") {
      if (ctx.throttle.otpFailCount >= cfg.otpMaxAttempts) return deny("SMS_OTP_TOO_MANY_ATTEMPTS", { retryAfterSeconds: 3600 });
    }

    return allow({ accountState });
  }

  // KYC + payout gating (selected or invited/submitted/approved)
  if (action === "KYC_VIEW" || action === "KYC_SUBMIT" || action === "PREFERRED_PAYMENT_CURRENCY_SET" || action === "PAYOUT_REQUEST") {
    const selected =
      ctx.user.userTier === "SELECTED" ||
      ctx.user.contenderTier === "SELECTED_REAL_CAPITAL" ||
      !!ctx.user.selectedAt ||
      ["INVITED", "SUBMITTED", "APPROVED", "REJECTED"].includes(ctx.kyc.status);

    if (!selected) return deny("KYC_NOT_SELECTED", { redirectTo: "/profile" });

    if (action === "KYC_SUBMIT") {
      if (!(ctx.kyc.status === "INVITED" || ctx.kyc.status === "REJECTED")) {
        return deny("KYC_STATE_INVALID");
      }
    }

    if (action === "PREFERRED_PAYMENT_CURRENCY_SET") {
      if (ctx.kyc.status !== "APPROVED") return deny("KYC_NOT_APPROVED", { redirectTo: "/profile" });
    }

    return allow({ accountState });
  }

  // Active states: trades are allowed, other actions handled above.
  if (isTradeAction(action)) return allow({ accountState });

  return allow({ accountState });
}

export function featureGates(ctx: DecisionContext, cfg: PolicyConfig = DEFAULT_POLICY_CONFIG) {
  const st = resolveAccountState(ctx, cfg);
  const contender = computeContenderEligibility(ctx, cfg);
  const selected =
    ctx.user.userTier === "SELECTED" ||
    ctx.user.contenderTier === "SELECTED_REAL_CAPITAL" ||
    !!ctx.user.selectedAt ||
    ["INVITED", "SUBMITTED", "APPROVED", "REJECTED"].includes(ctx.kyc.status);

  return {
    accountState: st,
    contenderEligible: contender.eligible,

    canTradeOpenOrIncrease: decidePolicy("TRADE_OPEN_OR_INCREASE", ctx, cfg).allowed,
    canTradeCloseOrReduce: decidePolicy("TRADE_CLOSE_OR_REDUCE", ctx, cfg).allowed,
    canCancelPending: decidePolicy("TRADE_CANCEL_PENDING", ctx, cfg).allowed,
    canModifySltp: decidePolicy("TRADE_MODIFY_SLTP", ctx, cfg).allowed,

    canStartSms: decidePolicy("PHONE_VERIFY_START", ctx, cfg).allowed,

    canViewKyc: selected && decidePolicy("KYC_VIEW", ctx, cfg).allowed,
    canSetPreferredPaymentCurrency: selected && decidePolicy("PREFERRED_PAYMENT_CURRENCY_SET", ctx, cfg).allowed,

    // Keep payout request conservative: only SELECTED tier (you can expand when payout pipeline is complete)
    canRequestPayout: ctx.user.userTier === "SELECTED",
  };
}

// server/middleware/requirePolicy.ts

import { Request, Response, NextFunction } from "express";
import { decidePolicy, type PolicyAction } from "../../shared/policyDecision";
import { buildDecisionContext } from "../policy/buildDecisionContext";

export function requirePolicy(action: PolicyAction) {
  return async (req: Request, res: Response, next: NextFunction) => {
    try {
      const userId = (req.session as any)?.userId;
      if (!userId) return res.status(401).json({ message: "Not authenticated" });

      const ctx = await buildDecisionContext({
        userId,
        nowMs: Date.now(),
        tradeIntent: (req as any).tradeIntent,
      });

      const decision = decidePolicy(action, ctx);
      if (decision.allowed) return next();

      return res.status(403).json({
        message: decision.message ?? "Action not allowed by policy",
        deny_code: decision.deny_code ?? "POLICY_DENIED",
        accountState: decision.accountState,
        showLockedBanner: decision.showLockedBanner ?? (String(decision.deny_code ?? "").startsWith("EMAIL_")),
        redirectTo: decision.redirectTo ?? null,
        retryAfterSeconds: decision.retryAfterSeconds ?? null,
      });
    } catch (err) {
      console.error("Policy enforcement error:", err);
      return res.status(500).json({ message: "Policy enforcement failed." });
    }
  };
}

const adminTierUpdateSchema = z.object({
  userTier: z.enum(["CANDIDATE", "PERFORMER", "SELECTED"]).optional(),
  contenderTier: z.enum(["NONE", "CANDIDATE_EMAIL_ONLY", "CANDIDATE_SMS_REQUIRED", "VERIFIED_SMS", "SELECTED_REAL_CAPITAL"]).optional(),
  reason: z.string().max(300).optional(),
});

app.post("/api/admin/users/:id/tier", requireAdmin, async (req: Request, res: Response) => {
  try {
    const userId = Number(req.params.id);
    if (!Number.isFinite(userId)) return res.status(400).json({ message: "Invalid user id" });

    const parsed = adminTierUpdateSchema.safeParse(req.body);
    if (!parsed.success) return res.status(400).json({ message: "Invalid payload", issues: parsed.error.issues });

    const actorAdminId = (req.session as any)?.userId ?? null;

    const user = await storage.getUserById(userId);
    if (!user) return res.status(404).json({ message: "User not found" });

    const beforeUserTier = (user as any).userTier ?? "CANDIDATE";

    // Load verification row (create if missing)
    let verification = await db.query.userVerification.findFirst({
      where: (uv, { eq }) => eq(uv.userId, userId),
    });

    if (!verification) {
      await db.insert(userVerification).values({
        userId,
        contenderTier: "NONE",
        emailResendCountDay: 0,
        smsSendCountDay: 0,
        smsVerifyFailCount: 0,
        createdAt: new Date(),
        updatedAt: new Date(),
      });
      verification = await db.query.userVerification.findFirst({
        where: (uv, { eq }) => eq(uv.userId, userId),
      });
    }

    const beforeContenderTier = (verification as any)?.contenderTier ?? "NONE";

    // Update users tier if provided
    if (parsed.data.userTier) {
      await db
        .update(users)
        .set({
          userTier: parsed.data.userTier,
          tierPromotedAt: new Date(),
          tierPromotedBy: actorAdminId,
        } as any)
        .where(eq(users.id, userId));
    }

    // Update contenderTier if provided
    if (parsed.data.contenderTier) {
      await db
        .update(userVerification)
        .set({
          contenderTier: parsed.data.contenderTier,
          updatedAt: new Date(),
        })
        .where(eq(userVerification.userId, userId));
    }

    // Audit: adminActions + identityAudit
    await storage.logAdminAction({
      adminId: actorAdminId,
      action: "TIER_CHANGED",
      targetType: "USER",
      targetId: userId,
      details: {
        reason: parsed.data.reason ?? null,
        before: { userTier: beforeUserTier, contenderTier: beforeContenderTier },
        after: {
          userTier: parsed.data.userTier ?? beforeUserTier,
          contenderTier: parsed.data.contenderTier ?? beforeContenderTier,
        },
      },
    });

    await appendIdentityAudit({
      userId,
      category: "TIER",
      eventType: "TIER_CHANGED",
      actorType: "ADMIN",
      actorUserId: actorAdminId,
      payload: {
        reason: parsed.data.reason ?? null,
        before: { userTier: beforeUserTier, contenderTier: beforeContenderTier },
        after: {
          userTier: parsed.data.userTier ?? beforeUserTier,
          contenderTier: parsed.data.contenderTier ?? beforeContenderTier,
        },
      },
    });

    return res.json({ ok: true });
  } catch (e) {
    console.error("Error updating tier:", e);
    return res.status(500).json({ message: "Failed to update tier" });
  }
});

import { useEffect, useMemo, useState } from "react";

function getTokenFromUrl(): string | null {
  const url = new URL(window.location.href);
  return url.searchParams.get("token");
}

export default function VerifyEmail() {
  const token = useMemo(() => (typeof window !== "undefined" ? getTokenFromUrl() : null), []);
  const [status, setStatus] = useState<"idle" | "verifying" | "ok" | "error">("idle");
  const [message, setMessage] = useState<string>("");

  useEffect(() => {
    let cancelled = false;

    async function run() {
      if (!token) {
        setStatus("error");
        setMessage("Missing verification token.");
        return;
      }
      setStatus("verifying");
      try {
        const res = await apiRequest("POST", "/api/verification/email/verify", { token });
        const data = await res.json();
        if (cancelled) return;

        setStatus("ok");
        setMessage(data?.message ?? "Email verified.");

        // Redirect to profile shortly (no timers needed; allow user click too)
      } catch (e: any) {
        if (cancelled) return;
        setStatus("error");
        setMessage(e?.message ?? "Verification failed.");
      }
    }

    run();
    return () => {
      cancelled = true;
    };
  }, [token]);

  return (
    <div className="max-w-xl mx-auto p-4">
      <Card>
        <CardHeader>
          <CardTitle>Email Verification</CardTitle>
        </CardHeader>
        <CardContent className="space-y-3">
          {status === "verifying" && <div>Verifying token…</div>}
          {status === "ok" && <div className="text-sm">{message}</div>}
          {status === "error" && <div className="text-sm text-red-600">{message}</div>}

              <Link href="/profile">Go to Profile Settings</Link>
            </Button>
            <Button asChild variant="outline">
              <Link href="/">Go to Dashboard</Link>
