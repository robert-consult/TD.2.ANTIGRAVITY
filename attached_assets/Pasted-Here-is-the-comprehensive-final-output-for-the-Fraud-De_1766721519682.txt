// ----------------------------
// Client identity headers
// ----------------------------
const HEADER_DEVICE_FP = "x-device-fp";
const HEADER_DEVICE_INSTALL = "x-device-install-id";
const HEADER_CLIENT_TZ = "x-client-tz";
const HEADER_CLIENT_LANG = "x-client-lang";

export type ClientIdentityContext = {
  deviceFp?: string;
  deviceInstallId?: string;
  clientTz?: string;
  clientLang?: string;
};

export type GeoContext = {
  countryCode?: string;
  region?: string;
  city?: string;
  latitude?: number;
  longitude?: number;
  inferredTz?: string;
};

export type DeviceContext = {
  deviceType?: string;
  browser?: string;
  os?: string;
};

function readHeader(req: any, name: string): string | undefined {
  const v = req.headers?.[name] ?? req.headers?.[name.toLowerCase()];
  if (!v) return undefined;
  if (Array.isArray(v)) return String(v[0] ?? "");
  return String(v);
}

function sanitizeId(value: string | undefined, maxLen = 256): string | undefined {
  if (!value) return undefined;
  const s = value.trim();
  if (!s) return undefined;
  return s.slice(0, maxLen);
}

// Helpers for Routes
export function getClientIp(req: any): string {
  const xff = req.headers?.["x-forwarded-for"];
  if (typeof xff === "string" && xff.length > 0) return xff.split(",")[0].trim();
  if (Array.isArray(xff) && xff.length > 0) return String(xff[0]).split(",")[0].trim();
  return req.ip || req.socket?.remoteAddress || "0.0.0.0";
}

export function getUserAgent(req: any): string {
  return String(req.headers?.["user-agent"] || "unknown");
}

export function getClientIdentity(req: any): ClientIdentityContext {
  return {
    deviceFp: sanitizeId(readHeader(req, HEADER_DEVICE_FP), 128),
    deviceInstallId: sanitizeId(readHeader(req, HEADER_DEVICE_INSTALL), 128),
    clientTz: sanitizeId(readHeader(req, HEADER_CLIENT_TZ), 64),
    clientLang: sanitizeId(readHeader(req, HEADER_CLIENT_LANG), 32),
  };
}

export function parseDevice(userAgent: string | null | undefined): DeviceContext {
  if (!userAgent) return { deviceType: undefined, browser: undefined, os: undefined };
  const parser = new UAParser(userAgent);
  const device = parser.getDevice();
  const browser = parser.getBrowser();
  const os = parser.getOS();

  let deviceType = device.type || "desktop";
  const browserName = browser.name ? `${browser.name} ${browser.version || ""}`.trim() : undefined;
  const osName = os.name ? `${os.name} ${os.version || ""}`.trim() : undefined;

  return { deviceType, browser: browserName, os: osName };
}

export function buildGeoContext(ip: string | null | undefined): GeoContext {
  if (!ip || ip === "0.0.0.0") return {};
  try {
    const geo = geoip.lookup(ip);
    if (!geo) return {};
    let inferredTz: string | undefined;
    if (geo.ll && geo.ll.length === 2) {
      try { inferredTz = tzlookup(geo.ll[0], geo.ll[1]); } catch {}
    }
    return {
      countryCode: geo.country,
      region: geo.region,
      city: geo.city,
      latitude: geo.ll?.[0],
      longitude: geo.ll?.[1],
      inferredTz,
    };
  } catch {
    return {};
  }
}

export async function createUserSession(params: {

    deviceFp: params.identity?.deviceFp,
    deviceInstallId: params.identity?.deviceInstallId,
    clientTz: params.identity?.clientTz,
    clientLang: params.identity?.clientLang,
    // Parsed
    deviceType: device.deviceType,
    browser: device.browser,
    os: device.os,
    // Geo
    countryCode: geo.countryCode,
    region: geo.region,
    city: geo.city,
    latitude: geo.latitude,
    longitude: geo.longitude,
    inferredTz: geo.inferredTz,
  }).onConflictDoUpdate({
    target: userSessions.sessionId,
    set: {
      isCurrent: true,
      lastActiveAt: now,
      ip: params.ip,
      userAgent: params.userAgent,
      deviceFp: params.identity?.deviceFp,
      deviceInstallId: params.identity?.deviceInstallId,
    },
  });

  // Log successful login
  await db.insert(userLoginHistory).values({

    success: true,

export async function touchSession(sessionId: string, ctx?: { ip?: string; userAgent?: string; identity?: ClientIdentityContext }) {
  const updates: any = { lastActiveAt: new Date(), isCurrent: true };
  
  if (ctx?.ip) {
    updates.ip = ctx.ip;
    const geo = buildGeoContext(ctx.ip);
    if (geo.countryCode) {
      updates.countryCode = geo.countryCode;
      updates.region = geo.region;
      updates.city = geo.city;
      updates.latitude = geo.latitude;
      updates.longitude = geo.longitude;
      updates.inferredTz = geo.inferredTz;
    }
  }
  
  if (ctx?.userAgent) {
    updates.userAgent = ctx.userAgent;
    const dev = parseDevice(ctx.userAgent);
    updates.deviceType = dev.deviceType;
    updates.browser = dev.browser;
    updates.os = dev.os;
  }

  if (ctx?.identity) {
    if (ctx.identity.deviceFp) updates.deviceFp = ctx.identity.deviceFp;
    if (ctx.identity.deviceInstallId) updates.deviceInstallId = ctx.identity.deviceInstallId;
    if (ctx.identity.clientTz) updates.clientTz = ctx.identity.clientTz;
    if (ctx.identity.clientLang) updates.clientLang = ctx.identity.clientLang;
  }

  await db.update(userSessions)
    .set(updates)
    .where(and(eq(userSessions.sessionId, sessionId), isNull(userSessions.revokedAt)));
}

export async function endSession(sessionId: string, reason: string = "LOGOUT") {
  const now = new Date();
  
  // 1. Mark session as not current
  const session = await db.update(userSessions)
    .set({ isCurrent: false, lastActiveAt: now })
    .where(eq(userSessions.sessionId, sessionId))
    .returning({ createdAt: userSessions.createdAt })
    .get();

  // 2. Update the corresponding login history row to record duration
  if (session) {
    const createdAt = new Date(session.createdAt);
    const duration = Math.floor((now.getTime() - createdAt.getTime()) / 1000);
    
    // Find the latest successful login for this session
    const lastHistory = await db.select({ id: userLoginHistory.id })
      .from(userLoginHistory)
      .where(and(eq(userLoginHistory.sessionId, sessionId), eq(userLoginHistory.eventType, "LOGIN_SUCCESS")))
      .orderBy(desc(userLoginHistory.createdAt))
      .limit(1)
      .get();

    if (lastHistory) {
      await db.update(userLoginHistory)
        .set({ logoutAt: now, sessionLengthSec: duration })
        .where(eq(userLoginHistory.id, lastHistory.id));
    }
  }

  // 3. Insert a LOGOUT event
  // (We skip strict user lookups for brevity, assuming sessionId is valid context enough or passed in)
}

import { db } from "@db";
import {

export type FraudSeverity = "LOW" | "MEDIUM" | "HIGH" | "CRITICAL";

type IdentityEvent = {

  ip?: string;
  userAgent?: string;
  geo?: GeoContext;
  identity?: ClientIdentityContext;
  eventType: "LOGIN_SUCCESS" | "REGISTER_SUCCESS" | "TRADE_OPEN";
  // trade context
  tradeId?: number;
  symbolId?: number;
  side?: "BUY" | "SELL";
};

// --- Helpers ---
function now() { return new Date(); }
function safeJson(v: any) { try { return JSON.stringify(v); } catch { return "{}"; } }
function parseJson(v: string) { try { return JSON.parse(v); } catch { return {}; } }

// --- Core Logic ---

function upsertIdentityLink(params: { linkType: string; linkKey: string; userId: number; ip?: string; country?: string }) {
  const existing = db.select().from(fraudIdentityLinks)
    .where(and(
      eq(fraudIdentityLinks.linkType, params.linkType),
      eq(fraudIdentityLinks.linkKey, params.linkKey),
      eq(fraudIdentityLinks.userId, params.userId)
    )).get();

  if (existing) {
    db.update(fraudIdentityLinks)
      .set({
        lastSeenAt: now(),
        seenCount: (existing.seenCount || 0) + 1,
        lastIp: params.ip,
        lastCountryCode: params.country
      })
      .where(eq(fraudIdentityLinks.id, existing.id)).run();
  } else {
    db.insert(fraudIdentityLinks).values({
      linkType: params.linkType,
      linkKey: params.linkKey,
      userId: params.userId,
      firstSeenAt: now(),
      lastSeenAt: now(),
      seenCount: 1,
      lastIp: params.ip,
      lastCountryCode: params.country
    }).run();
  }
}

function upsertAlert(alert: {
  dedupeKey: string; userId: number; severity: FraudSeverity; category: string;
  score: number; title: string; details: any; relatedUserIds?: number[];
}) {
  try {
    db.insert(fraudAlerts).values({
      dedupeKey: alert.dedupeKey,
      userId: alert.userId,
      severity: alert.severity,
      category: alert.category,
      status: "OPEN",
      score: alert.score,
      title: alert.title,
      detailsJson: safeJson(alert.details),
      relatedUserIdsJson: safeJson(alert.relatedUserIds || []),
      createdAt: now(),
      updatedAt: now()
    }).run();
  } catch {
    // Already exists; update timestamp
    db.update(fraudAlerts)
      .set({ updatedAt: now(), score: alert.score, severity: alert.severity })
      .where(eq(fraudAlerts.dedupeKey, alert.dedupeKey)).run();
  }
}

function getOtherUsersForLink(type: string, key: string, excludeUserId: number): number[] {
  const rows = db.select({ uid: fraudIdentityLinks.userId })
    .from(fraudIdentityLinks)
    .where(and(eq(fraudIdentityLinks.linkType, type), eq(fraudIdentityLinks.linkKey, key), ne(fraudIdentityLinks.userId, excludeUserId)))
    .all();
  return rows.map(r => r.uid);
}

// --- Public Handlers ---

export function onIdentityEvent(evt: IdentityEvent) {
  // 1. Link Identity
  if (evt.identity?.deviceInstallId) {
    upsertIdentityLink({
      linkType: "DEVICE_INSTALL",
      linkKey: evt.identity.deviceInstallId,
      userId: evt.userId,
      ip: evt.ip,
      country: evt.geo?.countryCode
    });
    
    // Check Multi-Account (Strong)
    const others = getOtherUsersForLink("DEVICE_INSTALL", evt.identity.deviceInstallId, evt.userId);
    if (others.length > 0) {
      upsertAlert({
        dedupeKey: `MULTI:${evt.userId}:INSTALL:${evt.identity.deviceInstallId}`,
        userId: evt.userId,
        severity: "HIGH",
        category: "MULTI_ACCOUNT",
        score: 80,
        title: `Shared Device Install (${others.length + 1} accounts)`,
        details: { linkKey: evt.identity.deviceInstallId, others },
        relatedUserIds: [evt.userId, ...others]
      });
    }
  }

  if (evt.identity?.deviceFp) {
    upsertIdentityLink({
      linkType: "DEVICE_FP",
      linkKey: evt.identity.deviceFp,
      userId: evt.userId,
      ip: evt.ip,
      country: evt.geo?.countryCode
    });
    // Check Multi-Account (Weak)
    const others = getOtherUsersForLink("DEVICE_FP", evt.identity.deviceFp, evt.userId);
    if (others.length >= 2) { // Tolerance of 2
      upsertAlert({
        dedupeKey: `MULTI:${evt.userId}:FP:${evt.identity.deviceFp}`,
        userId: evt.userId,
        severity: "MEDIUM",
        category: "MULTI_ACCOUNT",
        score: 40,
        title: `Shared Device Fingerprint (${others.length + 1} accounts)`,
        details: { linkKey: evt.identity.deviceFp, others },
        relatedUserIds: [evt.userId, ...others]
      });
    }
  }

  if (evt.ip) {
    upsertIdentityLink({ linkType: "IP", linkKey: evt.ip, userId: evt.userId, ip: evt.ip, country: evt.geo?.countryCode });
  }

  // 2. Recompute Risk Score
  recomputeUserRisk(evt.userId);
}

export function onTradeOpened(params: { userId: number; tradeId: number; symbolId: number; side: "BUY" | "SELL"; identity?: ClientIdentityContext; ip?: string }) {
  // First, standard identity logging
  onIdentityEvent({ ...params, eventType: "TRADE_OPEN" });

  // Hedging Detection
  // Find linked users via device install ID (strongest link)
  const installId = params.identity?.deviceInstallId;
  if (!installId) return;

  const linkedUsers = getOtherUsersForLink("DEVICE_INSTALL", installId, params.userId);
  if (linkedUsers.length === 0) return;

  const oppositeSide = params.side === "BUY" ? "SELL" : "BUY";
  
  // Check if any linked user has an OPEN trade on the same symbol with OPPOSITE side
  const hedgingTrades = db.select({ id: trades.id, userId: trades.userId })
    .from(trades)
    .where(and(
      inArray(trades.userId, linkedUsers),
      eq(trades.symbolId, params.symbolId),
      eq(trades.type, oppositeSide),
      inArray(trades.status, ["OPEN", "PENDING"])
    )).all();

  if (hedgingTrades.length > 0) {
    const symbolRow = db.select().from(symbolConfigs).where(eq(symbolConfigs.id, params.symbolId)).get();
    const symbolStr = symbolRow?.symbol || String(params.symbolId);
    
    // Alert Current User
    upsertAlert({
      dedupeKey: `HEDGE:${params.userId}:${params.tradeId}`,
      userId: params.userId,
      severity: "CRITICAL",
      category: "HEDGING",
      score: 100,
      title: `Cross-Account Hedging Detected (${symbolStr})`,
      details: { tradeId: params.tradeId, oppositeTrades: hedgingTrades, installId },
      relatedUserIds: [params.userId, ...linkedUsers]
    });

    // Alert Linked Users
    for (const t of hedgingTrades) {
      upsertAlert({
        dedupeKey: `HEDGE:${t.userId}:${t.id}`,
        userId: t.userId,
        severity: "CRITICAL",
        category: "HEDGING",
        score: 100,
        title: `Cross-Account Hedging Detected (${symbolStr})`,
        details: { tradeId: t.id, triggeringTrade: params.tradeId, triggeringUser: params.userId },
        relatedUserIds: [t.userId, params.userId]
      });
      recomputeUserRisk(t.userId);
    }
    recomputeUserRisk(params.userId);
  }
}

export function recomputeUserRisk(userId: number) {
  // Simple scoring
  let score = 0;
  const reasons: string[] = [];

  // 1. Alert Score
  const alerts = db.select().from(fraudAlerts).where(and(eq(fraudAlerts.userId, userId), eq(fraudAlerts.status, "OPEN"))).all();
  for (const a of alerts) {
    score += a.score;
    if (a.score > 0) reasons.push(a.title);
  }

  // 2. Churn (IPs in last 24h)
  const ipCount = db.select({ count: sql<number>`count(distinct ${userSessions.ip})` })
    .from(userSessions)
    .where(and(eq(userSessions.userId, userId), sql`${userSessions.createdAt} > date('now', '-1 day')`))
    .get()?.count || 0;
  
  if (ipCount > 5) {
    score += 20;
    reasons.push(`High IP churn (${ipCount} in 24h)`);
  }

  const level: FraudSeverity = score > 80 ? "CRITICAL" : score > 50 ? "HIGH" : score > 20 ? "MEDIUM" : "LOW";

  // Upsert Risk Table
  const existing = db.select().from(fraudUserRisk).where(eq(fraudUserRisk.userId, userId)).get();
  if (existing) {
    db.update(fraudUserRisk).set({
      riskScore: score,
      riskLevel: level,
      reasonsJson: safeJson(reasons),
      openAlertsCount: alerts.length,
      lastEvaluatedAt: now()
    }).where(eq(fraudUserRisk.userId, userId)).run();
  } else {
    db.insert(fraudUserRisk).values({
      userId,
      riskScore: score,
      riskLevel: level,
      reasonsJson: safeJson(reasons),
      openAlertsCount: alerts.length,
      lastEvaluatedAt: now()
    }).run();
  }
}

export function listFraudUsers(limit = 100) {
  return db.select({
    userId: users.id,
    email: users.email,
    riskScore: fraudUserRisk.riskScore,
    riskLevel: fraudUserRisk.riskLevel,
    openAlerts: fraudUserRisk.openAlertsCount,
    reasons: fraudUserRisk.reasonsJson
  })
  .from(users)
  .leftJoin(fraudUserRisk, eq(users.id, fraudUserRisk.userId))
  .orderBy(desc(fraudUserRisk.riskScore))
  .limit(limit)
  .all()
  .map(u => ({ ...u, reasons: parseJson(u.reasons || '[]') }));
}

import { v4 as uuidv4 } from 'uuid';

export function getDeviceInstallId(): string {
  const key = 'device_install_id';
  let id = localStorage.getItem(key);
  if (!id) {
    id = uuidv4();
    localStorage.setItem(key, id);
  }
  return id;
}

export async function getDeviceFingerprint(): Promise<string> {
  // Simple entropy collection (Canvas + UserAgent + Screen + Timezone)
  // In a real app, use a library like FingerprintJS.
  // This is a "light" version for demonstration.
  const parts = [
    navigator.userAgent,
    navigator.language,
    new Date().getTimezoneOffset(),
    screen.width + 'x' + screen.height,
    // Add canvas noise if needed
  ];
  
  const str = parts.join('|');
  const encoder = new TextEncoder();
  const data = encoder.encode(str);
  const hashBuffer = await crypto.subtle.digest('SHA-256', data);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
}
