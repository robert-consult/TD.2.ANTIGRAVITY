import { Request, Response, NextFunction } from "express";
import { buildDecisionContext } from "../policy/buildDecisionContext";
import {
  decidePolicy,
  type PolicyAction,
  type PolicyDecision,
  type DecisionContext,
} from "../../shared/policyDecision";
import { appendIdentityAudit } from "../services/identityAudit";

export interface PolicyEnforcedRequest extends Request {
  policyDecision?: PolicyDecision;
  policyAction?: PolicyAction;
}

type PolicyActionResolver = PolicyAction | ((req: PolicyEnforcedRequest) => PolicyAction);

type PolicyOverrideBuilder =
  | ((req: PolicyEnforcedRequest, baseCtx: DecisionContext) => Promise<Partial<DecisionContext>> | Partial<DecisionContext>)
  | undefined;

export function requirePolicy(actionOrResolver: PolicyActionResolver, buildOverrides?: PolicyOverrideBuilder) {
  return async (req: PolicyEnforcedRequest, res: Response, next: NextFunction) => {
    const session: any = (req as any).session;
    const userId = session?.userId;
    const actorAdminId = session?.realAdminId ?? null; // present when admin is impersonating

    if (!userId) {
      return res.status(401).json({ message: "Not authenticated" });
    }

    try {
      const action = typeof actionOrResolver === "function" ? actionOrResolver(req) : actionOrResolver;

      // Base ctx
      let ctx = await buildDecisionContext({
        userId,
        nowMs: Date.now(),
      });

      // Overrides (tradeIntent, etc.)
      if (buildOverrides) {
        const overrides = await buildOverrides(req, ctx);
        ctx = await buildDecisionContext({
          userId,
          nowMs: Date.now(),
          overrides,
          tradeIntent: (overrides as any)?.tradeIntent ?? ctx.tradeIntent,
        });
      }

      const decision = decidePolicy(action, ctx);
      req.policyDecision = decision;
      req.policyAction = action;

      if (!decision.allowed) {
        try {
          appendIdentityAudit({
            userId,
            actorAdminId,
            email: ctx.user.email,
            username: ctx.user.username ?? undefined,
            category: "POLICY",
            type: "ACTION_DENIED",
            title: `Policy denied: ${action}`,
            description: `Deny code: ${decision.deny_code}, Account state: ${decision.accountState}`,
            ip: (req.ip || (req.headers["x-forwarded-for"] as string) || undefined) as any,
            userAgent: req.headers["user-agent"],
          });
        } catch (auditErr) {
          console.error("Error writing policy deny audit:", auditErr);
        }

        const httpStatus = getHttpStatusForDenyCode(decision.deny_code);
        return res.status(httpStatus).json({
          message: getMessageForDenyCode(decision.deny_code),
          deny_code: decision.deny_code,
          accountState: decision.accountState,
          showLockedBanner: decision.showLockedBanner,
          redirectTo: decision.redirectTo,
        });
      }

      next();
    } catch (error) {
      console.error("Policy enforcement error:", error);
      return res.status(500).json({ message: "Policy check failed" });
    }
  };
}

function getHttpStatusForDenyCode(code?: string): number {
  switch (code) {
    case "ACCOUNT_SUSPENDED":
      return 403;
    case "EMAIL_NOT_VERIFIED":
    case "EMAIL_REVERIFY_OVERDUE":
      return 403;
    case "EMAIL_SEND_RATE_LIMIT":
    case "EMAIL_SEND_COOLDOWN":
    case "SMS_RATE_LIMIT":
    case "SMS_OTP_TOO_MANY_ATTEMPTS":
      return 429;
    case "SMS_NOT_ELIGIBLE":
    case "SMS_NOT_ELIGIBLE_TIER":
    case "KYC_NOT_SELECTED":
    case "KYC_STATE_INVALID":
    case "ACTION_NOT_ALLOWED_IN_STATE":
      return 403;
    default:
      return 403;
  }
}

function getMessageForDenyCode(code?: string): string {
  switch (code) {
    case "ACCOUNT_SUSPENDED":
      return "Your account has been suspended. Please contact support.";
    case "EMAIL_NOT_VERIFIED":
      return "Please verify your email address to continue trading.";
    case "EMAIL_REVERIFY_OVERDUE":
      return "Your email verification has expired. Please re-verify your email.";
    case "EMAIL_SEND_RATE_LIMIT":
      return "Daily email limit reached. Please try again tomorrow.";
    case "EMAIL_SEND_COOLDOWN":
      return "Please wait before requesting another verification email.";
    case "SMS_NOT_ELIGIBLE":
      return "Phone verification is not available for your account type.";
    case "SMS_NOT_ELIGIBLE_TIER":
      return "Phone verification requires account progression. Keep trading to unlock.";
    case "SMS_RATE_LIMIT":
      return "SMS limit reached. Please try again later.";
    case "SMS_OTP_TOO_MANY_ATTEMPTS":
      return "Too many failed attempts. Please wait before trying again.";
    case "KYC_NOT_SELECTED":
      return "KYC verification is only available for selected traders.";
    case "KYC_STATE_INVALID":
      return "KYC submission is not allowed in the current state.";
    case "ACTION_NOT_ALLOWED_IN_STATE":
      return "This action is not allowed in your current account state.";
    default:
      return "Action not permitted.";
  }
}

### FULL FILE

Create `server/services/contenderProgress.ts`:

```ts
import { and, eq } from "drizzle-orm";
import { db } from "../db/index";
import { users, userVerification } from "../../shared/schema";
import { computeContenderEligibility, type ContenderTier, type UserTier } from "../../shared/policyDecision";
import { buildDecisionContext } from "../policy/buildDecisionContext";
import { appendIdentityAudit } from "./identityAudit";

/**
 * Persists contender eligibility + tier progression safely (additive-only).
 * - Ensures user_verification row exists
 * - Advances contenderTier to CANDIDATE_SMS_REQUIRED once eligible+emailVerified (unless already beyond)
 * - Auto-promotes users.userTier: CANDIDATE -> PERFORMER once eligible+emailVerified
 */
export async function syncContenderProgress(userId: number, actorAdminId: number | null = null) {
  await db.insert(userVerification).values({ userId }).onConflictDoNothing();

  const nowMs = Date.now();
  const nowSec = Math.floor(nowMs / 1000);

  const ctx = await buildDecisionContext({ userId, nowMs });
  const contender = computeContenderEligibility(ctx);

  if (!contender.eligible) return;

  // Load current verification row to check contenderEligibleAt existence
  const uv = await db.query.userVerification.findFirst({
    where: eq(userVerification.userId, userId),
  });

  const patch: any = {};

  if (uv && !uv.contenderEligibleAt) {
    patch.contenderEligibleAt = nowSec;
  }

  if (ctx.user.emailVerifiedAt) {
    const current = (uv?.contenderTier ?? ctx.user.contenderTier ?? "NONE") as ContenderTier;
    const alreadyBeyond = current === "VERIFIED_SMS" || current === "SELECTED_REAL_CAPITAL";

    if (!alreadyBeyond) {
      const desired: ContenderTier = ctx.user.phoneVerifiedAt ? "VERIFIED_SMS" : "CANDIDATE_SMS_REQUIRED";
      if (current !== desired) {
        patch.contenderTier = desired;
      }
    }
  }

  if (Object.keys(patch).length > 0) {
    await db.update(userVerification).set(patch).where(eq(userVerification.userId, userId));
  }

  if (ctx.user.emailVerifiedAt && ctx.user.userTier === "CANDIDATE") {
    const updated = await db
      .update(users)
      .set({ userTier: "PERFORMER" as UserTier })
      .where(and(eq(users.id, userId), eq(users.userTier, "CANDIDATE")))
      .returning({ id: users.id });

    if (updated.length > 0) {
      appendIdentityAudit({
        userId,
        actorAdminId,
        email: ctx.user.email,
        username: ctx.user.username ?? undefined,
        category: "TIER",
        type: "TIER_CHANGED",
        title: "Auto promotion to PERFORMER",
        description: "Eligible contender + email verified. Auto-progression applied.",
      });
    }
  }
}
```
