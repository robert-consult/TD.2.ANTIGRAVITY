import type { Request } from "express";
import { db } from "@db";
import { systemConfig } from "@shared/schema";
import { eq } from "drizzle-orm";
import { getClientIp } from "../lib/auditContext";

export type CaptchaProvider = "TURNSTILE" | "HCAPTCHA";

export async function getCaptchaPolicy(): Promise<{
  enforceSignupCaptcha: boolean;
  provider: CaptchaProvider;
}> {
  const cfg = db.select().from(systemConfig).where(eq(systemConfig.id, 1)).get();
  return {
    enforceSignupCaptcha: Boolean((cfg as any)?.signupCaptchaEnforce ?? false),
    provider: ((cfg as any)?.captchaProvider ?? "TURNSTILE") as CaptchaProvider,
  };
}

export async function verifySignupCaptcha(
  req: Request,
  captchaToken: string | undefined | null,
): Promise<{ ok: true } | { ok: false; message: string }> {
  const { enforceSignupCaptcha, provider } = await getCaptchaPolicy();
  if (!enforceSignupCaptcha) return { ok: true };

  if (!captchaToken || captchaToken.trim().length < 10) {
    return { ok: false, message: "CAPTCHA_REQUIRED" };
  }

  const ip = getClientIp(req) || undefined;

  try {
    if (provider === "TURNSTILE") {
      const secret = process.env.TURNSTILE_SECRET_KEY || process.env.CAPTCHA_TURNSTILE_SECRET;
      if (!secret) return { ok: false, message: "CAPTCHA_SECRET_NOT_CONFIGURED" };

      const form = new URLSearchParams();
      form.set("secret", secret);
      form.set("response", captchaToken);
      if (ip) form.set("remoteip", ip);

      const resp = await fetch("https://challenges.cloudflare.com/turnstile/v0/siteverify", {
        method: "POST",
        headers: { "content-type": "application/x-www-form-urlencoded" },
        body: form.toString(),
      });

      const data: any = await resp.json().catch(() => ({}));
      if (!data?.success) {
        const codes = Array.isArray(data?.["error-codes"]) ? data["error-codes"].join(",") : "unknown";
        return { ok: false, message: `CAPTCHA_FAILED:${codes}` };
      }
      return { ok: true };
    }

    if (provider === "HCAPTCHA") {
      const secret = process.env.HCAPTCHA_SECRET_KEY || process.env.CAPTCHA_HCAPTCHA_SECRET;
      if (!secret) return { ok: false, message: "CAPTCHA_SECRET_NOT_CONFIGURED" };

      const form = new URLSearchParams();
      form.set("secret", secret);
      form.set("response", captchaToken);
      if (ip) form.set("remoteip", ip);

      const resp = await fetch("https://hcaptcha.com/siteverify", {
        method: "POST",
        headers: { "content-type": "application/x-www-form-urlencoded" },
        body: form.toString(),
      });

      const data: any = await resp.json().catch(() => ({}));
      if (!data?.success) {
        const codes = Array.isArray(data?.["error-codes"]) ? data["error-codes"].join(",") : "unknown";
        return { ok: false, message: `CAPTCHA_FAILED:${codes}` };
      }
      return { ok: true };
    }

    return { ok: false, message: "CAPTCHA_PROVIDER_UNSUPPORTED" };
  } catch (e: any) {
    return { ok: false, message: `CAPTCHA_VERIFY_ERROR:${e?.message ?? "unknown"}` };
  }
}

export function ensureLegalComplianceSchema() {
  const db = new Database("trading_app.db");

  const hasColumn = (table: string, column: string) => {
    const cols = db.prepare(`PRAGMA table_info(${table})`).all() as any[];
    return cols.some((c) => c?.name === column);
  };

  try {
    // --- system_config: add missing columns for legal + captcha
    if (!hasColumn("system_config", "legal_coverage_enforce")) {
      db.exec(`ALTER TABLE system_config ADD COLUMN legal_coverage_enforce INTEGER NOT NULL DEFAULT 0;`);
    }
    if (!hasColumn("system_config", "signup_captcha_enforce")) {
      db.exec(`ALTER TABLE system_config ADD COLUMN signup_captcha_enforce INTEGER NOT NULL DEFAULT 0;`);
    }
    if (!hasColumn("system_config", "captcha_provider")) {
      db.exec(`ALTER TABLE system_config ADD COLUMN captcha_provider TEXT NOT NULL DEFAULT 'TURNSTILE';`);
    }

    // --- users: region/country capture
    if (!hasColumn("users", "country_iso2")) {
      db.exec(`ALTER TABLE users ADD COLUMN country_iso2 TEXT;`);
    }
    if (!hasColumn("users", "region_key")) {
      db.exec(`ALTER TABLE users ADD COLUMN region_key TEXT;`);
    }

    // --- legal_documents
    db.exec(`
      CREATE TABLE IF NOT EXISTS legal_documents (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        doc_set TEXT NOT NULL,
        doc_type TEXT NOT NULL,
        jurisdiction_type TEXT NOT NULL,
        jurisdiction_key TEXT NOT NULL,
        version TEXT NOT NULL,
        sha256 TEXT NOT NULL,
        content TEXT NOT NULL,
        notes TEXT,
        created_at INTEGER NOT NULL DEFAULT (CAST(strftime('%s','now') AS INTEGER) * 1000),
        created_by TEXT
      );
    `);

    // --- legal_doc_pointers
    db.exec(`
      CREATE TABLE IF NOT EXISTS legal_doc_pointers (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        doc_set TEXT NOT NULL,

export function generateDoc1TermsToken(payload: Doc1TermsTokenPayload): string {
  const payloadJson = stableStringify(payload);
  const payloadB64 = base64UrlEncode(payloadJson);
  const sigHex = hmacSha256Hex(requireSecret(), payloadB64);
  return `${payloadB64}.${sigHex}`;
}

export function verifyDoc1TermsToken(
  token: string,
  opts?: { expectedCountryIso2?: string; maxAgeMs?: number },
): { ok: true; payload: Doc1TermsTokenPayload } | { ok: false; error: string } {
  if (!token || typeof token !== "string") return { ok: false, error: "TOKEN_MISSING" };

  const parts = token.split(".");
  if (parts.length !== 2) return { ok: false, error: "TOKEN_FORMAT_INVALID" };

  const [payloadB64, sigHex] = parts;
  if (!payloadB64 || !sigHex) return { ok: false, error: "TOKEN_FORMAT_INVALID" };

  const expected = hmacSha256Hex(requireSecret(), payloadB64);
  if (!hmacVerify(expected, sigHex)) return { ok: false, error: "TOKEN_SIGNATURE_INVALID" };

  let payload: any;
  try {
    payload = JSON.parse(base64UrlDecode(payloadB64));
  } catch {
    return { ok: false, error: "TOKEN_PAYLOAD_INVALID" };
  }

  if (payload?.v !== 1) return { ok: false, error: "TOKEN_VERSION_UNSUPPORTED" };
  if (!payload?.countryIso2) return { ok: false, error: "TOKEN_COUNTRY_MISSING" };
  if (!payload?.combinedSha256) return { ok: false, error: "TOKEN_COMBINED_SHA_MISSING" };
  if (!payload?.ts || typeof payload.ts !== "number") return { ok: false, error: "TOKEN_TS_INVALID" };

  if (opts?.expectedCountryIso2 && payload.countryIso2 !== opts.expectedCountryIso2) {
    return { ok: false, error: "TOKEN_COUNTRY_MISMATCH" };
  }

  if (opts?.maxAgeMs) {
    const age = Date.now() - payload.ts;
    if (age < 0 || age > opts.maxAgeMs) return { ok: false, error: "TOKEN_EXPIRED" };
  }

  return { ok: true, payload: payload as Doc1TermsTokenPayload };
}

export function PhoneNumberInput({ countryIso2, value, onChange }: Props) {
  const [raw, setRaw] = useState(value);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    try {
      if (!raw) {
        onChange("", false);
        return;
      }

      const phone = parsePhoneNumberFromString(raw, countryIso2 as any);

        setError("Invalid phone number");
        onChange(raw, false);
        return;

      setError(null);
      onChange(phone.number, true); // E.164
    } catch {
      setError("Invalid phone number");
      onChange(raw, false);
    }
  }, [raw, countryIso2]);

  const prefix = countryIso2

    : "";

  return (
    <div className="space-y-1">
      <label className="text-sm">Phone Number</label>
      <input
        value={raw}
        onChange={(e) => setRaw(e.target.value)}
        placeholder={`${prefix} 7XXXXXXXX`}
        className="w-full rounded bg-black/40 px-3 py-2 border border-neutral-700"
      />
      {error && <div className="text-xs text-red-500">{error}</div>}
    </div>
  );
}

import {

  hmacSha256Hex,
  requireSecret,
  timingSafeEqualHex,
} from "./cryptoUtils";

export type TermsTokenPayloadV2 = {
  v: 2;
  countryCode: string; // ISO2
  globalId: number;
  addendumId: number;
  combinedSha256: string;
  issuedAt: number; // epoch ms
  expAt: number; // epoch ms
};

function normalizeIso2(iso2: string): string {
  return String(iso2 || "")
    .trim()
    .toUpperCase()
    .replace(/[^A-Z]/g, "")
    .slice(0, 2);
}

function signPayloadB64(payloadB64: string): string {
  const secret = requireSecret();
  return hmacSha256Hex(secret, payloadB64);
}

/**
 * v2 token for the Doc1 flow.
 * Format: base64url(JSON).HMAC_SHA256_HEX(base64url(JSON))
 */
export function createTermsToken(params: Omit<TermsTokenPayloadV2, "v">): string {
  const payload: TermsTokenPayloadV2 = {
    v: 2,
    countryCode: normalizeIso2(params.countryCode),
    globalId: Number(params.globalId),
    addendumId: Number(params.addendumId),
    combinedSha256: String(params.combinedSha256 || ""),
    issuedAt: Number(params.issuedAt),
    expAt: Number(params.expAt),
  };

  const payloadJson = JSON.stringify(payload);
  const payloadB64 = base64UrlEncode(payloadJson);
  const sig = signPayloadB64(payloadB64);
  return `${payloadB64}.${sig}`;
}

export function verifyTermsTokenV2(token: string): {
  valid: boolean;
  payload?: TermsTokenPayloadV2;
  reason?: string;
} {
  try {
    const raw = String(token || "").trim();
    const [payloadB64, sig] = raw.split(".");
    if (!payloadB64 || !sig) return { valid: false, reason: "MALFORMED" };

    const expected = signPayloadB64(payloadB64);
    if (!timingSafeEqualHex(expected, sig)) return { valid: false, reason: "BAD_SIGNATURE" };

    const payloadJson = base64UrlDecode(payloadB64);
    const payload = JSON.parse(payloadJson) as TermsTokenPayloadV2;

    if (!payload || payload.v !== 2) return { valid: false, reason: "BAD_VERSION" };
    if (!payload.countryCode || !/^[A-Z]{2}$/.test(payload.countryCode)) {
      return { valid: false, reason: "BAD_COUNTRY" };
    }
    if (!Number.isFinite(payload.globalId) || !Number.isFinite(payload.addendumId)) {
      return { valid: false, reason: "BAD_DOC_IDS" };
    }
    if (!payload.combinedSha256 || !/^[a-f0-9]{64}$/i.test(payload.combinedSha256)) {
      return { valid: false, reason: "BAD_HASH" };
    }
    if (!Number.isFinite(payload.issuedAt) || !Number.isFinite(payload.expAt)) {
      return { valid: false, reason: "BAD_TIME" };
    }
    if (payload.expAt < Date.now()) return { valid: false, reason: "EXPIRED" };

    return { valid: true, payload };
  } catch {
    return { valid: false, reason: "ERROR" };
  }
}

/**
 * Append a tamper-evident legal acceptance record.
 *
 * Notes:
 * - ledgerSeq is monotonic.
 * - ledgerHash is SHA-256(prevLedgerHash | canonical_payload)
 * - acceptedAt is stored as sqlite timestamp (seconds) via Drizzle.
 */
export function appendLegalAcceptance(params: AppendLegalAcceptanceParams) {
  const last = db
    .select({ ledgerSeq: legalAcceptances.ledgerSeq, ledgerHash: legalAcceptances.ledgerHash })
    .from(legalAcceptances)
    .orderBy(desc(legalAcceptances.ledgerSeq))
    .limit(1)
    .get();

  const ledgerSeq = Number(last?.ledgerSeq ?? 0) + 1;
  const prevLedgerHash = String(last?.ledgerHash ?? "GENESIS");

  const acceptedAt = new Date();
  const acceptedAtSec = Math.floor(acceptedAt.getTime() / 1000);

  const canonicalPayload = {
    ledgerSeq,
    prevLedgerHash,
    userId: params.userId,
    emailAtAcceptance: params.emailAtAcceptance,
    countryIso2: params.countryIso2,
    regionKey: params.regionKey,
    globalDocId: params.globalDocId,
    globalDocVersion: params.globalDocVersion,
    globalDocSha256: params.globalDocSha256,
    addendumId: params.addendumId,
    addendumVersion: params.addendumVersion,
    addendumSha256: params.addendumSha256,
    combinedSha256: params.combinedSha256,
    acceptedAt: acceptedAtSec,
    ipAddress: params.ipAddress || null,
    userAgent: params.userAgent || null,
    sessionId: params.sessionId || null,
  };

  const ledgerHash = sha256(`${prevLedgerHash}|${stableStringify(canonicalPayload)}`);

  db.insert(legalAcceptances)
    .values({
      userId: params.userId,
      emailAtAcceptance: params.emailAtAcceptance,
      countryIso2: params.countryIso2,
      regionKey: params.regionKey,

      globalDocId: params.globalDocId,
      globalDocVersion: params.globalDocVersion,
      globalDocSha256: params.globalDocSha256,

      addendumId: params.addendumId,
      addendumVersion: params.addendumVersion,
      addendumSha256: params.addendumSha256,

      combinedSha256: params.combinedSha256,
      combinedText: params.combinedText,

      ledgerSeq,
      prevLedgerHash,
      ledgerHash,

      ipAddress: params.ipAddress || null,
      userAgent: params.userAgent || null,
      sessionId: params.sessionId || null,

      // legacy fields for compatibility/exports
      termsToken: params.termsToken || null,
      termsTokenVerified: Boolean(params.termsTokenVerified),
      acceptedFromIp: params.ipAddress || null,
      acceptedUserAgent: params.userAgent || null,

      acceptedAt,
    })
    .run();

  return { ledgerSeq, prevLedgerHash, ledgerHash };
}

/**
 * Left-to-right “slider captcha” UX.
 * NOTE: This is a UI friction control, not a cryptographic CAPTCHA.
 */
export function SliderCaptcha({
  disabled,
  className,
  onVerified,
  resetKey,
  label = "Slide to verify",
  helpText = "",
}: SliderCaptchaProps) {
  const trackRef = useRef<HTMLDivElement | null>(null);
  const handleRef = useRef<HTMLButtonElement | null>(null);

  const [isDragging, setIsDragging] = useState(false);
  const [isVerified, setIsVerified] = useState(false);
  const [x, setX] = useState(0);
  const [trackW, setTrackW] = useState(0);
  const [handleW, setHandleW] = useState(0);

  const maxX = useMemo(() => Math.max(0, trackW - handleW), [trackW, handleW]);
  const pct = useMemo(() => (maxX > 0 ? Math.min(1, x / maxX) : 0), [x, maxX]);

  const measure = useCallback(() => {
    const track = trackRef.current;
    const handle = handleRef.current;
    if (!track || !handle) return;
    const tw = track.getBoundingClientRect().width;
    const hw = handle.getBoundingClientRect().width;
    setTrackW(tw);
    setHandleW(hw);
  }, []);

  useEffect(() => {
    measure();
    const onResize = () => measure();
    window.addEventListener("resize", onResize);
    return () => window.removeEventListener("resize", onResize);
  }, [measure]);

  useEffect(() => {
    // External reset
    setIsDragging(false);
    setIsVerified(false);
    setX(0);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [resetKey]);

  const startDrag = useCallback(
    (e: React.PointerEvent) => {
      if (disabled || isVerified) return;
      setIsDragging(true);
      (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
    },
    [disabled, isVerified]
  );

  const onMove = useCallback(
    (e: React.PointerEvent) => {
      if (!isDragging || disabled || isVerified) return;
      const track = trackRef.current;
      const handle = handleRef.current;
      if (!track || !handle) return;
      const rect = track.getBoundingClientRect();
      const hw = handle.getBoundingClientRect().width;
      // Center handle under pointer.
      const nx = e.clientX - rect.left - hw / 2;
      setX(Math.max(0, Math.min(maxX, nx)));
    },
    [disabled, isDragging, isVerified, maxX]
  );

  const finish = useCallback(() => {
    if (disabled || isVerified) return;
    setIsDragging(false);
    const hitEnd = x >= maxX - 2;
    if (hitEnd) {
      setX(maxX);
      setIsVerified(true);
      onVerified?.();
    } else {
      setX(0);
    }
  }, [disabled, isVerified, maxX, onVerified, x]);

  return (
    <div className={className}>
      <div className="mb-2 flex items-center justify-between">
        <div className="text-sm font-medium text-muted-foreground">{label}</div>
        {isVerified ? (

            <CheckCircle2 className="h-4 w-4" /> Verified
          </div>
        ) : null}
      </div>

      <div
        ref={trackRef}
        className={
          "relative h-11 w-full overflow-hidden rounded-md border border-border bg-muted/30" +
          (disabled ? " opacity-60" : "")
        }
      >
        {/* Filled progress */}
        <div
          className="absolute left-0 top-0 h-full bg-emerald-500/20"
          style={{ width: `${Math.round(pct * 100)}%` }}
        />

        {/* Hint text */}
        <div className="pointer-events-none absolute inset-0 flex items-center justify-center text-sm text-muted-foreground">
          {isVerified ? "Verified" : "Slide right"}
        </div>

        {/* Handle */}
        <button
          ref={handleRef}
          type="button"
          aria-label="Slider verification"
          disabled={disabled || isVerified}
          onPointerDown={startDrag}
          onPointerMove={onMove}
          onPointerUp={finish}
          onPointerCancel={finish}
          className={
            "absolute left-0 top-0 h-11 w-14 select-none rounded-md border border-border bg-background shadow-sm transition-transform" +
            (isDragging ? " ring-2 ring-emerald-500/40" : "")
          }
          style={{ transform: `translateX(${x}px)` }}
        >
          <div className="flex h-full w-full items-center justify-center text-muted-foreground">
            {isVerified ? (
              <CheckCircle2 className="h-5 w-5 text-emerald-400" />
            ) : (
              <ChevronRight className="h-5 w-5" />
            )}
          </div>
        </button>
      </div>

      {helpText ? <div className="mt-2 text-xs text-muted-foreground">{helpText}</div> : null}
    </div>
  );
}

This verifies the HMAC token that the render engine issues, prevents spoofed signup acceptance:

import crypto from "crypto";
import { base64UrlDecode, hmacSign, stableStringify } from "./cryptoUtils";

export type TermsTokenPayloadDb = {
  did: string;
  gdid: string;
  gv: string;
  gsha: string;
  aid: string;
  av: string;
  asha: string;
  cSha: string;
  cty: string;
  rg?: string | null;
  iat: number;
};

const TOKEN_SECRET = process.env.LEGAL_TERMS_HMAC_SECRET || "dev-secret-change-me";
const SIG_PREFIX_HEX_LEN = 32;

function timingSafeEqualHexPrefix(a: string, b: string): boolean {
  const aBuf = Buffer.from(a, "hex");
  const bBuf = Buffer.from(b, "hex");
  if (aBuf.length !== bBuf.length) return false;
  return crypto.timingSafeEqual(aBuf, bBuf);
}

export function verifyTermsTokenDb(
  token: string,
  opts?: { maxAgeSeconds?: number; expectedCombinedSha?: string }
):
  | { ok: true; payload: TermsTokenPayloadDb; canonicalPayload: string }
  | { ok: false; reason: string } {
  try {
    const parts = token.split(".");
    if (parts.length !== 2) return { ok: false, reason: "TOKEN_FORMAT" };
    const [payloadB64, sigPrefix] = parts;
    if (!payloadB64 || !sigPrefix) return { ok: false, reason: "TOKEN_FORMAT" };

    const canonicalPayload = base64UrlDecode(payloadB64);
    const expectedSig = hmacSign(canonicalPayload, TOKEN_SECRET).slice(0, SIG_PREFIX_HEX_LEN);

    if (sigPrefix.length !== expectedSig.length) return { ok: false, reason: "TOKEN_SIG_LEN" };
    if (!timingSafeEqualHexPrefix(sigPrefix, expectedSig)) return { ok: false, reason: "TOKEN_SIG" };

    const payload = JSON.parse(canonicalPayload) as TermsTokenPayloadDb;

    if (!payload?.did || !payload?.gdid || !payload?.gv || !payload?.gsha || !payload?.aid || !payload?.av || !payload?.asha || !payload?.cSha) {
      return { ok: false, reason: "TOKEN_PAYLOAD" };
    }
    if (!payload?.cty || typeof payload.cty !== "string" || payload.cty.length !== 2) {
      return { ok: false, reason: "TOKEN_COUNTRY" };
    }
    if (!payload?.iat || typeof payload.iat !== "number") {
      return { ok: false, reason: "TOKEN_IAT" };
    }

    const maxAge = opts?.maxAgeSeconds ?? 2 * 60 * 60;
    const nowSec = Math.floor(Date.now() / 1000);
    if (payload.iat > nowSec + 60) return { ok: false, reason: "TOKEN_FUTURE" };
    if (nowSec - payload.iat > maxAge) return { ok: false, reason: "TOKEN_EXPIRED" };

    if (opts?.expectedCombinedSha && payload.cSha !== opts.expectedCombinedSha) {
      return { ok: false, reason: "TOKEN_SHA_MISMATCH" };
    }

    const normalized = stableStringify(payload);
    JSON.parse(normalized);

    return { ok: true, payload, canonicalPayload: normalized };
  } catch {
    return { ok: false, reason: "TOKEN_ERROR" };
  }
}

export type LegalAcceptanceAppendInput = {

  regionKey?: string | null;
  globalDocId: string;

const sqlite = new Database("trading_app.db");

export function appendLegalAcceptanceLedger(input: LegalAcceptanceAppendInput): {
  id: number;
  ledgerSeq: number;
  prevLedgerHash: string | null;
  ledgerHash: string;
} {
  const last = sqlite
    .prepare(`SELECT id, ledgerSeq, ledgerHash FROM legal_acceptances ORDER BY ledgerSeq DESC LIMIT 1`)
    .get() as { id: number; ledgerSeq: number; ledgerHash: string } | undefined;

  const nextSeq = (last?.ledgerSeq ?? 0) + 1;
  const prevHash = last?.ledgerHash ?? null;

  const hashInput = stableStringify({
    ledgerSeq: nextSeq,
    prevLedgerHash: prevHash,
    ...input,
  });

  const ledgerHash = sha256(`${prevHash ?? ""}|${hashInput}`);

  const stmt = sqlite.prepare(`
    INSERT INTO legal_acceptances (
      ledgerSeq,
      prevLedgerHash,
      ledgerHash,
      userId,
      emailAtAcceptance,
      countryIso2,
      regionKey,
      globalDocId,
      globalVersion,
      globalSha256,
      addendumId,
      addendumVersion,
      addendumSha256,
      combinedSha256,
      combinedText,
      termsToken,
      ip,
      userAgent,
      sessionId
    ) VALUES (
      @ledgerSeq,
      @prevLedgerHash,
      @ledgerHash,
      @userId,
      @emailAtAcceptance,
      @countryIso2,
      @regionKey,
      @globalDocId,
      @globalVersion,
      @globalSha256,
      @addendumId,
      @addendumVersion,
      @addendumSha256,
      @combinedSha256,
      @combinedText,
      @termsToken,
      @ip,
      @userAgent,
      @sessionId
    )
  `);
