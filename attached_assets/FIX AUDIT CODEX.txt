## 1) `server/security/captcha.ts` (FULL REPLACEMENT)

```ts
import type { Request } from "express";
import fetch from "node-fetch";
import Database from "better-sqlite3";

export type CaptchaProvider = "TURNSTILE" | "SLIDER" | "DISABLED";

export interface CaptchaPolicy {
  enforceSignupCaptcha: boolean;
  provider: CaptchaProvider;
}

export interface SignupPolicy extends CaptchaPolicy {
  enforceSignupPhone: boolean;
}

export const SLIDER_CAPTCHA_TTL_MS = 15 * 60 * 1000; // 15 minutes
export const SLIDER_CAPTCHA_MIN_SOLVE_MS = 800; // Heuristic anti-bot; not security-critical

const DB_PATH = process.env.SQLITE_DB_PATH || "trading_app.db";

function toBool(v: any): boolean {
  return v === 1 || v === true || v === "1";
}

function normalizeProvider(v: any): CaptchaProvider {
  const s = String(v || "").toUpperCase();
  if (s === "TURNSTILE") return "TURNSTILE";
  if (s === "SLIDER") return "SLIDER";
  if (s === "DISABLED") return "DISABLED";
  return "TURNSTILE";
}

/**
 * Returns platform-wide signup policy settings.
 * Defensive query so missing columns do not break runtime.
 */
export function getSignupPolicy(): SignupPolicy {
  const defaults: SignupPolicy = {
    enforceSignupCaptcha: false,
    provider: "TURNSTILE",
    enforceSignupPhone: false,
  };

  let db: Database.Database | null = null;
  try {
    db = new Database(DB_PATH);

    const row = db
      .prepare(
        `SELECT
           captcha_provider AS captchaProvider,
           signup_captcha_enforce AS signupCaptchaEnforce,
           signup_phone_enforce AS signupPhoneEnforce
         FROM system_config
         WHERE id = 1`
      )
      .get() as any;

    if (!row) return defaults;

    return {
      enforceSignupCaptcha: toBool(row.signupCaptchaEnforce),
      provider: normalizeProvider(row.captchaProvider),
      enforceSignupPhone: toBool(row.signupPhoneEnforce),
    };
  } catch {
    return defaults;
  } finally {
    try {
      db?.close();
    } catch {
      // ignore
    }
  }
}

export function getCaptchaPolicy(): CaptchaPolicy {
  const p = getSignupPolicy();
  return { enforceSignupCaptcha: p.enforceSignupCaptcha, provider: p.provider };
}

type CaptchaVerifyResult = { ok: boolean; message: string };

/**
 * Verifies signup CAPTCHA in the backend.
 *
 * TURNSTILE: verifies token against Cloudflare Turnstile.
 * SLIDER: requires a session-bound verified slider captcha (see /api/captcha/slider/*).
 * DISABLED: always fails if enforcement is enabled (misconfiguration).
 */
export async function verifySignupCaptcha(req: Request, captchaToken?: string | null): Promise<CaptchaVerifyResult> {
  const policy = getCaptchaPolicy();
  if (!policy.enforceSignupCaptcha) {
    return { ok: true, message: "CAPTCHA_NOT_ENFORCED" };
  }

  if (policy.provider === "DISABLED") {
    return { ok: false, message: "CAPTCHA_PROVIDER_DISABLED" };
  }

  if (policy.provider === "SLIDER") {
    const s = (req.session as any)?.signupCaptcha as
      | {
          provider?: string;
          captchaId?: string;
          issuedAt?: number;
          verifiedAt?: number;
          consumedAt?: number;
        }
      | undefined;

    if (!s || String(s.provider || "").toUpperCase() !== "SLIDER") {
      return { ok: false, message: "CAPTCHA_REQUIRED" };
    }

    const now = Date.now();
    const issuedAt = typeof s.issuedAt === "number" ? s.issuedAt : 0;
    const verifiedAt = typeof s.verifiedAt === "number" ? s.verifiedAt : 0;

    if (!s.captchaId || !issuedAt || !verifiedAt) {
      return { ok: false, message: "CAPTCHA_NOT_VERIFIED" };
    }
    if (s.consumedAt) {
      return { ok: false, message: "CAPTCHA_ALREADY_USED" };
    }

    if (now - issuedAt > SLIDER_CAPTCHA_TTL_MS || now - verifiedAt > SLIDER_CAPTCHA_TTL_MS) {
      return { ok: false, message: "CAPTCHA_EXPIRED" };
    }

    if (verifiedAt - issuedAt < SLIDER_CAPTCHA_MIN_SOLVE_MS) {
      return { ok: false, message: "CAPTCHA_SOLVE_TOO_FAST" };
    }

    // Single-use: consume immediately to prevent reuse across repeated attempts.
    (req.session as any).signupCaptcha = { ...s, consumedAt: now };

    return { ok: true, message: "CAPTCHA_OK" };
  }

  // TURNSTILE
  if (!captchaToken) {
    return { ok: false, message: "CAPTCHA_TOKEN_REQUIRED" };
  }

  const secretKey = process.env.TURNSTILE_SECRET_KEY;
  if (!secretKey) {
    return { ok: false, message: "CAPTCHA_SECRET_NOT_CONFIGURED" };
  }

  const response = await fetch("https://challenges.cloudflare.com/turnstile/v0/siteverify", {
    method: "POST",
    headers: { "Content-Type": "application/x-www-form-urlencoded" },
    body: `secret=${encodeURIComponent(secretKey)}&response=${encodeURIComponent(captchaToken)}`,
  });

  const data = (await response.json()) as any;
  if (!data?.success) {
    return { ok: false, message: "CAPTCHA_VERIFICATION_FAILED" };
  }

  return { ok: true, message: "CAPTCHA_OK" };
}
```

```ts
import { Router } from "express";
import Database from "better-sqlite3";
import crypto from "crypto";

import { extractGriftContext } from "../grift/griftGeo";
import { recordAuthEvent } from "../grift/griftEngine";
import { getCaptchaPolicy, SLIDER_CAPTCHA_MIN_SOLVE_MS, SLIDER_CAPTCHA_TTL_MS } from "../security/captcha";

const DB_PATH = process.env.SQLITE_DB_PATH || "trading_app.db";

// Dedicated sqlite connection for lightweight auth event logging.
const authDb = new Database(DB_PATH);

function logCaptchaEvent(req: any, eventType: string, success: boolean, failureReason?: string, metadata?: Record<string, any>) {
  try {
    const ctx = extractGriftContext(req);
    recordAuthEvent(authDb, eventType, ctx, success, failureReason, metadata);
  } catch {
    // Never block signup flow on logging failure.
  }
}

export const captchaSliderRouter = Router();

/**
 * Session-bound slider CAPTCHA (P1 fix):
 * - start(): issues captchaId and persists to session
 * - complete(): marks verifiedAt (same session)
 *
 * Enforcement happens in /api/auth/register via verifySignupCaptcha() (provider=SLIDER).
 */

captchaSliderRouter.post("/slider/start", (req, res) => {
  const policy = getCaptchaPolicy();
  if (policy.provider !== "SLIDER") {
    return res.status(409).json({ ok: false, message: "SLIDER_CAPTCHA_NOT_CONFIGURED" });
  }

  const now = Date.now();
  const captchaId = crypto.randomUUID();

  (req.session as any).signupCaptcha = {
    provider: "SLIDER",
    captchaId,
    issuedAt: now,
    verifiedAt: undefined,
    consumedAt: undefined,
  };

  logCaptchaEvent(req, "CAPTCHA_ISSUED", true, undefined, {
    provider: "SLIDER",
    purpose: "SIGNUP",
    captchaId,
  });

  return res.json({ ok: true, captchaId, ttlMs: SLIDER_CAPTCHA_TTL_MS });
});

captchaSliderRouter.post("/slider/complete", (req, res) => {
  const policy = getCaptchaPolicy();
  if (policy.provider !== "SLIDER") {
    return res.status(409).json({ ok: false, message: "SLIDER_CAPTCHA_NOT_CONFIGURED" });
  }

  const { captchaId } = (req.body || {}) as { captchaId?: string };
  if (!captchaId) {
    logCaptchaEvent(req, "CAPTCHA_FAILED", false, "CAPTCHA_ID_REQUIRED", { provider: "SLIDER" });
    return res.status(400).json({ ok: false, message: "CAPTCHA_ID_REQUIRED" });
  }

  const s = (req.session as any).signupCaptcha as
    | {
        provider?: string;
        captchaId?: string;
        issuedAt?: number;
        verifiedAt?: number;
        consumedAt?: number;
      }
    | undefined;

  if (!s || String(s.provider || "").toUpperCase() !== "SLIDER") {
    logCaptchaEvent(req, "CAPTCHA_FAILED", false, "CAPTCHA_SESSION_MISSING", { provider: "SLIDER", captchaId });
    return res.status(400).json({ ok: false, message: "CAPTCHA_SESSION_MISSING" });
  }

  if (s.captchaId !== captchaId) {
    logCaptchaEvent(req, "CAPTCHA_FAILED", false, "CAPTCHA_ID_MISMATCH", { provider: "SLIDER", captchaId });
    return res.status(400).json({ ok: false, message: "CAPTCHA_ID_MISMATCH" });
  }

  const now = Date.now();
  const issuedAt = typeof s.issuedAt === "number" ? s.issuedAt : 0;

  if (!issuedAt || now - issuedAt > SLIDER_CAPTCHA_TTL_MS) {
    logCaptchaEvent(req, "CAPTCHA_FAILED", false, "CAPTCHA_EXPIRED", { provider: "SLIDER", captchaId });
    return res.status(400).json({ ok: false, message: "CAPTCHA_EXPIRED" });
  }

  if (now - issuedAt < SLIDER_CAPTCHA_MIN_SOLVE_MS) {
    logCaptchaEvent(req, "CAPTCHA_FAILED", false, "CAPTCHA_SOLVE_TOO_FAST", {
      provider: "SLIDER",
      captchaId,
      solveMs: now - issuedAt,
    });
    return res.status(400).json({ ok: false, message: "CAPTCHA_SOLVE_TOO_FAST" });
  }

  (req.session as any).signupCaptcha = {
    ...s,
    verifiedAt: now,
    consumedAt: undefined,
  };

  logCaptchaEvent(req, "CAPTCHA_VERIFIED", true, undefined, { provider: "SLIDER", captchaId });

  return res.json({ ok: true });
});

// Optional: client can call this on reset to force a re-issue.
captchaSliderRouter.post("/slider/reset", (req, res) => {
```

```ts
  provider: CaptchaProvider;

// Fail-open defaults: do not block signup if config lookup fails.
const DEFAULT_CONFIG: CaptchaConfig = {
  enforceSignupCaptcha: false,
  provider: "TURNSTILE",
};

export async function getCaptchaConfig(): Promise<CaptchaConfig> {
  try {
    const row = await db
      .select({
        enforceSignupCaptcha: systemConfig.signupCaptchaEnforce,
        provider: systemConfig.captchaProvider,
      })
      .from(systemConfig)
      .where(eq(systemConfig.id, 1))
      .get();

    return {

      provider: row?.provider === "SLIDER" ? "SLIDER" : "TURNSTILE",
    };
  } catch (error) {

    return DEFAULT_CONFIG;

async function verifyTurnstileToken(token: string): Promise<boolean> {
  const secretKey = process.env.TURNSTILE_SECRET_KEY;

  // If no secret is set, fail-open (dev-friendly, avoids lockout).
  if (!secretKey) {
    console.warn(
      "TURNSTILE_SECRET_KEY is not set; allowing signup without Turnstile verification (fail-open)."
    );
    return true;
  }

  try {
    const resp = await axios.post(
      "https://challenges.cloudflare.com/turnstile/v0/siteverify",
      new URLSearchParams({ secret: secretKey, response: token }).toString(),
      {
        headers: { "Content-Type": "application/x-www-form-urlencoded" },
        timeout: 10_000,
      }
    );

    return !!resp.data?.success;
  } catch (error) {
    console.error("Turnstile verification error:", error);
    return false;
  }
}

export type VerifyCaptchaResult = {
  ok: boolean;
  provider: CaptchaProvider;
  error?: string;
};

const SLIDER_MIN_SOLVE_MS = 800;
const SLIDER_MAX_AGE_MS = 15 * 60 * 1000;

/**
 * Verifies signup CAPTCHA based on current system config.
 *
 * TURNSTILE: validates captchaToken via Cloudflare endpoint.
 * SLIDER: requires a session-bound solve via /api/captcha/slider/start + /api/captcha/slider/complete.
 */
export async function verifySignupCaptcha(
  captchaToken: string | undefined,
  req: Request
): Promise<VerifyCaptchaResult> {
  const config = await getCaptchaConfig();

  if (!config.enforceSignupCaptcha) {
    return { ok: true, provider: config.provider };
  }

  if (config.provider === "TURNSTILE") {
    if (!captchaToken || typeof captchaToken !== "string") {
      return { ok: false, provider: "TURNSTILE", error: "CAPTCHA_REQUIRED" };
    }

    const ok = await verifyTurnstileToken(captchaToken);
    return ok
      ? { ok: true, provider: "TURNSTILE" }
      : { ok: false, provider: "TURNSTILE", error: "CAPTCHA_INVALID" };
  }

  // SLIDER mode: must be session-bound and completed.
  const sess: any = (req as any).session;
  const slider = sess?.signupCaptcha?.slider;
  const captchaId = slider?.captchaId;
  const issuedAt = slider?.issuedAt;
  const verifiedAt = slider?.verifiedAt;

  if (
    !captchaId ||
    typeof captchaId !== "string" ||
    typeof issuedAt !== "number" ||
    typeof verifiedAt !== "number"
  ) {
    return { ok: false, provider: "SLIDER", error: "CAPTCHA_REQUIRED" };
  }

  const now = Date.now();
  if (now - issuedAt > SLIDER_MAX_AGE_MS) {
    return { ok: false, provider: "SLIDER", error: "CAPTCHA_EXPIRED" };
  }

  const solveMs = verifiedAt - issuedAt;
  if (solveMs < SLIDER_MIN_SOLVE_MS) {
    return { ok: false, provider: "SLIDER", error: "CAPTCHA_TOO_FAST" };
  }

  return { ok: true, provider: "SLIDER" };
}
```

```ts
export const SLIDER_CAPTCHA_VERIFY_TTL_MS = 10 * 60 * 1000; // verified window for signup submit

export type CaptchaProvider = "TURNSTILE" | "HCAPTCHA" | "SLIDER";

export async function getCaptchaPolicy(): Promise<{
  enforceSignupCaptcha: boolean;
  provider: CaptchaProvider;
}> {

  return {

    provider: (cfg?.captchaProvider ?? "TURNSTILE") as CaptchaProvider,
  };
}

export async function verifySignupCaptcha(
  req: Request,
  captchaToken: string | undefined | null,
): Promise<{ ok: true } | { ok: false; message: string }> {
  const { enforceSignupCaptcha, provider } = await getCaptchaPolicy();
  if (!enforceSignupCaptcha) return { ok: true };

  // SLIDER captcha is verified via session-bound start/complete endpoints.
  if (provider === "SLIDER") {
    const s: any = (req as any).session;
    const slider = s?.captchaSlider;
    const now = Date.now();

    if (!slider?.id || !slider?.issuedAtMs) {
      return { ok: false, message: "CAPTCHA_REQUIRED" };
    }

    // Issuance must be recent; otherwise force restart.
    if (typeof slider.issuedAtMs === "number" && now - slider.issuedAtMs > SLIDER_CAPTCHA_ISSUE_TTL_MS) {
      return { ok: false, message: "CAPTCHA_EXPIRED" };
    }

    // Must be verified server-side.
    if (!slider?.verifiedAtMs || typeof slider.verifiedAtMs !== "number") {
      return { ok: false, message: "CAPTCHA_REQUIRED" };
    }

    // Verification must be fresh.
    if (now - slider.verifiedAtMs > SLIDER_CAPTCHA_VERIFY_TTL_MS) {
      return { ok: false, message: "CAPTCHA_EXPIRED" };
    }

    // Mark consumed for this request (best-effort).
    slider.consumedAtMs = now;
    s.captchaSlider = slider;

    return { ok: true };
  }

  if (!captchaToken || captchaToken.trim().length < 10) {
    return { ok: false, message: "CAPTCHA_REQUIRED" };
  }

  const ip = getClientIp(req) || undefined;

  try {
    if (provider === "TURNSTILE") {
      const secret = process.env.TURNSTILE_SECRET_KEY || process.env.CAPTCHA_TURNSTILE_SECRET;
      if (!secret) return { ok: false, message: "CAPTCHA_SECRET_NOT_CONFIGURED" };

      const form = new URLSearchParams();
      form.set("secret", secret);
      form.set("response", captchaToken);
      if (ip) form.set("remoteip", ip);

      const resp = await fetch("https://challenges.cloudflare.com/turnstile/v0/siteverify", {
        method: "POST",
        headers: { "content-type": "application/x-www-form-urlencoded" },
        body: form.toString(),
      });

      const data: any = await resp.json().catch(() => ({}));
      if (!data?.success) {
        const codes = Array.isArray(data?.["error-codes"]) ? data["error-codes"].join(",") : "unknown";
        return { ok: false, message: `CAPTCHA_FAILED:${codes}` };
      }
      return { ok: true };
    }

    if (provider === "HCAPTCHA") {
      const secret = process.env.HCAPTCHA_SECRET_KEY || process.env.CAPTCHA_HCAPTCHA_SECRET;
      if (!secret) return { ok: false, message: "CAPTCHA_SECRET_NOT_CONFIGURED" };

        method: "POST",
        headers: { "content-type": "application/x-www-form-urlencoded" },

        return { ok: false, message: `CAPTCHA_FAILED:${codes}` };
      }
      return { ok: true };
    }

    return { ok: false, message: "CAPTCHA_PROVIDER_UNSUPPORTED" };
  } catch (e: any) {
    return { ok: false, message: `CAPTCHA_VERIFY_ERROR:${e?.message ?? "unknown"}` };
  }
}
```

```tsx
import { useEffect, useRef, useState } from "react";
import { parsePhoneNumberFromString, getCountryCallingCode, CountryCode, AsYouType } from "libphonenumber-js";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";

type Props = {
  countryIso2: string;
  value: string;
  onChange: (e164: string, valid: boolean) => void;
  disabled?: boolean;
  required?: boolean;
};

export function PhoneNumberInput({ countryIso2, value, onChange, disabled, required = false }: Props) {
  const [raw, setRaw] = useState(value);
  const [error, setError] = useState<string | null>(null);
  const prevCountryRef = useRef(countryIso2);

  useEffect(() => {
    if (countryIso2 && countryIso2 !== prevCountryRef.current) {
      setRaw("");
      setError(null);
      onChange("", false);
    }
    prevCountryRef.current = countryIso2;
  }, [countryIso2, onChange]);

  useEffect(() => {
    try {
      if (!raw || raw.trim() === "") {
        onChange("", false);
        setError(null);
        return;
      }

      const country = countryIso2 as CountryCode;
      const phone = parsePhoneNumberFromString(raw, country);

      if (!phone || !phone.isValid()) {
        setError("Invalid phone number for selected country");
        onChange(raw, false);
        return;
      }

      setError(null);
      onChange(phone.number, true);
    } catch {
      setError("Invalid phone number");
      onChange(raw, false);
    }
  }, [raw, countryIso2, onChange]);

  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const inputValue = e.target.value;
    const digitsOnly = inputValue.replace(/[^\d\s\-()]/g, "");

    if (countryIso2) {
      try {
        const formatter = new AsYouType(countryIso2 as CountryCode);
        const formatted = formatter.input(digitsOnly.replace(/\D/g, ""));
        setRaw(formatted);
      } catch {
        setRaw(digitsOnly);
      }
    } else {
      setRaw(digitsOnly);
    }
  };

  const handleKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    const allowedKeys = [
      "Backspace",
      "Delete",
      "Tab",
      "Escape",
      "Enter",
      "ArrowLeft",
      "ArrowRight",
      "ArrowUp",
      "ArrowDown",
      "Home",
      "End",
    ];
    if (allowedKeys.includes(e.key)) return;
    if (e.ctrlKey || e.metaKey) return;
    if (!/[\d\s\-()]/.test(e.key)) {
      e.preventDefault();
    }
  };

  let prefix = "";
  try {
    if (countryIso2 && countryIso2.length === 2) {
      prefix = `+${getCountryCallingCode(countryIso2 as CountryCode)}`;
    }
  } catch {
    prefix = "";
  }

  return (
    <div className="space-y-1">
      <Label htmlFor="phone">
        Phone Number{required ? <span className="text-red-500"> *</span> : <span className="text-muted-foreground"> (optional)</span>}
      </Label>
      <div className="flex items-center gap-2">
        {prefix && (
          <span className="text-sm text-muted-foreground bg-muted px-2 py-2 rounded border">
            {prefix}
          </span>
        )}
        <Input
          id="phone"
          type="tel"
          inputMode="tel"
          value={raw}
          onChange={handleInputChange}
          onKeyDown={handleKeyDown}
          placeholder={prefix ? "Enter phone number" : "Select country first"}
          disabled={disabled || !countryIso2}
          className="flex-1"
        />
      </div>
      {error && <div className="text-xs text-red-500">{error}</div>}
      <p className="text-xs text-muted-foreground">Phone can be verified later in your profile settings.</p>
    </div>
  );
}
```

```ts
const SLIDER_CAPTCHA_ISSUE_TTL_MS = 2 * 60 * 1000;   // start->complete max age
const SLIDER_CAPTCHA_VERIFY_TTL_MS = 10 * 60 * 1000; // complete->register max age

function requireVerifiedSliderCaptcha(req: Request): { ok: true } | { ok: false; message: string } {
  const now = Date.now();
  const s: any = (req as any).session;
  const c = s?.captchaSlider;

  if (!c?.id || typeof c.issuedAtMs !== "number") {
    return { ok: false, message: "CAPTCHA_REQUIRED" };
  }
  if (now - c.issuedAtMs > SLIDER_CAPTCHA_ISSUE_TTL_MS) {
    return { ok: false, message: "CAPTCHA_EXPIRED" };
  }
  if (!c.verifiedAtMs || typeof c.verifiedAtMs !== "number") {
    return { ok: false, message: "CAPTCHA_REQUIRED" };
  }
  if (now - c.verifiedAtMs > SLIDER_CAPTCHA_VERIFY_TTL_MS) {
    return { ok: false, message: "CAPTCHA_EXPIRED" };
  }
  if (c.consumedAtMs) {
    return { ok: false, message: "CAPTCHA_ALREADY_USED" };
  }

  // single-use: consume now
  c.consumedAtMs = now;
  s.captchaSlider = c;
  return { ok: true };
}

function normalizeSignupPhone(phone: any, countryIso2: any): { e164: string | null; ok: boolean } {
  if (typeof phone !== "string") return { e164: null, ok: true };
  const trimmed = phone.trim();
  if (!trimmed) return { e164: null, ok: true };

  try {
    const parsed = parsePhoneNumberFromString(trimmed, String(countryIso2 || "").toUpperCase() as any);
    if (!parsed || !parsed.isValid()) return { e164: null, ok: false };
    return { e164: parsed.number, ok: true };
  } catch {
    return { e164: null, ok: false };
  }
}
```
