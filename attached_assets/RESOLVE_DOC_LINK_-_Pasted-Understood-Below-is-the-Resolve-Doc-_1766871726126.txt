```ts
// FILE: /server/routes/adminLegalAcceptances.ts
// ADD THESE IMPORTS at top with others:
import { inArray } from "drizzle-orm";
import { legalDocuments } from "../../shared/schema";

// ... existing code ...

// POST /api/admin/legal-acceptances/resolve-doc-links
// body: { ids: number[] }
// returns: { [acceptanceId]: { global: {best, matches[]}, addendum: {best, matches[]}} }
adminLegalAcceptancesRouter.post("/resolve-doc-links", (req, res) => {
  try {
    const idsRaw = (req.body || {}).ids;
    if (!Array.isArray(idsRaw)) return res.status(400).json({ ok: false, error: "ids must be an array." });

    const ids = idsRaw
      .map((x) => Number(x))
      .filter((n) => Number.isFinite(n) && n > 0)
      .slice(0, 200);

    if (ids.length === 0) return res.status(400).json({ ok: false, error: "No valid ids provided." });

    // Pull acceptance rows for those IDs
    const acceptances = db
      .select({
        id: legalAcceptances.id,
        globalDocSha256: legalAcceptances.globalDocSha256,
        addendumSha256: legalAcceptances.addendumSha256,
      })
      .from(legalAcceptances)
      .where(inArray(legalAcceptances.id, ids))
      .all();

    // Collect unique shas
    const globalShas = Array.from(
      new Set(acceptances.map((a) => String(a.globalDocSha256 || "")).filter(Boolean))
    );
    const addendumShas = Array.from(
      new Set(acceptances.map((a) => String(a.addendumSha256 || "")).filter(Boolean))
    );

    // Fetch candidate legalDocuments by sha
    const globalDocs =
      globalShas.length === 0
        ? []
        : db
            .select({
              id: legalDocuments.id,
              docSet: legalDocuments.docSet,
              docType: legalDocuments.docType,
              jurisdictionType: legalDocuments.jurisdictionType,
              jurisdictionKey: legalDocuments.jurisdictionKey,
              version: legalDocuments.version,
              sha256: legalDocuments.sha256,
              createdAt: legalDocuments.createdAt,
            })
            .from(legalDocuments)
            .where(
              and(
                eq(legalDocuments.docSet, "DOC1"),
                eq(legalDocuments.docType, "GLOBAL_MASTER"),
                inArray(legalDocuments.sha256, globalShas)
              ) as any
            )
            .orderBy(desc(legalDocuments.createdAt))
            .all();

    const addendumDocs =
      addendumShas.length === 0
        ? []
        : db
            .select({
              id: legalDocuments.id,
              docSet: legalDocuments.docSet,
              docType: legalDocuments.docType,
              jurisdictionType: legalDocuments.jurisdictionType,
              jurisdictionKey: legalDocuments.jurisdictionKey,
              version: legalDocuments.version,
              sha256: legalDocuments.sha256,
              createdAt: legalDocuments.createdAt,
            })
            .from(legalDocuments)
            .where(
              and(
                eq(legalDocuments.docSet, "DOC1"),
                eq(legalDocuments.docType, "ADDENDUM"),
                inArray(legalDocuments.sha256, addendumShas)
              ) as any
            )
            .orderBy(desc(legalDocuments.createdAt))
            .all();

    // Build sha -> docs[] maps
    const globalMap = new Map<string, any[]>();
    for (const d of globalDocs) {
      const key = String(d.sha256);
      if (!globalMap.has(key)) globalMap.set(key, []);
      globalMap.get(key)!.push(d);
    }

    const addMap = new Map<string, any[]>();
    for (const d of addendumDocs) {
      const key = String(d.sha256);
      if (!addMap.has(key)) addMap.set(key, []);
      addMap.get(key)!.push(d);
    }

    // Choose best match:
    // - If single match => it
    // - If multiple => latest createdAt (already ordered desc)
    const chooseBest = (arr: any[] | undefined) => {
      if (!arr || arr.length === 0) return null;
      if (arr.length === 1) return arr[0];
      return arr[0]; // latest createdAt
    };

    const resolved: Record<
      number,
      {
        global: { best: any | null; matches: any[] };
        addendum: { best: any | null; matches: any[] };
      }
    > = {};

    for (const a of acceptances) {
      const gSha = String(a.globalDocSha256 || "");
      const adSha = String(a.addendumSha256 || "");

      const gMatches = globalMap.get(gSha) || [];
      const aMatches = addMap.get(adSha) || [];

      resolved[Number(a.id)] = {
        global: { best: chooseBest(gMatches), matches: gMatches },
        addendum: { best: chooseBest(aMatches), matches: aMatches },
      };
    }

    return res.json({ ok: true, resolved });
  } catch (e: any) {
    return res.status(400).json({ ok: false, error: e?.message || "Resolve failed." });
  }
});
```

# 3) What “Resolve” does operationally

* For each acceptance row:

  * `globalDocSha256` is matched to `legalDocuments` where `docSet=DOC1` and `docType=GLOBAL_MASTER`
  * `addendumSha256` is matched to `legalDocuments` where `docSet=DOC1` and `docType=ADDENDUM`
* If multiple matches exist (rare but possible if identical content was inserted for different targets), the UI:

  * Shows `G(n)` / `A(n)` counts
  * Picks “best” as **most recently created** doc (deterministic)
