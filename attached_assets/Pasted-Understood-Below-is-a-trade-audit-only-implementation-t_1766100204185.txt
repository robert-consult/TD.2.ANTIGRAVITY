Understood. Below is a **trade-audit–only** implementation that delivers exactly the four requirements:

1. **CorrelationId on every order attempt**
2. **order_intent_audit** capturing **RECEIVED** + **DECISION (PASS/REJECT)** with **reject_check / reject_reason**
3. **trade_audit** events for **market creation**, **pending placement**, **cancel**, **manual close**, **targets update**
4. **Tamper-evident hash chain** via **SHA-256** (`prev_hash` → `event_hash`) with **canonical JSON payloads**

This is designed to drop into most Node/Express + SQLite (better-sqlite3) backends. If your project uses different paths, only adjust the import paths and route wiring.

```ts
export function canonicalJson(value: any): string {
  // Stable, deterministic JSON serialization (sorted keys), suitable for hashing.
  if (value === null || value === undefined) return "null";
  const t = typeof value;
  if (t === "number" || t === "boolean") return JSON.stringify(value);
  if (t === "string") return JSON.stringify(value);
  if (Array.isArray(value)) return `[${value.map(canonicalJson).join(",")}]`;
  if (t === "object") {
    const keys = Object.keys(value).sort();
    const pairs = keys
      .filter((k) => value[k] !== undefined)
      .map((k) => `${JSON.stringify(k)}:${canonicalJson(value[k])}`);
    return `{${pairs.join(",")}}`;
  }
  // fallback
  return JSON.stringify(String(value));
}
```

```ts
import type { Request, Response, NextFunction } from "express";

declare global {

export function attachAuditContext(req: Request, res: Response, next: NextFunction) {

    userId: (req as any).user?.id ?? null,
    sessionId,
    ip: req.ip ?? null,
    userAgent: req.get("user-agent") ?? null,
  };

  res.setHeader("x-correlation-id", correlationId);
  next();
```

```ts
function calcSpread(bid?: number | null, ask?: number | null) {
  if (bid == null || ask == null) return { spreadPoints: null, spreadPips: null };
  const spreadPoints = ask - bid;
  return { spreadPoints, spreadPips: spreadPoints }; // if you treat 1 point == 1 pip for some symbols, adjust here
}

function signedSlippage(side: string | null | undefined, ref: number | null, fill: number | null) {
  if (ref == null || fill == null) return { slippagePoints: null, slippagePips: null };
  const raw = fill - ref;
  // Signed convention: for BUY, positive = worse (paid higher); for SELL, positive = better (sold higher)
  const signed = side === "BUY" ? raw : -raw;
  return { slippagePoints: signed, slippagePips: signed };
}
```

# /backend/src/routes/trades.ts

This is the **minimal route wiring** showing how to guarantee:

* correlationId exists (middleware)
* order_intent_audit RECEIVED/DECISION are written (risk middleware hooks)
* trade_audit events are written for the lifecycle events you listed

```ts
import { Router } from "express";
import type Database from "better-sqlite3";
import crypto from "node:crypto";
import { attachAuditContext } from "../middleware/auditContext";
import { logOrderReceived, logOrderDecision } from "../risk/riskAuditHelpers";
import {
  auditOrderPlaced,
  auditOrderFilled,
  auditOrderCanceled,
  auditPositionClosed,
  auditTargetsUpdated
} from "./trades.auditWiring";

type DB = Database.Database;
const uuid = () => crypto.randomUUID();

export function tradeRoutes(db: DB) {
  const r = Router();

  // Replace with your auth
  const requireAuth = (req: any, res: any, next: any) => {
    if (!req.user?.id) return res.status(401).json({ error: "UNAUTHENTICATED" });
    next();
  };

  // ---- POST /trades (order attempt) ----
  r.post("/trades", attachAuditContext, requireAuth, (req, res) => {
    const userId = req.user.id as string;
    const ctx = req.auditContext!;
    ctx.userId = userId;

    const b = req.body || {};
    const symbol = String(b.symbol || "");
    const side = String(b.side || "");
    const orderType = String(b.orderType || "MARKET");
    const tif = b.timeInForce ? String(b.timeInForce) : null;
    const qtyLots = Number(b.lots ?? 0);

    const requestedPrice = b.requestedPrice != null ? Number(b.requestedPrice) : null;
    const limitPrice = b.limitPrice != null ? Number(b.limitPrice) : null;
    const stopPrice = b.stopPrice != null ? Number(b.stopPrice) : null;
    const takeProfit = b.takeProfit != null ? Number(b.takeProfit) : null;
    const stopLoss = b.stopLoss != null ? Number(b.stopLoss) : null;

    const quoteBid = b.quoteBid != null ? Number(b.quoteBid) : null;
    const quoteAsk = b.quoteAsk != null ? Number(b.quoteAsk) : null;
    const quoteMid = b.quoteMid != null ? Number(b.quoteMid) : (quoteBid != null && quoteAsk != null ? (quoteBid + quoteAsk) / 2 : null);
    const quoteTsUtcMs = b.quoteAt != null ? Number(b.quoteAt) : null;

    // (1) RECEIVED (always)
    logOrderReceived(db, ctx, {
      userId, symbol, side, orderType, timeInForce: tif ?? undefined,
      qtyLots, requestedPrice: requestedPrice ?? undefined,
      limitPrice: limitPrice ?? undefined, stopPrice: stopPrice ?? undefined,
      takeProfit: takeProfit ?? undefined, stopLoss: stopLoss ?? undefined,
      quoteBid: quoteBid ?? undefined, quoteAsk: quoteAsk ?? undefined, quoteMid: quoteMid ?? undefined, quoteTsUtcMs: quoteTsUtcMs ?? undefined,
      payload: { route: "POST /trades" },
    });

    // (2) Minimal PASS/REJECT decision example (plug in your actual risk engine)
    // NOTE: this section is purposely small to keep scope on audit. Replace with your real checks.
    const effectiveLimits = { maxLotsTotal: 50, maxTradesTotal: 5, maxTradesPerSymbol: 3 };
    const observed = { concurrentLots: b.concurrentLots ?? null, concurrentTrades: b.concurrentTrades ?? null };
    const snapshot = { hint: "attach your real risk snapshot here" };

    const reject = (check: string, reason: string) => {
      logOrderDecision(db, ctx, {
        userId,
        decision: "REJECT",
        rejectCheck: check,
        rejectReason: reason,
        symbol, side, orderType, timeInForce: tif ?? undefined, qtyLots,
        requestedPrice: requestedPrice ?? undefined, limitPrice: limitPrice ?? undefined, stopPrice: stopPrice ?? undefined,
        takeProfit: takeProfit ?? undefined, stopLoss: stopLoss ?? undefined,
        quoteBid: quoteBid ?? undefined, quoteAsk: quoteAsk ?? undefined, quoteMid: quoteMid ?? undefined, quoteTsUtcMs: quoteTsUtcMs ?? undefined,
        riskLimit: effectiveLimits,
        riskObserved: observed,
        riskSnapshot: snapshot,
      });
      return res.status(400).json({ error: "RISK_REJECT", check, reason, correlationId: ctx.correlationId });
    };

    if (!symbol || !side || qtyLots <= 0) return reject("BAD_REQUEST", "Missing symbol/side/lots");

    // Example reject: max lots (replace with actual computed exposure)
    if (qtyLots > effectiveLimits.maxLotsTotal) return reject("MAX_CONCURRENT_LOTS", "Requested lots exceed max concurrent lots");

    // PASS decision
    logOrderDecision(db, ctx, {
      userId,
      decision: "PASS",
      symbol, side, orderType, timeInForce: tif ?? undefined, qtyLots,
      requestedPrice: requestedPrice ?? undefined, limitPrice: limitPrice ?? undefined, stopPrice: stopPrice ?? undefined,
      takeProfit: takeProfit ?? undefined, stopLoss: stopLoss ?? undefined,
      quoteBid: quoteBid ?? undefined, quoteAsk: quoteAsk ?? undefined, quoteMid: quoteMid ?? undefined, quoteTsUtcMs: quoteTsUtcMs ?? undefined,
      riskLimit: effectiveLimits,
      riskObserved: observed,
      riskSnapshot: snapshot,
    });

    // (3) Create trade (your table may differ; key requirement is persisting correlation_id)
    const tradeId = uuid();
    const orderId = uuid(); // allocator-grade separation: order_id distinct from trade_id
    const positionId = orderType === "MARKET" ? uuid() : null;

    const status = orderType === "MARKET" ? "OPEN" : "PENDING";
    const fillPrice = orderType === "MARKET" ? (requestedPrice ?? quoteMid) : null;

    db.prepare(`
      INSERT INTO trades (
        id, user_id, symbol, side, order_type, time_in_force, status, lots,
        requested_price, limit_price, stop_price, take_profit, stop_loss,
        fill_price, correlation_id, order_id, position_id
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `).run(
      tradeId, userId, symbol, side, orderType, tif,
      status, qtyLots,
      requestedPrice, limitPrice, stopPrice, takeProfit, stopLoss,
      fillPrice,
      ctx.correlationId,
      orderId,
      positionId
    );

    // (4) trade_audit events:
    // ORDER_PLACED always
    auditOrderPlaced(db, req, {
      id: tradeId,
      user_id: userId,
      symbol,
      side,
      order_type: orderType,
      time_in_force: tif,
      lots: qtyLots,
      requested_price: requestedPrice,
      limit_price: limitPrice,
      stop_price: stopPrice,
      correlation_id: ctx.correlationId,
      order_id: orderId,
      position_id: positionId,
    }, { bid: quoteBid, ask: quoteAsk, mid: quoteMid });

    // ORDER_FILLED only for MARKET on creation
    if (status === "OPEN") {
      auditOrderFilled(db, req, {
        id: tradeId,
        user_id: userId,
        symbol,
        side,
        order_type: orderType,
        time_in_force: tif,
        lots: qtyLots,
        requested_price: requestedPrice,
        fill_price: fillPrice,
        correlation_id: ctx.correlationId,
        order_id: orderId,
        execution_id: uuid(),
        position_id: positionId,
      }, { bid: quoteBid, ask: quoteAsk, mid: quoteMid });
    }

    return res.json({ ok: true, tradeId, orderId, positionId, status, correlationId: ctx.correlationId });
  });

  // ---- POST /trades/:id/cancel (pending cancel) ----
  r.post("/trades/:id/cancel", attachAuditContext, requireAuth, (req, res) => {
    const userId = req.user.id as string;
    const ctx = req.auditContext!;
    ctx.userId = userId;

    const tradeId = String(req.params.id);
    const t = db.prepare(`SELECT * FROM trades WHERE id = ?`).get(tradeId) as any;
    if (!t) return res.status(404).json({ error: "NOT_FOUND" });
    if (t.user_id !== userId) return res.status(403).json({ error: "FORBIDDEN" });

    db.prepare(`UPDATE trades SET status = 'CANCELED' WHERE id = ?`).run(tradeId);

    auditOrderCanceled(db, req, {
      id: t.id,
      user_id: t.user_id,
      symbol: t.symbol,
      side: t.side,
      order_type: t.order_type,
      time_in_force: t.time_in_force,
      lots: t.lots,
      requested_price: t.requested_price,
      correlation_id: t.correlation_id,
      order_id: t.order_id,
      position_id: t.position_id,
    }, "User canceled pending order");

    res.json({ ok: true, tradeId, correlationId: ctx.correlationId });
  });

  // ---- POST /trades/:id/close (manual close) ----
  r.post("/trades/:id/close", attachAuditContext, requireAuth, (req, res) => {
    const userId = req.user.id as string;
    const ctx = req.auditContext!;
    ctx.userId = userId;

    const tradeId = String(req.params.id);
    const t = db.prepare(`SELECT * FROM trades WHERE id = ?`).get(tradeId) as any;
    if (!t) return res.status(404).json({ error: "NOT_FOUND" });
    if (t.user_id !== userId) return res.status(403).json({ error: "FORBIDDEN" });

    db.prepare(`UPDATE trades SET status = 'CLOSED' WHERE id = ?`).run(tradeId);

    auditPositionClosed(db, req, {
      id: t.id,
      user_id: t.user_id,
      symbol: t.symbol,
      side: t.side,
      order_type: t.order_type,
      lots: t.lots,
      correlation_id: t.correlation_id,
      position_id: t.position_id,
    }, "Manual close");

    res.json({ ok: true, tradeId, correlationId: ctx.correlationId });
  });

  // ---- POST /trades/:id/targets (TP/SL update) ----
  r.post("/trades/:id/targets", attachAuditContext, requireAuth, (req, res) => {
    const userId = req.user.id as string;

    ctx.userId = userId;

      position_id: t.position_id,
    }, { takeProfit, stopLoss });

  });

}
```

## What this gives you immediately (auditor-grade for the four items)

### 1) CorrelationId on every attempt

* Generated if missing; returned as `x-correlation-id` response header
* Stored in `order_intent_audit.correlation_id` and `trades.correlation_id` and `trade_audit.correlation_id`

### 2) order_intent_audit (RECEIVED + DECISION)

* Always writes:

  * `RECEIVED` at request entry
  * `DECISION PASS/REJECT` at risk conclusion
* For rejects: writes `reject_check` and `reject_reason`

### 3) trade_audit lifecycle events

* Market creation: `ORDER_PLACED` then `ORDER_FILLED`
* Pending placement: `ORDER_PLACED`
* Cancel: `ORDER_CANCELED`
* Manual close: `POSITION_CLOSED`
* TP/SL update: `TARGETS_UPDATED`

### 4) Tamper-evident SHA-256 chaining

* Both tables are **append-only**
* Each stream is hash-chained:

  * `order_intent_audit` chained per `correlation_id`
  * `trade_audit` chained per `trade_id`
* Stored fields:

  * `prev_hash`
  * `event_hash`
  * `payload_json` is **canonical** and is exactly what is hashed
