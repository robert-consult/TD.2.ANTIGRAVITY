1) Client Device ID (stable per device)

File: client/src/lib/deviceId.ts (NEW)

// client/src/lib/deviceId.ts
const STORAGE_KEY = "tc_device_id_v1";

function fallbackUuid(): string {
  // RFC4122-ish fallback (good enough)
  const s: string[] = [];
  const hex = "0123456789abcdef";
  for (let i = 0; i < 36; i++) s.push(hex[Math.floor(Math.random() * 16)]);
  s[14] = "4";
  // @ts-ignore
  s[19] = hex[(parseInt(s[19], 16) & 0x3) | 0x8];
  s[8] = s[13] = s[18] = s[23] = "-";
  return s.join("");
}

export function getDeviceId(): string {
  try {
    const existing = localStorage.getItem(STORAGE_KEY);
    if (existing && existing.length >= 16) return existing;

    const id =
      typeof crypto !== "undefined" && "randomUUID" in crypto
        ? (crypto as any).randomUUID()
        : fallbackUuid();

    localStorage.setItem(STORAGE_KEY, id);
    return id;
  } catch {
    // If localStorage blocked, still return something stable-ish for this session:
    return typeof crypto !== "undefined" && "randomUUID" in crypto
      ? (crypto as any).randomUUID()
      : fallbackUuid();
  }
}

2) Ensure all requests send X-Device-Id

Because I’m not inspecting your ZIP, I’m giving you a repo-agnostic approach that will work even if you have multiple callers.

File: client/src/lib/api.ts (NEW)

Use this everywhere in the admin UI (and you can gradually migrate other calls later).

// client/src/lib/api.ts
import { getDeviceId } from "./deviceId";

export async function apiFetch<T>(
  url: string,
  opts?: {
    method?: string;
    body?: any;
    headers?: Record<string, string>;
  }
): Promise<T> {
  const method = opts?.method ?? "GET";
  const headers: Record<string, string> = {
    ...(opts?.headers ?? {}),
    "X-Device-Id": getDeviceId(),
  };

  let body: any = undefined;
  if (opts?.body !== undefined) {
    headers["Content-Type"] = "application/json";
    body = JSON.stringify(opts.body);
  }

  const res = await fetch(url, {
    method,
    headers,
    body,
    credentials: "include",
  });

  if (!res.ok) {
    const text = await res.text().catch(() => "");
    throw new Error(`API ${res.status}: ${text || res.statusText}`);
  }

  return (await res.json()) as T;
}

Optional global patch (recommended)

If your app uses a centralized fetch wrapper like client/src/lib/queryClient.ts, add the X-Device-Id header there too.

Example pattern:

// Wherever your default fetcher lives
import { getDeviceId } from "./deviceId";

const headers = {
  "X-Device-Id": getDeviceId(),
  ...(yourExistingHeaders),
};

  if (!ts) return "—";

function TierBadge({ tier }: { tier: string }) {
  const t = (tier || "").toUpperCase();
  const cls =
    t === "CRITICAL"
      ? "bg-red-600 text-white"
      : t === "HIGH"
      ? "bg-orange-600 text-white"
      : t === "MED"
      ? "bg-yellow-500 text-black"
      : "bg-muted text-foreground";
  return <span className={`px-2 py-1 rounded text-xs font-semibold ${cls}`}>{t || "LOW"}</span>;
}

                    </table>
                  ) : (
                    <div className="p-3 text-sm text-muted-foreground">No signals.</div>
                  )}
                </div>

              </div>
            )}
          </div>
        </div>
      )}
    </div>
  );
}
