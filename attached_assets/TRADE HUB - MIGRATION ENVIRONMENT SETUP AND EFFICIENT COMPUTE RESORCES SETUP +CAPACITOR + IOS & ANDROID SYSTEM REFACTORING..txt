import { EventEmitter } from "events";
import crypto from "crypto";
import type { Redis } from "ioredis";
import { getValkey } from "./valkey";

export interface LiveEvent {
  type: string;
  userId?: number;
  payload?: any;
  ts?: number;
}

type WireEvent = LiveEvent & { origin: string };

const emitter = new EventEmitter();
const ORIGIN = process.env.LIVE_BUS_INSTANCE_ID ?? crypto.randomUUID();
const CHANNEL = process.env.LIVE_BUS_CHANNEL ?? "tradehub:live";

let pub: Redis | null = null;
let sub: Redis | null = null;
let valkeyReady = false;
let initStarted = false;

async function initValkeyBridge() {
  if (initStarted) return;
  initStarted = true;

  const url = process.env.VALKEY_URL;
  if (!url) return;

  try {
    pub = await getValkey("livebus-pub");
    sub = await getValkey("livebus-sub");

    await sub.subscribe(CHANNEL);
    sub.on("message", (_chan, message) => {
      try {
        const evt = JSON.parse(message) as WireEvent;
        if (!evt || evt.origin === ORIGIN) return;
        const { origin: _o, ...localEvt } = evt;
        emitter.emit("event", localEvt);
      } catch {
        // ignore malformed
      }
    });

    valkeyReady = true;
    // eslint-disable-next-line no-console
    console.log(`[LiveBus] Valkey bridge enabled channel=${CHANNEL}`);
  } catch (e) {
    valkeyReady = false;
    // eslint-disable-next-line no-console
    console.warn("[LiveBus] Valkey bridge init failed; falling back to in-proc emitter:", e);
  }
}

// start bridge lazily
void initValkeyBridge();

export function onLiveEvent(handler: (event: LiveEvent) => void) {
  emitter.on("event", handler);
  return () => emitter.off("event", handler);
}

// Throttle by (type,userId) for high-frequency UI triggers (keep minimal)
const throttleMs = 250;
const lastSent = new Map<string, number>();

export function publishLiveEvent(event: LiveEvent) {
  const now = Date.now();
  const enriched: LiveEvent = { ...event, ts: now };

  // Keep throttle for account/trade UI updates; allow quotes batches through.
  const throttleKey = `${event.type}:${event.userId ?? "global"}`;
  if (!event.type.startsWith("quotes:")) {
    const last = lastSent.get(throttleKey) ?? 0;
    if (now - last < throttleMs) return;
    lastSent.set(throttleKey, now);
  }

  emitter.emit("event", enriched);

  if (!valkeyReady || !pub) return;

  const wire: WireEvent = { ...enriched, origin: ORIGIN };
  // fire-and-forget
  void pub.publish(CHANNEL, JSON.stringify(wire)).catch(() => {});
}

import React, { createContext, useCallback, useContext, useEffect, useMemo, useRef, useState } from "react";

type Handler = (msg: any) => void;

type LiveCtx = {
  connected: boolean;
  subscribe: (type: string, handler: Handler) => () => void;

  // quotes
  subscribeQuotes: (handler: (rows: any[], kind: "snapshot" | "batch") => void) => () => void;
};

const Ctx = createContext<LiveCtx | null>(null);

function wsUrl() {
  const proto = window.location.protocol === "https:" ? "wss:" : "ws:";
  return `${proto}//${window.location.host}/ws`;
}

export function LiveUpdatesProvider({ children }: { children: React.ReactNode }) {
  const [connected, setConnected] = useState(false);

  const wsRef = useRef<WebSocket | null>(null);
  const reconnectRef = useRef<number>(0);
  const reconnectTimer = useRef<number | null>(null);

  const handlers = useRef<Map<string, Set<Handler>>>(new Map());
  const quoteHandlers = useRef<Set<(rows: any[], kind: "snapshot" | "batch") => void>>(new Set());
  const quotesSubscribed = useRef(false);

  const emit = useCallback((type: string, msg: any) => {
    const set = handlers.current.get(type);
    if (!set || set.size === 0) return;
    for (const fn of set) {
      try { fn(msg); } catch { /* ignore */ }
    }
  }, []);

  const emitQuotes = useCallback((rows: any[], kind: "snapshot" | "batch") => {
    for (const fn of quoteHandlers.current) {
      try { fn(rows, kind); } catch { /* ignore */ }
    }
  }, []);

  const send = useCallback((payload: any) => {
    const ws = wsRef.current;
    if (!ws || ws.readyState !== WebSocket.OPEN) return;
    ws.send(JSON.stringify(payload));
  }, []);

  const ensureQuotesSubscribed = useCallback(() => {
    if (quotesSubscribed.current) return;
    quotesSubscribed.current = true;
    send({ type: "quotes:subscribe" });
  }, [send]);

  const maybeUnsubscribeQuotes = useCallback(() => {
    if (!quotesSubscribed.current) return;
    if (quoteHandlers.current.size > 0) return;
    quotesSubscribed.current = false;
    send({ type: "quotes:unsubscribe" });
  }, [send]);

  const connect = useCallback(() => {
    if (reconnectTimer.current) {
      window.clearTimeout(reconnectTimer.current);
      reconnectTimer.current = null;
    }

    const ws = new WebSocket(wsUrl());
    wsRef.current = ws;

    ws.onopen = () => {
      setConnected(true);
      reconnectRef.current = 0;

      // re-subscribe quotes if needed
      if (quoteHandlers.current.size > 0) {
        quotesSubscribed.current = false;
        ensureQuotesSubscribed();
      }
    };

    ws.onclose = () => {
      setConnected(false);
      wsRef.current = null;

      // reconnect with backoff
      const attempt = Math.min(reconnectRef.current + 1, 8);
      reconnectRef.current = attempt;
      const delay = Math.min(30000, 500 * Math.pow(2, attempt)); // up to 30s

      reconnectTimer.current = window.setTimeout(() => connect(), delay);
    };

    ws.onerror = () => {
      try { ws.close(); } catch { /* ignore */ }
    };

    ws.onmessage = (ev) => {
      let msg: any;
      try { msg = JSON.parse(String(ev.data)); } catch { return; }
      const t = String(msg?.type ?? "");
      if (!t) return;

      if (t === "quotes:snapshot") {
        const rows = Array.isArray(msg?.rows) ? msg.rows : [];
        emitQuotes(rows, "snapshot");
        return;
      }
      if (t === "quotes:batch") {
        const rows = Array.isArray(msg?.rows) ? msg.rows : [];
        emitQuotes(rows, "batch");
        return;
      }

      emit(t, msg);
    };
  }, [emit, emitQuotes, ensureQuotesSubscribed]);

  useEffect(() => {
    connect();
    return () => {
      if (reconnectTimer.current) window.clearTimeout(reconnectTimer.current);
      const ws = wsRef.current;
      wsRef.current = null;
      try { ws?.close(); } catch { /* ignore */ }
    };
  }, [connect]);

  const subscribe = useCallback((type: string, handler: Handler) => {
    if (!handlers.current.has(type)) handlers.current.set(type, new Set());
    handlers.current.get(type)!.add(handler);
    return () => handlers.current.get(type)?.delete(handler);
  }, []);

  const subscribeQuotes = useCallback((handler: (rows: any[], kind: "snapshot" | "batch") => void) => {
    quoteHandlers.current.add(handler);
    ensureQuotesSubscribed();

    return () => {
      quoteHandlers.current.delete(handler);
      maybeUnsubscribeQuotes();
    };
  }, [ensureQuotesSubscribed, maybeUnsubscribeQuotes]);

  const value = useMemo<LiveCtx>(() => ({
    connected,
    subscribe,
    subscribeQuotes,
  }), [connected, subscribe, subscribeQuotes]);

  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
}

export function useLiveUpdates() {
  const v = useContext(Ctx);
  if (!v) throw new Error("useLiveUpdates must be used within LiveUpdatesProvider");
  return v;
}

import { EventEmitter } from "events";
import crypto from "crypto";
import Redis from "ioredis";

export type LiveEvent = {
  type: string;
  userId?: number;
  payload?: Record<string, any> | null;
  ts?: number;
};

type WireEvent = LiveEvent & { origin: string };

const emitter = new EventEmitter();
emitter.setMaxListeners(1000);

const ORIGIN = process.env.LIVE_BUS_INSTANCE_ID ?? crypto.randomUUID();
const CHANNEL = process.env.LIVE_BUS_CHANNEL ?? "tradehub:live";

let pub: Redis | null = null;
let sub: Redis | null = null;
let valkeyReady = false;

const lastEmitByKey = new Map<string, number>();

function shouldEmit(event: LiveEvent): boolean {
  // Keep your existing throttles
  if (event.type === "account:updated" && typeof event.userId === "number") {
    const key = `account:updated:${event.userId}`;
    const now = Date.now();
    const last = lastEmitByKey.get(key) ?? 0;
    const minIntervalMs = Number(process.env.LIVE_ACCOUNT_THROTTLE_MS ?? 1000);
    if (now - last < minIntervalMs) return false;
    lastEmitByKey.set(key, now);
  }
  return true;
}

async function initValkeyBridge() {
  const url = process.env.VALKEY_URL;
  if (!url) return;

  try {
    pub = new Redis(url, { lazyConnect: true, maxRetriesPerRequest: 1 });
    sub = new Redis(url, { lazyConnect: true, maxRetriesPerRequest: 1 });

    await pub.connect();
    await sub.connect();

    await sub.subscribe(CHANNEL);
    sub.on("message", (_ch, message) => {
      try {
        const evt = JSON.parse(message) as WireEvent;
        if (!evt || evt.origin === ORIGIN) return;
        const { origin: _o, ...local } = evt;
        emitter.emit("event", local);
      } catch {
        // ignore
      }
    });

    valkeyReady = true;
    // eslint-disable-next-line no-console
    console.log(`[LiveBus] Valkey enabled channel=${CHANNEL}`);
  } catch (e) {
    valkeyReady = false;
    // eslint-disable-next-line no-console
    console.warn("[LiveBus] Valkey init failed; using in-proc emitter only:", e);
  }
}

void initValkeyBridge();

export function onLiveEvent(handler: (event: LiveEvent) => void) {
  emitter.on("event", handler);
  return () => emitter.off("event", handler);
}

export function publishLiveEvent(event: LiveEvent) {
  const ev: LiveEvent = { ...event, ts: event.ts ?? Date.now() };
  if (!shouldEmit(ev)) return;

  // local
  emitter.emit("event", ev);

  // cross-pod
  if (!valkeyReady || !pub) return;
  const wire: WireEvent = { ...(ev as any), origin: ORIGIN };
  void pub.publish(CHANNEL, JSON.stringify(wire)).catch(() => {});
}

import BetterSQLite3 from "better-sqlite3";
import { ensureMarketDailyCloseTable } from "../utils/marketDailyClose";
import { computeCurrentSessionDay } from "../utils/quoteSession";

export function readEnhancedQuotesLatest() {
  const db = new BetterSQLite3("./trading_app.db");
  try { db.pragma("busy_timeout = 5000"); } catch {}

  ensureMarketDailyCloseTable(db);

  let staleThresholdMs = 30000;
  let fxRolloverTz = "America/New_York";
  let fxRolloverTime = "17:00";

  try {
    const cfg = db
      .prepare(`SELECT stale_threshold_ms AS staleMs, fx_rollover_tz AS tz, fx_rollover_time AS time FROM system_config WHERE id = 1`)
      .get() as any;
    if (cfg?.staleMs) staleThresholdMs = Number(cfg.staleMs);
    if (cfg?.tz) fxRolloverTz = String(cfg.tz);
    if (cfg?.time) fxRolloverTime = String(cfg.time);
  } catch {}

  const currentSessionDay = computeCurrentSessionDay({ tz: fxRolloverTz, time: fxRolloverTime });
  const nowMs = Date.now();

  const quotes = db.prepare(`
    SELECT
      q.symbol,
      q.bid,
      q.ask,
      q.price AS lastPrice,
      COALESCE(q.is_stale, 0) AS isStale,
      COALESCE(q.last_api_update, ?) AS lastApiUpdate,
      COALESCE(
        (
          SELECT dc.close
          FROM market_daily_close dc
          WHERE dc.symbol = q.symbol AND dc.session_day < ?
          ORDER BY dc.session_day DESC
          LIMIT 1
        ),
        (
          SELECT dc2.close
          FROM market_daily_close dc2
          WHERE dc2.symbol = q.symbol
          ORDER BY dc2.session_day DESC
          LIMIT 1
        )
      ) AS prevClose
    FROM quotes q
  `).all(nowMs, currentSessionDay);

  db.close();

  return quotes.map((quote: any) => {
    const bid = typeof quote.bid === "number" ? quote.bid : null;
    const ask = typeof quote.ask === "number" ? quote.ask : null;
    const lastPrice = typeof quote.lastPrice === "number" ? quote.lastPrice : null;
    const midPrice = bid != null && ask != null ? (bid + ask) / 2 : lastPrice;
    const spread = bid != null && ask != null ? Math.abs(ask - bid) : null;
    const prevClose = typeof quote.prevClose === "number" ? quote.prevClose : null;

    let pctChange = 0;
    if (prevClose != null && prevClose > 0 && Number.isFinite(midPrice)) {
      pctChange = ((midPrice - prevClose) / prevClose) * 100;
      pctChange = Math.round(pctChange * 100) / 100;
    }

    const change = Number.isFinite(midPrice) && prevClose != null ? midPrice - prevClose : 0;
    const lastUpdate = quote.lastApiUpdate || nowMs;
    const ageMs = nowMs - lastUpdate;
    const dbIsStale = quote.isStale === 1 || quote.isStale === true;
    const isStale = dbIsStale || ageMs > staleThresholdMs;

    return {
      symbol: quote.symbol,
      bid,
      ask,
      price: midPrice,
      spread,
      prevClose: prevClose ?? midPrice,
      change,
      pctChange,
      isStale,
      lastApiUpdate: lastUpdate,
      dataAge: ageMs,
      timestamp: Math.floor(nowMs / 1000),
    };
  });
}

7.1 Create client/src/live/LiveUpdatesProvider.tsx
import React, { createContext, useCallback, useContext, useEffect, useMemo, useRef, useState } from "react";

type QuoteHandler = (rows: any[], kind: "snapshot" | "batch") => void;
type Handler = (msg: any) => void;

type LiveCtx = {
  connected: boolean;
  subscribe: (type: string, handler: Handler) => () => void;
  subscribeQuotes: (handler: QuoteHandler) => () => void;
};

const Ctx = createContext<LiveCtx | null>(null);

function wsUrl() {
  const proto = window.location.protocol === "https:" ? "wss:" : "ws:";
  return `${proto}//${window.location.host}/ws`;
}

export function LiveUpdatesProvider({ children }: { children: React.ReactNode }) {
  const [connected, setConnected] = useState(false);

  const wsRef = useRef<WebSocket | null>(null);
  const reconnectAttempt = useRef(0);
  const handlers = useRef<Map<string, Set<Handler>>>(new Map());
  const quoteHandlers = useRef<Set<QuoteHandler>>(new Set());
  const quotesSubscribed = useRef(false);

  const send = useCallback((payload: any) => {
    const ws = wsRef.current;
    if (!ws || ws.readyState !== WebSocket.OPEN) return;
    ws.send(JSON.stringify(payload));
  }, []);

  const ensureQuotesSubscribed = useCallback(() => {
    if (quotesSubscribed.current) return;
    quotesSubscribed.current = true;
    send({ type: "quotes:subscribe" });
  }, [send]);

  const maybeUnsubscribeQuotes = useCallback(() => {
    if (!quotesSubscribed.current) return;
    if (quoteHandlers.current.size > 0) return;
    quotesSubscribed.current = false;
    send({ type: "quotes:unsubscribe" });
  }, [send]);

  const connect = useCallback(() => {
    const ws = new WebSocket(wsUrl());
    wsRef.current = ws;

    ws.onopen = () => {
      setConnected(true);
      reconnectAttempt.current = 0;

      if (quoteHandlers.current.size > 0) {
        quotesSubscribed.current = false;
        ensureQuotesSubscribed();
      }
    };

    ws.onclose = () => {
      setConnected(false);
      wsRef.current = null;

      const a = Math.min(reconnectAttempt.current + 1, 8);
      reconnectAttempt.current = a;
      const delay = Math.min(30_000, 500 * Math.pow(2, a));
      window.setTimeout(connect, delay);
    };

    ws.onerror = () => {
      try { ws.close(); } catch {}
    };

    ws.onmessage = (ev) => {
      let msg: any;
      try { msg = JSON.parse(String(ev.data)); } catch { return; }
      const type = String(msg?.type ?? "");
      if (!type) return;

      if (type === "quotes:snapshot") {
        const rows = Array.isArray(msg?.rows) ? msg.rows : [];
        for (const fn of quoteHandlers.current) fn(rows, "snapshot");
        return;
      }
      if (type === "quotes:batch") {
        const rows = Array.isArray(msg?.rows) ? msg.rows : [];
        for (const fn of quoteHandlers.current) fn(rows, "batch");
        return;
      }

      const set = handlers.current.get(type);
      if (!set) return;
      for (const fn of set) {
        try { fn(msg); } catch {}
      }
    };
  }, [ensureQuotesSubscribed]);

  useEffect(() => {
    connect();
    return () => { try { wsRef.current?.close(); } catch {} };
  }, [connect]);

  const subscribe = useCallback((type: string, handler: Handler) => {
    if (!handlers.current.has(type)) handlers.current.set(type, new Set());
    handlers.current.get(type)!.add(handler);
    return () => handlers.current.get(type)?.delete(handler);
  }, []);

    quoteHandlers.current.add(handler);
    ensureQuotesSubscribed();
    return () => {
      quoteHandlers.current.delete(handler);
      maybeUnsubscribeQuotes();
    };
  }, [ensureQuotesSubscribed, maybeUnsubscribeQuotes]);

  const value = useMemo(() => ({ connected, subscribe, subscribeQuotes }), [connected, subscribe, subscribeQuotes]);
  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;

    });

  subscribeQuotes: (symbols?: string[]) => void;
  unsubscribeQuotes: () => void;
};

const Ctx = createContext<LiveCtx | null>(null);

function wsUrlFromLocation() {
  const proto = window.location.protocol === "https:" ? "wss" : "ws";
  return `${proto}://${window.location.host}/ws`;
}

export function LiveUpdatesProvider({ children }: { children: React.ReactNode }) {
  const [quotesBySymbol, setQuotesBySymbol] = useState<Record<string, QuoteRow>>({});
  const subscribedRef = useRef<string[] | null>(null);

  const { isConnected, sendMessage } = useWebSocket(wsUrlFromLocation(), {
    enabled: true,
    onOpen: () => {
      // resubscribe on reconnect

        sendMessage({ type: "quotes:subscribe", symbols: subscribedRef.current });
      }
    },
    onMessage: (msg) => {

// server/services/valkeySessionStore.ts
import session from "express-session";
import type IORedis from "ioredis";

function ttlSecondsFromSess(sess: any, defaultTtlSeconds: number): number {
  // express-session cookie maxAge is ms
  const maxAgeMs = Number(sess?.cookie?.maxAge ?? 0);
  if (Number.isFinite(maxAgeMs) && maxAgeMs > 0) return Math.max(60, Math.floor(maxAgeMs / 1000));

  // cookie.expires might exist
  const exp = sess?.cookie?.expires ? new Date(sess.cookie.expires).getTime() : 0;
  if (Number.isFinite(exp) && exp > Date.now()) return Math.max(60, Math.floor((exp - Date.now()) / 1000));

  return defaultTtlSeconds;
}

export class ValkeySessionStore extends session.Store {
  private redis: IORedis;
  private prefix: string;
  private defaultTtlSeconds: number;

  constructor(opts: { redis: IORedis; prefix?: string; defaultTtlSeconds?: number }) {
    super();
    this.redis = opts.redis;
    this.prefix = opts.prefix ?? "sess:";
    this.defaultTtlSeconds = Number(opts.defaultTtlSeconds ?? 86400);
  }

  private key(sid: string) {
    return `${this.prefix}${sid}`;
  }

  get(sid: string, cb: (err?: any, session?: any | null) => void) {
    void (async () => {
      try {
        const raw = await this.redis.get(this.key(sid));
        if (!raw) return cb(undefined, null);
        return cb(undefined, JSON.parse(raw));
      } catch (e) {
        return cb(e, null);
      }
    })();
  }

  set(sid: string, sess: any, cb?: (err?: any) => void) {
    void (async () => {
      try {
        const ttl = ttlSecondsFromSess(sess, this.defaultTtlSeconds);
        await this.redis.set(this.key(sid), JSON.stringify(sess), "EX", ttl);
        cb?.();
      } catch (e) {
        cb?.(e);
      }
    })();
  }

  destroy(sid: string, cb?: (err?: any) => void) {
    void (async () => {
      try {
        await this.redis.del(this.key(sid));
        cb?.();
      } catch (e) {
        cb?.(e);
      }
    })();
  }

  touch(sid: string, sess: any, cb?: (err?: any) => void) {
    void (async () => {
      try {
        const ttl = ttlSecondsFromSess(sess, this.defaultTtlSeconds);
        await this.redis.expire(this.key(sid), ttl);
        cb?.();
      } catch (e) {
        cb?.(e);

          WHERE session_day < ?

          ORDER BY symbol ASC, session_day DESC
        `
        )

      for (const r of closeRows) {
        const sym = String(r.symbol);
        if (!prevClose.has(sym) && typeof r.close === "number") {
          prevClose.set(sym, r.close);
        }
      }
    } catch {}
  }

  const staleThreshold = dynamicConfig?.staleThresholdMs ?? DEFAULT_STALE_MS;

  const enhanced: CachedQuote[] = rows.map((r) => {
    const symbol = String(r.symbol);
    const bid = typeof r.bid === "number" ? r.bid : null;
    const ask = typeof r.ask === "number" ? r.ask : null;

    const price =
      typeof r.price === "number"
        ? r.price
        : bid != null && ask != null
        ? (bid + ask) / 2
        : bid != null
        ? bid
        : ask != null
        ? ask
        : null;

    const lastApiUpdate =
      typeof r.lastUpdated === "number"
        ? r.lastUpdated
        : typeof r.last_api_update === "number"
        ? r.last_api_update
        : now;

    const dataAge = Math.max(0, now - lastApiUpdate);
    const isStale = !!r.isStale || dataAge > staleThreshold;

    const pc = prevClose.get(symbol);
    const prev = typeof pc === "number" ? pc : (price ?? 0);

    const change = price != null ? price - prev : null;

    const spread =
      bid != null && ask != null ? Math.max(0, ask - bid) : null;

      symbol,
      bid,
      ask,
      price,
      spread,
      prevClose: prev,
      change,
      pctChange,
      isStale,
      lastApiUpdate,
      dataAge,
      timestamp: now,
    };
  });

export function useQuotes() {
  const { user } = useAuth();

  const [quoteMap, setQuoteMap] = useState<QuoteMap>({});
  const subscribedRef = useRef<string>("");

  const wsUrl = useMemo(() => getWsUrl(), []);

  const { isConnected, sendMessage } = useWebSocket(wsUrl, {
    onMessage: (msg) => {
      if (!msg || typeof msg !== "object") return;

      if (msg.type === "quotes:snapshot" && Array.isArray(msg?.payload?.quotes)) {
        const incoming: Quote[] = msg.payload.quotes;
        setQuoteMap((prev) => {
          const next = { ...prev };
          for (const q of incoming) next[toKey(q.symbol)] = q;
          return next;
        });
        return;
      }

      if (msg.type === "quotes:update" && msg?.payload?.quote) {
        const q: Quote = msg.payload.quote;
        setQuoteMap((prev) => ({ ...prev, [toKey(q.symbol)]: q }));
        return;
      }
    },
  });

  // Bootstrap symbol list (used for WS subscribe)
  const { data: symbols = [] } = useQuery<{ id: number; symbol: string; enabled: boolean }[]>({
    queryKey: ["/api/symbols"],
    staleTime: 60_000,
  });

  const enabledSymbols = useMemo(() => {
    return (symbols || []).filter((s) => s.enabled).map((s) => toKey(s.symbol));
  }, [symbols]);

  // Authenticate WS (server already verifies cookie session; this is optional metadata)
  useEffect(() => {
    if (user && sendMessage) {
      sendMessage({ type: "auth", userId: user.id });
    }
  }, [user, sendMessage]);

  // Subscribe to enabled symbols once WS is up (and whenever symbol set changes)
  useEffect(() => {
    if (!user) return;
    if (!isConnected) return;

    const key = enabledSymbols.join(",");
    if (!key) return;

    if (subscribedRef.current !== key) {
      subscribedRef.current = key;
      sendMessage?.({ type: "quotes:subscribe", symbols: enabledSymbols });
    }
  }, [user, isConnected, enabledSymbols, sendMessage]);

  // REST bootstrap once (and slow fallback if WS is down)
  const bootstrap = useQuery<Quote[]>({

    quoteMap,
    isConnected,
    isLoading,

  };

let client: Redis | null = null;
let pub: Redis | null = null;
let sub: Redis | null = null;

export function getValkey(): Redis | null {
  const url = process.env.VALKEY_URL;
  if (!url) return null;
  if (client) return client;

  client = new Redis(url, {
    enableReadyCheck: true,
    maxRetriesPerRequest: 1,
    lazyConnect: true,
  });

  client.on("error", () => {
    // Silent by design; callers should handle null-ish behavior via wrappers.
  });

  return client;
}

function newClient(): Redis | null {
  const url = process.env.VALKEY_URL;
  if (!url) return null;
  const c = new Redis(url, {
    enableReadyCheck: true,
    maxRetriesPerRequest: 1,
    lazyConnect: true,
  });
  c.on("error", () => {});
  return c;
}

export function getValkeyPub(): Redis | null {
  if (pub) return pub;
  pub = newClient();
  return pub;
}

export function getValkeySub(): Redis | null {

function key(prefix: string, sid: string) {

function ttlSeconds(sess: any): number {
  const maxAgeMs = Number(sess?.cookie?.maxAge ?? 0);
  if (Number.isFinite(maxAgeMs) && maxAgeMs > 0) return Math.max(1, Math.ceil(maxAgeMs / 1000));

  const exp = sess?.cookie?.expires;
  if (exp) {
    const expMs = new Date(exp).getTime();
    const diffMs = expMs - Date.now();
    if (Number.isFinite(diffMs) && diffMs > 0) return Math.max(1, Math.ceil(diffMs / 1000));
  }

  // Fallback: 1 day
  return 86400;
}

    // @ts-ignore
    super();
    this.prefix = opts?.prefix ?? DEFAULT_PREFIX;
  }

      .then((sess) => cb(undefined, sess))
      .catch((err) => cb(err));
  }

      .catch((err) => cb(err));

// server/security/valkeySessionStore.ts
import session from "express-session";

import { getValkey } from "../services/valkey";

export const VALKEY_SESSION_PREFIX = process.env.SESSION_KEY_PREFIX ?? "sess:";

function keyFor(sid: string) {
  return `${VALKEY_SESSION_PREFIX}${sid}`;
}

function ttlSecondsFromSess(sess: any): number {
  // express-session cookie maxAge is ms; cookie.expires is Date.
  const maxAgeMs =
    typeof sess?.cookie?.maxAge === "number" ? sess.cookie.maxAge : null;

  const expires =
    sess?.cookie?.expires ? new Date(sess.cookie.expires).getTime() : null;

  const now = Date.now();

  let ttlMs: number | null = null;

  if (expires && Number.isFinite(expires)) ttlMs = Math.max(0, expires - now);
  else if (maxAgeMs && Number.isFinite(maxAgeMs)) ttlMs = Math.max(0, maxAgeMs);

  // Default 24h if not present
  if (!ttlMs || ttlMs <= 0) ttlMs = 24 * 60 * 60 * 1000;

  return Math.max(60, Math.floor(ttlMs / 1000)); // minimum 60s
}

/**
 * Minimal Valkey-backed express-session Store.
 * Compatible with Valkey (Redis protocol) using ioredis client.
 */
export class ValkeySessionStore extends session.Store {
  private client: Redis;

  constructor(opts?: { client?: Redis }) {
    super();
    this.client = opts?.client ?? getValkey();
  }

  override get(sid: string, cb: (err?: any, session?: any | null) => void) {
    this.client
      .get(keyFor(sid))
      .then((raw) => {
        if (!raw) return cb(null, null);
        try {
          return cb(null, JSON.parse(raw));
        } catch (e) {
          return cb(e, null);
        }
      })
      .catch((err) => cb(err, null));
  }

  override set(sid: string, sess: any, cb?: (err?: any) => void) {
    const ttlSec = ttlSecondsFromSess(sess);
    const payload = JSON.stringify(sess ?? {});
    this.client
      .set(keyFor(sid), payload, "EX", ttlSec)
      .then(() => cb?.())
      .catch((err) => cb?.(err));
  }

  override destroy(sid: string, cb?: (err?: any) => void) {
    this.client
      .del(keyFor(sid))
      .then(() => cb?.())
      .catch((err) => cb?.(err));
  }

  override touch(sid: string, sess: any, cb?: (err?: any) => void) {
    const ttlSec = ttlSecondsFromSess(sess);
    this.client
      .expire(keyFor(sid), ttlSec)
      .then(() => cb?.())
      .catch((err) => cb?.(err));
  }
}

// client/src/lib/origins.ts

function stripTrailingSlash(s: string) {
  return s.endsWith("/") ? s.slice(0, -1) : s;
}

function isAbsoluteUrl(url: string) {
  return /^https?:\/\//i.test(url);
}

export function apiOrigin(): string {
  // For Capacitor/mobile you can set VITE_API_ORIGIN="https://app.yourdomain.com"
  const env = (import.meta as any).env?.VITE_API_ORIGIN as string | undefined;
  if (env && typeof env === "string" && env.trim()) return stripTrailingSlash(env.trim());
  return ""; // same-origin
}

export function apiUrl(pathOrUrl: string): string {
  if (!pathOrUrl) return pathOrUrl;
  if (isAbsoluteUrl(pathOrUrl)) return pathOrUrl;

  const origin = apiOrigin();
  if (!origin) return pathOrUrl;

  if (pathOrUrl.startsWith("/")) return origin + pathOrUrl;
  return origin + "/" + pathOrUrl;
}

function httpToWs(originOrUrl: string) {
  return originOrUrl
    .replace(/^http:\/\//i, "ws://")
    .replace(/^https:\/\//i, "wss://");
}

export function wsUrl(): string {
  const envUrl = (import.meta as any).env?.VITE_WS_URL as string | undefined;
  if (envUrl && typeof envUrl === "string" && envUrl.trim()) return envUrl.trim();

  const envOrigin = (import.meta as any).env?.VITE_WS_ORIGIN as string | undefined;
  if (envOrigin && typeof envOrigin === "string" && envOrigin.trim()) {
    // allow http(s) origin; convert to ws(s)
    const base = httpToWs(stripTrailingSlash(envOrigin.trim()));
    return base + "/ws";
  }

  // default: same origin as the app is served from
  const loc = window.location;
  const base = loc.protocol === "https:" ? "wss://" : "ws://";
  return `${base}${loc.host}/ws`;
}

export class ApiError extends Error {
  status: number;

  constructor(message: string, status: number) {
    super(message);
    this.status = status;
    this.name = "ApiError";
  }
}

async function throwIfResNotOk(res: Response) {
  if (!res.ok) {
    const text = await res.text();
    let message = text || res.statusText;
    let errorData: any = null;
    try {
      errorData = JSON.parse(text);
      message = errorData.message || text || res.statusText;
    } catch {
      // non-JSON
    }

    // Global compliance gate: prompt for legal re-acceptance when required.
    if (errorData && (errorData.code === "LEGAL_REACCEPT_REQUIRED" || errorData.message === "LEGAL_REACCEPT_REQUIRED")) {
      try {
        window.dispatchEvent(new CustomEvent("legal:reaccept-required", { detail: errorData }));
      } catch {
        // ignore
      }
    }
    throw new ApiError(message, res.status);
  }
}

export async function apiRequest(
  method: string,
  url: string,
  data?: unknown | undefined,
): Promise<Response> {
  const identityHeaders = await getIdentityHeaders();
  const baseHeaders: Record<string, string> = {
    ...(data ? { "Content-Type": "application/json" } : {}),
    ...identityHeaders,
  };

  const finalUrl = apiUrl(url);

  const doFetch = (extra?: Record<string, string>) =>
    fetch(finalUrl, {
      method,
      headers: { ...baseHeaders, ...(extra ?? {}) },
      body: data ? JSON.stringify(data) : undefined,
      credentials: "include",
    });

  let res = await doFetch();

  // Bot challenge loop (428 Precondition Required)
  if (res.status === 428) {
    const cloned = res.clone();
    let payload: any = null;
    try {
      payload = await cloned.json();
    } catch {
      payload = null;
    }

    if (payload?.code === "BOT_CHALLENGE_REQUIRED" && payload?.challenge) {
      const proof = await solveBotChallenge(payload.challenge, baseHeaders);
      res = await doFetch({ "x-bot-proof": proof });
    }
  }

  await throwIfResNotOk(res);
  return res;
}

type UnauthorizedBehavior = "returnNull" | "throw";
export const getQueryFn: <T>(options: {
  on401: UnauthorizedBehavior;
}) => QueryFunction<T> =
  ({ on401: unauthorizedBehavior }) =>
  async ({ queryKey }) => {
    const identityHeaders = await getIdentityHeaders();
    const raw = queryKey[0] as string;
    const url = apiUrl(raw);

    const res = await fetch(url, {
      credentials: "include",
      headers: identityHeaders,
    });

    if (unauthorizedBehavior === "returnNull" && res.status === 401) {
      return null;
    }

    await throwIfResNotOk(res);
    return await res.json();
  };

import axios, { AxiosHeaders } from "axios";
import { getIdentityHeaders } from "./identity";

let installed = false;

export function installAxiosIdentityHeaders() {
  if (installed) return;
  installed = true;

  // Mobile/Capacitor: allow absolute API origin without rewriting all axios calls.
  const origin = apiOrigin();
  if (origin && !axios.defaults.baseURL) {
    axios.defaults.baseURL = origin;
  }

  // Always include cookies for session auth when same-origin or properly configured cross-origin.
  axios.defaults.withCredentials = true;

  axios.interceptors.request.use(async (config) => {
    const headers = await getIdentityHeaders();
    const merged = AxiosHeaders.from(config.headers ?? {});
    for (const [key, value] of Object.entries(headers)) {
      merged.set(key, value);
    }
    config.headers = merged;
    return config;
  });
}

import React, { createContext, useCallback, useContext, useEffect, useMemo, useRef, useState } from "react";
import { useQuery } from "@tanstack/react-query";
import { useAuth } from "@/hooks/use-auth";
import { wsUrl } from "@/lib/origins";

type QuoteRow = {
  symbol: string;
  price?: number;
  bid?: number;
  ask?: number;
  prevClose?: number;
  change?: number;
  pctChange?: number;
  percent_change?: number;
  isStale?: boolean;
  lastApiUpdate?: number;
  timestamp?: number;
  dataAge?: number;
};

type SymbolConfig = {
  id: number;
  symbol: string;
  name: string;
  enabled?: boolean;
};

export type QuoteView = {
  symbol: string;
  name: string;
  price: number;
  change: number;
  time: number;
  bid?: number;
  ask?: number;
  spread?: number;
  prevClose?: number;
  changePct?: number;
  percent_change?: number;
  isStale?: boolean;
  lastApiUpdate?: number;
  dataAge?: number;
};

type LiveState = {
  isConnected: boolean;      // WS transport connected
  isReady: boolean;          // server sent ws:ready
  lastMessageAt?: number;
  quotes: QuoteView[];
  hasStaleData: boolean;
  // Optional controls for later screens to reduce subscription set
  subscribeSymbols: (symbols: string[]) => void;
  unsubscribeSymbols: (symbols: string[]) => void;
};

const LiveUpdatesContext = createContext<LiveState | null>(null);

function normSymbol(s: string) {
  return String(s ?? "").replace("/", "").trim().toUpperCase();
}

function calculatePctChange(current: number, previous: number): number {
  if (!previous || previous === 0) return 0;
  return ((current - previous) / previous) * 100;
}

function safeNum(v: any, fallback = 0) {
  const n = typeof v === "number" ? v : parseFloat(String(v));
  return Number.isFinite(n) ? n : fallback;
}

export function LiveUpdatesProvider({ children }: { children: React.ReactNode }) {
  const { isAuthenticated } = useAuth();

  const [isConnected, setIsConnected] = useState(false);
  const [isReady, setIsReady] = useState(false);
  const [lastMessageAt, setLastMessageAt] = useState<number | undefined>(undefined);

  const wsRef = useRef<WebSocket | null>(null);
  const pingTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const reconnectTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const attemptRef = useRef(0);

  const pendingSendRef = useRef<any[]>([]);
  const desiredSubsRef = useRef<Set<string>>(new Set());
  const lastSentSubsRef = useRef<Set<string>>(new Set());

  const quoteMapRef = useRef<Map<string, QuoteView>>(new Map());
  const [quotes, setQuotes] = useState<QuoteView[]>([]);

  const fallbackPollMs = useMemo(() => {
    const raw = (import.meta as any).env?.VITE_QUOTES_FALLBACK_POLL_MS;
    const n = raw != null ? Number(raw) : NaN;
    return Number.isFinite(n) && n >= 500 ? n : 5000; // default 5s fallback when WS down
  }, []);

  // Symbols (names + enabled set)
  const { data: symbols = [] } = useQuery<SymbolConfig[]>({
    queryKey: ["/api/config/symbols"],
    enabled: isAuthenticated,
  });

  const enabledSymbols = useMemo(() => {
    return symbols
      .filter((s) => s.enabled !== false)
      .map((s) => ({ symbol: normSymbol(s.symbol), name: s.name }))
      .filter((s) => s.symbol.length > 0);
  }, [symbols]);

  // Bootstrap quotes + REST fallback polling only when WS is down
  const { data: latestQuotesData } = useQuery<QuoteRow[]>({
    queryKey: ["/api/quotes/latest"],
    enabled: isAuthenticated,
    refetchInterval: () => (isConnected ? false : fallbackPollMs),
    refetchOnWindowFocus: true,
    staleTime: 0,
  });

  const rebuildQuotes = useCallback(() => {
    const m = quoteMapRef.current;
    const out: QuoteView[] = [];
    for (const s of enabledSymbols) {
      const q = m.get(s.symbol);
      if (q) out.push(q);
    }
    setQuotes(out);
  }, [enabledSymbols]);

  // Apply bootstrap/fallback REST payload into map (server already includes stale fallback when feed is down)
  useEffect(() => {
    if (!latestQuotesData || !Array.isArray(latestQuotesData) || enabledSymbols.length === 0) return;

    const enabledNameMap = new Map(enabledSymbols.map((x) => [x.symbol, x.name]));

    for (const row of latestQuotesData) {
      const symbol = normSymbol(row.symbol);
      if (!enabledNameMap.has(symbol)) continue;

      const prevClose = row.prevClose != null ? safeNum(row.prevClose, 0) : undefined;
      const price = safeNum(row.price, 0);
      const bid = row.bid != null ? safeNum(row.bid, NaN) : undefined;
      const ask = row.ask != null ? safeNum(row.ask, NaN) : undefined;

      const lastApiUpdate =
        typeof row.lastApiUpdate === "number"
          ? row.lastApiUpdate
          : typeof row.timestamp === "number"
          ? row.timestamp * 1000
          : Date.now();

      const dataAge = Math.max(0, Date.now() - lastApiUpdate);

      const pctFromServer =
        typeof row.pctChange === "number"
          ? row.pctChange
          : typeof row.percent_change === "number"
          ? row.percent_change
          : prevClose
          ? calculatePctChange(price, prevClose)
          : 0;

      const change =
        typeof row.change === "number"
          ? row.change
          : prevClose
          ? price - prevClose
          : 0;

      const spread =
        bid != null && ask != null && Number.isFinite(bid) && Number.isFinite(ask)
          ? Math.abs(ask - bid)
          : undefined;

      quoteMapRef.current.set(symbol, {
        symbol,
        name: enabledNameMap.get(symbol) ?? symbol,
        price,
        change,
        time: row.timestamp ? row.timestamp : Date.now(),
        bid,
        ask,
        spread,
        prevClose,
        changePct: pctFromServer,
        percent_change: pctFromServer,
        isStale: !!row.isStale,
        lastApiUpdate,
        dataAge,
      });
    }

    rebuildQuotes();
  }, [latestQuotesData, enabledSymbols, rebuildQuotes]);

  const clearTimers = useCallback(() => {
    if (pingTimerRef.current) {
      clearInterval(pingTimerRef.current);
      pingTimerRef.current = null;
    }
    if (reconnectTimerRef.current) {
      clearTimeout(reconnectTimerRef.current);
      reconnectTimerRef.current = null;
    }
  }, []);

  const sendJson = useCallback((payload: any) => {
    const ws = wsRef.current;
    if (ws && ws.readyState === WebSocket.OPEN) {
      ws.send(JSON.stringify(payload));
      return true;
    }
    pendingSendRef.current.push(payload);
    return false;
  }, []);

  const flushPending = useCallback(() => {
    const ws = wsRef.current;
    if (!ws || ws.readyState !== WebSocket.OPEN) return;
    const pending = pendingSendRef.current.splice(0, pendingSendRef.current.length);
    for (const p of pending) {
      try {
        ws.send(JSON.stringify(p));
      } catch {
        // ignore
      }
    }
  }, []);

  const scheduleReconnect = useCallback(() => {
    if (!isAuthenticated) return;

    const attempt = attemptRef.current;
    attemptRef.current = attempt + 1;

    const base = Math.min(30_000, 1000 * Math.pow(2, attempt));
    const jitter = Math.floor(Math.random() * 250);
    const delay = base + jitter;

    reconnectTimerRef.current = setTimeout(() => {
      connect();
    }, delay);
  }, [isAuthenticated]);

  const applyDesiredSubscriptions = useCallback(() => {
    // Send only diffs (subscribe/unsubscribe)
    const desired = desiredSubsRef.current;
    const lastSent = lastSentSubsRef.current;

    const toSub: string[] = [];
    const toUnsub: string[] = [];

    for (const s of desired) if (!lastSent.has(s)) toSub.push(s);
    for (const s of lastSent) if (!desired.has(s)) toUnsub.push(s);

    if (toSub.length) sendJson({ type: "quotes:subscribe", symbols: toSub });
    if (toUnsub.length) sendJson({ type: "quotes:unsubscribe", symbols: toUnsub });

    lastSentSubsRef.current = new Set(desired);
  }, [sendJson]);

  const handleMessage = useCallback(
    (msg: any) => {
      setLastMessageAt(Date.now());

      if (!msg || typeof msg !== "object") return;

      if (msg.type === "ws:ready") {
        setIsReady(true);
        applyDesiredSubscriptions();
        return;
      }

      if (msg.type === "pong") return;

      if (msg.type === "quotes:subscribed") {
        // server echo; optional to display/debug later
        return;
      }

      // Server broadcasts quotes:delta from liveBus payload: { quotes: [...] }
      if (msg.type === "quotes:delta") {
        const delta = msg?.payload?.quotes;
        if (!Array.isArray(delta)) return;

        for (const row of delta) {
          const symbol = normSymbol(row?.symbol);
          if (!symbol) continue;

          const existing = quoteMapRef.current.get(symbol);
          // Only update symbols we already know about or that are enabled
          const isEnabled = enabledSymbols.some((s) => s.symbol === symbol);
          if (!existing && !isEnabled) continue;

          const bid = row?.bid != null ? safeNum(row.bid, NaN) : existing?.bid;
          const ask = row?.ask != null ? safeNum(row.ask, NaN) : existing?.ask;
          const price =
            row?.price != null
              ? safeNum(row.price, 0)
              : existing?.price ?? 0;

          const lastApiUpdate =
            typeof row?.lastApiUpdate === "number" ? row.lastApiUpdate : Date.now();

          const prevClose = existing?.prevClose;
          const change = prevClose != null ? price - prevClose : existing?.change ?? 0;
          const pct = prevClose != null ? calculatePctChange(price, prevClose) : existing?.changePct ?? 0;

          const spread =
            bid != null && ask != null && Number.isFinite(bid) && Number.isFinite(ask)
              ? Math.abs(ask - bid)
              : existing?.spread;

          const dataAge = Math.max(0, Date.now() - lastApiUpdate);

          quoteMapRef.current.set(symbol, {
            symbol,
            name: existing?.name ?? symbol,
            price,
            change,
            time: Date.now(),
            bid,
            ask,
            spread,
            prevClose,
            changePct: pct,
            percent_change: pct,
            isStale: !!row?.isStale,
            lastApiUpdate,
            dataAge,
          });
        }

        rebuildQuotes();
        return;
      }
    },
    [applyDesiredSubscriptions, enabledSymbols, rebuildQuotes]
  );

  const connect = useCallback(() => {
    clearTimers();
    setIsReady(false);

    if (!isAuthenticated) return;

    // close any previous socket
    try {
      wsRef.current?.close();
    } catch {
      // ignore
    }
    wsRef.current = null;

    const url = wsUrl();
    const ws = new WebSocket(url);
    wsRef.current = ws;

    ws.onopen = () => {
      setIsConnected(true);
      attemptRef.current = 0;
      flushPending();

      // heartbeat
      pingTimerRef.current = setInterval(() => {
        sendJson({ type: "ping", ts: Date.now() });
      }, 25_000);

      // If server doesn't send ws:ready quickly, still attempt subscriptions after short delay.
      setTimeout(() => applyDesiredSubscriptions(), 500);
    };

    ws.onmessage = (evt) => {
      try {
        const msg = JSON.parse(evt.data);
        handleMessage(msg);
      } catch {
        // ignore
      }
    };

    ws.onclose = () => {
      setIsConnected(false);
      setIsReady(false);
      clearTimers();
      scheduleReconnect();
    };

    ws.onerror = () => {
      // error typically followed by close; keep logic simple
    };
  }, [applyDesiredSubscriptions, clearTimers, flushPending, handleMessage, isAuthenticated, scheduleReconnect, sendJson]);

  // Expose for reconnect timer closure
  (connect as any).displayName = "connect";

  // Connect/disconnect on auth change
  useEffect(() => {
    if (!isAuthenticated) {
      clearTimers();
      setIsConnected(false);
      setIsReady(false);
      attemptRef.current = 0;
      pendingSendRef.current = [];
      lastSentSubsRef.current = new Set();
      desiredSubsRef.current = new Set();

      try {
        wsRef.current?.close();
      } catch {
        // ignore
      }
      wsRef.current = null;
      return;
    }

    connect();

    return () => {
      clearTimers();
      try {
        wsRef.current?.close();
      } catch {
        // ignore
      }
      wsRef.current = null;
    };
  }, [clearTimers, connect, isAuthenticated]);

  // Default subscription: all enabled symbols (huge win vs global broadcast)
  useEffect(() => {
    if (!isAuthenticated) return;

    const next = new Set<string>();
    for (const s of enabledSymbols) next.add(s.symbol);

    desiredSubsRef.current = next;
    applyDesiredSubscriptions();
  }, [applyDesiredSubscriptions, enabledSymbols, isAuthenticated]);

  const subscribeSymbols = useCallback(
    (symbols: string[]) => {
      const desired = new Set(desiredSubsRef.current);
      for (const s of symbols) {
        const n = normSymbol(s);
        if (n) desired.add(n);
      }
      desiredSubsRef.current = desired;
      applyDesiredSubscriptions();
    },
    [applyDesiredSubscriptions]
  );

  const unsubscribeSymbols = useCallback(
    (symbols: string[]) => {
      const desired = new Set(desiredSubsRef.current);
      for (const s of symbols) {
        const n = normSymbol(s);
        if (n) desired.delete(n);
      }
      desiredSubsRef.current = desired;
      applyDesiredSubscriptions();
    },
    [applyDesiredSubscriptions]
  );

  const hasStaleData = useMemo(() => quotes.some((q) => q.isStale), [quotes]);

  const value: LiveState = useMemo(
    () => ({
      isConnected,
      isReady,
      lastMessageAt,
      quotes,
      hasStaleData,
      subscribeSymbols,
      unsubscribeSymbols,
    }),
    [hasStaleData, isConnected, isReady, lastMessageAt, quotes, subscribeSymbols, unsubscribeSymbols]
  );

  return <LiveUpdatesContext.Provider value={value}>{children}</LiveUpdatesContext.Provider>;
}

export function useLiveUpdates() {
  const ctx = useContext(LiveUpdatesContext);
  if (!ctx) throw new Error("useLiveUpdates must be used within LiveUpdatesProvider");
  return ctx;
}
