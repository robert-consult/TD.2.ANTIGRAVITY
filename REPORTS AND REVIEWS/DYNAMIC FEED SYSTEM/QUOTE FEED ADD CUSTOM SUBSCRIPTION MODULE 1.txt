app.get("/api/admin/quote-subscriptions/traders", requireAdmin, async (req: Request, res: Response) => {
  try {
    const q = String(req.query.q ?? "").trim();
    const limit = Math.min(Math.max(Number(req.query.limit ?? 50) || 50, 1), 200);
    const offset = Math.max(Number(req.query.offset ?? 0) || 0, 0);

    const sqlText = `
      SELECT u.id, u.email, u.username, u.is_admin, u.is_disabled,
             tqp.quote_mode
      FROM users u
      LEFT JOIN trader_quote_prefs tqp ON tqp.user_id = u.id
      WHERE (
        $1 = ''
        OR u.email ILIKE '%' || $1 || '%'
        OR u.username ILIKE '%' || $1 || '%'
      )
      ORDER BY u.id DESC
      LIMIT $2 OFFSET $3
    `;
    const rows = await dbClient.query(sqlText, [q, limit, offset]);
    res.json({
      users: rows.rows.map((r: any) => ({
        id: Number(r.id),
        email: r.email != null ? String(r.email) : null,
        username: r.username != null ? String(r.username) : null,
        isAdmin: Boolean(r.is_admin),
        isDisabled: Boolean(r.is_disabled),
        overrideMode: r.quote_mode ? (String(r.quote_mode) as QuoteMode) : null,
      })),
      limit,
      offset,
    });
  } catch (error) {
    console.error("Error searching users for quote subscriptions:", error);
    res.status(500).json({ message: "Internal server error" });
  }
});

// Bulk set/clear quote mode overrides.
app.put("/api/admin/quote-subscriptions/traders/modes", requireAdmin, async (req: Request, res: Response) => {
  try {
    const schema = z.object({
      userIds: z.array(z.number().int().positive()).min(1).max(2000),
      mode: quoteModeSchema.nullable(),
    });
    const body = schema.parse(req.body);

    const uniqueUserIds = Array.from(
      new Set(body.userIds.map((x) => Number(x)).filter((x) => Number.isFinite(x) && x > 0))
    );
    if (!uniqueUserIds.length) return res.status(400).json({ message: "No valid user IDs" });

    if (body.mode == null) {
      await dbClient.query("DELETE FROM trader_quote_prefs WHERE user_id = ANY($1)", [uniqueUserIds]);
    } else {
      await dbClient.query(
        `
          INSERT INTO trader_quote_prefs (user_id, quote_mode, updated_at)
          SELECT unnest($1::int[]), $2, extract(epoch from now())
          ON CONFLICT (user_id) DO UPDATE SET
            quote_mode = EXCLUDED.quote_mode,
            updated_at = EXCLUDED.updated_at
        `,
        [uniqueUserIds, body.mode],
      );
    }

    publishLiveEvent({ type: "quote-subscriptions:updated", payload: { action: "bulk-mode", userIds: uniqueUserIds } });
    res.json({ ok: true, updated: uniqueUserIds.length });
  } catch (error) {
    console.error("Error bulk updating quote modes:", error);
    res.status(400).json({ message: (error as Error).message });
  }
});

function buildQuoteView(
  quote: any,
  prevCloseMap: Map<string, number>,
  nowMs: number,
  staleThresholdMs: number,
) {
  const bid = typeof quote.bid === "number" ? quote.bid : null;
  const ask = typeof quote.ask === "number" ? quote.ask : null;
  const lastPrice =
    typeof quote.price === "number"
      ? quote.price
      : typeof quote.lastPrice === "number"
        ? quote.lastPrice
        : null;
  const midPrice = bid != null && ask != null ? (bid + ask) / 2 : lastPrice;
  const spread = bid != null && ask != null ? Math.abs(ask - bid) : null;
  const prevClose =
    prevCloseMap.get(String(quote.symbol)) ??
    (typeof quote.prevClose === "number" ? quote.prevClose : null);

  let pctChange = 0;
  if (prevClose != null && prevClose > 0 && Number.isFinite(midPrice)) {
    pctChange = ((midPrice - prevClose) / prevClose) * 100;
    pctChange = Math.round(pctChange * 100) / 100;
  }

  const change = Number.isFinite(midPrice) && prevClose != null ? midPrice - prevClose : 0;

  const rawLastUpdate = Number(
    quote.lastApiUpdate ??
      quote.last_api_update ??
      quote.lastUpdated ??
      quote.updatedAt ??
      quote.updated_at ??
      nowMs,
  );
  const lastUpdate = rawLastUpdate < 1e12 ? rawLastUpdate * 1000 : rawLastUpdate;
  const ageMs = nowMs - lastUpdate;
  const dbIsStale = quote.isStale === 1 || quote.isStale === true;
  const marketOpen = isMarketOpenForSymbol(String(quote.symbol), new Date(nowMs));
  const isStale = dbIsStale || (marketOpen && ageMs > staleThresholdMs);

  return {
    symbol: quote.symbol,
    bid,
    ask,
    price: midPrice,
    spread,
    prevClose: prevClose ?? midPrice,
    change,
    pctChange,
    marketOpen,
    isStale,
    lastApiUpdate: lastUpdate,
    dataAge: ageMs,
  };
}
